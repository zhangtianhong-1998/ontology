19def _table(root, name, columns, rows):
20 """Use a distinct domain and blank metadata; names are not ontology rules."""
21 base = {"schema": "operations", "table_name": name}
22 write_yaml(root / "schema/tables" / f"{name}.yaml", {**base, "table_comment": "",
23 "columns": [{"column_name": column, "ordinal_position": i + 1, "data_type": "text",
24 "column_comment": comment, "is_not_null": False, "default_value": None}
25 for i, (column, comment) in enumerate(columns.items())]})
26 write_yaml(root / "schema/constraints" / f"{name}.yaml", {**base, "constraints": [
27 {"constraint_name": "pk", "constraint_type": "p", "definition": "PRIMARY KEY (id)"}]})
28 write_yaml(root / "schema/foreign_keys" / f"{name}.yaml", {**base, "foreign_keys": []})
29 (root / "data").mkdir(exist_ok=True)
30 with (root / "data" / f"{name}.csv").open("w", newline="") as stream:
31 writer = csv.DictWriter(stream, fieldnames=columns)
32 writer.writeheader()
33 writer.writerows(rows)

36def _proof(data, table, row, column):
37 record_id = data.record_id(table, row)
38 key = "record:" + digest([data.snapshot_id, record_id, column])[:24]
39 data.evidence[key] = {"id": key, "origin": "observed_record", "raw_fragment": row[column],
40 "raw_fragment_truncated": False, "source_ref": {"snapshot_id": data.snapshot_id,
41 "table": table, "row": row["__r2_row"], "record_id": record_id, "column": column}}
42 return key

45def _membership_case(tmp_path):
46 root = tmp_path / "input"
47 _table(root, "x17", dict.fromkeys(["id", "c1", "c2", "c3"], ""), [
48 {"id": "R1", "c1": "组件", "c2": "组件是独立装配的可复用部件", "c3": "R2"},
49 {"id": "R2", "c1": "组件", "c2": "组件是独立装配的可复用部件", "c3": "R1"},
50 {"id": "R3", "c1": "组件", "c2": "组件是独立装配的可复用部件", "c3": "R2"},
51 ])
52 work = tmp_path / "work"
53 work.mkdir()
54 data = Dataset(root, work)
55 table = "operations.x17"
56 rows = list(data.rows(table))
57 proof = _proof(data, table, rows[0], "c2")
58 item = DerivedType(id="type:component", parent="GeneralObject", category="business_type",
59 label="组件", definition=rows[0]["c2"], evidence_ids=[proof], source_properties=[{
60 "role": "description", "source_table": table, "source_column": "c2", "evidence_ids": [proof]}])
61 semantics = [["description", "c2", rows[0]["c2"]], ["name", "c1", rows[0]["c1"]]]
62 template = {"id": "definition_template:" + digest([data.snapshot_id, table, item.id, semantics])[:24],
63 "type_id": item.id, "table": table, "representative_record_id": data.record_id(table, rows[0]),
64 "representative_row_number": 1, "semantic_fields": [
65 {"role": role, "column": field, "value": value} for role, field, value in semantics],
66 "semantic_fingerprint": digest(semantics), "evidence_ids": [proof]}
67 members = [{"id": "member:" + row["id"], "table": table, "row_number": row["__r2_row"],
68 "record_id": data.record_id(table, row), "type_id": item.id, "template_id": template["id"],
69 "snapshot_id": data.snapshot_id, "status": "definition_template_match", "evidence_ids": [proof],
70 "entity_identity_claim": False, "relationship_inheritance": False,
71 "verification": {"method": "complete_original_source_field_equality",
72 "semantic_fingerprint": template["semantic_fingerprint"], "columns": ["c2", "c1"]}}
73 for row in rows]
74 return data, BuildPlan(object_types=[item]), {"memberships": members, "definition_templates": [template]}
