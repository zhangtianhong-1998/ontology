23 if data.get("simulate_error"):
24 raise RuntimeError("Synthetic knowledge source failure")
25 query_tokens = tokens(query)
26 ranked = sorted(((len(query_tokens & tokens(d["title"] + " " + d["text"])), d) for d in docs.values()), key=lambda x: (-x[0], x[1]["id"]))
27 return {"hits": [{"id": d["id"], "title": d["title"], "snippet": d["text"][:200]} for score, d in ranked if score > 0][:max(0, min(limit, 20))]}
