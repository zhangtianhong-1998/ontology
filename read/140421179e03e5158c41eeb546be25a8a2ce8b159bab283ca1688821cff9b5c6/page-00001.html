17def rdf_cards(path):
18 graph = Graph().parse(path, format="turtle" if path.suffix.lower() == ".ttl" else "xml")
19 resources = set()
20 for kind in (OWL.Class, RDFS.Class, OWL.ObjectProperty, OWL.DatatypeProperty, RDF.Property):
21 resources.update(graph.subjects(RDF.type, kind))
22 for node in sorted(resources, key=str):
23 if not str(node).startswith(("http://", "https://")):
24 continue
25 labels = list(graph.objects(node, RDFS.label)) + list(graph.objects(node, SKOS.prefLabel))
26 definitions = list(graph.objects(node, RDFS.comment)) + list(graph.objects(node, SKOS.definition))
27 yield {"uri": str(node), "label": str(labels[0]) if labels else str(node).split("#")[-1].split("/")[-1], "aliases": [str(x) for x in labels[1:]] + [str(x) for x in graph.objects(node, SKOS.altLabel)], "definition": "\n".join(map(str, definitions)), "source_languages": sorted({str(x.language) for x in labels + definitions if getattr(x, "language", None)}), "parents": [str(x) for x in graph.objects(node, RDFS.subClassOf) if isinstance(x, URIRef)], "supported_semantics": "labels_definitions_named_parents", "unsupported_semantics": "OWL axioms are retained only in the original source"}

30def cdm_cards(path, base):
31 raw = path.read_bytes()
32 encoding = "utf-8-sig"
33 try:
34 text = raw.decode(encoding)
35 except UnicodeDecodeError:
36 encoding = "cp1252"
37 text = raw.decode(encoding)
38 content = json.loads(text)
39 for definition in content.get("definitions", []):
40 name = definition.get("entityName")
41 if name:
42 yield {"uri": "cdm:" + str(path.relative_to(base)) + "#" + name, "label": name, "definition": definition.get("description", ""), "parents": [definition.get("extendsEntity")] if definition.get("extendsEntity") else [], "source_encoding": encoding, "supported_semantics": "entity_name_description_parent_reference", "unsupported_semantics": "traits, attribute groups and imported constraints are not interpreted"}

101 def search_many(self, queries, limit=5):
102 scores, cards, channels, similarities = {}, {}, {}, {}
103 for query in dict.fromkeys(queries):
104 for rank, card in enumerate(self.search(query, limit), 1):
105 key = card["uri"]
106 cards[key] = card
107 scores[key] = scores.get(key, 0) + 1 / (60 + rank)
108 channels.setdefault(key, set()).add("fts5")
109 if self.embedding is not None and query.strip() and len(self.vectors):
110 for rank, (index, similarity) in enumerate(
111 top_cosine(self.vectors, self.embedding.query(query), limit, self.vector_norms), 1):
112 if similarity < self.embedding.config["min_cosine_similarity"]:
113 continue
114 card = self.cards[index]
115 key = card["uri"]
116 cards[key] = card
117 scores[key] = scores.get(key, 0) + 1 / (60 + rank)
118 channels.setdefault(key, set()).add("cosine")
