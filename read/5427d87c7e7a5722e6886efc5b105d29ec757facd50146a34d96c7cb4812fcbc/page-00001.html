27@pytest.mark.parametrize("batch_size", [1, 2])
28def test_exact_metric_response_is_repaired_before_any_review_or_acceptance(batch_size):
29 data, bundle = setup()
30 # Mirrors the live failure: the explanation promises a class and slots,
31 # but the actual action still attempts to create a type from an instance.
32 wrong = ConceptBundleDecision(
33 status="proposed", root_type="Metric", action="exact_definition",
34 label="苹果合格率", definition="苹果合格率按华东汇总",
35 ontology_level="type", classification_basis="business_driven_metric",
36 classification_quote="水果合格率统计质量合格比例。", business_object_quote="水果",
37 scope_roles={"x7": "applicability"},
38 alignments=[{"record_id": "row1", "mapping_kind": "exact", "quote": "苹果合格率"}],
39 reason="通用类与配置实例应分离，保留对象与度量槽位")

41 class LLM:
42 def __init__(self):
43 self.tasks = []
44
45 async def ask(self, task, payload, schema):
46 self.tasks.append(task)
47 if task == "concept_batch":
48 return ConceptBatchDecision.model_validate({"decisions": [
49 {"bundle_id": packet["bundle"]["bundle_id"], "decision": wrong.model_dump()}
50 for packet in payload["packets"]]})
51 if task == "concept_bundle":
52 if "compiler_error" not in payload:
53 return wrong
54 assert "requires project_template" in payload["compiler_error"]
55 assert "Metric and GeneralObject require project_template" in payload["repair_instruction"]
56 return ConceptBundleDecision(status="proposed", action="project_template", projection=decision())
57 # Invalid batch or single proposals must never reach the reviewer.
58 assert task == "group_review"
59 assert payload["candidate"]["action"] == "project_template"
60 return BundleReview(accepted=True)
61
62 bundles = [bundle]
63 if batch_size > 1:
64 bundles.append({**bundle, "bundle_id": "b2"})
65 llm = LLM()
66 result = asyncio.run(construct_from_bundles(
67 data, PROFILE, BuildPlan(), bundles, llm, enable_template_projection=True,
68 concept_batch_size=batch_size, max_repairs_per_bundle=1))
69 assert result["steps"][0]["status"] == "accepted"
70 assert result["steps"][0]["repair_calls"] == 1
71 assert len(result["template_projections"]) == 1
72 assert not result["concepts"] and not result["record_alignments"]
73 assert "苹果合格率" not in {item.label for item in result["plan"].object_types}
74 assert "group_review_batch" not in llm.tasks
