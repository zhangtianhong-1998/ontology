76class FixtureDecisions:
77 """Explicit expected decisions; never presented as measured model quality."""
78
79 async def ask(self, task, payload, schema):
80 if task == "concept_bundle":
81 seed = payload["bundle"]["records"][0]
82 name = seed["fields"]["name"][0]["value"]
83 definition = seed["fields"]["description"][0]["value"]
84 alignments = []
85 exact_allowed = set(payload["bundle"].get(
86 "exact_alignment_record_ids", [seed["record_id"]]))
87 for record in payload["bundle"]["records"]:
88 if (record["record_id"] in exact_allowed
89 and record.get("kind") == "definition"
90 and record.get("root_hint") == seed.get("root_hint")
91 and record.get("fields", {}).get("name", [{}])[0].get("value") == name
92 and record.get("unit", "") == seed.get("unit", "")):
93 alignments.append(RecordAlignmentDecision(
94 record_id=record["record_id"], mapping_kind="exact", quote=name))
95 return ConceptBundleDecision(
96 status="proposed", label=name, definition=definition,
97 root_type=seed["root_hint"], ontology_level="type",
98 classification_basis=("business_driven_metric" if seed["root_hint"] == "Metric"
99 else "reusable_measure" if seed["root_hint"] == "Measure"
100 else "other"),
101 classification_quote=definition,
102 business_object_quote=("水果销售" if seed["root_hint"] == "Metric" else ""),
103 alignments=alignments,
104 )
105 if task == "relation_bundle":
106 bundle = payload["bundle"]
107 pair = bundle["examples"]["positive"][0]
108 records = {item["record_id"]: item for item in bundle["records"]}
109 source = records[pair["source_record_id"]]
110 target = records[pair["target_record_id"]]
111 return RelationBundleDecision(
112 status="proposed", parent_relation="depends_on",
113 label="depends_on",
114 definition="水果销售收入指标的公式引用通用收入度量，并对收入求和",
115 source_quote=source["fields"]["name"][0]["value"],
116 target_quote=target["fields"]["name"][0]["value"],
117 )
118 if task == "group_review":
119 return BundleReview(accepted=True)
120 raise AssertionError(task)

123def run(output):
124 output = Path(output)
125 if output.exists() and any(output.iterdir()):
126 raise FileExistsError("Experiment output already contains files: " + str(output))
127 output.mkdir(parents=True, exist_ok=True)
128 root, work = output / "input", output / "work"
129 make_fixture(root)
130 work.mkdir()
131 profile = read_yaml(Path(__file__).resolve().parents[1] / "ontologies/internal_model.yaml")
132 data = Dataset(root, work)
133 try:
134 cards = build_semantic_cards(data, work / "cards.sqlite", max_cards=100)
135 index = SemanticCardIndex(cards["index_path"])
136 candidate = {
137 "candidate_id": "fixture:metric_measure_code",
138 "source": {"table": "fruit.fruit_metric_definition", "field": "measure_code"},
139 "target": {"table": "fruit.fruit_measure_definition", "field": "measure_code"},
140 "retrieval_channels": ["fixture_declared_candidate"],
141 "suggested_selector": {}, "suggested_scope_bindings": {},
142 "numeric_overlap_only": False,
143 }
144 checked = validate_candidate(data, candidate)
145 association = asyncio.run(build_association_rules(
146 data, {"candidates": [candidate], "checks": [checked]},
147 {"agent_enabled": False, "max_rules": 1}))
148 try:
149 packets = build_instance_bundles(data, index, association, {
150 "max_concept_bundles": 3, "max_relation_bundles": 1,
151 "max_candidates_per_bundle": 2, "max_joint_pairs_per_rule": 1,
152 })
153 finally:
154 index.close()
155 core, mapping = direct_mapping(data)
156 result = asyncio.run(construct_from_bundles(
157 data, profile, core, packets["bundles"], FixtureDecisions(),
158 max_bundles=4, review=True))
159 business_types = [item for item in result["plan"].object_types
160 if item.category == "business_type"]
161 relation_types = result["plan"].relation_types
162 if len(result["concept_relations"]) != 1:
163 raise AssertionError("Controlled positive pair did not lift to a business relation: "
164 + repr({"steps": result["steps"],
165 "derivations": result["concept_relation_derivations"]}))
166 sink = Sink(output)
167 try:
168 for item in data.evidence.values():
169 sink.put("evidence", item)
170 for item in result["concepts"]:
171 sink.put("objects", item)
172 for item in result["record_alignments"]:
173 sink.put("record_alignments", item)
174 for item in result["concept_relations"]:
175 sink.put("assertions", item)
176 extractor = Extractor(data, sink, result["plan"], FixtureDecisions(),
177 {"max_relation_records": 100})
178 materialized = 0
179 for table in result["plan"].tables:
180 for row in data.rows(table.table):
181 extractor.object(table.table, row)
182 materialized += 1
183 extraction = asyncio.run(extractor.execute())
184 counts = sink.export()
185 validation = check_output(sink)
186 finally:
187 sink.close()
188 if extraction["accepted"] != 1 or not validation["passed"]:
189 raise AssertionError("Controlled relation materialization failed structural checks")
190 manifest = {
191 "status": "complete", "experimental_scope": "controlled_synthetic_fixture_only",
192 "input_tables": len(data.tables), "input_records": 3,
193 "viewer": "viewer.html",
194 "llm": {"mode": "fixed_fixture_decisions", "calls": 0},
195 "semantic_quality": "unjudged; no live LLM or independent business Gold",
196 }
197 summary = {
198 "experiment": "controlled_synthetic_group_compilation",
199 "snapshot_id": data.snapshot_id,
200 "fixture_rows": sum(table["rows"] for table in data.tables.values()),
201 "semantic_cards": cards["coverage"]["cards_indexed"],
202 "checked_technical_rules": sum(rule["status"] == "checked_technical"
203 for rule in association["rules"]),
204 "concept_bundles": packets["coverage"]["bundles_by_task"]["concept_induction"],
205 "relation_bundles": packets["coverage"]["bundles_by_task"]["relation_meaning"],
206 "accepted_business_types": [{"id": item.id, "parent": item.parent,
207 "label": item.label} for item in business_types],
208 "accepted_relation_types": [{"id": item.id, "parent": item.parent,
209 "category": item.category,
210 "domain": item.domain, "range": item.range,
211 "evidence_scope": item.evidence_scope}
212 for item in relation_types],
213 "relation_level": "source_record_relation_and_one_exact_concept_pair",
214 "step_statuses": [item["status"] for item in result["steps"]],
215 "materialized_source_records": materialized,
216 "source_record_relation_assertions": extraction["accepted"],
217 "concept_relation_assertions": len(result["concept_relations"]),
218 "structural_validation": validation,
219 "viewer": "viewer.html",
220 "evaluation_scope": "engineering_contract_only; canned decisions; no live LLM or semantic Gold",
221 }
222 write_yaml(output / "semantic_card_coverage.yaml", cards["coverage"])
223 write_yaml(output / "association_rules.yaml", association)
224 write_yaml(output / "evidence_bundles.yaml", packets["bundles"])
225 write_yaml(output / "group_steps.yaml", result["steps"])
226 write_yaml(output / "evidence.yaml", list(data.evidence.values()))
227 write_yaml(output / "business_concepts.yaml", result["concepts"])
228 write_yaml(output / "record_alignments.yaml", result["record_alignments"])
229 write_yaml(output / "concept_relations.yaml", result["concept_relations"])
230 write_yaml(output / "concept_relation_derivations.yaml",
231 result["concept_relation_derivations"])
232 write_yaml(output / "extraction_plan.yaml", result["plan"].model_dump())
233 write_yaml(output / "ontology.yaml", ontology_from_plan(
234 result["plan"], profile, data, mapping, []))
235 write_yaml(output / "meta_graph.yaml", technical_graph(data))
236 write_yaml(output / "construction.yaml", {
237 "steps": [], "group_steps": result["steps"],
238 "direct_mapping_columns": sum(len(table["columns"]) for table in data.tables.values()),
239 })
240 write_yaml(output / "coverage.yaml", {"relation_plans": extraction["plans"],
241 "input_scope": "six_synthetic_definition_rows"})
242 write_yaml(output / "metrics.yaml", {"counts": counts, "extraction": extraction,
243 "materialized_records": materialized,
244 "semantic_quality": "unjudged"})
245 write_yaml(output / "validation.yaml", validation)
246 write_yaml(output / "manifest.yaml", manifest)
247 write_yaml(output / "summary.yaml", summary)
248 render_viewer(output, 50)
249 return summary
250 finally:
251 data.close()
