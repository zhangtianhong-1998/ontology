71def _configuration_case(tmp_path, wording):
72 data, plan, concepts, alignments, spec = _input(tmp_path)
73 table = data.tables[spec["configuration_table"]]
74 data.db.execute(f"UPDATE {qi(table['sql_name'])} SET relation_text=?", [wording])
75 rules = [_rule("metric-ref", spec, "metric_ref", spec["source_definition_table"], "metric_code"),
76 _rule("measure-ref", spec, "measure_ref", spec["target_definition_table"], "measure_code")]
77 for rule in rules:
78 rule["snapshot_id"] = data.snapshot_id
79 inferred = infer_configuration_specs(data, plan, concepts, alignments, rules)
80 candidate = next(item for item in discover_configuration_relations(
81 data, plan, concepts, alignments, [inferred["spec_candidates"][0]["spec"]])["candidates"]
82 if item["status"] == "endpoint_verified_candidate")
83 return data, plan, concepts, alignments, candidate, rules

86class _PurposeModel:
87 def __init__(self):
88 self.calls = 0
89
90 async def ask(self, stage, packet, model):
91 self.calls += 1
92 assert stage == "configuration_purpose"
93 evidence = next(item for item in packet["evidence"]
94 if item.get("column") == "relation_text" and item["kind"] == "configuration_value")
95 return model.model_validate({"status": "proposed", "role": "measure",
96 "evidence_id": evidence["evidence_id"], "purpose_quote": evidence["text"],
97 "reason": "The configuration declares which reusable quantity the performance definition uses."})

194def _numeric_bridge_case(tmp_path):
195 """A two-column PK whose components independently reference definition IDs."""
196 data, plan, concepts, alignments, spec = _input(tmp_path)
197 table = data.tables[spec["configuration_table"]]
198 data.db.execute(f"DELETE FROM {qi(table['sql_name'])} WHERE id <> '21'")
199 data.db.execute(f"UPDATE {qi(table['sql_name'])} SET metric_ref='1', measure_ref='11', "
200 "relation_text='度量提供指标的量定义'")
201 table["pk"] = ["metric_ref", "measure_ref"]
202 rules = []
203 for field, target in (("metric_ref", spec["source_definition_table"]),
204 ("measure_ref", spec["target_definition_table"])):
205 schema, name = target.split(".", 1)
206 table["foreign_keys"].append({"fk_name": "fk_" + field, "column_name": field,
207 "referenced_schema": schema, "referenced_table": name, "referenced_column": "id"})
208 rule = _rule("checked:" + field, spec, field, target, "id")
209 rule.update(numeric_overlap_only=True, snapshot_id=data.snapshot_id)
210 rule["verification"]["checks"].update(eligible_references=1, unique_matches=1)
211 rules.append(rule)
212 return data, plan, concepts, alignments, rules
