1"""Load an ontology-r2 DataHub metadata file into local DataHub Lite.
2
3Optional demo, tested with ``acryl-datahub[datahub-lite]==1.7.0.12``:
4
5 python scripts/import_datahub_lite.py run/datahub-metadata.json --catalog run/datahub-lite.duckdb
6
7Only datasetProperties and schemaMetadata are accepted. This script provides
8local storage and read-back verification; it does not expose graph traversal,
9lineage discovery, or the full DataHub GMS/UI server.
10"""
11
12import argparse
13import json
14import os
15from importlib.metadata import PackageNotFoundError, version
16from pathlib import Path
17
18
19DATAHUB_VERSION = "1.7.0.12"
20ALLOWED_ASPECTS = {"datasetProperties", "schemaMetadata"}
21
22# Lite is used as a local file store here. Disable the SDK's default usage
23# telemetry before importing any DataHub module.
24os.environ["DATAHUB_TELEMETRY_ENABLED"] = "false"
25
26
27def import_metadata_file(metadata_file: Path, catalog_file: Path) -> dict:
28 try:
29 installed_version = version("acryl-datahub")
30 except PackageNotFoundError as exc:
31 raise RuntimeError(f"Install acryl-datahub[datahub-lite]=={DATAHUB_VERSION} in this Python environment") from exc
32 if installed_version != DATAHUB_VERSION:
33 raise RuntimeError(f"Tested with acryl-datahub=={DATAHUB_VERSION}; found {installed_version}")
34
35 from datahub.emitter.mcp import MetadataChangeProposalWrapper
36 from datahub.lite.duckdb_lite import DuckDBLite
37 from datahub.lite.duckdb_lite_config import DuckDBLiteConfig
38
39 records = json.loads(metadata_file.read_text(encoding="utf-8"))
40 if not isinstance(records, list):
41 raise ValueError("Expected a DataHub metadata-file JSON array")
42
43 mcps = []
44 expected = {}
45 for record in records:
46 if not isinstance(record, dict) or record.get("entityType") != "dataset" or record.get("changeType") != "UPSERT" or record.get("aspectName") not in ALLOWED_ASPECTS:
47 raise ValueError("Only dataset UPSERTs with datasetProperties/schemaMetadata are supported")
48 mcp = MetadataChangeProposalWrapper.from_obj(record.copy())
49 if not isinstance(mcp, MetadataChangeProposalWrapper) or not mcp.validate():
50 raise ValueError("Invalid DataHub metadata proposal")
51 key = (mcp.entityUrn, mcp.aspectName)
52 if key in expected:
53 raise ValueError(f"Duplicate dataset aspect: {key}")
54 expected[key] = mcp.to_obj(simplified_structure=True)["aspect"]["json"]
55 mcps.append(mcp)
56
57 catalog_file.parent.mkdir(parents=True, exist_ok=True)
58 # DataHub Lite's close() reindexes its DuckDB, so keep the default
59 # read_only=False even for this read-back check.
60 lite = DuckDBLite(DuckDBLiteConfig(file=str(catalog_file)))
61 try:
62 for mcp in mcps:
63 lite.write(mcp)
64 for (urn, aspect_name), body in expected.items():
65 stored = lite.get(urn, [aspect_name])
66 if not stored or stored.get(aspect_name) != body:
67 raise RuntimeError(f"DataHub Lite read-back differs for {urn} / {aspect_name}")
68 finally:
69 lite.close()
70 return {"catalog": str(catalog_file), "datasets": len({urn for urn, _ in expected}), "aspects_verified": len(expected)}
71
72
73def main() -> None:
74 parser = argparse.ArgumentParser(description="Import table/column MCPs into local DataHub Lite and verify read-back")
75 parser.add_argument("metadata_file", type=Path, help="DataHub metadata-file JSON exported by route 2")
76 parser.add_argument("--catalog", type=Path, required=True, help="Local DuckDB catalog file")
77 args = parser.parse_args()
78 print(json.dumps(import_metadata_file(args.metadata_file, args.catalog), ensure_ascii=False))
79
80
81if __name__ == "__main__":
82 main()
