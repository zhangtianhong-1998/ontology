1"""Bounded, source-backed candidates for rows that look like business facts.
2
3An observation candidate is a distinct tuple *present in the imported CSV*.
4It is not an ontology type, a business-key assertion, or a Metric/Measure
5instance. The SQL groups only observed rows; it never produces a Cartesian
6product of coordinate values. Definition and unresolved tables remain in
7the coverage report rather than disappearing from the extraction pipeline.
8"""

22def _accepted_dimension_evidence(data, links):
23 """Use only caller-accepted, snapshot-matched Dimension links for ranking.
24
25 ``links`` is an optional iterable of mappings with source_table,
26 source_column, target_table, target_column, target_root_type='Dimension',
27 status='accepted', witness_snapshot_id, evidence_ids, and
28 full_input_verified=True. This function cannot itself prove semantic
29 acceptance; the caller must pass only accepted ontology relations.
30 """
31 accepted = {}
32 ignored = 0
33 for link in links or ():
34 if not isinstance(link, dict):
35 ignored += 1
36 continue
37 table = link.get("source_table")
38 field = link.get("source_column")
39 target = link.get("target_table")
40 target_field = link.get("target_column")
41 valid = (link.get("status") == "accepted"
42 and link.get("target_root_type") == "Dimension"
43 and link.get("witness_snapshot_id") == data.snapshot_id
44 and link.get("full_input_verified") is True
45 and bool(link.get("evidence_ids"))
46 and table in data.tables and target in data.tables
47 and field in data.tables[table]["column_names"]
48 and target_field in data.tables[target]["column_names"])
49 if not valid:
50 ignored += 1
51 continue
52 accepted.setdefault(table, {}).setdefault(field, []).append(link["evidence_ids"])
53 return accepted, ignored

56def _verified_technical_fields(data, links):
57 """Rank only: a checked raw-value match does not establish a Dimension."""
58 verified = {}
59 ignored = 0
60 for link in links or ():
61 if not isinstance(link, dict):
62 ignored += 1
63 continue
64 source, target = link.get("source"), link.get("target")
65 verification = link.get("verification") or {}
66 checks = verification.get("checks") or {}
67 eligible = checks.get("eligible_references")
68 valid = (link.get("status") == "checked_technical"
69 and link.get("snapshot_id") == data.snapshot_id
70 and verification.get("scan_scope") == "full_input"
71 and isinstance(eligible, int) and eligible > 0
72 and checks.get("unique_matches") == eligible
73 and checks.get("missing_scope", 0) == 0
74 and isinstance(source, dict) and isinstance(target, dict))
75 if not valid:
76 ignored += 1
77 continue
78 ends = ((source.get("table"), source.get("field")),
79 (target.get("table"), target.get("field")))
80 if any(table not in data.tables or field not in data.tables[table]["column_names"]
81 for table, field in ends):
82 ignored += 1
83 continue
84 for table, field in ends:
85 verified.setdefault(table, set()).add(field)
86 return verified, ignored

89def _field_priority(table, name, cue, *, accepted=(), technical=()):
90 profile = next((item for item in table.get("profiles", ())
91 if item.get("column") == name), {})
92 count = profile.get("approx_distinct_usable")
93 # Sorting only schedules a bounded candidate scan; it is not a semantic
94 # assertion or a proof that a selected set is the complete fact grain.
95 return (name not in accepted, name not in technical,
96 not bool(cue.search(name)),
97 name in table.get("pk", ()),
98 count if isinstance(count, (int, float)) else float("inf"),
99 name)

102def _selected_fields(table, purpose, *, max_dimensions, max_times, max_values,
103 accepted_dimensions=(), technical_links=()):
104 evidence = purpose["evidence_columns"]
105 all_times = sorted(set(evidence["business_time"]),
106 key=lambda name: _field_priority(
107 table, name, _TIME, technical=technical_links))
108 # A value column comment may mention the business object (for example
109 # "水果销售利润（元）") and therefore also match a dimension cue. Numeric
110 # value evidence takes precedence over that lexical mention.
111 all_values = sorted((name for name in set(evidence["numeric_business_value"])
112 if name not in all_times),
113 key=lambda name: _field_priority(table, name, _VALUE))
114 all_dimensions = sorted((name for name in set(evidence["dimension_coordinate"])
115 if name not in all_times and name not in all_values),
116 key=lambda name: _field_priority(
117 table, name, _DIMENSION, accepted=accepted_dimensions,
118 technical=technical_links))
119 # Time and dimension are disjoint; a column that matches both cues has the
120 # time role. This is a selection rule, not a semantic claim about the DB.
121 dimensions = all_dimensions[:max_dimensions]
122 times = all_times[:max_times]
123 values = all_values[:max_values]
124 return {
125 "dimensions": dimensions,
126 "business_times": times,
127 "values": values,
128 "omitted_dimensions": all_dimensions[max_dimensions:],
129 "omitted_business_times": all_times[max_times:],
130 "omitted_values": all_values[max_values:],
131 "selection_method": "accepted_dimension_link_then_verified_technical_hint_then_name_and_profile_rank",
132 "accepted_dimension_fields_selected": [name for name in dimensions
133 if name in accepted_dimensions],
134 "technical_link_fields_selected": [name for name in [*dimensions, *times]
135 if name in technical_links],
136 }

139def _query_observed(data, table, coordinates, value_column, limit):
140 """One bounded result set from exact GROUP BY over actual rows only."""
141 fields = [*coordinates, value_column]
142 complete = " AND ".join(
143 f"{qi(field)} IS NOT NULL AND length(trim({qi(field)})) > 0"
144 for field in fields
145 )
146 coordinate_sql = ", ".join(qi(field) for field in coordinates)
147 select_sql = ", ".join(f"{qi(field)} AS {qi(field)}" for field in coordinates)
148 using_sql = ", ".join(qi(field) for field in coordinates)
149 order_sql = "md5(concat_ws(chr(31), " + ", ".join(
150 [*(f"o.{qi(field)}" for field in coordinates), "o.observation_value"]) + "))"
151 sql = f"""
152 WITH observed AS (
153 SELECT {select_sql}, {qi(value_column)} AS observation_value,
154 count(*) AS source_row_count,
155 min(__r2_row) AS source_row_min,
156 max(__r2_row) AS source_row_max
157 FROM {qi(table['sql_name'])}
158 WHERE {complete}
159 GROUP BY {coordinate_sql}, {qi(value_column)}
160 ), coordinates AS (
161 SELECT {coordinate_sql}, count(*) AS distinct_value_count
162 FROM observed GROUP BY {coordinate_sql}
163 )
164 SELECT o.*, c.distinct_value_count,
165 count(*) OVER () AS all_observed_tuples,
166 sum(o.source_row_count) OVER () AS all_complete_rows
167 FROM observed o JOIN coordinates c USING ({using_sql})
168 ORDER BY {order_sql}, o.observation_value
169 LIMIT ?
170 """
171 cursor = data.db.execute(sql, [limit])
172 names = [item[0] for item in cursor.description]
173 return [dict(zip(names, row)) for row in cursor.fetchall()]

176def _estimate_tuple_count(data, table, coordinates, value_column):
177 fields = [*coordinates, value_column]
178 complete = " AND ".join(
179 f"{qi(field)} IS NOT NULL AND length(trim({qi(field)})) > 0"
180 for field in fields)
181 args = ", ".join(qi(field) for field in fields)
182 return data.db.execute(
183 f"SELECT approx_count_distinct(hash({args})) FROM {qi(table['sql_name'])} "
184 f"WHERE {complete}").fetchone()[0] or 0

187def _query_seeded(data, table, coordinates, value_column, limit):
188 """Search a bounded set of observed tuples, then verify their whole grain.
189
190 A selected coordinate is checked against every matching source row. Min/max
191 detects conflicting raw values without materializing all distinct values
192 of a high-cardinality coordinate. Coverage stays partial because other
193 source coordinates were not selected.
194 """
195 fields = [*coordinates, value_column]
196 complete = " AND ".join(
197 f"{qi(field)} IS NOT NULL AND length(trim({qi(field)})) > 0"
198 for field in fields)
199 coordinate_sql = ", ".join(qi(field) for field in coordinates)
200 raw_fields = ", ".join(qi(field) for field in fields)
201 priority_sql = f"md5(concat_ws(chr(31), {raw_fields}))"
202 matches_coord = " AND ".join(f"r.{qi(field)} = s.{qi(field)}" for field in coordinates)
203 joins_coord = " AND ".join(f"m.{qi(field)} = c.{qi(field)}" for field in coordinates)
204 grouped = ", ".join(f"s.{qi(field)}" for field in coordinates)
205 projection = ", ".join(f"m.{qi(field)}" for field in coordinates)
206 sql = f"""
207 WITH sampled AS (
208 SELECT DISTINCT {coordinate_sql}, {qi(value_column)} AS observation_value,
209 {priority_sql} AS candidate_rank
210 FROM {qi(table['sql_name'])}
211 WHERE {complete}
212 ORDER BY candidate_rank LIMIT ?
213 ), seeds AS (
214 SELECT DISTINCT {coordinate_sql}, observation_value, candidate_rank
215 FROM sampled ORDER BY candidate_rank LIMIT ?
216 ), seed_coordinates AS (
217 SELECT DISTINCT {coordinate_sql} FROM seeds
218 ), scoped AS (
219 SELECT {', '.join('r.' + qi(field) for field in coordinates)},
220 r.{qi(value_column)} AS observation_value, r.__r2_row
221 FROM {qi(table['sql_name'])} r JOIN seed_coordinates s
222 ON {matches_coord}
223 WHERE r.{qi(value_column)} IS NOT NULL
224 AND length(trim(r.{qi(value_column)})) > 0
225 ), coordinate_status AS (
226 SELECT {coordinate_sql}, min(observation_value) AS min_value,
227 max(observation_value) AS max_value
228 FROM scoped GROUP BY {coordinate_sql}
229 ), matches AS (
230 SELECT {grouped}, s.observation_value,
231 count(*) AS source_row_count,
232 min(r.__r2_row) AS source_row_min,
233 max(r.__r2_row) AS source_row_max
234 FROM seeds s JOIN scoped r ON {matches_coord}
235 AND s.observation_value = r.observation_value
236 GROUP BY {grouped}, s.observation_value
237 )
238 SELECT {projection}, m.observation_value, m.source_row_count,
239 m.source_row_min, m.source_row_max,
240 CASE WHEN c.min_value = c.max_value THEN 1 ELSE 2 END
241 AS distinct_value_count
242 FROM matches m JOIN coordinate_status c ON {joins_coord}
243 ORDER BY md5(concat_ws(chr(31), {projection}, m.observation_value))
244 LIMIT ?
245 """
246 cursor = data.db.execute(sql, [max(limit * 4, limit), limit, limit])
247 names = [item[0] for item in cursor.description]
248 return [dict(zip(names, row)) for row in cursor.fetchall()]

251def _unmodeled_grain_fields(table, selected):
252 known = set(table.get("pk", ())) | set(selected["dimensions"]) | set(selected["business_times"])
253 known.update(selected["omitted_dimensions"])
254 known.update(selected["omitted_business_times"])
255 known.update(selected["values"])
256 known.update(selected["omitted_values"])
257 profiles = {item["column"]: item for item in table.get("profiles", ())}
258 possible, unprofiled = [], []
259 for item in classify_columns(table):
260 name = item["column"]
261 if name in known or item["role"] not in ("semantic", "unknown"):
262 continue
263 profile = profiles.get(name)
264 if not profile or profile.get("scan_scope") != "full_input":
265 unprofiled.append(name)
266 elif (profile.get("approx_distinct_usable") or 0) > 1:
267 possible.append(name)
268 return possible, unprofiled
