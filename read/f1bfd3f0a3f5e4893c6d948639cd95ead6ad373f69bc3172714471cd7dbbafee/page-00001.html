86def predicate_endpoint_errors(relation, object_types):
87 """Validate the endpoint meaning of registered cross-root predicates."""
88 contract = _PREDICATE_ENDPOINTS.get(relation.predicate_name)
89 if contract is None:
90 return []
91 types = {item.id: item for item in object_types}
92
93 def root(key):
94 seen = set()
95 while key in types and key not in seen:
96 seen.add(key)
97 key = types[key].parent
98 return key
99
100 if (len(relation.domain) != 1 or len(relation.range) != 1
101 or root(relation.domain[0]) not in contract[0]
102 or root(relation.range[0]) not in contract[1]):
103 return ["Predicate endpoint roots disagree with the registered relation contract"]
104 return []

107def _classification_scope_property(item, key, data):
108 """Only original field metadata can identify a classification attribute.
109
110 A matching field name or a model-written type label is insufficient. The
111 value must have a complete source proof and its schema must explicitly say
112 category/classification. Tenant, version and unknown scopes stay strict.
113 """
114 if data is None:
115 return False
116 for prop in item.source_properties:
117 if prop.role != "scope" or prop.source_column != key:
118 continue
119 table = data.tables.get(prop.source_table, {})
120 column = next((value for value in table.get("columns", [])
121 if value.get("column_name") == key), {})
122 description = str(column.get("column_comment") or "")
123 if not re.search(r"分类|类别|\b(?:category|classification)\b", description, re.I):
124 continue
125 if re.search(r"租户|命名空间|版本|tenant|namespace|version", description, re.I):
126 continue
127 for eid in prop.evidence_ids:
128 proof = data.evidence.get(eid, {})
129 ref = proof.get("source_ref", {})
130 if (proof.get("origin") == "observed_record" and not proof.get("raw_fragment_truncated")
131 and ref.get("snapshot_id") == data.snapshot_id
132 and ref.get("table") == prop.source_table and ref.get("column") == key
133 and str(proof.get("raw_fragment")) == str(item.applicability_scope[key])):
134 return True
135 return False

138def relation_scope_conflicts(source, target, data=None):
139 """Return incompatible scope keys without treating categories as namespaces.
140
141 This permits relations between differently classified objects. It never
142 merges their definitions, relaxes identity qualifiers, or proves a relation.
143 """
144 conflicts = []
145 for key in source.applicability_scope.keys() & target.applicability_scope.keys():
146 if source.applicability_scope[key] == target.applicability_scope[key]:
147 continue
148 if all(_classification_scope_property(item, key, data) for item in (source, target)):
149 continue
150 conflicts.append(key)
151 for key in source.identity_qualifiers.keys() & target.identity_qualifiers.keys():
152 if source.identity_qualifiers[key] != target.identity_qualifiers[key]:
153 conflicts.append("identity:" + key)
154 return sorted(conflicts)

168def canonical_relation_label(parent: str, predicate_name: str | None = None) -> str:
169 """Validate a root or a registered derived predicate and return its name."""
170 try:
171 root = RELATION_VERBS[parent]

177 raise ValueError("Unknown derived predicate or incompatible relation root")

182 """Only reproducible calculation roles currently qualify relation identity."""

214def validate_calculation_parameter_evidence(parameters, formulas, target_names):
215 """Prove an operand role from a parsed binary expression, never prose."""
216 if not parameters:
217 return
218 wanted = parameters["operand_role"]
219 side_roles = {ast.Sub: ("minuend", "subtrahend"),
220 ast.Div: ("numerator", "denominator"),
221 ast.Add: ("addend", "addend"), ast.Mult: ("factor", "factor")}
222 names = {str(value).strip() for value in target_names}
223 for raw in formulas:
224 expression = unicodedata.normalize("NFKC", raw).strip()
225 if "=" in expression:
226 expression = expression.split("=", 1)[1].strip()
227 try:
228 tree = ast.parse(expression, mode="eval")
229 except (SyntaxError, ValueError):
230 continue
231 for node in ast.walk(tree):
232 if not isinstance(node, ast.BinOp) or type(node.op) not in side_roles:
233 continue
234 for side, role in zip((node.left, node.right), side_roles[type(node.op)]):
235 if role == wanted and isinstance(side, ast.Name) and side.id in names:
236 return
237 raise ValueError("Calculation operand role lacks parsed source formula evidence")

240def canonical_relation_id(parent: str, domain: str, range_type: str, *,
241 namespace: str = "business_relation",
242 qualifier: list[str] | tuple[str, ...] = (),
243 predicate_name: str | None = None,
244 semantic_parameters: dict[str, str] | None = None) -> str:
245 """Generate a reproducible machine ID from a directed type signature.
246
247 A qualifier is only for a source-record plan. Business predicate identity
248 instead distinguishes registered derived names and evidenced operand roles.
249 Legacy root-only signatures retain their IDs.
250 """
251 name = canonical_relation_label(parent, predicate_name)
252 parameters = relation_semantic_parameters(parent, predicate_name, semantic_parameters)
253 if namespace not in ("business_relation", "relation"):
254 raise ValueError("Unknown relation namespace")
255 if not domain or not range_type or not isinstance(domain, str) or not isinstance(range_type, str):
256 raise ValueError("Relation requires one ordered endpoint on each side")
257 if not isinstance(qualifier, (list, tuple)) or not all(
258 isinstance(item, str) and item for item in qualifier):
259 raise ValueError("Relation qualifier must be a list of nonempty strings")
260 if namespace == "business_relation" and qualifier:
261 raise ValueError("Business relation identity cannot depend on a source rule")
262 signature = [parent, domain, range_type, list(qualifier)]
263 if name != parent or parameters:
264 signature.extend([name, parameters])
265 payload = json.dumps(signature,
266 ensure_ascii=False, separators=(",", ":"))
267 token = hashlib.sha256(payload.encode("utf-8")).hexdigest()[:24]
268 return f"{namespace}:{parent}:{token}"

271def validate_proposed_relation_label(label: str, parent: str, *,
272 subject_label: str | None = None,
273 object_label: str | None = None,
274 predicate_name: str | None = None) -> None:
275 """Reject incompatible model labels before replacing them with a verb.
276
277 With known business endpoints only a bare verb or the exact directed
278 ``subject + verb + object`` form is accepted. This is deliberately stricter
279 than matching a cue somewhere in a free-form sentence.
280 """
281 name = canonical_relation_label(parent, predicate_name)
282 value = _normalized(label)
283 if not value or len(value) > 96 or re.search(r"[\n\r:;；。]", value):
284 raise ValueError("Relation label must be a short predicate phrase")
285 # The machine predicate is already validated above. A root's controlled
286 # display verb may describe its registered subtype without renaming it.
287 cues = (*_DERIVED_CUES[name], *_CUES[parent]) if name != parent else _CUES[parent]
288 allowed_cues = {_normalized(item) for item in cues}
289 if subject_label is not None and object_label is not None:
290 subject, object_ = _normalized(subject_label), _normalized(object_label)
291 if not subject or not object_:
292 raise ValueError("Relation label endpoint names are missing")
293 allowed = allowed_cues | {
294 subject + cue + object_ for cue in allowed_cues
295 }
296 if value not in allowed:
297 raise ValueError("Relation label is not a canonical directed predicate")
298 return
299 if value not in allowed_cues:
300 raise ValueError("Relation label must name the selected root predicate")
