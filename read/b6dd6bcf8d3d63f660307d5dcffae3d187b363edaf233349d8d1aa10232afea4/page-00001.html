82def _source_evidence(data, evidence_ids):
83 """Only complete input evidence can support a slot; model prose cannot."""
84 if not evidence_ids:
85 raise ValueError("Binding lacks source evidence")
86 for evidence_id in evidence_ids:
87 proof = data.evidence.get(evidence_id)
88 if not proof or proof.get("raw_fragment_truncated"):
89 raise ValueError("Binding source evidence is missing or truncated")
90 if proof.get("origin") not in ("observed_record", "schema", "source_schema", "metadata"):
91 # Schema evidence from Dataset is identified by schema: IDs and
92 # still must exist in this current Dataset's evidence registry.
93 if not evidence_id.startswith("schema:"):
94 raise ValueError("Binding requires source evidence, not an inferred claim")
95 snapshot = proof.get("source_ref", {}).get("snapshot_id")
96 if snapshot is not None and snapshot != data.snapshot_id:
97 raise ValueError("Binding evidence belongs to another snapshot")

111def _publish_binding(data, plan, source_id, slot, contract):
112 """One checked contract adds one type edge plus a scoped binding assertion."""
113 slot = OntologySlotBinding.model_validate(slot)
114 types = {item.id: item for item in plan.object_types}
115 _endpoints(types, source_id, slot.target_type_id, slot.role, data)
116 _source_evidence(data, slot.evidence_ids)
117 predicate, parent, _ = _slot_contract(types[source_id], types, slot.role)
118 proof_body = {
119 "snapshot_id": data.snapshot_id, "source_type_id": source_id,
120 "slot": slot.model_dump(), "contract": contract,
121 }
122 proof_id = "semantic_binding_proof:" + digest(proof_body)[:24]
123 data.evidence[proof_id] = {
124 "id": proof_id, "origin": "automatic_contract_check",
125 "inference_kind": "checked_structured_binding", "binding_proof": proof_body,
126 "basis_evidence_ids": slot.evidence_ids,
127 "source_ref": {"snapshot_id": data.snapshot_id},
128 }
129 relation_id = canonical_relation_id(parent, source_id, slot.target_type_id, predicate_name=predicate)
130 proposed = DerivedType(
131 id=relation_id, parent=parent, label=predicate, predicate_name=predicate,
132 definition=canonical_relation_definition(parent, predicate),
133 category="business_relation_type", domain=[source_id], range=[slot.target_type_id],
134 endpoint_basis="semantic_binding", evidence_scope="checked_structured_binding",
135 evidence_ids=sorted(set(slot.evidence_ids + [proof_id])),
136 )
137 old = next((item for item in plan.relation_types if item.id == relation_id), None)
138 if old:
139 if old.model_dump(exclude={"evidence_ids"}) != proposed.model_dump(exclude={"evidence_ids"}):
140 raise ValueError("Binding conflicts with an existing relation contract")
141 proposed.evidence_ids = sorted(set(old.evidence_ids + proposed.evidence_ids))
142 plan.relation_types[plan.relation_types.index(old)] = proposed
143 else:
144 plan.relation_types.append(proposed)
145 return {
146 "id": "semantic_binding:" + digest(proof_body)[:24],
147 "relation_type_id": relation_id, "source_type_id": source_id,
148 "target_type_id": slot.target_type_id, "slot": slot.model_dump(),
149 "contract": contract, "snapshot_id": data.snapshot_id,
150 "status": "accepted", "evidence_ids": proposed.evidence_ids,
151 "identity_claim": "none", "relationship_scope": "checked_definition_slot",

155def binding_relation_errors(data, relation, object_types):
156 """Replay validates endpoint and source proofs, not just a saved status flag."""
157 types = {item.id: item for item in object_types}
158 if len(relation.domain) != 1 or len(relation.range) != 1:
159 return ["structured binding requires one source and target type"]
160 found = False
161 for evidence_id in relation.evidence_ids:
162 evidence = data.evidence.get(evidence_id, {})
163 body = evidence.get("binding_proof")
164 if evidence.get("inference_kind") != "checked_structured_binding" or not isinstance(body, dict):
165 continue
166 try:
167 if body.get("snapshot_id") != data.snapshot_id:
168 raise ValueError("Binding contract belongs to another snapshot")
169 if evidence_id != "semantic_binding_proof:" + digest(body)[:24]:
170 raise ValueError("Binding contract hash differs")
171 slot = OntologySlotBinding.model_validate(body["slot"])
172 _endpoints(types, body["source_type_id"], slot.target_type_id, slot.role, data)
173 _source_evidence(data, slot.evidence_ids)
174 if (_root(types[body["source_type_id"]], types) == "GeneralObject"
175 and body.get("contract", {}).get("kind") == "template_projection"):
176 _check_resource_field_contract(data, types[slot.target_type_id],
177 body["contract"].get("resource_field_contract", {}))
178 predicate, parent, _ = _slot_contract(types[body["source_type_id"]], types, slot.role)
179 if (relation.domain != [body["source_type_id"]] or relation.range != [slot.target_type_id]
180 or relation.predicate_name != predicate or relation.parent != parent):
181 raise ValueError("Binding proof path differs from relation endpoints")
182 if not set(slot.evidence_ids) <= set(relation.evidence_ids):
183 raise ValueError("Binding relation omits its source proofs")
184 found = True
185 except (ValueError, KeyError) as exc:
186 return [str(exc)]
187 return [] if found else ["structured binding lacks a checked source contract"]

190def measure_slot_value_supported(data, target, value):

192 from .calculation_contracts import _aliases, _norm

207def _resource_slot_proof(data, projection, raw_slot, target, bindings):
208 """Do not promote a query topic or substring of a business-domain label.
209
210 This entry point supports explicit name-valued resource fields. Member
211 codes require the independently checked reference path channel; merely
212 observing EAST, for example, cannot prove which dimension owns it.
213 """
214 current = [item for item in bindings if item.get("template_id") == projection["template_id"]
215 and item.get("status") == "accepted" and item.get("mapping_kind") == "template_instance"
216 and item.get("snapshot_id") == data.snapshot_id
217 and item.get("object_type_id") == projection["object_type_id"]]
218 values = [item.get("slot_values", {}).get(raw_slot["name"]) for item in current]
219 if (not values or any(not isinstance(value, str) or not measure_slot_value_supported(data, target, value)

226 _check_resource_field_contract(data, target, contract)
227 return contract

249def compile_template_relations(data, profile, plan, template_projections, template_bindings=()):
250 """Publish reusable type edges once; observed member values remain bindings."""
251 from .validation import validate_plan
252
253 candidate = BuildPlan.model_validate(plan).model_copy(deep=True)
254 assertions, pending, seen = [], [], set()
255 for projection in template_projections:
256 if projection.get("status") != "accepted" or projection.get("root_type") not in _SLOT_CONTRACTS:
257 continue
258 for raw_slot in projection.get("slots", []):
259 if raw_slot.get("role") not in _ROLES:
260 continue
261 try:
262 if projection.get("snapshot_id") != data.snapshot_id:
263 raise ValueError("Template projection snapshot differs")
264 if not raw_slot.get("target_type_id"):
265 raise ValueError("Slot requires a targeted definition endpoint")
266 resource_contract = None
267 if projection.get("root_type") == "GeneralObject":
268 target = next((item for item in candidate.object_types if item.id == raw_slot["target_type_id"]), None)
269 if target is None:
270 raise ValueError("Resource slot target is not an accepted type")
271 resource_contract = _resource_slot_proof(data, projection, raw_slot, target, template_bindings)
272 variable = any("{" + raw_slot["name"] + "}" in field.get("template", "")
273 for field in projection.get("field_templates", []))
274 if raw_slot["role"] == "measure" and variable:
275 target = next((item for item in candidate.object_types if item.id == raw_slot["target_type_id"]), None)
276 values = [binding.get("slot_values", {}).get(raw_slot["name"])
277 for binding in template_bindings if binding.get("template_id") == projection["template_id"]]
278 if (target is None or not values or any(value is None or not
279 measure_slot_value_supported(data, target, value) for value in values)):
280 raise ValueError("Varying measure slot values do not all identify the proposed target definition")
281 slot = {key: raw_slot[key] for key in OntologySlotBinding.model_fields if key in raw_slot}
282 contract = {"kind": "template_projection", "template_id": projection["template_id"],
283 "contract_hash": projection["contract_hash"],
284 "source_table": projection.get("source_table")}
285 if resource_contract:
286 contract["resource_field_contract"] = resource_contract
287 assertion = _publish_binding(data, candidate, projection["object_type_id"], slot, contract)
288 if assertion["id"] not in seen:
289 assertions.append(assertion)
290 seen.add(assertion["id"])
291 except (ValueError, KeyError) as exc:
292 pending.append({"template_id": projection.get("template_id"), "slot": raw_slot,
293 "status": "unresolved", "reason": str(exc)})
294 errors = validate_plan(candidate, data, profile)
295 if errors:
296 raise ValueError("Invalid structured ontology bindings: " + "; ".join(errors))
297 return {"plan": candidate, "bindings": assertions, "pending": pending,
298 "coverage": {"accepted": len(assertions), "pending": len(pending),
299 "template_instance_bindings": len(template_bindings), "llm_calls": 0,
300 "partial": bool(pending)}}
