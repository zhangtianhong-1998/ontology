597 def pattern_window(self, limit, *, window_index=0, kind="definition"):
598 """Page distinct semantic *patterns*, preserving every card separately.
599
600 Representatives are selected by stable card_id. The order interleaves
601 source tables before paging, so the first page cannot be monopolized
602 by a single large table. A pattern is only a candidate scheduling unit;
603 its non-representative records remain unaligned.
604 """
605 if type(limit) is not int or not 0 < limit <= 10000:
606 raise ValueError("pattern window limit must be an integer in 1..10000")
607 if type(window_index) is not int or not 0 <= window_index <= 1000000:
608 raise ValueError("window_index must be an integer in 0..1000000")
609 if kind not in ("definition", "reference", "uncertain"):
610 raise ValueError("Unknown card kind")
611 total = self.db.execute(
612 "SELECT count(DISTINCT pattern_id) FROM cards WHERE kind=?", (kind,)).fetchone()[0]
613 total_cards = self.db.execute(
614 "SELECT count(*) FROM cards WHERE kind=?", (kind,)).fetchone()[0]
615 offset = window_index * limit
616 rows = self.db.execute("""
617 WITH grouped AS (
618 SELECT pattern_id, table_name, root_hint,
619 min(card_id) AS representative_card_id,
620 count(*) AS pattern_card_count,
621 sum(occurrence_count) AS pattern_record_count,
622 count(DISTINCT reference_signature) AS reference_variant_count
623 FROM cards WHERE kind=?
624 GROUP BY pattern_id, table_name, root_hint
625 ), numbered AS (
626 SELECT *, row_number() OVER (
627 PARTITION BY table_name ORDER BY pattern_id) AS table_rank
628 FROM grouped
629 )
630 SELECT pattern_id, representative_card_id, pattern_card_count,
631 pattern_record_count, reference_variant_count
632 FROM numbered
633 ORDER BY table_rank, table_name, pattern_id
634 LIMIT ? OFFSET ?
635 """, (kind, limit, offset)).fetchall()
636 before_cards = self.db.execute("""
637 WITH grouped AS (
638 SELECT pattern_id, table_name, count(*) AS card_count
639 FROM cards WHERE kind=? GROUP BY pattern_id, table_name
640 ), ranked AS (
641 SELECT *, row_number() OVER (
642 PARTITION BY table_name ORDER BY pattern_id) AS table_rank
643 FROM grouped
644 ), ordered AS (
645 SELECT card_count, row_number() OVER (
646 ORDER BY table_rank, table_name, pattern_id) AS global_rank
647 FROM ranked
648 )
649 SELECT coalesce(sum(card_count), 0) FROM ordered WHERE global_rank <= ?
650 """, (kind, offset)).fetchone()[0]
651 patterns = []
652 for row in rows:
653 card = self.get(row["representative_card_id"])
654 card["pattern_id"] = row["pattern_id"]
655 card["pattern_card_count"] = row["pattern_card_count"]
656 card["pattern_record_count"] = row["pattern_record_count"]
657 card["reference_variant_count"] = row["reference_variant_count"]
658 card["pattern_nonrepresentative_cards"] = row["pattern_card_count"] - 1
659 card["pattern_status"] = "candidate_only_not_identity_alignment"
660 patterns.append(card)
661 return {"patterns": patterns, "total_patterns": total,
662 "total_cards": total_cards,
663 "window_index": window_index, "window_limit": limit,
664 "patterns_before_window": min(total, offset),
665 "patterns_after_window": max(0, total - offset - len(patterns)),
666 "cards_before_window": before_cards,
667 "cards_in_window": sum(item["pattern_card_count"] for item in patterns),
668 "cards_after_window": max(0, total_cards - before_cards
669 - sum(item["pattern_card_count"] for item in patterns)),
670 "next_window_index": (window_index + 1 if offset + len(patterns) < total else None),
671 "ordering": "interleaved_by_table_rank_then_pattern_id"}

673 def seeds(self, limit, *, per_table=16, kind=None, window_index=0):
674 """Round-robin one stable, bounded window of cards from each table.
675
676 Window ``n`` reads card positions ``n * per_table`` through
677 ``(n + 1) * per_table - 1`` in card-id order. Repeated runs can visit
678 later definitions without loading the complete corpus into memory.
679 """
680 if any(type(value) is not int or not 0 < value <= 1000 for value in (limit, per_table)):
681 raise ValueError("limit and per_table must be integers in 1..1000")
682 if type(window_index) is not int or not 0 <= window_index <= 1000000:
683 raise ValueError("window_index must be an integer in 0..1000000")
684 if kind is not None and kind not in ("definition", "reference", "uncertain"):
685 raise ValueError("Unknown card kind")
686 tables = [row[0] for row in self.db.execute("SELECT DISTINCT table_name FROM cards ORDER BY table_name")]
687 queues = {}
688 for table in tables:
689 rows = self.db.execute(
690 "SELECT * FROM cards WHERE table_name=? AND (? IS NULL OR kind=?) "
691 "ORDER BY card_id LIMIT ? OFFSET ?",
692 (table, kind, kind, per_table, window_index * per_table)).fetchall()
693 queues[table] = [self._card(row) for row in rows]
694 result = []
695 for position in range(per_table):
696 for table in tables:
697 if position < len(queues[table]):
698 result.append(queues[table][position])
699 if len(result) >= limit:
700 return result
701 return result

703 def search(self, query, *, limit=10, kind=None, table=None, scope=None,
704 root_hint=None, exclude_card_id=None, exclude_pattern_id=None):
705 """Return exact-name/alias and BM25 candidates with explicit filters."""
706 if type(limit) is not int or not 0 < limit <= 100:
707 raise ValueError("limit must be an integer in 1..100")
708 if not isinstance(query, str) or len(query) > 512:
709 raise ValueError("query must be a string of at most 512 characters")
710 if kind is not None and kind not in ("definition", "reference", "uncertain"):
711 raise ValueError("Unknown card kind")
712 clauses, params = [], []
713 for field, value in (("kind", kind), ("table_name", table), ("root_hint", root_hint)):
714 if value is not None:
715 clauses.append(f"c.{field}=?")
716 params.append(value)
717 if exclude_card_id is not None:
718 clauses.append("c.card_id<>?")
719 params.append(exclude_card_id)
720 if exclude_pattern_id is not None:
721 clauses.append("c.pattern_id<>?")
722 params.append(exclude_pattern_id)
723 if isinstance(scope, dict):
724 clauses.append("c.scope_json=?")
725 params.append(json.dumps(scope, ensure_ascii=False, sort_keys=True))
726 elif isinstance(scope, str):
727 clauses.append("EXISTS (SELECT 1 FROM json_each(c.scope_json) WHERE value=?)")
728 params.append(scope)
729 elif scope is not None:
730 raise ValueError("scope must be a string, mapping, or null")
731 where = (" AND " + " AND ".join(clauses)) if clauses else ""
732 query_norm = _norm(query)
733 if not query_norm:
734 return []
735 results = {}
736 # Resolve names/aliases through their inverted indexes first. The
737 # CROSS JOIN keeps this small ID set outermost; otherwise SQLite may
738 # choose a full kind scan before testing the alias correlation.
739 exact = self.db.execute(
740 "WITH exact_ids AS MATERIALIZED ("
741 "SELECT card_id FROM cards INDEXED BY cards_name WHERE name_norm=? "
742 "UNION SELECT card_id FROM aliases INDEXED BY aliases_name WHERE name_norm=?) "
743 "SELECT c.*, 0.0 AS bm25 FROM exact_ids e CROSS JOIN cards c "
744 "ON c.card_id=e.card_id WHERE 1=1"
745 + where + " ORDER BY c.card_id LIMIT ?", [query_norm, query_norm, *params, limit]).fetchall()
746 for row in exact:
747 card = self._card(row)
748 card.update(retrieval_channels=["exact_name_or_alias"], bm25=None)
749 results[card["card_id"]] = card
750 terms = _terms(query_norm)
751 if terms:
752 expression = " OR ".join('"' + term + '"' for term in terms)
753 lexical = self.db.execute(
754 "SELECT c.*, bm25(card_fts) AS bm25 FROM card_fts JOIN cards c "
755 "ON c.card_id=card_fts.card_id WHERE card_fts MATCH ?" + where
756 + " ORDER BY bm25(card_fts), c.card_id LIMIT ?",
757 [expression, *params, limit]).fetchall()
758 for rank, row in enumerate(lexical, 1):
759 card = results.get(row["card_id"])
760 if card is None:
761 card = self._card(row)
762 card.update(retrieval_channels=[], bm25=row["bm25"])
763 results[card["card_id"]] = card
764 else:
765 card["bm25"] = row["bm25"]
766 card["retrieval_channels"].append("bm25_zh_2_3gram")
767 card["lexical_rank"] = rank
768 ordered = sorted(results.values(), key=lambda item: (
769 "exact_name_or_alias" not in item["retrieval_channels"],
770 item["bm25"] if item["bm25"] is not None else 0,
771 item["card_id"]))[:limit]
772 for item in ordered:
773 item["candidate_status"] = "unjudged"
774 item["retrieval_scope"] = "indexed_cards_only"
775 return ordered
