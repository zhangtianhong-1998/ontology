24_HAN = re.compile(r"[\u3400-\u9fff]+")
25_WORDS = re.compile(r"[a-z0-9]+")
26_REFERENCE = re.compile(r"(?:^|_)(?:code|id|key|source|target|ref|reference|type|field|value)(?:_|$)", re.I)
27_CONTENT_REFERENCE = re.compile(r"(?:^|_)(?:code|field|value)(?:_|$)", re.I)
28_PURE_TECHNICAL = re.compile(r"(?:\d+(?:\.\d+)?|[0-9a-f]{16,}|\d{4}-\d\d-\d\d)", re.I)
29_CARD_ROLES = ("name", "alias", "description", "formula", "unit", "scope",
30 "provenance", "metadata", "identity")
31# Context stays in complete cards and provenance. It does not create additional
32# scheduling groups; definition qualifiers and unresolved content still do.
33_PATTERN_ROLES = ("name", "alias", "description", "formula", "unit", "scope", "identity", "unknown")
34_PHYSICAL_NAME_DECLARATION = re.compile(
35 r"^(?:物理(?:数据表|表|数据库|文件)(?:名称|名)|"

42def _definition_pattern_exclusions(table, roles):
43 """Separate only declared technical physical names from scheduling.
44
45 Generic resource/object names may denote business assets and remain intact.
46 Their complete values remain in source cards. A checked projection contract
47 may preserve them as instance values instead of fixing a class invariant;
48 this scheduling hint alone never proves membership.
49 """
50 names = set(roles.get("name", ()))
51 result = []
52 for column in table["columns"]:
53 raw = str(column.get("column_comment") or "")
54 declaration = _norm(raw)
55 if (column["column_name"] in names
56 and _PHYSICAL_NAME_DECLARATION.search(declaration)
57 and not _BUSINESS_NAME_DECLARATION.search(declaration)):
58 result.append({"column": column["column_name"], "role": "name",
59 "reason": "declared_physical_resource_name",
60 "declaration": raw,
61 "schema_evidence_id": f"schema:{table['name']}:{column['column_name']}",
62 "scope": "definition_pattern_scheduling_only",
63 "entity_identity_claim": False,
64 "template_membership_requires_complete_projection_contract": True,
65 "original_value_retained_as_source_or_instance_value": True})
66 return result

78def _terms(value: str) -> list[str]:
79 """Use opaque ASCII FTS terms so a two-Han-character query stays one token."""
80 value = _norm(value)
81 found = set()
82 for match in _HAN.finditer(value):
83 chunk = match.group()
84 for size in (2, 3):
85 for index in range(len(chunk) - size + 1):
86 found.add("g" + "".join(f"{ord(c):06x}" for c in chunk[index:index + size]))
87 if len(chunk) == 1:
88 found.add("g" + f"{ord(chunk):06x}")
89 for word in _WORDS.findall(value):
90 found.add("w" + word.encode("utf-8").hex())
91 return sorted(found)

94def _columns(table, max_unknown_fields_per_table):
95 from .column_role_inference import formula_fragment_context
96
97 roles = _field_roles(table)
98 classified = classify_columns(table)
99 safe = {item["column"] for item in classified if item["role"] not in (
100 "sensitive", "empty", "audit_time", "audit_metadata", "technical_identifier")}
101 fragments = {name: item for name, item in formula_fragment_context(table).items()
102 if name in safe}
103 for name in fragments:
104 if name in roles.get("formula", ()):
105 roles["formula"].remove(name)
106 # Calculation parameters distinguish the definition even when they
107 # are insufficient to construct a complete arithmetic expression.
108 if name not in roles.setdefault("scope", []):
109 roles["scope"].append(name)
110 # Metadata heuristics and model proposals share the same key/content
111 # boundary. Forced references include PKs.
112 conflicts = field_role_conflicts(table)
113 bindings = list(dict.fromkeys([*roles.pop("reference", []),
114 *(item["column"] for item in conflicts if item["binding_only"])]))
115 # Explicit unresolved role evidence must survive the generic unknown-field
116 # cap. Dropping a reviewed structured field would hide its meaning entirely.
117 unresolved = roles.get("unknown", [])
118 selected = {column for columns in roles.values() for column in columns}
119 reference = [item["column"] for item in classified
120 if item["role"] in ("semantic", "unknown") and item["column"] not in selected
121 and item["column"] not in table["pk"]
122 and _REFERENCE.search(item["column"])]
123 unknown = [item["column"] for item in classified if item["role"] == "unknown"]
124 # Every eligible reference participates in the complete card fingerprint.
125 # Packet admission bounds bytes later; it must not silently drop columns.
126 reference = list(dict.fromkeys([*bindings, *reference]))
127 selected.update(reference)
128 selected.update(unresolved)
129 unknown_available = [name for name in unknown if name not in selected]
130 fallback = list(dict.fromkeys([*unresolved, *unknown_available[:max_unknown_fields_per_table]]))
131 return roles, reference, fallback, {
132 "unknown_columns_considered": fallback,
133 "unknown_columns_not_examined": unknown_available[max_unknown_fields_per_table:],
134 "reference_columns_considered": reference,
135 "reference_columns_not_examined": [],
136 "reference_column_selection": "all_eligible_reference_columns",
137 "definition_pattern_exclusions": _definition_pattern_exclusions(table, roles),
138 "role_conflicts": conflicts,
139 "binding_columns_excluded_from_semantic_pattern": bindings,
140 "context_columns_excluded_from_semantic_pattern": {
141 role: list(roles.get(role, ())) for role in ("metadata", "provenance")},
142 "calculation_fragments": list(fragments.values()),
143 "source_verified_role_candidates": [
144 {"column": item["column"], "role": item["role"]}
145 for item in table.get("inferred_semantic_roles", ())
146 if item.get("status") == "source_verified_role_candidate"],
147 }

150def _structured_semantic_candidate(value):
151 """Recall named structured lexical/reference content for semantic review.
152
153 This is a scheduling signal only. Configuration rows are routed separately,
154 and a plain administrative JSON object does not satisfy this condition.
155 Preserve the entire original JSON rather than manufacturing a definition.
156 """
157 try:
158 content = json.loads(value)
159 except (ValueError, TypeError):
160 return False
161 if not isinstance(content, dict):
162 return False
163 semantic_key = re.compile(
164 r"(?:^|_)(?:meaning|definition|description|alias|aliases|synonym|synonyms|"
165 r"term|concept|formula|expression|reference|ref|code)(?:_|$)|词义|释义|同义|别名|定义|引用")
166 administrative_key = re.compile(
167 r"^(?:language|locale|status|record|source|system|owner|created|updated|schema)(?:_|$)")
168 return any(semantic_key.search(str(key).casefold())
169 and not administrative_key.search(str(key).casefold())
170 and raw not in (None, "", [], {}) for key, raw in content.items())

173def _row_card(table_name, row, roles, reference, fallback, max_field_chars, max_index_chars,
174 row_purpose="unresolved", pattern_name_exclusions=(), calculation_fragments=()):
175 if row_purpose == "business_fact":
176 return None
177 # A sampled operator column may contain a later expression. Check while
178 # doing the existing index scan; never erase that row's formula candidate.
179 from .column_role_inference import formula_fragment_value
180 for fragment in calculation_fragments:
181 name = fragment["column"]
182 value = row.get(name)
183 if value is None or not str(value).strip() or formula_fragment_value(
184 fragment["effective_role"], value):
185 continue
186 roles = {role: list(columns) for role, columns in roles.items()}
187 if name in roles.get("scope", ()):
188 roles["scope"].remove(name)
189 if name not in roles.setdefault("formula", []):
190 roles["formula"].append(name)
191 fields = {}
192 signature_fields = {}
193 for role, columns in [*((role, roles.get(role, ())) for role in _CARD_ROLES),
194 ("reference", reference), ("unknown", fallback)]:
195 values = []
196 originals = []
197 for column in columns:
198 raw = row.get(column)
199 if raw is None or not str(raw).strip():
200 continue
201 raw = str(raw)
202 if (role == "unknown" and column not in roles.get("unknown", ())
203 and (len(raw.strip()) < 3 or _PURE_TECHNICAL.fullmatch(raw.strip()))):
204 continue
205 originals.append([column, raw])
206 values.append({"column": column, "value": raw[:max_field_chars],
207 "truncated": len(raw) > max_field_chars,
208 "schema_evidence_id": f"schema:{table_name}:{column}"})
209 if values:
210 fields[role] = values
211 signature_fields[role] = originals
212 informative_reference = any(_CONTENT_REFERENCE.search(entry["column"])
213 for entry in fields.get("reference", ()))
214 structured_definition_candidate = bool(fields.get("name") and any(
215 _structured_semantic_candidate(str(row.get(entry["column"]) or ""))
216 for entry in fields.get("unknown", ())))
217 if row_purpose == "configuration_data":
218 kind = "reference"
219 elif row_purpose == "definition_data" and fields.get("name"):
220 kind = "definition"
221 elif structured_definition_candidate:
222 kind = "definition"
223 elif informative_reference:
224 kind = "reference"
225 elif fields.get("description") or fields.get("formula") or fields.get("alias") or fields.get("unknown"):
226 kind = "uncertain"
227 else:
228 return None
229 # Choose a readable display label without removing any other name from
230 # the source fields, search text, or complete pattern fingerprint.
231 names = sorted(fields.get("name", ()), key=lambda entry: (
232 not bool(re.search(r"(?:^|_)(?:cn|zh|chinese)(?:_|$)", entry["column"], re.I)),
233 not bool(_HAN.search(entry["value"])),
234 entry["column"] in {"schema_name", "physical_table_name"}))
235 aliases = [value["value"] for value in fields.get("alias", ())]
236 # Definition identity qualifiers must survive with their original values;
237 # administrative/lexical context stays visible outside business scope.
238 scope = {value["column"]: value["value"] for role in ("scope", "identity")
239 for value in fields.get(role, ())}
240 units = [value["value"] for value in fields.get("unit", ())]
241 search_text = " ".join(value["value"] for role in ("name", "alias", "description", "formula", "unknown")
242 for value in fields.get(role, ()))
243 reference_text = " ".join(value["value"] for value in fields.get("reference", ())
244 if _CONTENT_REFERENCE.search(value["column"]))
245 search_text = " ".join(part for part in (search_text, reference_text) if part)
246 index_text_truncated = len(search_text) > max_index_chars
247 search_text = search_text[:max_index_chars]
248 # Preview fields omit empty values, but reference identity also preserves
249 # NULL/empty differences and full values beyond the visible preview.
250 if reference:
251 signature_fields["reference"] = [[column, row.get(column)] for column in reference]
252 signature = digest([table_name, kind, signature_fields])
253 excluded_names = set(pattern_name_exclusions) if kind == "definition" else set()
254 semantic_fields = {role: [[column, _norm(raw)] for column, raw in signature_fields[role]
255 if not (role == "name" and column in excluded_names)]
256 for role in _PATTERN_ROLES if role in signature_fields}
257 semantic_preview_truncated = any(entry["truncated"] for role in _PATTERN_ROLES
258 for entry in fields.get(role, ()))
259 # A pattern is a retrieval/scheduling group, never a business identity.
260 # A truncated semantic preview cannot safely stand for another card, even
261 # though the unabridged source value was available when fingerprinting.
262 pattern_signature = digest([table_name, kind, semantic_fields,
263 signature if semantic_preview_truncated else None])
264 reference_signature = (digest(signature_fields["reference"])
265 if signature_fields.get("reference") else None)
266 return {"kind": kind, "table": table_name, "name": names[0]["value"] if names else "",
267 "aliases": aliases, "scope": scope, "unit": units[0] if units else "",
268 "fields": fields, "search_text": search_text,
269 "index_text_truncated": index_text_truncated, "signature": signature,
270 "pattern_signature": pattern_signature,
271 "reference_signature": reference_signature,
272 "semantic_preview_truncated": semantic_preview_truncated,
273 "structured_definition_candidate": structured_definition_candidate and kind == "definition"}

276def _root_hint(table):
277 if _context_table(table["table_name"]):
278 return "GeneralObject"
279 text = " ".join((table["table_name"], table.get("table_comment") or "")).casefold()
280 for pattern, root in ((r"metric|指标", "Metric"), (r"measure|度量", "Measure"),

293def _table_quotas(specs, max_cards):
294 """Reserve across tables before scanning; no first-table monopoly."""
295 quotas = {name: 0 for name, table, *_ in specs}
296 active = [(name, table, weight) for name, table, *_, weight in specs
297 if table["rows"] > 0 and weight > 0]
298 active.sort(key=lambda item: (-item[2], item[0]))
299 if max_cards < len(active):
300 for name, _, _ in active[:max_cards]:
301 quotas[name] = 1
302 return quotas
303 for name, _, _ in active:
304 quotas[name] = 1
305 remaining = max_cards - len(active)
306 while remaining:
307 candidates = [(name, table, weight) for name, table, weight in active
308 if quotas[name] < table["rows"]]
309 if not candidates:
310 break
311 total_weight = sum(weight for _, _, weight in candidates)
312 grants = {name: min(table["rows"] - quotas[name], remaining * weight // total_weight)
313 for name, table, weight in candidates}
314 if not any(grants.values()):
315 name = candidates[0][0]
316 grants[name] = 1
317 for name, grant in grants.items():
318 quotas[name] += grant
319 remaining -= grant
320 return quotas
