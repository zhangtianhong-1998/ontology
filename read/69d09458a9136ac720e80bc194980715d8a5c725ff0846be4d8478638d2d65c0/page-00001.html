19def _mapping(loader, node, deep=False):
20 out = {}
21 for key, value in node.value:
22 key = loader.construct_object(key, deep=deep)
23 if key in out:
24 raise ValueError(f"Duplicate YAML key: {key}")
25 out[key] = loader.construct_object(value, deep=deep)
26 return out

191 def rows_at(self, table, row_numbers, *, on_fetch=None):
192 """Read row locators in source order; on_fetch counts actual batch rows."""
193 numbers = list(row_numbers)
194 if any(type(number) is not int or number < 1 for number in numbers):
195 raise ValueError("Row locators must be positive integers")
196 if any(number > self.tables[table]["rows"] for number in numbers):
197 raise ValueError("Row locators exceed the source table row count")
198 numbers = sorted(set(numbers))
199 if not numbers:
200 return
201 sql_table = qi(self.tables[table]["sql_name"])
202 index = "row_idx_" + digest(table)[:20]
203 if index not in self.indexes:
204 self.db.execute(f"CREATE INDEX {qi(index)} ON {sql_table} (__r2_row)")
205 self.indexes.add(index)
206 for offset in range(0, len(numbers), 1000):
207 batch = numbers[offset:offset + 1000]
208 placeholders = ",".join("?" for _ in batch)
209 cursor = self.db.cursor().execute(
210 f"SELECT * FROM {sql_table} WHERE __r2_row IN ({placeholders}) ORDER BY __r2_row", batch)
211 names = [column[0] for column in cursor.description]
212 try:
213 fetched = cursor.fetchall()
214 if on_fetch is not None:
215 on_fetch(len(fetched))
216 for row in fetched:
217 yield dict(zip(names, row))
218 finally:
219 cursor.close()

222 def lookup(self, table, bindings):
223 info = self.tables[table]
224 if any(v is None or v == "" for _, v in bindings):
225 return []
226 cols = [c for c, _ in bindings]
227 if not set(cols) <= set(info["column_names"]):
228 raise ValueError("Unknown lookup columns")
229 index = "idx_" + digest([table, cols])[:20]
230 if index not in self.indexes:
231 self.db.execute(f"CREATE INDEX {qi(index)} ON {qi(info['sql_name'])} ({','.join(map(qi, cols))})")
232 self.indexes.add(index)
233 sql = f"SELECT * FROM {qi(info['sql_name'])} WHERE " + " AND ".join(f"{qi(c)} = ?" for c in cols) + " LIMIT 2"
234 cursor = self.db.execute(sql, [v for _, v in bindings])
235 names = [d[0] for d in cursor.description]
236 return [dict(zip(names, r)) for r in cursor.fetchall()]

238 def record_id(self, table, row):
239 # A source-local identity never claims cross-source business equality.
240 t = self.tables[table]
241 key = {k: row[k] for k in t["pk"]} if t["pk"] and all(row.get(k) not in (None, "") for k in t["pk"]) else {"file": t["csv_hash"], "row": row["__r2_row"]}
242 # The row position disambiguates invalid/repeated declared keys as well.
