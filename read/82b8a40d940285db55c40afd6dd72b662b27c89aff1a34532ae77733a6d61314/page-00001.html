75def _source_target_names(data, target):
76 """Only complete, same-snapshot name/alias evidence authorizes captures."""
77 values, evidence_ids = set(), set()
78 for prop in target.source_properties:
79 if prop.role not in {"name", "alias"}:
80 continue
81 for eid in prop.evidence_ids:
82 evidence = data.evidence.get(eid) or {}
83 ref = evidence.get("source_ref") or {}
84 raw = evidence.get("raw_fragment")
85 if (eid not in target.evidence_ids or evidence.get("origin") != "observed_record"
86 or evidence.get("raw_fragment_truncated") or not isinstance(raw, str)
87 or ref.get("snapshot_id") != data.snapshot_id
88 or ref.get("table") != prop.source_table or ref.get("column") != prop.source_column):
89 continue
90 names = [raw]
91 if prop.role == "alias":
92 try:
93 parsed = json.loads(raw)
94 except (ValueError, TypeError):
95 parsed = None
96 names = (parsed if isinstance(parsed, list) and all(isinstance(v, str) for v in parsed)
97 else re.split(r"[;,；，、|]", raw))
98 values.update(name.strip() for name in names if name.strip())
99 evidence_ids.add(eid)
100 return {"type_id": target.id, "accepted_values": sorted(values),
101 "evidence_ids": sorted(evidence_ids), "match": "literal_source_name_or_alias"}

113def _instance_columns(data, record, slots=()):
114 """Separate declared examples and technical bindings, not arbitrary JSON.
115
116 Unknown JSON may contain a formula or business rule. Only an explicit
117 example/sample declaration or an already checked binding role permits it
118 to vary without changing the type. Its actual value stays on the binding.
119 """
120 values, roles = _fields(record)
121 schema = getattr(data, "tables", {}).get(record["table"], {})
122 columns = {item["column_name"]: item for item in schema.get("columns", [])}
123 binding = {item["column"] for item in record.get("role_conflicts", []) if item.get("binding_only")}
124 coordinates = {item["column"] for item in schema.get("inferred_semantic_roles", [])
125 if item.get("status") == "source_verified_role_candidate"
126 and item.get("role") == "dimension_coordinate"}
127 direct_slots = {slot.get("source_column") or slot.get("value_evidence", {}).get("column"): slot
128 for slot in slots if slot.get("role") in {"record_identity", "dimension"}}
129 # Reuse the independent declaration check already used by card scheduling.
130 # A storage locator is instance data even when no model supplied its slot.
131 from .semantic_cards import _definition_pattern_exclusions
132 role_columns = {role: [column for column in values if role in roles[column]]
133 for role in {role for column_roles in roles.values() for role in column_roles}}
134 physical_names = {item["column"] for item in _definition_pattern_exclusions(
135 {**schema, "name": record["table"], "columns": schema.get("columns", [])}, role_columns)}
136 result = {}
137 for column, value in values.items():
138 if roles[column] & {"formula", "unit", "identity"}:
139 continue
140 if column in binding or "reference" in roles[column]:
141 result[column] = "source_checked_record_binding"
142 elif column in physical_names:
143 result[column] = "declared_physical_resource_name"
144 elif (column in direct_slots and direct_slots[column].get("value_evidence", {}).get("quote") == value
145 and (direct_slots[column]["role"] == "record_identity"
146 or column in coordinates
147 or re.search(r"(?:^|_)member_(?:code|name|cn_name|en_name|value)$", column, re.I)
148 or re.search(r"(?:维度|成员).*(?:取值|编码|名称)",
149 str(columns.get(column, {}).get("column_comment") or "")))):
150 result[column] = "source_checked_observed_slot_value"
151 elif (re.search(r"(?:^|_)(?:request|response)_(?:example|sample)(?:_|$)", column, re.I)
152 or re.search(r"(?:请求|响应|返回)(?:报文|数据)?(?:示例|样例)",
153 str(columns.get(column, {}).get("column_comment") or ""))):
154 try:
155 if isinstance(json.loads(value), (dict, list)):
156 result[column] = "declared_structured_example"
157 except (ValueError, TypeError):
158 pass
159 return result

162def _fields(record):
163 values, roles = {}, {}
164 for role, entries in record.get("fields", {}).items():
165 for entry in entries:
166 if entry.get("truncated"):
167 raise ValueError("Template projection requires complete source fields")
168 column, value = entry["column"], str(entry["value"])
169 if column in values and values[column] != value:
170 raise ValueError("One source field has inconsistent values")
171 values[column] = value
172 roles.setdefault(column, set()).add(role)
173 return values, roles

194def _match_text(pattern, value, slots, values, captures):
195 """Match escaped literals and bounded cells; never execute model regex."""
196 pieces, offset = [], 0
197 local = set()
198 for match in _SLOT.finditer(pattern):
199 name = match.group(1)
200 if name not in slots:
201 raise ValueError("Field template names an undeclared slot")
202 pieces.append(re.escape(pattern[offset:match.start()]))
203 slot = slots[name]
204 bound = captures.get(name)
205 if slot.get("source_column"):
206 bound = values.get(slot["source_column"])
207 if not bound:
208 return None
209 if name in captures and captures[name] != bound:
210 return None
211 if bound is not None:
212 pieces.append(re.escape(bound))
213 elif name in local:
214 pieces.append(f"(?P={name})")
215 else:
216 if offset and not pattern[offset:match.start()] and not slot.get("source_column"):
217 raise ValueError("Adjacent unbound slots are ambiguous")
218 pieces.append(f"(?P<{name}>.{{1,256}}?)")
219 local.add(name)
220 offset = match.end()
221 pieces.append(re.escape(pattern[offset:]))
222 matched = re.fullmatch("".join(pieces), value, flags=re.DOTALL)
223 if not matched:
224 return None
225 found = {**captures, **{key: value for key, value in matched.groupdict().items() if value is not None}}
226 for name, slot in slots.items():
227 if slot.get("source_column") and slot["source_column"] in values:
228 if name in found and found[name] != values[slot["source_column"]]:
229 return None
230 found[name] = values[slot["source_column"]]
231 return found

234def match_projection(template, record):
235 """Return an observed binding candidate, or None on any uncovered change."""
236 if record.get("table") != template["source_table"]:
237 return None
238 values, roles = _fields(record)
239 instance_columns = template.get("instance_columns", {})
240 if any(roles.get(column, set()) & {"formula", "unit", "identity"} for column in instance_columns):
241 return None # A later role change cannot bypass protected semantics.
242 semantic = {column: value for column, value in values.items()
243 if roles[column] & _SEMANTIC and column not in instance_columns}
244 if set(semantic) != set(template["semantic_columns"]):
245 return None
246 if any(semantic.get(column) != value for column, value in template["invariants"].items()):
247 return None
248 slots = {item["name"]: item for item in template["slots"]}
249 captures = {name: slot["fixed_value"] for name, slot in slots.items() if "fixed_value" in slot}
250 pending = list(template["field_templates"])
251 while pending:
252 progressed = False
253 for item in pending[:]:
254 if item["column"] not in values:
255 return None
256 try:
257 matched = _match_text(item["template"], values[item["column"]], slots, values, captures)
258 except ValueError as exc:
259 if str(exc) == "Adjacent unbound slots are ambiguous":
260 continue # An anchored field may bind these same slots first.
261 raise
262 if matched is None:
263 return None
264 captures = matched
265 pending.remove(item)
266 progressed = True
267 if not progressed:
268 raise ValueError("Adjacent unbound slots are ambiguous")
269 if not set(slots) <= set(captures):
270 return None
271 if _slot_target_mismatches(template["slots"], captures):
272 return None
273 return {"record_id": record["record_id"], "source_table": record["table"],
274 "row_number": record.get("row_number"), "source_card_id": record.get("card_id"),
275 "slot_values": captures,
276 "instance_values": {column: values[column] for column in instance_columns if column in values},
277 "reference_values": {column: value for column, value in values.items()
278 if "reference" in roles[column]},
279 "mapping_kind": "template_instance", "identity_claim": "none"}
