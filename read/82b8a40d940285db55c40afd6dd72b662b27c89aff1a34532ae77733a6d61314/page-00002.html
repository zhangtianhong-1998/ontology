662def reuse_projection(data, templates, bundle):
663 """Resolve one representative only; ambiguity must return to the model."""
664 allowed = set(bundle.get("exact_alignment_record_ids") or ())
665 seed = next((record for record in bundle.get("records", [])
666 if record.get("context_role") != "related_context"
667 and (not allowed or record["record_id"] in allowed)), None)
668 if seed is None:
669 return None
670 matches = [template for template in templates if match_projection(template, seed) is not None]
671 if len({item["object_type_id"] for item in matches}) != 1:
672 return None
673 endpoints = {}
674 for template in matches:
675 for slot in template["slots"]:
676 if slot.get("target_type_id"):
677 endpoints.setdefault((slot["role"], slot["name"]), set()).add(slot["target_type_id"])
678 if any(len(targets) > 1 for targets in endpoints.values()):
679 return None # One source row cannot silently choose conflicting accepted endpoints.
680 chosen = min(matches, key=lambda item: (
681 sum(slot["role"] in {"business_object", "measure", "dimension"}
682 and not slot.get("target_type_id") for slot in item["slots"]),
683 -sum(bool(slot.get("target_type_id")) for slot in item["slots"]), item["template_id"]))
684 return bind_projection(data, chosen, seed)
