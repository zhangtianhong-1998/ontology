7def _field(data, table, field):
8 if table not in data.tables or field not in data.tables[table]["column_names"]:
9 raise ValueError(f"Unknown field: {table}.{field}")

13def _row(data, table, row_number, key, fields):
14 info = data.tables[table]
15 shown = list(dict.fromkeys([key, *fields]))
16 needed = list(dict.fromkeys([*shown, *info["pk"]]))
17 columns = ", ".join(qi(field) for field in needed)
18 cursor = data.db.execute(
19 f"SELECT __r2_row, {columns} FROM {qi(info['sql_name'])} WHERE __r2_row = ?",
20 [row_number],
21 )
22 values = cursor.fetchone()
23 if values is None:
24 raise ValueError(f"Validated row is absent from snapshot: {table}:{row_number}")
25 record = dict(zip(["__r2_row", *needed], values))
26 return {"table": table, "row_number": row_number,
27 "record_id": data.record_id(table, record),
28 "fields": {field: record[field] for field in shown}}

31def joint_examples(data, candidate, check, *, limit=2, source_fields=(), target_fields=()):
32 """Return up to two uniquely matched original row pairs and one counterexample.
33
34 These rows illustrate a verified *field comparison*, not an accepted
35 business relation. Selector and scope predicates follow the exact check.
36 """
37 if isinstance(limit, bool) or not isinstance(limit, int) or limit < 0:
38 raise ValueError("limit must be a nonnegative integer")
39 if isinstance(source_fields, (str, bytes)) or isinstance(target_fields, (str, bytes)):
40 raise ValueError("projection fields must be sequences of column names")
41 source, target = candidate["source"], candidate["target"]
42 source_table, target_table = source["table"], target["table"]
43 source_key = _field(data, source_table, source["field"])
44 target_key = _field(data, target_table, target["field"])
45 source_fields, target_fields = tuple(source_fields), tuple(target_fields)
46 for field in source_fields:
47 _field(data, source_table, field)
48 for field in target_fields:
49 _field(data, target_table, field)
50 if (check.get("candidate_id") != candidate["candidate_id"]
51 or check.get("snapshot_id") != data.snapshot_id
52 or check.get("source") != source or check.get("target") != target):
53 raise ValueError("Candidate and check must refer to the same snapshot field pair")
54
55 counts = check.get("checks", {})
56 result = {
57 "candidate_id": candidate["candidate_id"], "snapshot_id": data.snapshot_id,
58 "source": source, "target": target,
59 "match_basis": "raw_value_identity", "semantic_relation": "unresolved",
60 "validation_ref": {
61 "candidate_id": check["candidate_id"], "snapshot_id": check["snapshot_id"],
62 "scan_scope": check.get("scan_scope"), "normalization": check.get("normalization"),
63 "selector": check.get("selector") or {},
64 "scope_bindings": check.get("scope_bindings") or {},
65 "eligible_references": counts.get("eligible_references"),
66 "unique_matches": counts.get("unique_matches"),
67 "ambiguous_matches": counts.get("ambiguous_matches"),
68 "missing_in_input": counts.get("missing_in_input"),
69 },
70 "matched_pairs": [], "counterexamples": [],
71 }
72 if (source_table == target_table or check.get("decision", {}).get("status") != "checked"
73 or check.get("normalization", "identity") not in {"identity", "nfkc_whitespace_casefold", "source_alias_items", "target_alias_items", "both_alias_items"}
74 or not isinstance(counts.get("unique_matches"), int)
75 or counts["unique_matches"] <= 0):
76 result["status"] = "not_applicable"
77 return result
78 if limit == 0:
79 result["status"] = "disabled_by_limit"
80 return result
81
82 selector = check.get("selector") or {}
83 scopes = check.get("scope_bindings") or {}
84 if not isinstance(selector, dict) or not isinstance(scopes, dict):
85 raise ValueError("selector and scope_bindings must be mappings")
86 for field in selector:
87 _field(data, source_table, field)
88 for target_field, source_field in scopes.items():
89 _field(data, target_table, target_field)
90 _field(data, source_table, source_field)
91
92 transform = check.get("normalization", "identity")
93 sql, params = association_match_sql(data, candidate, selector=selector,
94 scope_bindings=scopes, transform=transform)
95 # The matching SQL emits at most one target for each eligible source row.
96 # Sample different observed targets before repeating one popular target;
97 # this groups validated pairs in DuckDB, never expands a row/target product.
98 pairs = data.db.execute(f"""WITH matches AS ({sql}), ranked AS (
99 SELECT source_row_number, target_row_number,
100 row_number() OVER (PARTITION BY target_row_number
101 ORDER BY source_row_number) AS target_occurrence,
102 count(DISTINCT target_row_number) OVER () AS target_witness_count
103 FROM matches
104 ) SELECT source_row_number, target_row_number, target_witness_count FROM ranked
105 ORDER BY target_occurrence, source_row_number, target_row_number LIMIT ?""",
106 [*params, min(limit, 2)]).fetchall()
107 result["match_basis"] = transform
108 target_count = pairs[0][2] if pairs else 0
109 sampled_targets = len({pair[1] for pair in pairs})
110 result["sampling"] = {"method": "distinct_observed_target_round_robin",
111 "unique_target_witnesses": target_count,
112 "sampled_unique_target_witnesses": sampled_targets,
113 "unique_target_witnesses_not_sampled": target_count - sampled_targets,
114 "ambiguous_targets_sampled_as_positive": False}
115 for source_row, target_row, _ in pairs:
116 result["matched_pairs"].append({
117 "source_record": _row(data, source_table, source_row, source["field"], source_fields),
118 "target_record": _row(data, target_table, target_row, target["field"], target_fields),
119 "matching_raw_value": _row(data, source_table, source_row, source["field"], ())["fields"][source["field"]],
120 "source_raw_value": _row(data, source_table, source_row, source["field"], ())["fields"][source["field"]],
121 "target_raw_value": _row(data, target_table, target_row, target["field"], ())["fields"][target["field"]],
122 "transform": transform,
123 })
124 for example in counts.get("counterexample_rows", [])[:1]:
125 row_number = example["row_number"]
126 result["counterexamples"].append({
127 "reason": example["reason"],
128 "source_record": _row(data, source_table, row_number, source["field"], source_fields),
129 })
130 result["status"] = "examples" if result["matched_pairs"] else "no_sample"
131 return result
