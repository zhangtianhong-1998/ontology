21_MEMBER_CODE = re.compile(r"成员(?:编码|标识)|member\s+(?:code|identifier|key)", re.I)
22_LEXICAL = re.compile(r"名称|别名|同义词|\b(?:name|label|alias|aliases|synonym|synonyms|term|word)\b", re.I)
23_ALIAS = re.compile(r"别名|同义词|\b(?:alias|aliases|synonym|synonyms)\b", re.I)
24_CODE = re.compile(r"编码|标识|\b(?:code|identifier|key)\b", re.I)
25_MEMBER = re.compile(r"成员|\bmember\b", re.I)
26_SCOPE = re.compile(r"所属|内解释|作用域|within|scoped|namespace", re.I)
27_NEGATION = re.compile(r"不是|并非|不属于|非成员|not\s+(?:a\s+)?member|does\s+not\s+belong", re.I)

30def _normal(value):
31 return " ".join(unicodedata.normalize("NFKC", str(value)).split()).casefold()

34def _columns(data, table):
35 info = data.tables[table]

47def _member_specs(data):
48 """Recognize declared member keys and their explicitly named namespaces."""
49 output = []
50 for table in sorted(data.tables):
51 columns = _columns(data, table)
52 for column in columns:
53 declaration = str(column.get("column_comment") or "")
54 if not _MEMBER_CODE.search(declaration) or not _SCOPE.search(declaration):
55 continue
56 scopes = [item["column_name"] for item in columns
57 if item != column and re.search(
58 r"(?<![\w])" + re.escape(item["column_name"]) + r"(?![\w])", declaration)]
59 if len(scopes) == 1 and _lexical_columns(data, table):
60 output.append({"table": table, "field": column["column_name"],
61 "scope_column": scopes[0], "declaration": declaration})
62 return output

65def _lookup(data, table, bindings):
66 """Fresh bounded lookup, including on replay after source mutation."""
67 if not bindings or any(value in (None, "") for _, value in bindings):
68 return []
69 if any(column not in data.tables[table]["column_names"] for column, _ in bindings):
70 raise ValueError("Unknown semantic reference lookup column")
71 cursor = data.db.execute(f"SELECT * FROM {qi(data.tables[table]['sql_name'])} WHERE "
72 + " AND ".join(f"{qi(column)}=?" for column, _ in bindings) + " LIMIT 2",
73 [value for _, value in bindings])
74 return [dict(zip((item[0] for item in cursor.description), row)) for row in cursor.fetchall()]

77def _row(data, table, number):
78 cursor = data.db.execute(f"SELECT * FROM {qi(data.tables[table]['sql_name'])} WHERE __r2_row=?", [number])
79 values = cursor.fetchone()
80 if values is None:
81 raise ValueError("Semantic reference source row disappeared")
82 return dict(zip((item[0] for item in cursor.description), values))

85def _proof(data, table, row, column):
86 """Store the whole original cell; path extraction is independently replayed."""
87 record_id = data.record_id(table, row)
88 evidence_id = "record:" + digest([data.snapshot_id, record_id, column])[:24]
89 evidence = {"id": evidence_id, "origin": "observed_record", "raw_fragment": row[column],
90 "raw_fragment_truncated": False, "source_ref": {
91 "table": table, "column": column, "row": row["__r2_row"],
92 "record_id": record_id, "snapshot_id": data.snapshot_id}}
93 old = data.evidence.get(evidence_id)
94 if old and (old.get("raw_fragment") != evidence["raw_fragment"]
95 or old.get("source_ref") != evidence["source_ref"]):
96 raise ValueError("Conflicting source evidence for semantic reference")
97 data.evidence.setdefault(evidence_id, evidence)
98 return evidence_id

101def _identity_conflict(data, left_table, left, right_table, right):
102 """An explicit namespace/tenant/version is never discarded to obtain a match."""
103 sides = []
104 for table, row in ((left_table, left), (right_table, right)):
105 sides.append({column["column_name"]: row.get(column["column_name"])
106 for column in _columns(data, table) if declared_context_role(column) == "identity"})
107 return any(sides[0].get(key) in (None, "") or sides[1].get(key) in (None, "")
108 or sides[0][key] != sides[1][key] for key in set(sides[0]) | set(sides[1]))

111def _lexical_match(data, term, member_table, member):
112 for column in _lexical_columns(data, member_table):
113 name = column["column_name"]
114 value = member.get(name)
115 if not isinstance(value, str):
116 continue
117 declaration = str(column.get("column_comment") or "") + " " + name.replace("_", " ")
118 pieces = re.split(r"[,;，；、|]", value) if _ALIAS.search(declaration) else [value]
119 if _normal(term) in {_normal(part) for part in pieces if part.strip()}:
120 return name
121 return None

124def _membership_text(data, member_table, member, dimension_label):
125 """Require an affirmative statement naming the accepted Dimension itself."""
126 for column in _columns(data, member_table):
127 value = member.get(column["column_name"])

130 return column["column_name"]

149def build_semantic_references(data, plan, concepts=(), alignments=(), *, memberships=(), options=None):
150 """Return a copied plan, source proofs, typed instance paths and bounded coverage.
151
152 This known lexical/member contract needs no LLM. Unrecognized patterns are
153 pending, never guessed. Limits bound discovery and source-type rows; a
154 certified path still checks its code uniqueness against the whole table.
155 """
156 options = options or {}
157 defaults = {"max_source_rows_per_table": 10000, "max_paths": 64,
158 "max_candidates": 128, "max_target_fields": 128}
159 if set(options) - set(defaults):
160 raise ValueError("Unknown semantic reference option")
161 limits = {**defaults, **options}
162 if any(type(value) is not int or value < 1 for value in limits.values()):
163 raise ValueError("Semantic reference limits must be positive integers")
164 core = BuildPlan.model_validate(plan).model_copy(deep=True)
165 types = {item.id: item for item in core.object_types if item.category == "business_type"}
166 accepted, conflicts = _type_alignments(core, concepts, alignments, memberships)
167 wanted = {record_id: match for record_id, match in accepted.items()
168 if types[match["type_id"]].parent in {"Term", "Dimension"}}
169 sources, omissions = [], 0
170 for table in sorted({record_id.rpartition(":")[0] for record_id in wanted}):
171 if table not in data.tables:
172 continue
173 total = data.db.execute(f"SELECT count(*) FROM {qi(data.tables[table]['sql_name'])}").fetchone()[0]
174 omissions += max(0, total - limits["max_source_rows_per_table"])
175 for row in data.rows(table, limit=limits["max_source_rows_per_table"]):
176 match = wanted.get(data.record_id(table, row))
177 if match:
178 sources.append((table, row, types[match["type_id"]], match))
179 terms = [source for source in sources if source[2].parent == "Term"]
180 dimensions = [source for source in sources if source[2].parent == "Dimension"]
181 specs = _member_specs(data)
182 recognized = {spec["table"] for spec in specs}
183 skipped_member_tables = [{"table": table, "reason": "no_source_declared_member_key_and_named_namespace"}
184 for table in sorted(data.tables) if table not in recognized]
185 recalled = discover_json_reference_candidates(data,
186 source_tables={source[0] for source in terms},
187 target_fields={(spec["table"], spec["field"]) for spec in specs},
188 max_rows_per_table=limits["max_source_rows_per_table"], max_paths=limits["max_paths"],
189 max_candidates=limits["max_candidates"], max_target_fields=limits["max_target_fields"])
190 candidates = recalled["candidates"]
191 checks = [validate_candidate(data, candidate) for candidate in candidates]
192 by_source = defaultdict(list)
193 for candidate in candidates:
194 by_source[candidate["source"]["table"]].append(candidate)
195 nodes, assertions, paths, relation_proofs, pending = {}, {}, [], defaultdict(list), []
196 located_term_types = {source[2].id for source in terms}
197 pending.extend({"type_id": item.id, "reason": "accepted_term_type_has_no_verified_source_alignment"}
198 for item in types.values() if item.parent == "Term" and item.id not in located_term_types)
199 for table, row, term_type, alignment in terms:
200 matches = []
201 failure = "no_source_grounded_json_member_path"
202 lexical_fields = [column["column_name"] for column in _lexical_columns(data, table)
203 if _normal(row.get(column["column_name"], "")) == _normal(term_type.label)]
204 if not lexical_fields:
205 pending.append({"type_id": term_type.id, "reason": "term_label_not_in_source_lexical_field"})
206 continue
207 for candidate in by_source[table]:
208 source, target = candidate["source"], candidate["target"]
209 leaves = {tuple(path): value for path, value in json_scalar_paths(row.get(source["field"]))}
210 code = leaves.get(tuple(source["source_path"]))
211 if code is None:
212 continue
213 spec = next(item for item in specs if item["table"] == target["table"] and item["field"] == target["field"])
214 # No source namespace is invented. A code ambiguous anywhere in
215 # the target table needs an explicit namespace in the source JSON.
216 bindings = [(target["field"], code)]
217 namespace_values = [value for path, value in leaves.items() if path[-1] == spec["scope_column"]]
218 if namespace_values:
219 if len(set(namespace_values)) != 1:
220 failure = "conflicting_source_member_namespaces"
221 continue
222 bindings.append((spec["scope_column"], namespace_values[0]))
223 targets = _lookup(data, target["table"], bindings)
224 if len(targets) != 1:
225 failure = "member_code_missing_or_ambiguous_in_namespace"
226 continue
227 member = targets[0]
228 if _identity_conflict(data, table, row, target["table"], member):
229 failure = "lexical_member_identity_scope_conflict"
230 continue
231 lexical = _lexical_match(data, term_type.label, target["table"], member)
232 if lexical is None:
233 failure = "code_match_without_lexical_evidence"
234 continue
235 axes = []
236 for dt, dr, dimension, dm in dimensions:
237 if _identity_conflict(data, target["table"], member, dt, dr):
238 continue
239 text_column = _membership_text(data, target["table"], member, dimension.label)
240 if text_column is None:
241 continue
242 for column in _columns(data, dt):
243 name = column["column_name"]
244 if (_CODE.search(str(column.get("column_comment") or ""))
245 and member.get(spec["scope_column"]) not in (None, "")
246 and dr.get(name) == member[spec["scope_column"]]
247 and len(_lookup(data, dt, ((name, dr[name]),))) == 1):
248 axes.append((dt, dr, dimension, dm, name, text_column))
249 if len(axes) != 1:
250 failure = "member_dimension_missing_or_ambiguous"
251 continue
252 matches.append((candidate, spec, member, lexical, axes[0], bindings))
253 # Multiple candidate paths to the same witnessed member/axis are
254 # redundant evidence; distinct endpoints are a semantic ambiguity.
255 grouped = defaultdict(list)
256 for match in matches:
257 candidate, spec, member, lexical, axis, bindings = match
258 grouped[(data.record_id(spec["table"], member), axis[2].id)].append(match)
259 if len(grouped) != 1:
260 pending.append({"type_id": term_type.id, "source_record_id": data.record_id(table, row),
261 "reason": "multiple_member_paths" if grouped else failure})
262 continue
263 candidate, spec, member, lexical, axis, bindings = next(iter(grouped.values()))[0]
264 dt, dr, dimension, dm, dimension_key, membership_text = axis
265 source = candidate["source"]
266 cells = [(table, row, lexical_fields[0]), (table, row, source["field"]),
267 (spec["table"], member, spec["field"]), (spec["table"], member, lexical),
268 (spec["table"], member, spec["scope_column"]),
269 (spec["table"], member, membership_text), (dt, dr, dimension_key)]
270 evidence = sorted({_proof(data, *cell) for cell in cells} |
271 set(alignment["evidence_ids"]) | set(dm["evidence_ids"]))
272 schema_evidence = [f"schema:{owner}:{column}" for owner, column in (
273 (table, source["field"]),
274 (spec["table"], spec["field"]), (spec["table"], spec["scope_column"]),
275 (spec["table"], lexical), (dt, dimension_key))]
276 evidence = sorted(set(evidence) | set(schema_evidence))
277 term_node = _node(data, table, row, kind="lexical_term", label=term_type.label,
278 type_id=term_type.id, evidence_ids=evidence)
279 member_label = next((member[column["column_name"]] for column in _lexical_columns(data, spec["table"])
280 if member.get(column["column_name"]) and not _ALIAS.search(
281 str(column.get("column_comment") or "") + " " + column["column_name"].replace("_", " "))), member[lexical])
282 member_node = _node(data, spec["table"], member, kind="dimension_member",
283 label=member_label, type_id=dimension.id, evidence_ids=evidence)
284 dimension_node = _node(data, dt, dr, kind="dimension_definition", label=dimension.label,
285 type_id=dimension.id, evidence_ids=evidence)
286 proof = {"snapshot_id": data.snapshot_id, "candidate": candidate,
287 "member_spec": spec, "source_lexical_column": lexical_fields[0],
288 "member_lexical_column": lexical, "membership_text_column": membership_text,
289 "dimension_key_column": dimension_key, "member_lookup_bindings": bindings,
290 "term_type_id": term_type.id, "dimension_type_id": dimension.id,
291 "term_node": term_node["id"], "member_node": member_node["id"],
292 "dimension_node": dimension_node["id"], "evidence_ids": evidence}
293 for node in (term_node, member_node, dimension_node):
294 if node["id"] in nodes:
295 nodes[node["id"]]["evidence_ids"] = sorted(set(nodes[node["id"]]["evidence_ids"]) | set(evidence))
296 else:
297 nodes[node["id"]] = node
298 edge_ids = []
299 for left, right, parent, predicate in ((term_node, member_node, "points_to", "denotes"),
300 (member_node, dimension_node, "related_to", "member_of")):
301 aid = "semantic_reference:" + digest([data.snapshot_id, predicate, left["id"], right["id"]])[:24]
302 assertions[aid] = {"id": aid, "subject": left["id"], "object": right["id"],
303 "predicate": parent, "parent_relation": parent, "predicate_name": predicate,
304 "instance_scope": "source_bound_lexical_member_path_not_type_identity",
305 "definition": canonical_relation_definition(parent, predicate),
306 "evidence_scope": "verified_lexical_member_path", "proof": proof,
307 "evidence_ids": evidence, "decision": {"status": "accepted", "method": "replayed_source_path"}}
308 edge_ids.append(aid)
309 rid = canonical_relation_id("related_to", term_type.id, dimension.id, predicate_name="term_dimension_context")
310 path = {"id": "semantic_reference_path:" + digest([term_node["id"], member_node["id"], dimension_node["id"]])[:24],
311 "relation_type_id": rid, "source_type_id": term_type.id, "target_type_id": dimension.id,
312 "instance_ids": [term_node["id"], member_node["id"], dimension_node["id"]],
313 "assertion_ids": edge_ids, "proof": proof, "evidence_ids": evidence}
314 # Self-contained evidence lets ordinary validate_plan replay a type
315 # relation after reload, without a transient stage-result side channel.
316 path_evidence_id = "lexical_member_path:" + digest([data.snapshot_id, path["id"]])[:24]
317 data.evidence[path_evidence_id] = {
318 "id": path_evidence_id, "origin": "semantic_reference_path",
319 "raw_fragment": "Term denotes member; member belongs to Dimension",
320 "raw_fragment_truncated": False, "source_ref": {"snapshot_id": data.snapshot_id},
321 "path": path, "instances": [term_node, member_node, dimension_node],
322 "assertions": [assertions[aid] for aid in edge_ids]}
323 path["path_evidence_id"] = path_evidence_id
324 path["evidence_ids"] = [*evidence, path_evidence_id]
325 paths.append(path)
326 relation_proofs[rid].append(path)

328 for rid, group in relation_proofs.items():
329 core.relation_types = [item for item in core.relation_types if item.id != rid]
330 relation = DerivedType(id=rid, parent="related_to", category="business_relation_type",
331 label="term_dimension_context", predicate_name="term_dimension_context",
332 definition=canonical_relation_definition("related_to", "term_dimension_context"),
333 domain=[group[0]["source_type_id"]], range=[group[0]["target_type_id"]],

340 "relation_contracts": [{"relation_type_id": rid, "path_ids": [path["id"] for path in group],
341 "endpoint_semantics": "Term denotes member; member belongs to Dimension; no type identity claim"}

343 "coverage": {"accepted_term_source_records": len(terms), "paths_accepted": len(paths),
344 "unique_member_instances": sum(node["instance_kind"] == "dimension_member" for node in nodes.values()),
345 "relation_types": len(relation_proofs), "assertions": len(assertions),
346 "pending": len(pending), "source_rows_not_examined": omissions,

369def verify_semantic_reference_assertion(data, assertion, instances, plan):
370 """Re-read every proof cell and execute the two referenced keys before export."""
371 nodes = instances if isinstance(instances, dict) else {item["id"]: item for item in instances}
372 types = {item.id: item for item in plan.object_types}
373 proof = assertion["proof"]
374 if proof.get("snapshot_id") != data.snapshot_id or assertion.get("decision", {}).get("status") != "accepted":
375 raise ValueError("Semantic reference has no current accepted proof")
376 term, member, dimension = (nodes[proof[key]] for key in ("term_node", "member_node", "dimension_node"))
377 if (types[term["ontology_type_id"]].parent != "Term"
378 or types[dimension["ontology_type_id"]].parent != "Dimension"
379 or member["ontology_type_id"] != dimension["ontology_type_id"]):
380 raise ValueError("Semantic reference levels or accepted types changed")
381 rows = []
382 for node in (term, member, dimension):
383 ref = node["source_ref"]
384 row = _row(data, ref["table"], ref["row"])
385 if data.record_id(ref["table"], row) != ref["record_id"]:
386 raise ValueError("Semantic reference source identity changed")
387 rows.append(row)
388 for eid in proof["evidence_ids"]:
389 evidence = data.evidence.get(eid) or {}
390 if evidence.get("origin") != "observed_record":
391 continue
392 ref = evidence.get("source_ref") or {}
393 if (ref.get("snapshot_id") != data.snapshot_id or evidence.get("raw_fragment_truncated")
394 or _row(data, ref["table"], ref["row"]).get(ref["column"]) != evidence.get("raw_fragment")):
395 raise ValueError("Semantic reference source evidence changed")
