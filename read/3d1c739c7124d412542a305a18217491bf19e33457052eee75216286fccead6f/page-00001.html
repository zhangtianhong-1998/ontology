11def rule(identifier, *, source="s.a", target="s.b", field="label", target_field="alias",
12 selector=None, numeric=False, transform="identity", scope=None):
13 conditional = bool(selector)
14 return {"rule_id": identifier, "candidate_id": identifier,
15 "status": "checked_technical", "semantic_relation": "unresolved",
16 "source": {"table": source, "field": field},
17 "target": {"table": target, "field": target_field},
18 "selector": selector or {}, "scope_bindings": scope or {},
19 "transform": {"operator": transform}, "numeric_overlap_only": numeric,
20 "risk_flags": ["numeric_value_coincidence"] if numeric else [],
21 "verification": {"scan_scope": "full_input", "checks": {
22 "source_rows": 30, "selector_true": 10 if conditional else 30,
23 "selector_false": 20 if conditional else 0, "selector_unknown": 0,
24 "eligible_references": 10 if conditional else 30,
25 "unique_matches": 10 if conditional else 30,
26 "target_rows_with_complete_key": 10, "target_max_multiplicity": 1,
27 "ambiguous_matches": 0, "missing_in_input": 0}}}
