1"""Prepare a Windows x64 / CPython 3.14 wheel bundle on a networked machine.
2
3Example (macOS preparation host):
4
5 python scripts/prepare_windows_offline.py --bundle .cache/windows-offline --with-datahub-lite \
6 --pip-python .venv-datahub/bin/python
7
8The target computer needs no network, Docker, or source database. DataHub Lite
9is optional and must be installed in a separate environment from route 2.
10Only Python wheels, pinned requirements, and their hashes enter the bundle.
11"""

41def _run(command, *, capture=False, input_text=None):

46def _sha256(path):
47 hash_obj = hashlib.sha256()
48 with Path(path).open("rb") as stream:
49 for block in iter(lambda: stream.read(1024 * 1024), b""):
50 hash_obj.update(block)
51 return hash_obj.hexdigest()

54def _target_environment():
55 environment = default_environment()
56 environment.update(
57 os_name="nt",
58 sys_platform="win32",

61 platform_python_implementation="CPython",

70def _logical_lines(text):
71 pending = ""
72 for raw in text.splitlines():
73 line = raw.strip()
74 if not line or line.startswith("#"):
75 continue
76 if line.endswith("\\"):
77 pending += line[:-1].strip() + " "
78 continue
79 yield pending + line
80 pending = ""
81 if pending:
82 raise ValueError("Unterminated requirements continuation")

85def _target_requirements(exported, *, require_hashes):
86 selected = {}
87 for line in _logical_lines(exported):
88 head, *hashes = re.split(r"\s+--hash=", line)
89 requirement = Requirement(head.strip())
90 if requirement.marker and not requirement.marker.evaluate(environment=_target_environment()):
91 continue
92 if requirement.url or len(requirement.specifier) != 1 or next(iter(requirement.specifier)).operator != "==":
93 raise ValueError(f"uv.lock did not export an exact pin: {head}")
94 normalized_name = canonicalize_name(requirement.name)
95 if normalized_name in selected:
96 raise ValueError(f"Duplicate target package: {requirement.name}")
97 requirement_text = f"{normalized_name}=={next(iter(requirement.specifier)).version}"
98 if require_hashes:
99 valid_hashes = sorted(set(value.strip() for value in hashes))
100 if not valid_hashes or any(not re.fullmatch(r"sha256:[0-9a-f]{64}", value) for value in valid_hashes):
101 raise ValueError(f"Missing/invalid lock hash for {requirement.name}")
102 requirement_text += " " + " ".join("--hash=" + value for value in valid_hashes)
103 selected[normalized_name] = requirement_text
104 if not selected:
105 raise ValueError("No Windows CPython 3.14 requirements in uv.lock")
106 return [selected[key] for key in sorted(selected)]

113def _download(python, wheelhouse, requirements, cache, *, offline_only=False, spec=None):
114 command = _pip(python, "download", "--quiet", "--dest", str(wheelhouse), *TARGET, "--find-links", str(cache))
115 if spec:
116 command.append(spec)
117 else:
118 command.extend(["-r", str(requirements)])
119 # First try the existing local wheel cache without network. A failed
120 # attempt only means a wheel is missing; the online retry fills that gap.
121 try:
122 _run([*command, "--no-index"])
123 except subprocess.CalledProcessError:
124 if offline_only:
125 raise
126 _run(command)

129def _verify_offline(python, wheelhouse, requirements=None, spec=None):
130 with tempfile.TemporaryDirectory(prefix="ontology-win-wheel-verify-") as temp:
131 command = _pip(python, "download", "--quiet", "--no-index", "--find-links", str(wheelhouse), "--dest", temp, *TARGET)

135 command.append(spec)

139def _build_route2_wheel(wheelhouse, version, *, offline_only=False):
140 command = ["uv", "build", "--wheel", "--no-build-logs", "--out-dir", str(wheelhouse), str(PROJECT)]
141 if offline_only:
142 command.append("--offline")
143 _run(command)
144 wheels = sorted(wheelhouse.glob(f"ontology_route2-{version}-*.whl"))
145 if len(wheels) != 1:
146 raise ValueError(f"Expected exactly one ontology-route2 wheel for {version}")
147 wheel = wheels[0]
148 if not wheel.name.endswith("-py3-none-any.whl"):
149 raise ValueError(f"Route 2 wheel must be pure Python: {wheel.name}")
150 with zipfile.ZipFile(wheel) as archive:
151 names = archive.namelist()
152 metadata_prefix = f"ontology_route2-{version}.dist-info/"
153 if not names or any(not (name.startswith("ontology_r2/") or name.startswith(metadata_prefix)) for name in names):
154 raise ValueError("Route 2 wheel contains files outside its package/metadata")
155 if "ontology_r2/viewer.html" not in names:
156 raise ValueError("Route 2 wheel is missing the local viewer")
157 return wheel

160def prepare(bundle, cache, pip_python, *, with_datahub_lite=False, offline_only=False):
161 bundle, cache = Path(bundle).resolve(), Path(cache).resolve()
162 wheelhouse = bundle / "wheelhouse"
163 wheelhouse.mkdir(parents=True, exist_ok=True)
164 # Keep the venv's executable path: resolving its symlink can switch pip
165 # to the underlying base interpreter and lose the venv's site-packages.
166 pip_python = Path(pip_python).absolute()
167 if not pip_python.is_file():
168 raise FileNotFoundError(f"Preparation Python not found: {pip_python}")
169 _run(_pip(pip_python, "--version"))
170 project_meta = tomllib.loads((PROJECT / "pyproject.toml").read_text(encoding="utf-8"))
171 route2_version = project_meta["project"]["version"]
172
173 export = ["uv", "export", "--project", str(PROJECT), "--locked", "--no-dev", "--no-emit-project", "--format", "requirements.txt", "--no-header", "--no-annotate", "--quiet"]
174 if offline_only:
175 export.append("--offline")
176 simple = _target_requirements(_run([*export, "--no-hashes"], capture=True), require_hashes=False)
177 hashed = _target_requirements(_run(export, capture=True), require_hashes=True)
178 if [line.split(" ", 1)[0] for line in hashed] != simple:
179 raise ValueError("Hashed and unhashed lock exports differ")
180 locked_file = bundle / "requirements-route2-locked.txt"
181 locked_file.write_text("\n".join(simple) + "\n", encoding="utf-8")
182 hashed_file = bundle / "requirements-route2-hashed.txt"
183 hashed_file.write_text("\n".join(hashed) + "\n", encoding="utf-8")
184
185 route2_wheel = _build_route2_wheel(wheelhouse, route2_version, offline_only=offline_only)
186 _download(pip_python, wheelhouse, locked_file, cache, offline_only=offline_only)
187 _verify_offline(pip_python, wheelhouse, hashed_file)
188 _verify_offline(pip_python, wheelhouse, locked_file, f"ontology-route2=={route2_version}")
189
190 datahub_file = None
191 if with_datahub_lite:
192 datahub_file = bundle / "requirements-datahub-lite-locked.txt"
193 datahub_input = bundle / "requirements-datahub-lite.in"
194 datahub_input.write_text(DATAHUB_REQUIREMENT + "\n", encoding="utf-8")
195 compile_command = ["uv", "pip", "compile", str(datahub_input), "--python-platform", "x86_64-pc-windows-msvc", "--python-version", "3.14", "--only-binary", ":all:", "--no-header", "--no-annotate", "--output-file", str(datahub_file), "--quiet", "--find-links", str(cache)]
196 try:
197 _run([*compile_command, "--no-index"])
198 except subprocess.CalledProcessError:
199 if offline_only:
200 raise
201 _run(compile_command)
202 _download(pip_python, wheelhouse, datahub_file, cache, offline_only=offline_only)
203 # Verify DataHub's closure independently; it is installed in another
204 # venv and may legitimately require different transitive versions.
205 _verify_offline(pip_python, wheelhouse, datahub_file)
206 _verify_offline(pip_python, wheelhouse, spec=DATAHUB_REQUIREMENT)
207
208 files = [locked_file, hashed_file, route2_wheel, *sorted(wheelhouse.glob("*.whl"))]
209 if datahub_file:
210 files.extend([datahub_file, bundle / "requirements-datahub-lite.in"])
211 unique_files = sorted(set(files), key=lambda path: str(path.relative_to(bundle)))
212 manifest = {
213 "target": {"os": "Windows", "arch": "x86_64", "python": "CPython 3.14", "wheel_platform": "win_amd64"},
214 "route2": {"version": route2_version, "requirements": locked_file.name, "wheel": route2_wheel.name, "lock_sha256": _sha256(PROJECT / "uv.lock"), "offline_closure_verified": True},
215 "datahub_lite": {"enabled": with_datahub_lite, "version": "1.7.0.12" if with_datahub_lite else None, "separate_environment_required": with_datahub_lite, "offline_closure_verified": with_datahub_lite},
216 "files": [{"path": str(path.relative_to(bundle)), "size": path.stat().st_size, "sha256": _sha256(path)} for path in unique_files],
217 "contains_business_data": False,
218 }
219 manifest_file = bundle / "sha256-manifest.json"
220 manifest_file.write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
221 return {"bundle": str(bundle), "wheels": len(list(wheelhouse.glob("*.whl"))), "route2_requirements": len(simple), "datahub_lite": with_datahub_lite, "manifest": str(manifest_file)}
