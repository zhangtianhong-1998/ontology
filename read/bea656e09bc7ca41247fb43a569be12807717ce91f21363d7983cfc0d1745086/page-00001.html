38@pytest.mark.parametrize("case", SAVED["cases"], ids=lambda case: case["table"])
39def test_saved_v11_responses_replayed_against_original_cited_source_rows(tmp_path, case):
40 """Replay all proposals unchanged; sparse row numbers retain original citations."""
41 root = tmp_path / "input"
42 table_name = case["table"]
43 schema = case["schema"]
44 write_yaml(root / "schema/tables" / f"{table_name}.yaml", schema)
45 write_yaml(root / "schema/constraints" / f"{table_name}.yaml", case["constraints"])
46 write_yaml(root / "schema/foreign_keys" / f"{table_name}.yaml", {
47 "schema": schema["schema"], "table_name": table_name, "foreign_keys": []})
48 (root / "data").mkdir()
49 with (root / "data" / f"{table_name}.csv").open("w", newline="") as stream:
50 writer = csv.DictWriter(stream, fieldnames=[col["column_name"] for col in schema["columns"]])
51 writer.writeheader()
52 writer.writerows(item["values"] for item in case["source_rows"])
53 data = _open_data(root, tmp_path)
54 index = None
55 try:
56 name = f"{schema['schema']}.{table_name}"
57 table = data.tables[name]
58 positions = {i: item["source_row_number"] for i, item in enumerate(case["source_rows"], 1)}
59 # One CASE update avoids temporary collisions with another original row.
60 clauses = " ".join(f"WHEN {compact} THEN {original}" for compact, original in positions.items())
61 data.db.execute(f"UPDATE {qi(table['sql_name'])} SET __r2_row = CASE __r2_row {clauses} END")
62 observed = {(col, item["source_row_number"]): raw for item in case["source_rows"]
63 for col, raw in item["values"].items() if raw}
64 response = ColumnRoleInferenceResponse.model_validate(case["response"])
65 accepted, rejected = _validate_response(data, name, response, set(table["column_names"]), observed)
66 # The saved API response also mislabels a numeric table key as an
67 # operand; the pre-existing fragment validator correctly rejects it.
68 expected_rejected = ([{"column": "wide_table_id", "role": "operand_reference",
69 "reason": "fragment_role_conflicts_with_observed_values"}]
70 if table_name == "fruit_market_api" else [])
71 assert rejected == expected_rejected
72 assert len(accepted) + len(rejected) == len(response.proposals)
73 table["inferred_semantic_roles"] = accepted
74 for item in accepted:
75 if item.get("proposed_role") == "identity":
76 assert item["role"] == "reference"
77 assert item["role_evidence"]["semantic_identity_claim"] is False
78 assert item["observations"] # Original IDs/values remain traceable.
79 built = build_semantic_cards(data, tmp_path / "cards.sqlite")
80 index = SemanticCardIndex(built["index_path"])
81 expected_identity = {"version"} if table_name == "fruit_market_api" else set()
82 assert set(_field_roles(table).get("identity", ())) == expected_identity
83 cards = index.all_cards(100)["cards"]
84 assert cards
85 assert all({item["column"] for item in card["fields"].get("identity", ())}
86 == expected_identity for card in cards)
87 if table_name == "fruit_synonym_dict":
88 assert _field_roles(table)["unknown"] == ["type", "attr"]
89 assert built["coverage"]["rows_without_card"] == 0
90 assert built["coverage"]["structured_definition_candidate_rows"] == len(case["source_rows"])
91 packets = build_instance_bundles(data, index, {"rules": []}, {
92 "max_concept_bundles": 10, "max_relation_bundles": 0,
93 "max_candidates_per_bundle": 1, "max_joined_context_records": 0})
94 assert packets["bundles"]
95 assert packets["coverage"]["definition_patterns_unprocessed"] == 0
96 assert all("attr" in {item["column"] for item in card["fields"]["unknown"]}
97 for bundle in packets["bundles"] for card in bundle["records"])
98 # A reference JSON is evidence for review, not a manufactured meaning.
99 assert all(not card["fields"].get("description") for card in cards)
100 finally:
101 if index:
102 index.close()
103 data.close()
