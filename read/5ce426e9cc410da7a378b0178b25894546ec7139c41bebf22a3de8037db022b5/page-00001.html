1"""Export the existing technical graph as DataHub metadata-file MCPs.
2
3This is an offline interchange adapter. It reads only table/column metadata from
4``meta_graph.yaml``; it does not start DataHub, read CSV rows, or infer relations.
5The JSON shape follows DataHub's simplified metadata-file sink (MCP + aspect.json).
6"""

14def _urn_part(value, label):
15 if not isinstance(value, str) or not value or any(c in value for c in ",()"):
16 raise ValueError(f"Invalid DataHub {label}: {value!r}")
17 return value

20def dataset_urn(qualified_name, platform="postgres", environment="PROD"):
21 """The default DataHub dataset identity for one local source table.
22
23 This is an interchange identifier. It does not imply that a DataHub
24 service, graph index, or schema-field entity was created.
25 """
26 platform = _urn_part(platform, "platform")
27 qualified_name = _urn_part(qualified_name, "dataset name")
28 environment = _urn_part(environment, "environment")
29 return f"urn:li:dataset:(urn:li:dataPlatform:{platform},{qualified_name},{environment})"

32def schema_field_urn(qualified_name, field_path, platform="postgres", environment="PROD"):
33 """Stable DataHub schema-field identity; does not require a running server."""
34 field_path = _urn_part(field_path, "field path")
35 return f"urn:li:schemaField:({dataset_urn(qualified_name, platform, environment)},{field_path})"

38def _nullable(value):
39 if isinstance(value, bool):
40 return not value
41 if isinstance(value, str):
42 normalized = value.strip().lower()
43 if normalized in {"true", "t", "yes", "1"}:
44 return False
45 if normalized in {"false", "f", "no", "0"}:
46 return True
47 if value is None:
48 return True
49 raise ValueError(f"Invalid is_not_null value: {value!r}")

52def _datahub_type(native_type):
53 name = native_type.lower().strip()
54 # Arrays must be recognized before their scalar element type.
55 if name.endswith("[]") or name.startswith("array"):
56 return "ArrayType"
57 if re.match(r"^(bool|boolean)\b", name):
58 return "BooleanType"
59 if re.match(r"^(smallint|integer|int|bigint|numeric|decimal|real|float|double|money|serial|bigserial)\b", name):
60 return "NumberType"
61 if re.match(r"^date\b", name):
62 return "DateType"
63 if re.match(r"^(time|timestamp|timestamptz|timetz|datetime)\b", name):
64 return "TimeType"
65 if re.match(r"^(bytea|blob|binary|varbinary)\b", name):
66 return "BytesType"
67 # DataHub requires a generic type. Keep the source's full type verbatim in
68 # nativeDataType so this fallback cannot erase its original meaning.
69 return "StringType"

72def _mcp(urn, aspect_name, aspect, run_id):
73 return {
74 "entityType": "dataset",
75 "entityUrn": urn,
76 "changeType": "UPSERT",
77 "aspectName": aspect_name,
78 "aspect": {"json": aspect},
79 # DataHub Lite's DuckDB writer expects this field to exist in the
80 # serialized MCP. Zero means the source observation time is unknown.
81 "systemMetadata": {"lastObserved": 0, "runId": run_id, "lastRunId": run_id, "properties": {}},
82 }

85def _declared_primary_keys(table_id, graph_nodes, edges, column_names):
86 keys = []
87 for edge in edges:
88 if edge.get("source") != table_id or edge.get("type") != "has_declared_constraint":
89 continue
90 constraint = graph_nodes.get(edge.get("target"), {})
91 if constraint.get("constraint_type") != "p" and str(constraint.get("constraint_type_name", "")).upper() != "PRIMARY KEY":
92 continue
93 match = re.search(r"PRIMARY\s+KEY\s*\(([^)]+)\)", str(constraint.get("definition", "")), re.I)
94 if match:
95 names = [item.strip().strip('"') for item in match.group(1).split(",")]
96 if all(name in column_names for name in names):
97 keys = names
98 break
99 return keys

102def metadata_change_proposals(meta_graph, platform="postgres", environment="PROD"):
103 """Return deterministic DataHub MCP dictionaries for table and column metadata.
104
105 ``platform='postgres'`` is a practical default for GaussDB-compatible
106 catalogs; callers may choose another registered DataHub platform key.
107 ``meta_graph`` is the existing technical_graph() result or loaded YAML.
108 """
109 platform = _urn_part(platform, "platform")
110 environment = _urn_part(environment, "environment")
111 nodes = {node["id"]: node for node in meta_graph["nodes"]}
112 edges = meta_graph["edges"]
113 source_nodes = {node_id: node for node_id, node in nodes.items() if node.get("kind") == "Source"}
114 snapshots = sorted(node_id for node_id, node in nodes.items() if node.get("kind") == "DatasetSnapshot")
115 if len(snapshots) > 1:
116 raise ValueError("Expected at most one DatasetSnapshot")
117 snapshot_id = snapshots[0].removeprefix("snapshot:") if snapshots else None
118 run_id = "ontology-r2-" + (snapshot_id or "metadata-export")
119 proposals = []
120 for table_id, table in sorted(nodes.items()):
121 if table.get("kind") != "Table":
122 continue
123 qualified_name = _urn_part(table_id, "dataset name")
124 table_name = table.get("table_name")
125 if not isinstance(table_name, str) or not table_name:
126 raise ValueError(f"Missing table_name for {table_id}")
127 urn = dataset_urn(qualified_name, platform, environment)
128 table_edges = [edge for edge in edges if edge.get("source") == table_id]
129 columns = [nodes[edge["target"]] for edge in table_edges if edge.get("type") == "table_has_column" and edge.get("target") in nodes]
130 columns.sort(key=lambda col: (int(col.get("ordinal_position") or 0), col["column_name"]))
131 column_names = {col["column_name"] for col in columns}
132 primary_keys = _declared_primary_keys(table_id, nodes, table_edges, column_names)
133
134 provenance = {}
135 if snapshot_id:
136 provenance["source_snapshot_id"] = snapshot_id
137 for edge_type, label in (("documented_by", "source_schema"), ("sample_from", "source_csv")):
138 source = next((source_nodes.get(edge["target"]) for edge in table_edges if edge.get("type") == edge_type and edge.get("target") in source_nodes), None)
139 if source:
140 provenance[label + "_file"] = str(source["path"])
141 provenance[label + "_sha256"] = str(source["sha256"])
142 if table.get("relkind") is not None:
143 provenance["source_relkind"] = str(table["relkind"])
144 if table.get("observed_rows") is not None:
145 provenance["source_observed_rows"] = str(table["observed_rows"])
146
147 properties = {"name": table_name, "qualifiedName": qualified_name, "customProperties": provenance}
148 if table.get("table_comment"):
149 properties["description"] = str(table["table_comment"])
150 proposals.append(_mcp(urn, "datasetProperties", properties, run_id))
151
152 fields = []
153 for column in columns:
154 native = str(column.get("data_type") or "unknown")
155 generic = _datahub_type(native)
156 field = {
157 "fieldPath": column["column_name"],
158 "nullable": _nullable(column.get("is_not_null")),
159 "type": {"type": {f"com.linkedin.schema.{generic}": {}}},
160 "nativeDataType": native,
161 "recursive": False,
162 "isPartOfKey": column["column_name"] in primary_keys,
163 }
164 if column.get("column_comment"):
165 field["description"] = str(column["column_comment"])
166 fields.append(field)
167 schema_body = {
168 "schemaName": qualified_name,
169 "platform": f"urn:li:dataPlatform:{platform}",
170 "version": 0,
171 "hash": hashlib.sha256(json.dumps(fields, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode()).hexdigest(),
172 "platformSchema": {"com.linkedin.schema.OtherSchema": {"rawSchema": ""}},
173 "fields": fields,
174 }
175 if primary_keys:
176 schema_body["primaryKeys"] = primary_keys
177 proposals.append(_mcp(urn, "schemaMetadata", schema_body, run_id))
178 return proposals

181def export_datahub_metadata(meta_graph, output, platform="postgres", environment="PROD"):
182 """Write a DataHub metadata-file JSON array without a running server."""
183 proposals = metadata_change_proposals(meta_graph, platform, environment)
184 output = Path(output)
185 output.parent.mkdir(parents=True, exist_ok=True)
186 output.write_text(json.dumps(proposals, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
187 return {"path": str(output), "datasets": len(proposals) // 2, "aspects": len(proposals)}
