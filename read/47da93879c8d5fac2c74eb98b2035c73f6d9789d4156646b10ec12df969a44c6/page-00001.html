31def main():
32 parser = argparse.ArgumentParser()
33 parser.add_argument("--output", required=True, help="New YAML result file")
34 args = parser.parse_args()
35 output = Path(args.output)
36 if output.exists():
37 raise FileExistsError(output)
38 sql = Path(__file__).with_name("cases.sql").read_bytes()
39 with duckdb.connect() as db:
40 actual = dict(db.execute(sql.decode()).fetchall())
41 checks = [{"name": name, "actual": actual.get(name), "expected": expected,
42 "passed": name in actual and (actual[name] is None if expected is None
43 else actual[name] is not None and math.isclose(actual[name], expected))}
44 for name, expected in EXPECTED.items()]
45 passed = set(actual) == set(EXPECTED) and all(c["passed"] for c in checks)
46 report = {"status": "passed" if passed else "failed", "recorded_at": datetime.now(timezone.utc).isoformat(),
47 "synthetic": True, "llm_calls": 0, "duckdb_version": duckdb.__version__,
48 "sql_sha256": hashlib.sha256(sql).hexdigest(), "checks": checks,
49 "validation_scope": "SQL measurement definitions only; no field candidate discovery or business semantic accuracy claim"}
50 output.parent.mkdir(parents=True, exist_ok=True)
51 output.write_text(yaml.safe_dump(report, allow_unicode=True, sort_keys=False))
52 print({"status": report["status"], "checks": len(checks), "llm_calls": 0, "output": str(output)})
53 if not passed:
54 raise SystemExit(1)
