1205def _balanced_selection(bundles, limit):
1206 """Reserve bounded model calls for both concept and relation packets."""
1207 queues = {kind: [item for item in bundles if item.get("task_kind") == kind]
1208 for kind in ("concept_induction", "relation_meaning")}
1209 other = [item for item in bundles if item.get("task_kind") not in queues]
1210 selected = []
1211 while len(selected) < limit and any(queues.values()):
1212 for kind in queues:
1213 if queues[kind] and len(selected) < limit:
1214 selected.append(queues[kind].pop(0))
