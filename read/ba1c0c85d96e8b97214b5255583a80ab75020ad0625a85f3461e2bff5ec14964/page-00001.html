203def _measure_reuse_assessment(data, records, label=None):
204 """Every exact record needs its own positive, complete reuse declaration.
205
206 A null business_object_quote says nothing about the original definition.
207 Related records and formula/operator shapes cannot fill this evidence gap.
208 """
209 assessments = []
210 for record in records:
211 metadata = {item["column_name"]: item for item in
212 getattr(data, "tables", {}).get(record["table"], {}).get("columns", [])}
213 proofs, conflicts, standard_names = [], [], []
214 for role, entry in _entries(record):
215 if entry.get("truncated"):
216 continue
217 column, value = entry["column"], str(entry.get("value") or "").strip()
218 if not value:
219 continue
220 declaration = str(metadata.get(column, {}).get("column_comment") or "")
221 witness = {"column": column, "role": role, "value": value,
222 "declaration": declaration}
223 business_scope = role == "scope" and (
224 _BUSINESS_SCOPE_COLUMN.search(column) or _BUSINESS_SCOPE_COLUMN.search(declaration))
225 if role in ("description", "scope") and _generic_business_statement(value):
226 proofs.append({**witness, "basis": "explicit_business_independence"})
227 elif business_scope and _explicitly_unrestricted(value):
228 proofs.append({**witness, "basis": "unrestricted_business_object_field"})
229 elif business_scope:
230 conflicts.append({**witness, "reason": "explicit_business_object_binding"})
231 if role in ("name", "alias") and (
232 _STANDARD_NAME_COLUMN.search(column) or _STANDARD_NAME_COLUMN.search(declaration)):
233 standard_names.append(witness)
234 # A short alias cannot abstract its own more specific canonical
235 # definition. A separate generic definition must be the seed.
236 if (label and label.casefold() != value.casefold()
237 and label.casefold() in value.casefold()):
238 conflicts.append({**witness, "reason": "short_name_does_not_replace_standard_name"})
239 reasons = sorted({item["reason"] for item in conflicts})
240 if not proofs:
241 reasons.append("missing_explicit_business_independence")
242 assessments.append({"record_id": record["record_id"], "table": record["table"],
243 "supported": bool(proofs) and not conflicts,
244 "proofs": proofs, "conflicts": conflicts,
245 "standard_names": standard_names, "reasons": reasons})
246 return assessments

424 descriptions, names, formulas = set(), set(), []

427 for role, entry in _entries(record):

439 record_formulas.add(" ".join(value.split()))

443 raise ValueError("Conflicting source formulas cannot share an exact concept")
