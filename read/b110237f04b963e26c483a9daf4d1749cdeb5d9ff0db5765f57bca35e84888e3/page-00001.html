96class _LLM:
97 def __init__(self, decision=None):
98 self.calls = []
99 self.decision = decision

101 async def ask(self, task, payload, schema):
102 self.calls.append((task, payload))
103 assert task == "fact_type_binding"
104 assert "observed_value" not in str(payload)
105 return schema.model_validate(self.decision or {
106 "status": "bind", "type_id": "type:operating_profit",
107 "source_column_quote": payload["column_comment"],
108 "type_definition_quote": payload["type_candidates"][0]["definition"],
109 "source_definition_evidence_id": payload["type_candidates"][0][
110 "full_source_definitions"][0]["evidence_id"],
111 "type_source_quote": payload["type_candidates"][0][
112 "full_source_definitions"][0]["value"],
113 })
