49 record["fields"]["description"][0]["column"] = "measure_description"
50 record["fields"]["unit"][0]["value"] = "元/kg"

56 "aggregation_operator": None, "ontology_level": "type",

58 "scope_roles": {"domain_code": "applicability", "inference_type": "parameter", "period": "applicability"},
59 "alignments": [{"record_id": record_id, "mapping_kind": "exact", "quote": definition}],
