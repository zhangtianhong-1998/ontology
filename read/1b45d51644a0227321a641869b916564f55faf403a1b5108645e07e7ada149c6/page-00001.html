31def build_metadata_graph(data, association=None, *, platform="postgres", environment="PROD"):
32 """Build a deterministic graph from Dataset plus already checked rules.
33
34 ``association`` is a ``{"rules": [...]}`` result or a rule list. This
35 function does not perform a join or judge meaning. Rejected/candidate rules
36 and rules from another snapshot are recorded in coverage, never as edges.
37 Existing local table/column IDs are preserved for extraction compatibility;
38 their ``datahub_urn`` values supply stable interchange identities.
39 """
40 # Validate the platform before constructing a partial graph.
41 dataset_urn("metadata_identity_check", platform, environment)
42 nodes, edges = {}, []
43
44 def add_node(item):
45 if item["id"] in nodes and nodes[item["id"]] != item:
46 raise ValueError("Conflicting metadata node identity: " + item["id"])
47 nodes[item["id"]] = item
48
49 def source_edge(subject, kind, path):
50 if path in data.files:
51 edges.append({"source": subject, "type": kind, "target": "source:" + path})
52
53 snapshot_node = "snapshot:" + data.snapshot_id
54 add_node({"id": snapshot_node, "kind": "DatasetSnapshot", "snapshot_id": data.snapshot_id})
55 for path, sha in sorted(data.files.items()):
56 add_node({"id": "source:" + path, "kind": "Source", "path": path, "sha256": sha})
57 edges.append({"source": snapshot_node, "type": "includes_source", "target": "source:" + path})
58 for name, table in sorted(data.tables.items()):
59 metadata_file = data.evidence["schema:" + name]["source_ref"]["file"]
60 record_type, classification = source_record_type(name, table)
61 try:
62 urn = dataset_urn(name, platform, environment)
63 except ValueError:
64 urn = None
65 add_node({"id": name, "kind": "Table",
66 **{key: table.get(key) for key in ("schema", "table_name", "table_comment", "relkind",
67 "estimated_rows", "total_size", "data_size")},
68 "observed_rows": table["rows"], "declared_primary_key": list(table["pk"]),
69 "source_record_type": record_type.id, "datahub_urn": urn,
70 "datahub_urn_scope": "offline_interchange" if urn else "unsupported_local_identifier",
71 "evidence_ids": ["schema:" + name]})
72 add_node({"id": record_type.id, "kind": "SourceRecordType", "category": record_type.category,
73 "label": record_type.label, "definition": record_type.definition,
74 "parent": record_type.parent, "evidence_ids": record_type.evidence_ids,
75 "classification_basis": classification["basis"], "business_concept_inferred": False,
76 "observed_rows": table["rows"], "mapping_origin": "local_deterministic_source_structure"})
77 edges.append({"source": name, "type": "mapped_as_source_record_type", "target": record_type.id,
78 "semantic_status": "source_structure_only"})
79 source_edge(name, "documented_by", metadata_file)
80 source_edge(name, "sample_from", str(Path(table["csv_path"]).relative_to(data.root)))
81 for constraint in table["constraints"]:
82 cid = name + ":constraint:" + digest([constraint.get("constraint_name"),
83 constraint.get("constraint_type"),
84 constraint.get("definition")])[:16]
85 add_node({"id": cid, "kind": "Constraint", **deepcopy(constraint)})
86 edges.append({"source": name, "type": "has_declared_constraint", "target": cid})
87 source_edge(cid, "documented_by", metadata_file.replace("schema/tables/", "schema/constraints/", 1))
88 if (constraint.get("constraint_type") == "p" or
89 str(constraint.get("constraint_type_name", "")).upper() == "PRIMARY KEY"):
90 for position, column in enumerate(table["pk"], start=1):
91 edges.append({"source": cid, "type": "declared_key_column", "target": name + "." + column,
92 "key_position": position})
93 for column in table["columns"]:
94 field = column["column_name"]
95 try:
96 field_urn = schema_field_urn(name, field, platform, environment)
97 except ValueError:
98 field_urn = None
99 cid = name + "." + field
100 add_node({"id": cid, "kind": "Column", **deepcopy(column), "table": name,
101 "datahub_urn": field_urn, "datahub_field_path": field,
102 "is_declared_primary_key": field in table["pk"],
103 "evidence_ids": ["schema:" + name + ":" + field]})
104 edges.append({"source": name, "type": "table_has_column", "target": cid})
105 for fk in table["foreign_keys"]:
106 source = name + "." + fk["column_name"]
107 target = fk["referenced_schema"] + "." + fk["referenced_table"] + "." + fk["referenced_column"]
108 edges.append({"id": "declared_fk:" + digest([source, target, fk])[:24],
109 "source": source, "type": "declared_fk", "target": target,
110 "raw": deepcopy(fk), "status": "declared",
111 "meaning": "declared_foreign_key_only", "lineage_inferred": False,
112 "source_ref": {"file": metadata_file.replace("schema/tables/", "schema/foreign_keys/", 1)}})
113
114 rules = association.get("rules", []) if isinstance(association, dict) else association or []
115 omitted = defaultdict(int)
116 link_ids = set()
117 for rule in rules:
118 if rule.get("status") not in CHECKED_LINK_STATUSES:
119 omitted["not_checked"] += 1
120 continue
121 if rule.get("snapshot_id") not in (None, data.snapshot_id):
122 omitted["different_snapshot"] += 1
123 continue
124 source, target = rule.get("source") or {}, rule.get("target") or {}
125 endpoints = [str(value.get("table", "")) + "." + str(value.get("field", ""))
126 for value in (source, target)]
127 if any(nodes.get(endpoint, {}).get("kind") != "Column" for endpoint in endpoints):
128 omitted["unknown_field"] += 1
129 continue
130 link_id = technical_link_id(rule)
131 if link_id in link_ids:
132 omitted["duplicate_link"] += 1
133 continue
134 link_ids.add(link_id)
135 evidence_ids = list(dict.fromkeys(rule.get("evidence_ids", []) + [
136 "schema:" + value["table"] + ":" + value["field"] for value in (source, target)]))
137 edges.append({"id": link_id, "type": "technical_link", "source": endpoints[0], "target": endpoints[1],
138 "rule_id": rule.get("rule_id"), "candidate_id": rule.get("candidate_id"),
139 "status": rule["status"], "snapshot_id": data.snapshot_id,
140 **{key: deepcopy(rule.get(key) or {}) for key in
141 ("selector", "scope_bindings", "transform", "verification")},
142 "evidence_ids": evidence_ids,
143 "numeric_overlap_only": bool(rule.get("numeric_overlap_only", False)),
144 "risk_flags": deepcopy(rule.get("risk_flags") or []),
145 "semantic_relation": rule.get("semantic_relation", "unresolved"),
146 "meaning": "checked_field_association_only", "lineage_inferred": False})
147 for edge in edges:
148 for endpoint in (edge["source"], edge["target"]):
149 if endpoint not in nodes:
150 add_node({"id": endpoint, "kind": "ExternalColumnReference", "availability": "not_in_input"})
151 evidence_ids = {eid for item in [*nodes.values(), *edges] for eid in item.get("evidence_ids", [])}
152 return {"schema_version": "0.2", "snapshot_id": data.snapshot_id,
153 "nodes": list(nodes.values()), "edges": edges,
154 "evidence": {eid: deepcopy(data.evidence[eid]) for eid in sorted(evidence_ids) if eid in data.evidence},
155 "graph_source": "local_schema_and_csv_snapshot", "lineage_source_available": False,
156 "association_coverage": {"input_rules": len(rules), "included_links": len(link_ids),
157 "omitted": dict(omitted)},
158 "datahub_interchange": {"format": "metadata-file", "default_platform": platform,
159 "default_environment": environment, "service_backed": False}}

173class MetadataGraph:
174 """Read-only query index for source neighborhoods; returns detached values."""
175
176 def __init__(self, graph):
177 self.graph = deepcopy(graph)
178 self.nodes = {node["id"]: node for node in self.graph.get("nodes", [])}
179 self.edges = self.graph.get("edges", [])
180 self.owners = {edge["target"]: edge["source"] for edge in self.edges
181 if edge.get("type") == "table_has_column"}
182 self._field_edges = [edge for edge in self.edges if edge.get("type") in FIELD_EDGE_TYPES]
183 self._field_edge_positions = defaultdict(list)
184 for position, edge in enumerate(self._field_edges):
185 for owner in {self.owners.get(edge["source"]), self.owners.get(edge["target"])} - {None}:
186 self._field_edge_positions[owner].append(position)
187
188 def table(self, name):
189 item = self.nodes.get(name)
190 return deepcopy(item) if item and item.get("kind") == "Table" else None
191
192 def columns(self, table):
193 return deepcopy(sorted((self.nodes[column] for column, owner in self.owners.items() if owner == table),
194 key=lambda item: (item.get("ordinal_position") or 0, item["id"])))
195
196 def field_links(self, table=None, column=None, *, statuses=None, edge_types=None):
197 """Return directed field edges, retaining every selector/transform branch."""
198 kinds = FIELD_EDGE_TYPES if edge_types is None else set(edge_types)
199 ids = ({table + "." + column} if table and column else
200 {key for key, owner in self.owners.items() if owner == table} if table else None)
201 return deepcopy([edge for edge in self.edges if edge.get("type") in kinds
202 and (ids is None or edge["source"] in ids or edge["target"] in ids)
203 and (statuses is None or edge.get("status") in statuses)])

205 def field_link_context(self, tables, fields, *, statuses=None, rule_id=None,
206 with_coverage=False):
207 """Project selected links before copying; keep audit references and counts.
208
209 Table adjacency is indexed once. A prompt never copies unrelated links
210 or potentially large counterexample arrays from full validation reports.
211 Link/rule IDs and snapshot IDs locate those complete reports on disk.
212 """
213 positions = {position for table in tables for position in self._field_edge_positions.get(table, ())}
214 selected, available = [], 0
215 keys = ("id", "rule_id", "candidate_id", "snapshot_id", "source", "target", "type", "status",
216 "selector", "scope_bindings", "transform", "evidence_ids", "source_ref",
217 "numeric_overlap_only", "risk_flags", "semantic_relation", "meaning", "lineage_inferred")
218 for position in sorted(positions):
219 edge = self._field_edges[position]
220 if (edge["source"] not in fields or edge["target"] not in fields
221 or (statuses is not None and edge.get("status") not in statuses)):
222 continue
223 available += 1
224 # A relation packet judges one rule, not every conditional branch
225 # sharing its fields. Filter before copying validation evidence.
226 if rule_id is not None and edge.get("rule_id") != rule_id:
227 continue
228 item = {key: deepcopy(edge[key]) for key in keys if key in edge}
229 verification = edge.get("verification") or {}
230 item["verification"] = {key: deepcopy(verification[key]) for key in
231 ("scan_scope", "checks", "error") if key in verification}
232 item["verification"]["counterexample_count"] = len(verification.get("counterexamples") or [])
233 item["evidence_sources"] = {eid: {key: deepcopy(evidence[key]) for key in ("origin", "source_ref")
234 if key in evidence}
235 for eid in item.get("evidence_ids", [])
236 if (evidence := self.graph.get("evidence", {}).get(eid)) is not None}
237 selected.append(item)
238 if with_coverage:
239 return selected, {"available_links": available, "included_links": len(selected),
240 "omitted_links": available - len(selected)}
241 return selected

243 def neighbors(self, table, *, edge_types=None, max_hops=1, statuses=None):
244 """Find adjacent input table IDs; declarations and checked links are separate filters."""
245 if type(max_hops) is not int or max_hops < 0:
246 raise ValueError("max_hops must be a nonnegative integer")
247 if self.table(table) is None:
248 raise ValueError("Unknown metadata table: " + table)
249 adjacent = defaultdict(set)
250 for edge in self.field_links(statuses=statuses, edge_types=edge_types):
251 source, target = self.owners.get(edge["source"]), self.owners.get(edge["target"])
252 if source and target:
253 adjacent[source].add(target)
254 adjacent[target].add(source)
255 seen, queue = {table}, deque([(table, 0)])
256 while queue:
257 current, depth = queue.popleft()
258 if depth == max_hops:
259 continue
260 for neighbor in sorted(adjacent[current] - seen):
261 seen.add(neighbor)
262 queue.append((neighbor, depth + 1))
263 return sorted(seen - {table})

265 def packet(self, table_names, *, include_sources=False):
266 """Return the complete selected-table schema/links/evidence, without CSV values.
267
268 No internal truncation is applied. The bundle builder must budget or
269 split this packet explicitly. Source files stay in provenance unless
270 ``include_sources`` is requested, avoiding a graph dominated by files.
271 """
272 if isinstance(table_names, str):
273 table_names = [table_names]
274 selected = set(table_names)
275 unknown = sorted(name for name in selected if self.table(name) is None)
276 if unknown:
277 raise ValueError("Unknown metadata tables: " + ", ".join(unknown))
278 ids = set(selected)
279 for edge in self.edges:
280 if edge["source"] in selected and edge.get("type") in (
281 "table_has_column", "has_declared_constraint", "mapped_as_source_record_type"):
282 ids.add(edge["target"])
283 provenance = [deepcopy(edge) for edge in self.edges if edge["source"] in ids
284 and edge.get("type") in ("documented_by", "sample_from")]
285 if include_sources:
286 ids.update(edge["target"] for edge in provenance)
287 edges = [deepcopy(edge) for edge in self.edges if edge["source"] in ids and edge["target"] in ids]
288 nodes = [deepcopy(self.nodes[node_id]) for node_id in sorted(ids)]
289 evidence_ids = {eid for item in nodes + edges for eid in item.get("evidence_ids", [])}
290 return {"snapshot_id": self.graph.get("snapshot_id"), "tables": sorted(selected),
291 "nodes": nodes, "edges": edges, "provenance": provenance,
292 "evidence": {eid: deepcopy(self.graph["evidence"][eid]) for eid in sorted(evidence_ids)
293 if eid in self.graph.get("evidence", {})},
294 "scope": "selected_source_metadata_and_checked_field_links", "truncated": False}
