58def test_second_pass_preserves_core_and_resolves_only_checked_source_gaps():
59 data, bundles, prior = accepted_state()
60 tasks = build_targeted_repair_tasks(data, bundles, prior)
61 original_type = prior["template_projections"][0]["object_type_id"]

63 class RepairLLM:
64 calls = 0
65
66 async def ask(self, task, payload, schema):
67 self.calls += 1
68 assert task == "concept_bundle"
69 assert payload["targeted_repair"]["preserve_type_id"] == original_type
70 assert any(item["label"] == "合格率" for item in payload["component_type_candidates"])
71 proposed = decision()
72 proposed.existing_type_id = original_type
73 proposed.slots[1].target_type_id = "type:region"
74 return ConceptBundleDecision(status="proposed", action="project_template", projection=proposed)

76 llm = RepairLLM()
