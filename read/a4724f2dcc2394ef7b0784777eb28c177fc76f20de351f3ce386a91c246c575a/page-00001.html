48def model_digest(directory):
49 """Hash local weights, tokenizer and model configuration without loading them in RAM."""
50 root = Path(directory)
51 files = sorted(p for p in root.rglob("*") if p.is_file() and
52 not any(part.startswith(".") for part in p.relative_to(root).parts) and
53 p.suffix in (".safetensors", ".bin", ".model", ".json", ".txt"))
54 if not files or not any(p.suffix in (".safetensors", ".bin") for p in files):
55 raise ValueError("Embedding model has no local weight files")
56 digest = hashlib.sha256()
57 for path in files:
58 digest.update(path.relative_to(root).as_posix().encode())
59 with path.open("rb") as stream:
60 for chunk in iter(lambda: stream.read(4 * 1024 * 1024), b""):
61 digest.update(chunk)
62 return digest.hexdigest()

65def top_cosine(vectors, query, limit, vector_norms=None):
66 """Exact cosine top-k with stable tie order; no FAISS dependency for small corpora."""
67 import numpy as np
68
69 matrix = np.asarray(vectors, dtype=np.float32)
70 if matrix.size == 0:
71 return []
72 needle = np.asarray(query, dtype=np.float32)
73 if matrix.ndim != 2 or needle.ndim != 1 or matrix.shape[1] != needle.shape[0]:
74 raise ValueError("Embedding dimensions do not match")
75 norms = (np.asarray(vector_norms, dtype=np.float32) if vector_norms is not None
76 else np.linalg.norm(matrix, axis=1)) * np.linalg.norm(needle)
77 if norms.shape != (len(matrix),):
78 raise ValueError("Embedding norm count does not match vectors")
79 scores = np.divide(matrix @ needle, norms, out=np.zeros(len(matrix), dtype=np.float32), where=norms > 0)
80 return [(int(i), float(scores[i])) for i in np.argsort(-scores, kind="stable")[:limit]]

122 def _encode(self, texts, *, query=False):
123 import numpy as np
