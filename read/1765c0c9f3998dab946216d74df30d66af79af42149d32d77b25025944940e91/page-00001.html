55def test_nested_json_with_different_key_names_forms_replayable_lexical_member_path(tmp_path):
56 data, plan, concepts, alignments = fixture(tmp_path)
57 try:
58 result = build_semantic_references(data, plan, concepts, alignments)
59 assert len(result["paths"]) == 1
60 assert result["coverage"]["unique_member_instances"] == 1
61 assert result["coverage"]["llm_calls"] == 0
62 assert not result["pending"]
63 assert not plan.relation_types # The accepted caller core is immutable.
64 relation = result["plan"].relation_types[0]
65 assert (relation.domain, relation.range, relation.predicate_name) == (["type:Term"], ["type:Dimension"], "term_dimension_context")
66 nodes = {node["id"]: node for node in result["instances"]}
67 assert {edge["predicate_name"] for edge in result["assertions"]} == {"denotes", "member_of"}
68 lexical = next(node for node in nodes.values() if node["instance_kind"] == "lexical_term")
69 member = next(node for node in nodes.values() if node["instance_kind"] == "dimension_member")
70 assert lexical["ontology_type_id"] == "type:Term"
71 assert member["ontology_type_id"] == "type:Dimension"
72 assert member["type_role"] == "member_of_dimension"
73 candidate = result["json_reference_candidates"][0]
74 assert candidate["source"]["source_path"] == ["link", "personnel_key"]
75 assert candidate["target"]["field"] == "local_key"
76 sql, params = association_match_sql(data, candidate)
77 assert data.db.execute(sql, params).fetchall() == [(1, 1)]
78 assert result["checked_reference_rules"][0]["checks"]["unique_matches"] == 1
79 assert result["reference_plans"][0]["source_path"] == ["link", "personnel_key"]
80 # Reloaded evidence, rather than an ephemeral stage result, validates
81 # the Term→Dimension summary without erasing the member intermediary.
82 data.evidence = json.loads(json.dumps(data.evidence))
83 verify_semantic_reference_relation(data, relation, result["plan"])
84 for edge in result["assertions"]:
85 verify_semantic_reference_assertion(data, edge, nodes, result["plan"])
86 finally:
87 data.close()
