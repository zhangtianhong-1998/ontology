13def test_fixed_formula_unit_fields_remain_exact_invariants_and_parameters():
14 data, bundle = setup()
15 proposed = decision()
16 proposed.field_templates.extend([
17 ProjectionField(column="x10", template="passed / inspected"),
18 ProjectionField(column="x11", template="%")])
19 proposed.definition_parameters = {"formula": "passed / inspected", "unit": "%"}
20 _, template, bindings = compile_projection(data, PROFILE, BuildPlan(), bundle, proposed)
21 assert len(bindings) == 2
22 assert template["invariants"]["x10"] == "passed / inspected"
23 assert template["invariants"]["x11"] == "%"
24 changed = deepcopy(bundle)
25 changed["records"][1]["fields"]["formula"][0]["value"] = "failed / inspected"
26 with pytest.raises(ValueError, match="Every witness must match"):
27 compile_projection(type(data)(snapshot_id="changed", evidence={}, tables={}), PROFILE,
28 BuildPlan(), changed, proposed)
29 assert reuse_projection(data, [template], {"records": [record("row3", formula="failed / inspected")]}) is None
30 proposed.field_templates[-2] = ProjectionField(column="x10", template="{product}")
31 with pytest.raises(ValueError, match="cannot be wildcarded"):
32 compile_projection(data, PROFILE, BuildPlan(), bundle, proposed)

46def test_slot_context_quote_must_support_its_actual_matching_path():
47 data, bundle = setup()
48 proposed = decision()
49 proposed.slots[0].evidence = ProjectionQuote.model_validate(quote("x3", "苹果合格率按华东汇总"))
50 _, template, bindings = compile_projection(data, PROFILE, BuildPlan(), bundle, proposed)
51 product = next(slot for slot in template["slots"] if slot["name"] == "product")

54 assert bindings[0]["slot_values"]["product"] == "苹果"

62def test_anchored_fields_bind_adjacent_slots_before_matching_title():
63 data, bundle = setup()
64 proposed = decision()
65 for rec in bundle["records"]:
66 fruit = next(e["value"] for e in rec["fields"]["scope"] if e["column"] == "x8")
67 region = next(e["value"] for e in rec["fields"]["scope"] if e["column"] == "x9")
68 rec["fields"]["name"][0]["value"] = region + fruit + "合格率"
69 proposed.slots[0].source_column = None
70 proposed.slots[1].source_column = None
71 proposed.field_templates[0] = ProjectionField(column="x1", template="{region}{product}合格率")
72 _, _, bindings = compile_projection(data, PROFILE, BuildPlan(), bundle, proposed)
73 assert bindings[1]["slot_values"]["product"] == "橙子"
74 # With no anchored field and no source-column binding, ambiguity remains.
75 broken = decision()
76 broken.slots = broken.slots[:2]
77 broken.components = broken.components[:1]
78 for slot in broken.slots:
79 slot.source_column = None
80 broken.field_templates = [ProjectionField(column="x1", template="{region}{product}合格率")]
81 broken.witness_record_ids = ["row1"]
82 with pytest.raises(ValueError, match="Adjacent unbound"):
83 compile_projection(data, PROFILE, BuildPlan(), bundle, broken)
