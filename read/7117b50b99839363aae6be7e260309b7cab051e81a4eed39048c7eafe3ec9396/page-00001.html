5def classification_basis(data, record, column, value):
6 """Require complete source content and an unambiguous schema declaration.
7
8 Column names, model labels and category-like values do not establish this
9 role. Namespace/version, calculation and restrictive scope remain strict.
10 """
11 fields = record.get("fields", {})
12 if any(entry.get("column") == column for role in ("identity", "formula", "unit")
13 for entry in fields.get(role, [])):
14 return None
15 if not str(value).strip() or not any(
16 entry.get("column") == column and not entry.get("truncated")
17 and str(entry.get("value")) == str(value) for entry in fields.get("scope", [])):
18 return None
19 metadata = next((item for item in getattr(data, "tables", {}).get(record["table"], {}).get("columns", [])
20 if item.get("column_name") == column), {})
21 declaration = str(metadata.get("column_comment") or "")
22 if (not re.search(r"分类|类别|\b(?:category|classification)\b", declaration, re.I)
23 or re.search(r"租户|命名空间|版本|粒度|计算|聚合|统计期间|适用范围|筛选|过滤|"
24 r"\b(?:tenant|namespace|version|granularity|calculation|aggregation|filter|restriction)\b",
25 declaration, re.I)):
26 return None
27 return {"basis": "source_column_classification_declaration", "record_id": record["record_id"],
28 "table": record["table"], "column": column, "value": str(value),
29 "declaration": declaration, "schema_evidence_id": f"schema:{record['table']}:{column}"}

32def source_classifications(data, records):
33 """Return only classification fields shared by every complete witness."""
34 if not records:
35 return {}, {}
36 candidates = {entry["column"]: str(entry["value"])
