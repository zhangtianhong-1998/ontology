71def _unit_key(value):
72 """Normalize literal unit spellings, never scales or observation values.
73
74 SI symbol case matters: MW and mW must not become the same unit. Chinese
75 denominator syntax and same-scale names such as 千克/kg are equivalent.
76 """
77 text = "".join(unicodedata.normalize("NFKC", str(value or "")).split()).replace("每", "/")
78 aliases = {"千克": "kg", "公斤": "kg", "克": "g", "吨": "吨",
79 "ton": "吨", "tons": "吨", "tonne": "吨", "tonnes": "吨",
80 "千瓦时": "kWh", "千瓦·时": "kWh", "千瓦": "kW", "瓦": "W",
81 "立方米": "m3", "平方米": "m2", "摄氏度": "°C"}
82 if text in aliases:
83 return aliases[text]
84 return "/".join(aliases.get(atom, atom) for atom in text.split("/"))

87def _unit_from_comment(comment):
88 """Read literal declared units; retain conflicting declarations separately."""
89 text = str(comment or "")
90 units = {_unit_key(match.group(1)) for match in _UNIT_MARKER.finditer(text)}
91 compact = [_unit_key(match.group(1)) for match in _COMPACT_UNIT_MARKER.finditer(text)]
92 units.update(value for value in compact
93 if all(_UNIT_TOKEN.fullmatch(atom) for atom in re.split(r"[/·*]", value)))
94 # A whole measurement token delimited by punctuation is a declaration
95 # (e.g. "power, kWh"). Prose, identifiers and arbitrary parentheses are not.
96 for fragment in re.split(r"[，,。；;、（）()\[\]【】]", text):
97 token = _unit_key(fragment)
98 atoms = re.split(r"[/·*]", token)
99 if atoms and all(_UNIT_TOKEN.fullmatch(atom) for atom in atoms):
100 units.add(token)
101 return {value for value in units if _norm(value) not in _MISSING_UNIT}

176def _terms(value):
177 words = set(_WORD.findall(_norm(value).replace("_", " ")))
178 grams = set()
179 for word in words:
180 if _HAN.search(word):
181 grams.update(word[i:i + 2] for i in range(len(word) - 1))
182 return words | grams

185def _source_names(item, data):
186 """Use accepted name/alias cells, never formula operands as quantity names."""
187 names = [{"value": item.label or "", "basis": "accepted_type_label", "evidence_ids": []}]
188 for source in item.source_properties:
189 if source.role not in ("name", "alias"):
190 continue
191 for eid in source.evidence_ids:
192 evidence = data.evidence.get(eid) or {}
193 ref = evidence.get("source_ref") or {}
194 raw = evidence.get("raw_fragment")
195 if (eid not in item.evidence_ids or evidence.get("origin") != "observed_record"
196 or evidence.get("raw_fragment_truncated")
197 or ref.get("snapshot_id") != data.snapshot_id
198 or ref.get("table") != source.source_table or ref.get("column") != source.source_column
199 or not ref.get("record_id") or not isinstance(raw, str) or len(raw) > 2048):
200 continue
201 # A list of explicitly declared aliases supplies separate surfaces;
202 # splitting a formula or arbitrary description would be unsafe.
203 for value in re.split(r"[;,；，、|\n]", raw) if source.role == "alias" else [raw]:
204 if value.strip():
205 names.append({"value": value.strip(), "basis": "source_" + source.role,
206 "source_table": source.source_table,
207 "source_column": source.source_column, "evidence_ids": [eid]})
208 return names

211def _name_in_declaration(name, declaration):
212 normalized, text = _norm(name), _norm(declaration)
213 if len(normalized) < 2:
214 return False
215 if _HAN.search(normalized):
216 return normalized in text
217 # Preserve word boundaries while allowing declared identifier separators.
218 # "load_kw" matches "load_kw_hourly"; "gas" does not match "gasoline".
219 name = re.sub(r"[_\W]+", " ", unicodedata.normalize("NFKC", str(name)).casefold()).strip()
220 text = re.sub(r"[_\W]+", " ", unicodedata.normalize("NFKC", str(declaration)).casefold()).strip()
221 return bool(name and re.search(r"(?<!\w)" + r"\s*".join(map(re.escape, name.split())) + r"(?!\w)", text))

229def _recall_types(data, core, table, column, *, max_candidates, max_definition_chars):
230 by_id = {item.id: item for item in core.object_types}
231 comment = str(column.get("column_comment") or "").strip()
232 source = f"{column['column_name']} {comment}"
233 source_terms = _terms(source)
234 found = []
235 complete_name_ids = set()
236 excluded = Counter()
237 for item in core.object_types:
238 # A measured quantity need not be a composed business Metric. Its
239 # observation keeps the source coordinates; binding a Measure never
240 # promotes it to a Metric or invents an object identity. An operator
241 # annotation describes its definition; it never runs an aggregation.
242 if (item.category != "business_type" or item.derivation_kind == "shared_supertype"
243 or _root_of(item, by_id) not in {"Metric", "Measure"}
244 or not item.label or not item.definition):
245 continue
246 if item.evidence_scope == "source_schema" and not any(
247 source.source_table == table["name"] and source.source_column == column["column_name"]
248 for source in item.source_properties):
249 # A field declaration is scoped to its own source, not a cross-table
250 # identity assertion between equally named columns.
251 continue
252 label = _norm(item.label)
253 if not label:
254 continue
255 names = _source_names(item, data)
256 exact_name = any(_name_in_declaration(entry["value"], source) for entry in names)
257 overlap = source_terms & set().union(*(_terms(entry["value"]) for entry in names))
258 score = (100 if exact_name or item.evidence_scope == "source_schema" else 0) + 2 * len(overlap)
259 if score == 0:
260 continue
261 evidence, invalid_reason = _source_definition_evidence(
262 item, data, max_chars=max_definition_chars)
263 if invalid_reason:
264 # An exact-label peer with hidden or conflicting source meanings
265 # could change the identity decision. Do not silently ignore it.
266 if exact_name:
267 excluded[invalid_reason] += 1
268 continue
269 found.append((score, item, evidence))
270 if exact_name or item.evidence_scope == "source_schema":
271 complete_name_ids.add(item.id)
272 found.sort(key=lambda entry: (-entry[0], entry[1].id))
273 if complete_name_ids:
274 # Only a complete declared name can pass the binding compiler. Keep
275 # every such competitor; fragment-only leads cannot displace one or
276 # force a false candidate-cap failure after a source alias matched.
277 found = [entry for entry in found if entry[1].id in complete_name_ids]
278 return found[:max_candidates], max(0, len(found) - max_candidates), dict(excluded)

281def _scope_consistent(item, table, column_comment, coordinates, scope_bindings=None,
282 constant_scope_bindings=None, scope_constants=()):
283 scope_bindings = scope_bindings or {}
284 constants = {entry['column']: entry['value'] for entry in scope_constants}
285 fixed = constant_scope_bindings or {}
286 # Tenant/version identity qualifiers must bind to the observed coordinates
287 # just as applicability scope does; matching labels cannot cross them.
288 identity = getattr(item, "identity_qualifiers", {})
289 if any(key in item.applicability_scope and item.applicability_scope[key] != value
290 for key, value in identity.items()):
291 return False
292 for key, value in {**item.applicability_scope, **identity}.items():
293 if key in fixed:
294 if fixed[key] not in constants or str(constants[fixed[key]]) != str(value):
295 return False
296 continue
297 field = scope_bindings.get(key, key)
298 if field in coordinates:
299 if str(coordinates[field]) != str(value):
300 return False
301 else:
302 # Text mentioning a place is not a binding to the row's coordinate.
303 # Differently named coordinates require an explicit field mapping.
304 return False
305 return True

314def field_scope_mapping_candidates(data, table, item, coordinate_columns=()):
315 """Propose mappings from source declarations, never from value overlap.
316
317 These are names of existing columns. Constancy and coordinate values are
318 independently verified; a missing domain/namespace cannot be synthesized.
319 """
320 result = []
321 for key, expected in {**item.applicability_scope, **item.identity_qualifiers}.items():
322 declarations = []
323 for source in item.source_properties:
324 if source.role not in ("scope", "identity") or source.source_column != key:
325 continue
326 definition_table = data.tables.get(source.source_table, {})
327 declaration = next((entry for entry in definition_table.get("columns", [])
328 if entry["column_name"] == key), {})
329 if declaration.get("column_comment"):
330 declarations.append({"source_table": source.source_table, "source_column": key,
331 "column_comment": declaration["column_comment"],
332 "evidence_id": f"schema:{source.source_table}:{key}"})
333 for column in table["columns"]:
334 name, comment = column["column_name"], column.get("column_comment") or ""
335 if is_sensitive_column(column) or name in table.get("semantic_excluded_columns", ()):
336 continue
337 basis = None
338 if name == key:
339 basis = "same_declared_column_name"
340 elif _scope_surface(name) == _scope_surface(key):
341 basis = "same_scope_identifier_stem"
342 elif comment and any(_scope_surface(comment) == _scope_surface(value['column_comment'])
343 for value in declarations):
344 basis = "same_source_scope_declaration"
345 if basis:
346 result.append({"scope_key": key, "column": name, "expected_value": expected,
347 "kind": "coordinate" if name in coordinate_columns else "requires_constant_check",
348 "basis": basis, "column_comment": comment,
349 "definition_scope_declarations": declarations,
350 "evidence_ids": [f"schema:{table['name']}:{name}",
351 *(value['evidence_id'] for value in declarations)]})
352 return result

392def field_unit_evidence(data, table, column):
393 """Read a declared/source-checked unit column over the current snapshot.
394
395 The returned singleton is usable only when every source row has the same
396 nonblank unit. A bounded sample cannot certify units for a whole field.
397 """
398 metadata = next((item for item in table["columns"] if item["column_name"] == column), None)
399 if (metadata is None or is_sensitive_column(metadata)
400 or column in table.get("semantic_excluded_columns", ())):
401 return None
402 role_supported = any(item.get("column") == column and item.get("role") == "unit"
403 and item.get("status") == "source_verified_role_candidate"
404 for item in table.get("inferred_semantic_roles", []))
405 declared_unit_column = (re.search(r"(?:^|[_\s])(?:unit|uom)(?:[_\s]|$)", column, re.I)
406 or re.fullmatch(r"(?:计量|度量|测量|数值)?单位(?:名称|编码|字段)?",
407 str(metadata.get("column_comment") or "").strip()))
408 if not role_supported and not declared_unit_column:
409 return None
410 sql = f"SELECT {qi(column)}, count(*), min(__r2_row) FROM {qi(table['sql_name'])} GROUP BY {qi(column)} ORDER BY {qi(column)} LIMIT 3"
411 values = data.db.execute(sql).fetchall()
412 if len(values) != 1 or values[0][0] is None or not str(values[0][0]).strip():
413 return {"column": column, "status": "mixed_or_missing_units"}
414 raw, count, row_number = values[0]
415 if count != table["rows"] or len(str(raw)) > 64:
416 return {"column": column, "status": "incomplete_unit_column"}
417 evidence_id = "unit_observation:" + digest([data.snapshot_id, table["name"], column, row_number])[:24]
418 data.evidence[evidence_id] = {
419 "id": evidence_id, "origin": "observed_record", "raw_fragment": str(raw),
420 "raw_fragment_truncated": False,
421 "source_ref": {"table": table["name"], "column": column, "row": row_number,
422 "snapshot_id": data.snapshot_id},
423 "verification": {"scan_scope": "full_input", "same_nonblank_unit_rows": count},
424 }
425 return {"column": column, "status": "verified_single_unit", "value": str(raw),
426 "evidence_id": evidence_id}
