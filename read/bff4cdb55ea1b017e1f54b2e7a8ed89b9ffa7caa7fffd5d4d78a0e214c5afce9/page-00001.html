15def main():

20 parser.add_argument("--kind", choices=["identifiers", "text", "import"], default="identifiers")

22 root = Path(args.output).resolve()
23 root.mkdir(parents=True, exist_ok=False)

25 make_demo(dataset, args.rows, "linked")
26 project = Path(__file__).resolve().parents[1]
27 config = load_config(project / "config/runtime.mock.yaml")
28 config.update(dataset=str(dataset), memory_limit="512MB", data_scope="complete_export")
29 config["llm"].update(responses=str(dataset / "mock_llm.yaml"), max_calls=30)
30 config["mcp"].update(command=sys.executable, args=["-m", "ontology_r2.mock_mcp", "--documents", str(dataset / "mock_documents.yaml")])
31 config["processing"].update(materialize_all_objects=args.rows <= args.record_cap, max_relation_records=args.record_cap)
32 if args.kind == "text":
33 responses = read_yaml(config["llm"]["responses"])
34 for task in ("plan", "final_plan"):
35 responses[task]["relations"][0].update(mode="text", source_column="description", evidence_ids=["schema:demo.records:description"])
36 write_yaml(config["llm"]["responses"], responses)
37 start = time.monotonic()
38 if args.kind == "import":
39 work = root / "import"
40 work.mkdir()

42 result = {"input_records": sum(t["rows"] for t in data.tables.values()), "llm_calls": 0, "stage": "import_and_profiles_only"}
43 data.close()
44 else:
45 result = asyncio.run(build(config, root / "run"))
46 result["extraction"] = read_yaml(root / "run/metrics.yaml").get("extraction") if (root / "run/metrics.yaml").exists() else None
47 report = {"synthetic": True, "scenario": args.kind, "requested_records": args.rows, "record_cap": args.record_cap, "elapsed_seconds": round(time.monotonic() - start, 3), "process_peak_rss_raw": resource.getrusage(resource.RUSAGE_SELF).ru_maxrss, "rss_unit": "bytes" if sys.platform == "darwin" else "KiB", "platform": sys.platform, "result": result, "semantic_validation": "mock responses only; not a real LLM quality measurement"}
48 write_yaml(root / "benchmark.yaml", report)
49 print(json.dumps({k: v for k, v in report.items() if k != "result"} | {"status": result.get("status", "import_complete"), "llm": result.get("llm"), "extraction": result.get("extraction")}, ensure_ascii=False))
