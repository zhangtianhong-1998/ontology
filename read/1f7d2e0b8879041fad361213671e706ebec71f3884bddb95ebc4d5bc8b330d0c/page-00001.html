1"""Read-only coverage of indexed definition source rows across binding paths.
2
3Coverage counts existing accepted artifacts; it does not rejudge their semantics
4or certify the source classifier. A template-only denominator is reported beside
5the full definition-card inventory, never substituted for it.
6"""
7
8from collections import Counter
9from contextlib import closing
10import json
11from pathlib import Path
12import sqlite3
13
14
15def _mapping(item):
16 return item.model_dump() if hasattr(item, "model_dump") else item
17
18
19def summarize_definition_coverage(index, group, *, definition_memberships=(),
20 template_bindings=None, template_projections=None,
21 plan=None, data=None):
22 """Return per-table/global unique-(table,row) coverage without source writes.
23
24 ``index`` is a SemanticCardIndex or its SQLite path; ``group`` supplies
25 concepts/record_alignments and, by default, template bindings/projections.
26 Iterables may stream memberships and bindings. Pass the final BuildPlan (or
27 its dict) to exclude types no longer accepted. ``data`` optionally verifies
28 the snapshot; no source values, models, or network calls are needed.
29
30 Missing template scope is not a failure: an accepted exact alignment or
31 membership can cover the same row. Counts measure binding presence only;
32 reference variants and inherited business relations remain unverified here.
33 """
34 path = Path(getattr(index, "path", index)).resolve()
35 bindings = group.get("template_bindings", ()) if template_bindings is None else template_bindings
36 projections = group.get("template_projections", ()) if template_projections is None else template_projections
37 concepts = {c["id"]: c for c in group.get("concepts", ())}
38 if plan is not None:
39 accepted_types = {t["id"] for value in _mapping(plan).get("object_types", ())
40 if (t := _mapping(value)).get("category") == "business_type"}
41 else:
42 accepted_types = None
43
44 # Temporary disk-backed SQL state bounds Python memory. ATTACH uses mode=ro
45 # even when the caller passes a live index object; its connection is untouched.
46 with closing(sqlite3.connect("", uri=True)) as db:
47 try:
48 db.execute("ATTACH DATABASE ? AS source", (path.as_uri() + "?mode=ro",))
49 meta = db.execute("SELECT value FROM source.metadata WHERE key='coverage'").fetchone()
50 if meta is None:
51 raise ValueError("Definition coverage requires completed index metadata")
52 source_coverage = json.loads(meta[0])
53 snapshot = source_coverage.get("snapshot_id")
54 for supplied in (group.get("snapshot_id"), getattr(data, "snapshot_id", None)):
55 if supplied is not None and supplied != snapshot:
56 raise ValueError("Definition coverage snapshot differs from its source index")
57 if not snapshot:
58 raise ValueError("Definition coverage index has no snapshot")
59 if db.execute("SELECT 1 FROM source.cards WHERE snapshot_id<>? LIMIT 1", (snapshot,)).fetchone():
60 raise ValueError("Definition coverage index mixes snapshots")
61 db.executescript("""
62 CREATE TABLE inventory (
63 table_name TEXT NOT NULL, row_number INTEGER NOT NULL,
64 channels INTEGER NOT NULL DEFAULT 0,
65 PRIMARY KEY(table_name,row_number));
66 INSERT OR IGNORE INTO inventory(table_name,row_number)
67 SELECT c.table_name,s.row_number FROM source.card_sources s
68 JOIN source.cards c ON c.card_id=s.card_id WHERE c.kind='definition';
69 CREATE TABLE template_scope(table_name TEXT PRIMARY KEY);
70 """)
71 rejected, artifacts = Counter(), Counter()
72
73 def allowed_type(type_id):
74 return bool(type_id) and (accepted_types is None or type_id in accepted_types)
75
76 def valid_snapshot(item):
77 return item.get("snapshot_id", snapshot) == snapshot
78
79 for projection in projections:
80 if (projection.get("status") == "accepted" and valid_snapshot(projection)
81 and allowed_type(projection.get("object_type_id"))
82 and projection.get("source_table")):
83 db.execute("INSERT OR IGNORE INTO template_scope VALUES (?)", (projection["source_table"],))
84 else:
85 rejected["template_scope:not_accepted_or_current_type"] += 1
86
87 def mark(item, channel, type_id):
88 artifacts[channel + "_seen"] += 1
89 if not allowed_type(type_id):
90 rejected[channel + ":no_accepted_business_type"] += 1
91 return
92 if not valid_snapshot(item):
93 rejected[channel + ":snapshot_mismatch"] += 1
94 return
95 record_id = item.get("source_record_id", item.get("record_id"))
96 if not record_id:
97 rejected[channel + ":missing_record_id"] += 1
98 return
99 clauses, parameters = ["s.record_id=?", "c.kind='definition'"], [record_id]
100 for column, value in (("c.card_id", item.get("source_card_id", item.get("card_id"))),
101 ("c.table_name", item.get("source_table", item.get("table"))),
102 ("s.row_number", item.get("row_number"))):
103 if value is not None:
104 clauses.append(column + "=?")
105 parameters.append(value)
106 rows = db.execute("SELECT DISTINCT c.table_name,s.row_number FROM source.card_sources s "
107 "JOIN source.cards c ON c.card_id=s.card_id WHERE " + " AND ".join(clauses)
108 + " LIMIT 2", parameters).fetchall()
109 if len(rows) != 1:
110 reason = "ambiguous_source_row" if rows else "source_row_not_in_definition_index"
111 rejected[channel + ":" + reason] += 1
112 return
113 bit = {"exact": 1, "membership": 2, "template": 4}[channel]
114 db.execute("UPDATE inventory SET channels=channels|? WHERE table_name=? AND row_number=?",
115 (bit, *rows[0]))
116 artifacts[channel + "_accepted"] += 1
117
118 for alignment in group.get("record_alignments", ()):
119 concept = concepts.get(alignment.get("concept_id"), {})
120 if (alignment.get("mapping_kind") != "exact"
121 or alignment.get("decision") not in {"accepted", "accepted_by_automatic_checks"}
122 or concept.get("ontology_level") != "type"
123 or concept.get("decision") not in {"accepted", "accepted_by_automatic_checks"}):
124 rejected["exact:not_accepted_type_alignment"] += 1
125 continue
126 mark(alignment, "exact", concept.get("ontology_type_id"))
127 for membership in definition_memberships:
128 if (membership.get("mapping_kind") != "shares_definition_type_template"
129 or membership.get("status") != "definition_template_match"):
130 rejected["membership:not_accepted"] += 1
131 continue
132 mark(membership, "membership", membership.get("type_id"))
133 for binding in bindings:
134 if binding.get("status") != "accepted" or binding.get("mapping_kind") != "template_instance":
135 rejected["template:not_accepted"] += 1
136 continue
137 mark(binding, "template", binding.get("object_type_id"))
138
139 by_table = {}
140 # Channel counts overlap; union is counted once per source row.
141 for row in db.execute("""
142 SELECT i.table_name, count(*), sum(channels&1 != 0), sum(channels&2 != 0),
143 sum(channels&4 != 0), sum(channels != 0), sum(channels = 0),
144 sum(channels IN (3,5,6,7)), t.table_name IS NOT NULL
145 FROM inventory i LEFT JOIN template_scope t ON t.table_name=i.table_name
146 GROUP BY i.table_name ORDER BY i.table_name
147 """):
148 table, total, exact, membership, template, bound, unbound, overlap, in_scope = row
149 by_table[table] = {
150 "indexed_definition_rows": total, "exact_alignment_rows": exact,
151 "membership_rows": membership, "template_binding_rows": template,
152 "rows_with_type_binding": bound, "rows_without_type_binding": unbound,
153 "rows_with_multiple_binding_paths": overlap,
154 "type_binding_fraction": bound / total if total else None,
155 "in_template_scope": bool(in_scope),
156 "rows_outside_template_scope": 0 if in_scope else total,
157 "bound_rows_outside_template_scope": 0 if in_scope else bound,
158 "template_scope_rows_without_type_binding": unbound if in_scope else 0,
159 }
160 count_fields = [key for key in next(iter(by_table.values()), {})
161 if key not in {"type_binding_fraction", "in_template_scope"}]
162 totals = {key: sum(report[key] for report in by_table.values()) for key in count_fields}
163 # Stable zero-valued fields even for an empty index.
164 for key in ("indexed_definition_rows", "exact_alignment_rows", "membership_rows",
165 "template_binding_rows", "rows_with_type_binding", "rows_without_type_binding",
166 "rows_with_multiple_binding_paths", "rows_outside_template_scope",
167 "bound_rows_outside_template_scope", "template_scope_rows_without_type_binding"):
168 totals.setdefault(key, 0)
169 scoped = totals["indexed_definition_rows"] - totals["rows_outside_template_scope"]
170 scoped_template = sum(r["template_binding_rows"] for r in by_table.values() if r["in_template_scope"])
171 totals.update({
172 "definition_tables": len(by_table),
173 "template_scope_definition_tables": sum(r["in_template_scope"] for r in by_table.values()),
174 "template_scope_rows": scoped,
175 "type_binding_fraction": (totals["rows_with_type_binding"] / totals["indexed_definition_rows"]
176 if totals["indexed_definition_rows"] else None),
177 "template_binding_fraction_within_template_scope": scoped_template / scoped if scoped else None,
178 })
179 scope_tables = [r[0] for r in db.execute("SELECT table_name FROM template_scope ORDER BY table_name")]
180 return {
181 "scope": "unique_source_table_and_row_in_definition_card_index",
182 "snapshot_id": snapshot, "llm_calls": 0,
183 "type_acceptance_basis": "final_plan_business_types" if plan is not None else "supplied_accepted_artifacts",
184 "semantic_validation": "not_rejudged; binding_presence_only; no_reference_or_relation_completeness_claim",
185 "source_index_partial": bool(source_coverage.get("partial")),
186 "source_index_rows_omitted_cap": source_coverage.get("rows_omitted_cap", 0),
187 "source_index_rows_without_card": source_coverage.get("rows_without_card", 0),
188 "definition_classification_scope": "index_definition_cards_only; excludes_reference_uncertain_and_business_fact_rows",
189 "totals": totals, "by_table": by_table,
190 "template_scope_tables": scope_tables,
191 "template_scope_tables_without_indexed_definitions": sorted(set(scope_tables) - set(by_table)),
192 "tables_outside_template_scope": [t for t, r in by_table.items() if not r["in_template_scope"]],
193 "artifact_counts": dict(artifacts), "excluded_artifact_counts": dict(rejected),
194 }
195 finally:
196 # A sqlite connection context manager commits/rolls back, but does
197 # not close it; close explicitly so its temporary database vanishes.
198 db.close()
