102def test_config_row_is_only_evidence_and_two_endpoints_are_exact(tmp_path):
103 data, plan, concepts, alignments, spec = _input(tmp_path)
104 try:
105 result = discover_configuration_relations(data, plan, concepts, alignments, [spec])
106 by_codes = {(item["source_code"], item["target_code"]): item
107 for item in result["candidates"]}
108 assert result["coverage"]["specs"][0]["input_rows"] == 6
109 assert result["coverage"]["specs"][0]["distinct_code_pairs"] == 5
110 accepted = by_codes[("I1", "M1")]
111 assert accepted["status"] == "endpoint_verified_candidate"
112 assert accepted["source_type_id"] == "type:profit"
113 assert accepted["target_type_id"] == "type:revenue"
114 assert accepted["configuration_rows_with_same_code_pair"] == 2
115 assert accepted["configuration_record_id"] not in {
116 accepted["source_definition"]["record_id"],
117 accepted["target_definition"]["record_id"],
118 }
119 assert accepted["predicate_status"] == "unjudged"
120 assert accepted["predicate_literal"] == "经营利润依赖收入"
121 assert accepted["semantic_status"] == "code_references_do_not_prove_relation_meaning"
122 assert any(data.evidence[e]["source_ref"].get("column") == "metric_ref"
123 for e in accepted["evidence_ids"] if e in data.evidence)
124 assert by_codes[("I1", "M2")]["target_definition"]["status"] == (
125 "definition_without_exact_accepted_type")
126 assert by_codes[("I1", "M_DUP")]["target_definition"]["status"] == (
127 "ambiguous_definition")
128 assert by_codes[("I_MISSING", "M1")]["source_definition"]["status"] == (
129 "missing_definition")
130 assert by_codes[(None, "M1")]["source_definition"]["status"] == "blank_code"
131 assert all(item["source_type_id"] is None and item["target_type_id"] is None
132 for item in by_codes.values() if item["status"] == "unresolved")
133 finally:
134 data.close()

186def test_infer_spec_from_two_checked_rules_and_keep_direction_unjudged(tmp_path):
187 data, plan, concepts, alignments, spec = _input(tmp_path)
188 try:
189 rules = [
190 _rule("a-metric", spec, "metric_ref", spec["source_definition_table"],
191 "metric_code"),
192 _rule("b-measure", spec, "measure_ref", spec["target_definition_table"],
193 "measure_code"),
194 ]
195 inferred = infer_configuration_specs(data, plan, concepts, alignments, rules)
196 assert inferred["coverage"]["rules_inspected"] == 2
197 assert inferred["coverage"]["specs_emitted"] == 1
198 item = inferred["spec_candidates"][0]
199 assert item["rule_ids"] == ["a-metric", "b-measure"]
200 assert item["direction_status"] == "unjudged"
201 assert item["spec"]["relation_text_column"] == "relation_text"
202 assert item["semantic_status"] == (
203 "two_verified_reference_rules_are_not_a_business_predicate")
204 resolved = discover_configuration_relations(
205 data, plan, concepts, alignments, [item["spec"]])
206 verified = next(x for x in resolved["candidates"]
207 if x["status"] == "endpoint_verified_candidate")
208 assert verified["direction_status"] == verified["predicate_status"] == "unjudged"
209
210 # Same-root references can be relationship candidates when a source
211 # statement exists; this does not repair an invalid endpoint lookup.
212 same_root = [rules[0], _rule(
213 "c-metric-alias", spec, "measure_ref", spec["source_definition_table"],
214 "metric_code")]
215 blocked = infer_configuration_specs(data, plan, concepts, alignments, same_root)
216 assert len(blocked["spec_candidates"]) == 1
217 unresolved = discover_configuration_relations(data, plan, concepts, alignments,
218 [blocked["spec_candidates"][0]["spec"]])
219 assert unresolved["coverage"]["endpoint_verified_candidates"] == 0
220
221 conditioned = [{**rules[0], "selector": {"business_type": "API"}}, rules[1]]
222 rejected = infer_configuration_specs(data, plan, concepts, alignments, conditioned)
223 assert rejected["spec_candidates"] == []
224 assert rejected["coverage"]["skipped_reasons"][
225 "invisible_or_unknown_column"] == 1
226
227 conditioned = [{**rules[0], "selector": {"metric_ref": "I1"}}, rules[1]]
228 supported = infer_configuration_specs(data, plan, concepts, alignments, conditioned)
229 assert supported["spec_candidates"][0]["spec"]["selector"] == {"metric_ref": "I1"}
230 scoped = discover_configuration_relations(
231 data, plan, concepts, alignments, [supported["spec_candidates"][0]["spec"]])
232 assert all(item["source_code"] == "I1" for item in scoped["candidates"])
233 assert scoped["coverage"]["specs"][0]["input_rows"] == 4
234 finally:
235 data.close()
