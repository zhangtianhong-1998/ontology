51# Context is retained as evidence, without becoming an applicability condition.
52# Identity qualifiers remain separate because removing a tenant or version can
53# merge equal labels that denote different definitions.
54CONTEXT_ROLES = frozenset(("provenance", "metadata", "identity"))

68def declared_context_role(column: dict[str, Any]) -> str | None:
69 """Separate explicit context from scope, without using value cardinality.
70
71 Opaque names may receive these roles through checked model proposals.
72 Generic status/type/source fields remain undecided because they can be
73 business coordinates or polymorphic references.
74 """
75 name = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", str(column.get("column_name") or "")).casefold()
76 comment = str(column.get("column_comment") or "")
77 if (_IDENTITY_CONTEXT.fullmatch(name)
78 or re.search(r"定义(?:版本|命名空间)|租户(?:标识|编码)|\b(?:definition version|tenant identifier|namespace)\b", comment, re.I)):
79 return "identity"
80 if (_PROVENANCE_CONTEXT.fullmatch(name)
81 or re.search(r"数据(?:来源|血缘)|来源系统|\b(?:source system|data provenance|data lineage)\b", comment, re.I)):
82 return "provenance"
83 if (_METADATA_CONTEXT.fullmatch(name)
84 or re.search(r"(?:词条|标签|显示|定义文本)(?:语言|语种)|负责人|负责团队|同步方式|同步类型|生命周期状态|\b(?:lexical language|display locale|record status)\b", comment, re.I)):
85 return "metadata"
86 return None

89def resolve_context_role(table: dict[str, Any], candidate: dict[str, Any]) -> dict[str, Any] | None:
90 """Check context semantics independently of a model's rationale.
91
92 Source-value verification proves a quotation, not definition identity.
93 A source key remains a binding; an unsupported qualifier remains visible
94 as unknown content. Profiles only support a *candidate*, never key truth.
95 This also runs when consuming saved role candidates, so old responses
96 cannot bypass the boundary by carrying a previously verified status.
97 """
98 proposed = candidate.get("proposed_role", candidate.get("role"))
99 if proposed not in {*CONTEXT_ROLES, "scope"}:
100 return None
101 field = candidate.get("column")
102 column = next((item for item in table.get("columns", ())
103 if item.get("column_name") == field), None)
104 if column is None:
105 return None
106 declared = declared_context_role(column)
107 normalized = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", field).casefold()
108 profile = next((item for item in table.get("profiles", ())
109 if item.get("column") == field), {})
110 samples = [str(value).strip() for value in profile.get("distinct_sample", ())
111 if value is not None and str(value).strip()]
112 # Observations have already been source-checked by the inference stage.
113 # Available profile values are a contradiction check against cherry-picking.
114 samples += [str(item.get("value", "")).strip()
115 for item in candidate.get("observations", ()) if str(item.get("value", "")).strip()]
116 effective, reason = proposed, None
117 if declared:
118 effective, reason = declared, "explicit_context_declaration"
119 elif proposed == "scope" and field in table.get("pk", ()):
120 # A model cannot evade the record-key boundary by renaming identity
121 # to scope. Non-PK dimension codes can legitimately delimit a scope.
122 effective, reason = "reference", "declared_record_primary_key_not_business_applicability"
123 elif proposed == "identity":
124 identifier_declaration = bool(re.search(
125 r"编号|编码|标识符|引用键|主键|外键|\b(?:identifier|primary key|foreign key|reference key)\b",
126 str(column.get("column_comment") or ""), re.I))
127 identifier_name = bool(re.search(r"(?:^|_)(?:id|code|key|uuid|guid|sn|no|number)$", normalized))
128 # Opaque columns can still carry explicit source-level qualifier text.
129 # Arbitrary A/B, numeric uniqueness or an LLM explanation is insufficient.
130 explicit_values = bool(samples) and all(re.fullmatch(
131 r"(?:tenant|namespace|definition[-_ ]version|definition[-_ ]revision|租户|命名空间|定义版本)"
132 r"\s*[:=/_-]\s*\S.*", value, re.I) for value in samples)
133 usable = profile.get("usable_count") or 0
134 near_unique = (usable >= 2 and profile.get("scan_scope") == "full_input"
135 and ((profile.get("sample_exhaustive_in_input") and len(set(samples)) == usable)
136 or (profile.get("approx_distinct_usable", profile.get("approx_distinct")) or 0)
137 >= 0.9 * usable))
138 code_values = bool(samples) and all(re.fullmatch(
139 r"(?:[+-]?\d+|(?=[A-Za-z0-9_-]*\d)[A-Za-z0-9][A-Za-z0-9_-]+)", value)
140 for value in samples)
141 if field in table.get("pk", ()) or identifier_declaration or identifier_name:
142 effective, reason = "reference", "source_record_or_reference_key_not_definition_qualifier"
143 elif explicit_values and profile.get("scan_scope") == "full_input":
144 effective, reason = "identity", "explicit_qualifier_values_with_profile_support"
145 elif near_unique and code_values:
146 effective, reason = "reference", "profile_supported_record_key_candidate_not_definition_identity"
147 else:
148 effective, reason = "unknown", "definition_identity_lacks_independent_source_support"
149 elif proposed in {"metadata", "provenance"}:
150 # Nonempty structured content can contain lexical meaning, references,
151 # constraints or formulas. Do not discard its semantics as generic admin
152 # metadata, or invent a description from the model's explanation.
153 structured = False
154 for value in samples:
155 try:
156 parsed = json.loads(value)
157 except (ValueError, TypeError):
158 continue
159 structured |= isinstance(parsed, (dict, list)) and bool(parsed)
160 if structured:
161 effective, reason = "unknown", "structured_content_requires_semantic_review"
162 elif proposed == "metadata" and samples and all(re.fullmatch(
163 r"(?:zh|en|de|fr|ja|ko|es|it|pt|ru|ar)(?:[-_](?:[A-Za-z]{2}|[A-Za-z]{4})(?:[-_][A-Za-z]{2})?)?",
164 value, re.I) for value in samples) and profile.get("scan_scope") == "full_input":
165 effective, reason = "metadata", "lexical_locale_values_with_profile_support"
166 elif proposed == "provenance" and samples and all(re.fullmatch(
167 r"(?:source|system|instrument|registry|upstream|来源系统|数据来源)\s*[:=/_-]\s*\S.*",
168 value, re.I) for value in samples) and profile.get("scan_scope") == "full_input":
169 effective, reason = "provenance", "explicit_source_values_with_profile_support"
170 else:
171 # Unknown scalar flags can distinguish actual/budget values, rule
172 # modes or lexical categories. A model's admin label alone must
173 # not erase them from complete template compatibility checks.
174 effective, reason = "unknown", "context_role_lacks_independent_source_support"
175 if reason is None:
176 return None
177 return {"column": field, "proposed_role": proposed, "effective_role": effective,
178 "reason": reason, "resolution_origin": "independent_source_role_evidence",
179 "binding_only": effective == "reference", "semantic_identity_claim": False,
180 "schema_evidence_id": f"schema:{_table_name(table)}:{field}",
181 "profile_field_id": profile.get("field_id"),
182 "validation_scope": "source_declaration_and_available_profile_values",
183 "semantic_status": "candidate_not_semantic_truth"}

186def is_sensitive_column(column: dict[str, Any]) -> bool:
187 """Conservative value-redaction boundary for common credential fields."""
188 name = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", str(column.get("column_name") or ""))

201def _empty_in_input(profile: dict[str, Any] | None) -> bool:
202 if not profile or profile.get("scan_scope") != "full_input":
203 return False
204 rows, usable = profile.get("row_count"), profile.get("usable_count")
205 stat = profile.get("statistics", {}).get("usable_count", {})
206 return (
207 isinstance(rows, int) and not isinstance(rows, bool) and rows > 0
208 and usable == 0 and stat.get("value") == 0
209 and stat.get("exact") is True and stat.get("status") == "complete"
210 and stat.get("rows_scanned") == rows
211 )

214def _literal_type(data_type: str) -> str:
215 sql_type = data_type.lower().strip()
216 if "timestamp" in sql_type or sql_type.startswith("datetime"):
217 return "datetime"
218 if sql_type == "date":
219 return "date"
220 if sql_type in ("bool", "boolean"):
221 return "boolean"
222 if re.match(r"^(smallint|integer|bigint|int\d*|serial\d*)$", sql_type):
223 return "integer"
224 if re.match(r"^(numeric|decimal|real|double|float)", sql_type):
225 return "decimal"
226 return "string"

229def classify_columns(table: dict[str, Any]) -> list[dict[str, Any]]:
230 """Classify each declared column without dropping it from source mappings.
231
232 ``include_in_semantic_prompt=False`` applies only to prompt construction;
233 every field remains in the deterministic direct mapping. A binding is
234 provided for columns whose meaning is settled by schema/profile evidence.
235 Unknown and semantic columns stay visible so uncertain business fields are
236 never silently discarded.
237 """
238 table_name = _table_name(table)
239 pk = set(table.get("pk") or ())
240 excluded = set(table.get("semantic_excluded_columns") or ())
241 profiles = {p["column"]: p for p in table.get("profiles", ()) if p.get("column")}
242 result = []
243 for column in table.get("columns", ()):
244 name = str(column["column_name"])
245 normalized = name.lower()
246 comment = str(column.get("column_comment") or "")
247 profile = profiles.get(name)
248 join_eligible = normalized in {"uuid", "guid", "pk"} or bool(
249 re.search(r"(?:^|_)(?:id|code|key)$", normalized)
250 )
251
252 if name in excluded or is_sensitive_column(column):
253 role, reason = "sensitive", "excluded from semantic processing by configuration or credential-like name"
254 join_eligible = False
255 elif _empty_in_input(profile):
256 role, reason = "empty", "no usable values in the full imported CSV snapshot"
257 elif normalized in _AUDIT_NAMES and not _BUSINESS_TIME_CUES.search(comment):
258 role, reason = "audit_time", "explicit audit timestamp name without business-time evidence"
259 elif normalized in _AUDIT_METADATA_NAMES and not _BUSINESS_TIME_CUES.search(comment):
260 role, reason = "audit_metadata", "explicit audit metadata name without business-context evidence"
261 elif (name in pk and (normalized in _TECHNICAL_ID_NAMES or normalized.endswith("_id"))
262 and not _BUSINESS_TIME_CUES.search(comment)):
263 role, reason = "technical_identifier", "declared primary key with a technical ID name"
264 elif (_SEMANTIC_NAME_CUES.search(normalized) or _SEMANTIC_COMMENT_CUES.search(comment)
265 or join_eligible):
266 role, reason = "semantic", "name or comment indicates business meaning or a reference key"
267 else:
268 role, reason = "unknown", "insufficient evidence to remove from semantic analysis"
269
270 deterministic = role in ("empty", "audit_time", "audit_metadata", "technical_identifier", "sensitive")
271 result.append({
272 "column": name,
273 "role": role,
274 "reason": reason,
275 "include_in_semantic_prompt": not deterministic,
276 "deterministic_binding": ({
277 "relation_type": "has",
278 "source_table": table_name,
279 "source_column": name,
280 "declared_data_type": column.get("data_type"),
281 "literal_type": _literal_type(str(column.get("data_type") or "")),
282 } if deterministic else None),
283 "join_eligible": join_eligible,
284 "evidence": {
285 "schema_id": f"schema:{table_name}:{name}",
286 "profile_field_id": profile.get("field_id") if profile else None,
287 "row_count": profile.get("row_count") if profile else None,
288 "usable_count": profile.get("usable_count") if profile else None,
289 "scan_scope": profile.get("scan_scope") if profile else None,
290 "input_scope": profile.get("input_scope") if profile else None,
291 },
292 })
293 return result
