40def test_no_comments_source_checked_roles_enter_definition_cards(tmp_path):
41 data = _opaque_data(tmp_path)
42 try:
43 assert not _field_roles(data.tables["fruit.opaque_one"]).get("name")
44 llm = FakeLLM({"proposals": [
45 {"column": "c1", "role": "name", "observations": [
46 {"row_number": 1, "value": "苹果销售额"}], "rationale": "人可读名称"},
47 {"column": "c2", "role": "description", "observations": [
48 {"row_number": 1, "value": "苹果销售产生的收入"}], "rationale": "解释性文字"},
49 ], "unresolved_columns": []})
50 result = asyncio.run(infer_column_role_candidates(data, llm, {"enabled": True}))
51 table = data.tables["fruit.opaque_one"]
52 assert _field_roles(table) == {"name": ["c1"], "description": ["c2"]}
53 assert result["coverage"]["source_verified_candidates"] == 2
54 assert result["coverage"]["model_calls_attempted"] == 1
55 assert result["candidates"][0]["observations"][0]["record_id"].startswith("fruit.opaque_one:")
56 assert "password" not in str(llm.packets)
57 assert "creation_date" not in str(llm.packets)
58 assert "sk_SECRET_MARKER" not in str(llm.packets)
59 assert "c3" not in str(llm.packets)
60 assert llm.packets[0]["sample_rows"][0]["values"]["c1"] == "苹果销售额"
61 built = build_semantic_cards(data, tmp_path / "cards.sqlite")
62 index = SemanticCardIndex(built["index_path"])
63 try:
64 cards = index.search("苹果销售额", kind="definition")
65 assert any(card["name"] == "苹果销售额" for card in cards)
66 assert next(card for card in cards if card["name"] == "苹果销售额")[
67 "fields"]["description"][0]["value"] == "苹果销售产生的收入"
68 assert index.search("苹果销售额", kind="uncertain") == []
69 assert built["coverage"]["by_table"]["fruit.opaque_one"]["row_purpose"]["purpose"] == "definition_data"
70 # Role candidates only widen recall; concept compilation and Judge
71 # still have to accept the actual definition record.
72 finally:
73 index.close()
74 recall = recall_concept_candidates(data)
75 assert recall["coverage"]["definition_tables_considered"] == 1
76 finally:
77 data.close()

101def test_disabled_missing_model_and_table_budget_report_scope(tmp_path):
102 data = _opaque_data(tmp_path, two=True)
103 try:
104 llm = FakeLLM(error=RuntimeError("model unavailable"))
105 off = asyncio.run(infer_column_role_candidates(data, llm, {}))
