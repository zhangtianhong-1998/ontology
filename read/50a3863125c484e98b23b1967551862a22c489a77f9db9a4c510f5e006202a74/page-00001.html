132async def complete(model, messages, *, timeout, on_delta, **kwargs):
133 """A single deadline includes opening and consuming the entire SSE stream."""
134 stream, final = None, None
135 async with asyncio.timeout(timeout) as deadline:
136 try:
137 response = await model(messages, **kwargs)

139 stream = response
140 async for chunk in stream:
141 if chunk.is_last:
142 final = chunk

146 on_delta(content)

148 final = response
149 # AgentScope can turn task cancellation into an interrupted response.
