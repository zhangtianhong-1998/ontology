145def test_seed_schedule_uses_full_population_not_equal_truncated_pool():
146 roots = {"Metric": ("metric_main", 16000),
147 "Measure": ("measure_main", 5000),
148 "Dimension": ("dimension_main", 80),
149 "GeneralObject": ("general_main", 3700)}
150 pool = [{"card_id": f"{root}-{i}", "table": table, "root_hint": root,
151 "fields": {"description": [{"value": "definition"}]}}
152 for root, (table, _) in roots.items() for i in range(50)]
153 # Every root has exactly 50 pool cards, but the indexed corpus is skewed.
154 chosen = _select_seeds(pool, 80,

159 assert counts["GeneralObject"] > counts["Dimension"]
