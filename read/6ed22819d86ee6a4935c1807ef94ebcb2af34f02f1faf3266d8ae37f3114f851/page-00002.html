520def discover_configuration_relations(
521 data, plan: BuildPlan, concepts, alignments, specs, *, max_pairs_per_spec=1000, memberships=(),
522):
523 """Return bounded, evidence-backed relation *candidates* from explicit codes.
524
525 The SQL scans every configuration row and groups only observed code tuples.
526 Each definition code must match exactly one row in the full imported table,
527 and that row must have an accepted type alignment or definition membership. Missing, multiple,
528 or unaligned endpoints remain unresolved. Even two verified endpoints do
529 not authorize a business predicate without a separate semantic decision.
530 """
531 if type(max_pairs_per_spec) is not int or max_pairs_per_spec < 1:
532 raise ValueError("max_pairs_per_spec must be a positive integer")
533 plan = BuildPlan.model_validate(plan)
534 parsed = [ConfigurationRelationSpec.model_validate(item) for item in specs]
535 if len({item.spec_id for item in parsed}) != len(parsed):
536 raise ValueError("Duplicate configuration relation spec ID")
537 align_by_record, conflicting_alignments = _type_alignments(plan, concepts, alignments, memberships)
538 candidates, reports = [], []
539 lookup_cache = {}
540
541 def endpoint(table_name, code_column, code, scope):
542 if code is None or str(code).strip() == "":
543 return {"status": "blank_code"}
544 if any(value in (None, "") for value in scope.values()):
545 return {"status": "missing_scope"}
546 cache_key = (table_name, code_column, code, tuple(sorted(scope.items())))
547 if cache_key in lookup_cache:
548 return lookup_cache[cache_key]
549 matches = data.lookup(table_name, ((code_column, code), *sorted(scope.items())))
550 if not matches:
551 result = {"status": "missing_definition"}
552 elif len(matches) > 1:
553 result = {"status": "ambiguous_definition", "matching_rows_at_least": 2}
554 else:
555 record_id = data.record_id(table_name, matches[0])
556 alignment = align_by_record.get(record_id)
557 result = ({"status": "accepted_type", "record_id": record_id,
558 **alignment} if alignment else
559 {"status": "definition_without_exact_accepted_type",
560 "record_id": record_id,
561 "conflicting_exact_alignments": record_id in conflicting_alignments})
562 lookup_cache[cache_key] = result
563 return result

565 for spec in parsed:
566 for table, column in (
567 (spec.configuration_table, spec.source_code_column),
568 (spec.configuration_table, spec.target_code_column),
569 (spec.source_definition_table, spec.source_definition_code_column),

583 raise ValueError("Configuration source and target code columns must differ")

585 total_groups = rows[0]["all_distinct_code_pairs"] if rows else 0
586 total_rows = rows[0]["all_configuration_rows"] if rows else 0
587 statuses = Counter()
588 for item in rows:
589 witness = _witness(data, spec, item["witness_row_number"])
590 columns = [spec.source_code_column, spec.target_code_column]
591 columns.extend(set(spec.selector) | set(spec.source_scope_bindings) | set(spec.target_scope_bindings))
592 if spec.relation_text_column:
593 columns.append(spec.relation_text_column)

602 status = "endpoint_verified_candidate" if verified else "unresolved"
603 candidate = {
604 "id": "configuration_relation_candidate:" + digest([
605 data.snapshot_id, spec.spec_id, item["source_code"], item["target_code"],
606 item.get("relation_literal"), source_scope, target_scope, spec.selector])[:24],
607 "spec_id": spec.spec_id, "status": status,
608 "snapshot_id": data.snapshot_id,
609 "predicate_status": "unjudged", "direction_status": "unjudged",
610 "predicate_literal": item.get("relation_literal"),
611 "configuration_table": spec.configuration_table,
612 "source_code_column": spec.source_code_column,
613 "target_code_column": spec.target_code_column,
614 "relation_text_column": spec.relation_text_column,

616 "source_scope_bindings": spec.source_scope_bindings,

618 "source_rule_id": spec.source_rule_id, "target_rule_id": spec.target_rule_id,

641 statuses[candidate["status"]] += 1
642 candidates.append(candidate)
643 reports.append({"spec_id": spec.spec_id, "configuration_table": spec.configuration_table,
644 "scan_scope": "full_input", "input_rows": total_rows,
645 "distinct_code_pairs": total_groups, "pairs_emitted": len(rows),
646 "pairs_omitted_limit": total_groups - len(rows),
647 "configuration_rows_in_emitted_pairs": sum(item["configuration_rows"] for item in rows),
648 "statuses_in_emitted_pairs": dict(statuses),
649 "partial": total_groups > len(rows)})
650 return {"method": "explicit_dual_code_full_snapshot_unique_lookup",
651 "semantic_status": "endpoint_candidates_only_predicate_unjudged",
652 "snapshot_id": data.snapshot_id,
653 "limits": {"max_pairs_per_spec": max_pairs_per_spec},
654 "candidates": candidates,
655 "coverage": {"specs": reports, "candidates_emitted": len(candidates),
656 "endpoint_verified_candidates": sum(
657 item["status"] == "endpoint_verified_candidate" for item in candidates),
658 "partial": any(item["partial"] for item in reports)}}
