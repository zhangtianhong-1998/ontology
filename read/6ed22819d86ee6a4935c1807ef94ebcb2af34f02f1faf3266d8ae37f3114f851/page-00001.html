59 """Bind text to this pair, without accepting its predicate or direction.
60
61 Complete sentences may identify both current endpoints by their codes or
62 names. A generic purpose statement can describe the sole schema-declared

64 table-wide document must never be borrowed by arbitrary column pairings.
65 """
66 literal = candidate.get("predicate_literal")
67 if not candidate.get("relation_text_column") or literal is None:
68 return None
69 if not isinstance(literal, str) or not literal.strip() or _structured_document(literal):
70 raise ValueError("configuration_text_is_not_a_statement_for_the_selected_reference_pair")

72 def mentioned(anchor):
73 if not isinstance(anchor, str) or len(anchor.strip()) < 2:
74 return False
75 # Do not match Q1 inside Q10. This is only recall; compilation checks
76 # the full positive clause and preserves direction and qualifiers.
77 return re.search(r"(?<![A-Za-z0-9_])" + re.escape(anchor.strip()) +
78 r"(?![A-Za-z0-9_])", literal, re.I) is not None
79
80 anchors = []
81 for side in ("source", "target"):
82 endpoint = candidate.get(side + "_definition", {})
83 item = types.get(candidate.get(side + "_type_id"))
84 label = item.label if item is not None else None
85 values = [candidate.get(side + "_code"), label, *endpoint.get("lookup_scope", {}).values()]
86 anchors.append({value for value in values if mentioned(value)})
87 if any(left != right for left in anchors[0] for right in anchors[1]):
88 return {"kind": "literal_endpoint_anchors", "column": candidate["relation_text_column"],
89 "source_anchors": sorted(anchors[0]), "target_anchors": sorted(anchors[1])}
90 table = data.tables[candidate["configuration_table"]]
91 declared = {item["column_name"] for item in table["columns"]
92 if re.search(r"引用|对应.*(?:编码|定义|表)|\b(?:references?|foreign key)\b",
93 str(item.get("column_comment") or ""), re.I)}
94 declared.update(fk.get("column_name") for fk in table.get("foreign_keys", ()))
95 selected = {candidate.get("source_code_column"), candidate.get("target_code_column")}

97 return {"kind": "sole_schema_declared_reference_pair", "column": candidate["relation_text_column"],

102def verify_configuration_pair(data, plan, candidate, witness):
103 """Recheck cheap pair constraints before semantic judgement or replay."""
104 left = candidate.get("source_definition", {}).get("record_id")
105 right = candidate.get("target_definition", {}).get("record_id")

132def _relation_text_candidate(data, table_name, reference_columns):
133 table = data.tables[table_name]
134 visible = []
135 for column in table["columns"]:
136 name = column["column_name"]
137 # Identifier names often contain "rule"; that does not make their
138 # numeric IDs a relationship statement.
139 if name in reference_columns or name in table.get("pk", []):
140 continue
141 try:
142 _check_column(data, table_name, name)
143 except ValueError:
144 continue
145 visible.append(name)
146 # Opaque names and absent comments do not erase an explicit source
147 # statement. This is bounded recall only: compilation still checks the
148 # entire positive clause, both endpoint names and the exact witness.
149 if visible:
150 select = ", ".join(f"substr(CAST({qi(name)} AS VARCHAR), 1, 1025)" for name in visible)
151 rows = data.db.execute(f"SELECT {select} FROM {qi(table['sql_name'])} ORDER BY __r2_row LIMIT 16").fetchall()
152 scored = []
153 for position, name in enumerate(visible):
154 values = [row[position] for row in rows if isinstance(row[position], str)
155 and row[position].strip() and len(row[position]) <= 1024]
156 if not values or all(re.fullmatch(r"[\d\s.+-]+", value) or _structured_document(value)
157 for value in values):
158 continue
159 column = next(value for value in table["columns"] if value["column_name"] == name)
160 declared = bool(_RELATION_TEXT.search(name + " " + str(column.get("column_comment") or "")))
161 stated = any(_RELATION_VALUE.search(value) for value in values)
162 if declared or stated:
163 scored.append((int(stated), int(declared), -position, name))
164 if scored:
165 return max(scored)[-1]
166 return None

169def declared_reference_evidence(data, rule, *, definition_tables=None):
170 """Resolve an original FK or explicitly named target column declaration.
171
172 A bridge column may be both PK and FK. This exception proves the reference
173 path only; full-input uniqueness and the business predicate still need
174 their independent checks. Bare target columns require one definition
175 table in the supplied accepted-type scope, never a best name match.
176 """
177 source, target = rule.get("source") or {}, rule.get("target") or {}
178 local = data.tables.get(source.get("table"))
179 remote = data.tables.get(target.get("table"))
180 if local is None or remote is None:
181 return None
182 for fk in local.get("foreign_keys", ()):
183 if (fk.get("column_name") == source.get("field")
184 and fk.get("referenced_schema") == remote.get("schema")
185 and fk.get("referenced_table") == remote.get("table_name")
186 and fk.get("referenced_column") == target.get("field")):
187 return {"kind": "declared_foreign_key", "source_table": source["table"],
188 "declaration": dict(fk)}
189 column = next((item for item in local["columns"] if item["column_name"] == source.get("field")), {})
190 quote = str(column.get("column_comment") or "")
191 scope = sorted(set(data.tables if definition_tables is None else definition_tables))
192 target_roots = definition_tables if isinstance(definition_tables, dict) else {}
193 for clause in re.split(r"[,;，；。\n]", quote):
194 if re.search(r"不|未|可能|或|\b(?:not|may|or|unless)\b", clause, re.I):
195 continue
196 # Conditional branches are literal discriminator values, not a
197 # guessed interpretation of labels such as "dimension-like".
198 conditional = re.fullmatch(r"\s*([A-Za-z_][\w-]*)\s*(?:引用|references?)\s*([A-Za-z_][\w.]*)\s*", clause, re.I)
199 condition = {}
200 if conditional:
201 branch, token = conditional.groups()
202 fields = [item["column_name"] for item in local["columns"]
203 if (rule.get("selector") or {}).get(item["column_name"]) == branch
204 and re.search(r"类型|种类|\b(?:type|kind|discriminator)\b",
205 str(item.get("column_comment") or ""), re.I)]
206 if len(fields) != 1:
207 continue
208 condition = {fields[0]: branch}
209 else:
210 match = re.search(r"(?:引用|对应|\breferences?\s+)[^A-Za-z_]*([A-Za-z_][\w.]*)", clause, re.I)
211 if not match:
212 # "quantity code, corresponds to the measure definition"
213 # declares the local code's role. Without a named definition
214 # root below this cannot fall back to same-column matching.
215 token = source.get("field", "")
216 else:
217 token = match.group(1).rstrip(".")
218 declared_roots = {root for root, name in (("Dimension", "维度"), ("Measure", "度量"), ("Metric", "指标"))
219 if re.search(r"(?:引用|对应|references?).*(?:" + name + "|" + root +
220 r").*(?:定义|definition)", clause, re.I)}
221 if not conditional and match is None and (len(declared_roots) != 1 or not
222 re.search(r"编码|\bcode\b", quote, re.I)):
223 continue
224 parts = token.split(".")
225 if len(parts) > 3 or parts[-1] != target.get("field"):
226 continue
227 matches = []
228 for name in scope:
229 table = data.tables.get(name, {})
230 if declared_roots and (len(declared_roots) != 1 or target_roots.get(name) != declared_roots):
231 continue
232 if parts[-1] not in table.get("column_names", []):
233 continue
234 if len(parts) == 3 and name != ".".join(parts[:2]):
235 continue
236 if len(parts) == 2 and table.get("table_name") != parts[0]:
237 continue
238 matches.append(name)
239 if matches != [target.get("table")]:
240 continue
241 return {"kind": "declared_column_reference", "source_table": source["table"],
242 "column": source["field"], "quote": quote, "target_token": token,
243 "target": dict(target), "condition": condition,
244 "definition_table_scope": scope, "declared_target_roots": sorted(declared_roots)}
245 return None

248def infer_configuration_specs(
249 data, plan: BuildPlan, concepts, alignments, checked_rules, *, max_specs=100, memberships=(),
250):
251 """Pair two checked config→definition code references, never assert a relation.
252
253 Each input rule must have exact raw-code matching over the full snapshot,
254 no missing/ambiguous referenced rows, and no selector/scope condition that
255 this simple dual-code contract cannot represent. Distinct accepted target
256 business roots without an explicit statement remain alignment-only leads.
257 The returned `spec` mappings can feed `discover_configuration_relations`;
258 their evidence remains candidate-level until both endpoints and the
259 predicate have been judged separately.
260 """
261 if type(max_specs) is not int or max_specs < 1:
262 raise ValueError("max_specs must be a positive integer")
263 plan = BuildPlan.model_validate(plan)
264 target_roots = _accepted_definition_roots(data, plan, concepts, alignments, memberships)
265 by_table = defaultdict(list)
266 skipped = Counter()
267 seen = 0
268 for rule in checked_rules:
269 seen += 1
270 if rule.get("status") != "checked_technical":
271 skipped["not_checked_technical"] += 1
272 continue
273 declaration = declared_reference_evidence(data, rule)
274 if rule.get("numeric_overlap_only") and declaration is None:
275 skipped["numeric_overlap_without_reference_semantics"] += 1
276 continue
277 if (rule.get("transform", {}).get("operator") != "identity"
278 or rule.get("verification", {}).get("scan_scope") != "full_input"):
279 skipped["not_full_input_raw_identity"] += 1
280 continue
281 checks = rule.get("verification", {}).get("checks") or {}
282 eligible, unique = checks.get("eligible_references"), checks.get("unique_matches")
283 if (type(eligible) is not int or eligible <= 0 or type(unique) is not int
284 or unique != eligible or any(checks.get(key, 0) for key in
285 ("ambiguous_matches", "missing_in_input", "missing_scope"))):
286 skipped["incomplete_or_ambiguous_full_input_check"] += 1
287 continue
288 source, target = rule.get("source") or {}, rule.get("target") or {}
289 source_table, target_table = source.get("table"), target.get("table")
290 source_field, target_field = source.get("field"), target.get("field")
291 if not all((source_table, target_table, source_field, target_field)):
292 skipped["missing_rule_endpoint"] += 1
293 continue
294 try:
295 _check_column(data, source_table, source_field)
296 _check_column(data, target_table, target_field)
297 for field in rule.get("selector") or {}:
298 _check_column(data, source_table, field)
299 for local, remote in (rule.get("scope_bindings") or {}).items():
300 _check_column(data, source_table, local)
301 _check_column(data, target_table, remote)
302 except ValueError:
303 skipped["invisible_or_unknown_column"] += 1
304 continue
305 # Table-name heuristics cannot veto two independently verified code
306 # references. Their purpose is still unjudged until the next stage.
307 if classify_row_purpose(data.tables[source_table])["purpose"] == "business_fact":
308 skipped["business_fact_is_not_configuration_witness"] += 1
309 continue
310 if (rule["source"]["field"] in data.tables[source_table].get("pk", [])
311 and declaration is None):
312 skipped["configuration_identity_is_not_business_reference"] += 1
313 continue
314 roots = target_roots.get(target_table, set())
315 if len(roots) != 1:
316 skipped["target_lacks_single_accepted_business_root"] += 1
317 continue
318 by_table[source_table].append((rule, next(iter(roots))))
319
320 output, total_pairs = [], 0
321 for table_name in sorted(by_table):
322 rules = sorted(by_table[table_name], key=lambda pair: str(pair[0].get("rule_id")))
323 for (left, left_root), (right, right_root) in combinations(rules, 2):
324 total_pairs += 1
325 left_source, right_source = left["source"], right["source"]
326 left_selector, right_selector = left.get("selector") or {}, right.get("selector") or {}
327 if any(left_selector[key] != right_selector[key] for key in left_selector.keys() & right_selector.keys()):
328 skipped["incompatible_reference_conditions"] += 1
329 continue
330 if left_source["field"] == right_source["field"]:
331 skipped["same_configuration_column"] += 1
332 continue
333 text_column = _relation_text_candidate(
334 data, table_name, {left_source["field"], right_source["field"]})
335 if left_root == right_root and text_column is None:
336 # Two code paths into records of one business root are often
337 # alternate descriptions of the same type, not a predicate.
338 skipped["same_business_root_alignment_lead"] += 1
339 continue
340 if left["target"]["table"] == right["target"]["table"] and text_column is None:
341 skipped["same_definition_table_alignment_lead"] += 1
342 continue
343 spec = ConfigurationRelationSpec(
344 spec_id="config_spec:" + digest([
345 data.snapshot_id, table_name, left["rule_id"], right["rule_id"]])[:24],
346 configuration_table=table_name,
347 source_code_column=left_source["field"],
348 target_code_column=right_source["field"],
349 source_definition_table=left["target"]["table"],
350 source_definition_code_column=left["target"]["field"],
351 target_definition_table=right["target"]["table"],
352 target_definition_code_column=right["target"]["field"],
353 relation_text_column=text_column,
354 selector={**left_selector, **right_selector},
355 source_scope_bindings=left.get("scope_bindings") or {},
356 target_scope_bindings=right.get("scope_bindings") or {},
357 source_rule_id=left["rule_id"], target_rule_id=right["rule_id"],
358 )
359 output.append({
360 "spec": spec.model_dump(), "status": "candidate_only",
361 "direction_status": "unjudged",
362 "rule_ids": [left["rule_id"], right["rule_id"]],
363 "target_root_hints": [left_root, right_root],
364 "relation_text_basis": (
365 "visible_column_name_or_comment_candidate" if text_column
366 else "none_predicate_remains_unjudged"),
367 "evidence_ids": sorted({
368 f"schema:{table_name}:{left_source['field']}",
369 f"schema:{table_name}:{right_source['field']}",
370 f"schema:{left['target']['table']}:{left['target']['field']}",
371 f"schema:{right['target']['table']}:{right['target']['field']}",
372 }),
373 "semantic_status": "two_verified_reference_rules_are_not_a_business_predicate",
374 })
375 # Spend the bounded spec allowance across tables rather than letting the
376 # lexically first table consume it through many equivalent rule pairings.
377 queues = defaultdict(deque)
378 for item in output:
379 queues[item["spec"]["configuration_table"]].append(item)
380 selected = []
381 while len(selected) < max_specs and any(queues.values()):
382 for key in sorted(queues):
383 if queues[key] and len(selected) < max_specs:
384 selected.append(queues[key].popleft())
385 skipped["spec_limit"] = len(output) - len(selected)
386 output = selected
387 return {
388 "method": "pair_full_input_checked_config_references_with_separate_semantic_adjudication",
389 "snapshot_id": data.snapshot_id,
390 "spec_candidates": output,
391 "coverage": {"rules_inspected": seen, "eligible_rule_groups": len(by_table),
392 "rule_pairs_considered": total_pairs, "specs_emitted": len(output),
393 "specs_omitted_limit": skipped["spec_limit"],
394 "skipped_reasons": dict(skipped),
395 "partial": skipped["spec_limit"] > 0},
396 }

410def _type_alignments(plan, concepts, alignments, memberships=()):
411 accepted = {item.id for item in plan.object_types if item.category == "business_type"}
412 by_concept = {item["id"]: item for item in concepts}
413 by_record = {}
414 conflicting = set()
415 for alignment in alignments:
416 if alignment.get("mapping_kind") != "exact":
417 continue
418 concept = by_concept.get(alignment.get("concept_id"))
419 if not concept or concept.get("ontology_level") != "type":
420 continue
421 type_id = concept.get("ontology_type_id")
422 if type_id not in accepted:
423 continue
424 record_id = alignment.get("source_record_id")
425 if not record_id:
426 continue
427 old = by_record.get(record_id)
428 if old and old["type_id"] != type_id:
429 conflicting.add(record_id)
430 else:
431 by_record[record_id] = {
432 "type_id": type_id, "concept_id": concept["id"],
433 "alignment_id": alignment.get("id"),
434 "evidence_ids": alignment.get("evidence_ids", []),
435 "mapping_kind": "exact",
436 }
437 for member in memberships:
438 record_id, type_id = member.get("record_id"), member.get("type_id", member.get("object_type_id"))
439 if (not record_id or type_id not in accepted
440 or not (member.get("status") == "definition_template_match"
441 or (member.get("mapping_kind") == "template_instance"
442 and member.get("status") == "accepted"))
443 or not member.get("evidence_ids")):
444 continue
445 old = by_record.get(record_id)
446 if old and old["type_id"] != type_id:
447 conflicting.add(record_id)
448 elif old is None:
449 by_record[record_id] = {"type_id": type_id, "concept_id": member.get("concept_id"),
450 "alignment_id": member["id"], "evidence_ids": member.get("evidence_ids", []),
451 "snapshot_id": member.get("snapshot_id"),
452 "mapping_kind": member.get("mapping_kind", "shares_definition_type_template")}
453 for record_id in conflicting:
454 by_record.pop(record_id, None)
455 return by_record, conflicting
