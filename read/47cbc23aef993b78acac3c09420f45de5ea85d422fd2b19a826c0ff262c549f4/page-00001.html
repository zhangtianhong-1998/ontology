33@pytest.mark.parametrize("role", ["identity", "parameter", "applicability", "classification"])
34def test_source_category_repairs_scope_role_without_erasing_attribute(role):
35 data, packet, proposed = classification_fixture(proposed_role=role)
36 concept, _ = compile_concept(data, PROFILE, packet, proposed, {})
37 item = _compiled_object_type(concept)
38 assert item.classification_attributes == {"c3": "ASSET"}
39 assert item.applicability_scope == item.definition_parameters == item.identity_qualifiers == {}
40 assert concept["scope_roles"] == {"c3": "classification"}
41 assert concept["proposed_scope_roles"] == {"c3": role}
42 prop = next(p for p in item.source_properties if p.role == "classification")
43 assert data.evidence[prop.evidence_ids[0]]["raw_fragment"] == "ASSET"

56def test_exact_and_projection_reuse_preserve_the_same_classification():
57 data, packet, exact = classification_fixture()
58 concept, _ = compile_concept(data, PROFILE, packet, exact, {})
59 accepted = _compiled_object_type(concept)
60 projection = TemplateProjectionDecision(root_type="Dimension", label=exact.label, definition=exact.definition,
61 witness_record_ids=["r1"], label_evidence=[{"record_id": "r1", "column": "c1", "quote": exact.label}],
62 applicability_scope={"c3": "ASSET"})
63 plan, template, _ = compile_projection(data, PROFILE, BuildPlan(object_types=[accepted]), packet, projection)
64 assert template["object_type_id"] == accepted.id
65 assert template["classification_attributes"] == accepted.classification_attributes
66 assert template["applicability_scope"] == {}
67 assert len(plan.object_types) == 1
68 other = deepcopy(packet["records"][0])
69 other["record_id"] = "r2"
70 other["scope"]["c3"] = other["fields"]["scope"][0]["value"] = "CONTRACT"
71 packet["records"].append(other)
72 projection.witness_record_ids.append("r2")
73 with pytest.raises(ValueError, match="different source classification"):
74 compile_projection(data, PROFILE, BuildPlan(), packet, projection)
