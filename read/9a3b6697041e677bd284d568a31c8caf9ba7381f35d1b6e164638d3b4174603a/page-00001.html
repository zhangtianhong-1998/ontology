13def _fixture(*, descriptions=None, formulas=None, roots=None, units=None,
14 scopes=None, truncated=False):
15 descriptions = descriptions or (
16 "水果销售利润是水果销售收入扣除水果销售成本后的利润",
17 "水果销售利润是水果销售收入扣除水果销售成本后的利润",
18 )
19 formulas = formulas if formulas is not None else (
20 "水果销售利润 = 水果销售收入 - 水果销售成本",
21 "水果销售利润 = 水果销售收入 - 水果销售成本",
22 )
23 roots = roots or tuple("Metric" for _ in descriptions)
24 units = units or tuple("元" for _ in descriptions)
25 scopes = scopes or tuple({} for _ in descriptions)
26 evidence, types = {}, []
27 for i, description in enumerate(descriptions):
28 props, evidence_ids = [], []
29 for role, value in (("description", description), ("formula", formulas[i])):
30 if value is None:
31 continue
32 evidence_id = f"record:e{i}-{role}"
33 evidence[evidence_id] = {
34 "id": evidence_id, "origin": "observed_record",
35 "raw_fragment": value,
36 "raw_fragment_truncated": bool(truncated and i == 1 and role == "description"),
37 "source_ref": {"table": f"fruit.metric_{i}", "column": role,
38 "record_id": f"r{i}", "snapshot_id": "snap"},
39 }
40 props.append(SourceProperty(role=role, source_table=f"fruit.metric_{i}",
41 source_column=role, evidence_ids=[evidence_id]))
42 evidence_ids.append(evidence_id)
43 types.append(DerivedType(
44 id=f"type:metric_{i}", parent=roots[i],
45 label="水果销售利润", definition=description, category="business_type",
46 unit=units[i], applicability_scope=scopes[i],
47 derivation_kind="exact_definition", evidence_scope="definition_record",
48 source_properties=props, evidence_ids=evidence_ids,
49 ))
50 return SimpleNamespace(snapshot_id="snap", evidence=evidence), BuildPlan(object_types=types)

53def _decision(data, candidate):
54 source_id, target_id = candidate["source_type_id"], candidate["target_type_id"]
55 left, right = int(source_id.rsplit("_", 1)[1]), int(target_id.rsplit("_", 1)[1])
56 result = {
57 "status": "proposed", "source_type_id": source_id,
58 "target_type_id": target_id,
59 "source_description_evidence_id": f"record:e{left}-description",
60 "target_description_evidence_id": f"record:e{right}-description",
61 "source_description_quote": data.evidence[f"record:e{left}-description"]["raw_fragment"],
62 "target_description_quote": data.evidence[f"record:e{right}-description"]["raw_fragment"],
63 "semantic_equivalence_explanation": "双方指向同一销售利润计算口径和单位",
64 }
65 if f"record:e{left}-formula" in data.evidence:
66 result.update(source_formula_evidence_id=f"record:e{left}-formula",
67 source_formula_quote=data.evidence[f"record:e{left}-formula"]["raw_fragment"])
68 if f"record:e{right}-formula" in data.evidence:
69 result.update(target_formula_evidence_id=f"record:e{right}-formula",
70 target_formula_quote=data.evidence[f"record:e{right}-formula"]["raw_fragment"])
71 return result

74def test_complete_bilateral_description_and_formula_support_one_pair():
75 data, core = _fixture()
76 candidates = propose_equivalence_candidates(core, data, 5)
77 assert len(candidates) == 1 and candidates[0]["status"] == "candidate_only"
78 assertion = compile_equivalence(data, core, candidates[0], _decision(data, candidates[0]))
79 assert assertion["source_type_id"] == "type:metric_0"
80 assert assertion["target_type_id"] == "type:metric_1"
81 assert assertion["canonical_type_id"] == "type:metric_0"
82 assert len(assertion["evidence_ids"]) == 4
83 assert assertion["source_description_quote"] == data.evidence["record:e0-description"]["raw_fragment"]
84 assert len(core.object_types) == 2

93def test_definition_parameters_prevent_recall_and_stale_pair_acceptance(left, right):
94 data, core = _fixture()

142@pytest.mark.parametrize("change,error", [
