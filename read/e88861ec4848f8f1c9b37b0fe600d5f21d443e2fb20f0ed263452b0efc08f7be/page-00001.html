24def test_conditional_branches_and_target_families_precede_same_branch_variants():
25 branches = ['01', '02', '甲', '乙', 'unknown']
26 items = [condition(branch, target, field)
27 for branch in branches
28 for target in ['u.a_definition', 'u.b', 'u.z']
29 for field in ['x', 'y', 'z']]
30 before = copy.deepcopy(items)
31 # A name hint alone must not displace an observed value-supported family.
32 shared = {(('u.a', 'opaque_ref'), ('u.z', 'x')): (10, True)}
33 ordered = _conditioned_order(items, shared)
34 assert {r['suggested_selector']['opaque_partition'] for r in ordered[:5]} == set(branches)
35 assert {r['target'] for r in ordered[:5]} == {('u.z', 'x')}
36 for branch in branches:
37 first_three = [r for r in ordered if r['suggested_selector']['opaque_partition'] == branch][:3]
38 assert len({r['target'][0] for r in first_three}) == 3
39 assert items == before # Selectors, scope and numeric risk are unchanged.
40 assert all(r['numeric_overlap_only'] for r in ordered)
41 shuffled = items[:]
42 random.Random(71).shuffle(shuffled)
43 assert _conditioned_order(shuffled, shared) == ordered

95def directions(leads):
96 candidates, proposals = {}, []
97 for lead in leads:
98 for reverse in (False, True):
99 cid = lead['candidate_id'] + ('r' if reverse else 'f')
100 source, target = (lead['target'], lead['source']) if reverse else (lead['source'], lead['target'])
101 candidates[cid] = {'source': source, 'target': target,
102 'alias_recall_candidate_id': lead['candidate_id']}
103 proposals.append((RuleProposal(candidate_id=cid), 'value_alias_recall'))
104 return candidates, proposals

200@pytest.mark.parametrize('branches', [('A01', 'A02'), ('01', '02'), ('甲一', '甲二')])
201@pytest.mark.parametrize('budget', [2, 4])
202def test_validation_keeps_complete_selector_order_and_full_checks(monkeypatch, branches, budget):
203 """Do not collapse distinct literals into one prefix during validation."""
204 from ontology_r2 import discovery
205 data = SmallDataset({
206 'u.feed': (['opaque', 'bucket'], [('101', branches[0]), ('202', branches[1]),
207 ('999', 'other')] * 2, []),
208 'u.left': (['x'], [('101',), ('202',)], ['x']),
209 'u.right': (['y'], [('101',), ('202',)], ['y']),
210 })
211 try:
212 found = discovery.propose_candidates(data, value_index_mode='full_distinct',
213 max_indexed_fields=0, max_conditional_pair_checks=100,
214 max_conditional_candidates=100, max_candidates_total=200)
215 # Real recall output, with ordinary pairs removed to isolate the bounded
216 # conditional validation queue. Values, selectors and risks are untouched.
217 found['candidates'] = [c for c in found['candidates'] if c['source'] ==
218 {'table': 'u.feed', 'field': 'opaque'} and c.get('suggested_selector')]
219 recalled = copy.deepcopy(found['candidates'])
220 assert len(recalled) == 4
221 assert {c['suggested_selector']['bucket'] for c in recalled[:2]} == set(branches)
222 expected = [discovery.validate_candidate(data, c, selector=c['suggested_selector'],
223 scope_bindings={target: source for source, target in
224 c['suggested_scope_bindings'].items()}) for c in recalled]
225 monkeypatch.setattr(discovery, 'propose_candidates', lambda *a, **k: copy.deepcopy(found))
226 result = discovery.discover_and_check(data, {'max_candidate_validations': budget})
227 assert result['checks'] == expected[:budget]
228 assert result['coverage']['candidates_not_attempted'] == 4 - budget
229 assert result['coverage']['candidates_checked'] == budget
230 assert result['coverage']['candidates_not_checked'] == 4 - budget
231 assert all(c['numeric_overlap_only'] and c['risk_flags'] == ['numeric_value_coincidence']
232 for c in result['candidates'])
233 assert all(c['decision']['semantic_relation'] == 'unresolved' for c in result['candidates'])
234 assert all(c['checks']['eligible_references'] == c['checks']['unique_matches'] == 2
235 for c in result['checks'])
236 if budget == 4:
237 # Untruncated validation produces the same complete checks, not a
238 # looser approximation. Only the bounded queue order was changed.
239 assert {c['candidate_id'] for c in result['checks']} == {
240 c['candidate_id'] for c in recalled}
241 finally:
242 data.close()
