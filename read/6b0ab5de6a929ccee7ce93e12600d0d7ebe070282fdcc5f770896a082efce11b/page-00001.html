41def test_graph_keeps_conditional_transforms_and_declarations_separate(data):
42 data.tables["fruit.records"]["foreign_keys"] = [{
43 "column_name": "ref", "referenced_schema": "fruit",
44 "referenced_table": "definitions", "referenced_column": "code",
45 "constraint_name": "records_ref_fkey"}]

47 transform={"operator": "source_alias_items"})]

157 class IndexedEdges(list):
158 lookups = 0
159
160 def __getitem__(self, item):
161 self.lookups += 1
162 return super().__getitem__(item)
163
164 class NeverCopy:
165 def __deepcopy__(self, memo):
166 raise AssertionError('Validation examples must not be copied into every prompt')
167
168 class NeverScan(list):
169 def __iter__(self):
170 raise AssertionError('Per-packet metadata lookup must not scan the complete graph')
171
172 graph._field_edges = IndexedEdges(graph._field_edges)
173 graph._field_edges[0]['verification']['counterexamples'] = [NeverCopy(), NeverCopy()]
174 graph._field_edges.lookups = 0
175 graph.edges = NeverScan(graph.edges)
176 records = [{'table': table, 'fields': {'reference': [{'column': 'key', 'value': 'A'}]}}
177 for table in ('s.a0', 's.b0')]
178 # An operation-count assertion is stable across machines, unlike a tight
179 # wall-clock threshold: 100 packets inspect 100 edges, not 300,000 edges.
180 for _ in range(100):
181 packet = _metadata_context(SimpleNamespace(metadata_graph=graph), records)
182 assert len(packet['field_links']) == 1
183 assert graph._field_edges.lookups == 100
184 edge = packet['field_links'][0]
185 assert edge['rule_id'] == 'rule:0' and edge['candidate_id'] == 'candidate:0'
186 assert edge['snapshot_id'] == 'snapshot'
187 assert edge['verification'] == {'scan_scope': 'full_input', 'checks': {'unique_matches': 1},
188 'counterexample_count': 2}
189 assert edge['evidence_sources']['evidence:1']['source_ref']['file'] == 'schema/tables/a0.yaml'
190 edge['selector']['kind'] = 'MUTATED'
191 assert graph._field_edges[0]['selector']['kind'] == 'API'
192 only_source = _metadata_context(SimpleNamespace(metadata_graph=graph), records[:1])
193 assert only_source['field_links'] == []
