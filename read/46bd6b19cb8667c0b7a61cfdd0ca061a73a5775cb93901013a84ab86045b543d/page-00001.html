1"""Bounded actual-row context from checked technical links, without identity claims.
2
3A rule is materialized once for the complete seed window. Queries preserve its
4selector, scope and transform, and only uniquely matched forward links enter
5an adjacency table. Reverse expansion samples those same links; it never forms
6an independent many-to-many join.
7"""

15class JoinedDefinitionContext:
16 def __init__(self, data, index, association, seeds, *, max_records_per_seed=3,
17 max_rules=64, max_value_chars=2048, progress=None):
18 self.data, self.index = data, index
19 self.limit, self.max_chars = max_records_per_seed, max_value_chars
20 if any(type(value) is not int or value < 0 for value in
21 (max_records_per_seed, max_rules, max_value_chars)) or max_value_chars == 0:
22 raise ValueError('Joined context limits must be nonnegative; value characters must be positive')
23 self.tables, self.skipped, self.seed_coverage = [], [], []
24 self.stats = Counter()
25 by_table = defaultdict(set)
26 for seed in seeds:
27 by_table[seed['table']].add(seed['row_number'])
28 rules = [rule for rule in association.get('rules', [])
29 if rule.get('status') == 'checked_technical' and
30 (rule['source']['table'] in by_table or rule['target']['table'] in by_table)]
31 rules.sort(key=lambda rule: (rule.get('numeric_overlap_only', False), rule['rule_id']))
32 self.stats['relevant_checked_rules'] = len(rules)
33 self.stats['rules_not_materialized_due_to_limit'] = max(0, len(rules)-max_rules)
34 if not self.limit:
35 self.stats['rules_not_materialized_due_to_limit'] = len(rules)
36 return
37 task = progress.task('物化定义关联邻域', min(len(rules), max_rules)) if progress else None
38 for rule in rules[:max_rules]:
39 if task:
40 task.advance(detail=rule['source']['table'] + ' → ' + rule['target']['table'])
41 counts = rule.get('verification', {}).get('checks', {})
42 if (rule.get('snapshot_id') != data.snapshot_id
43 or rule.get('verification', {}).get('scan_scope') != 'full_input'
44 or counts.get('eligible_references', 0) <= 0
45 or counts.get('unique_matches') != counts.get('eligible_references')
46 or any(counts.get(key, 0) for key in ('ambiguous_matches', 'missing_scope', 'missing_in_input'))):
47 self.skipped.append({'rule_id': rule['rule_id'], 'reason': 'invalid_full_input_check'})
48 continue
49 table_name = '__r2_definition_context_' + digest([
50 data.snapshot_id, rule['rule_id'], sorted((name, sorted(rows)) for name, rows in by_table.items())])[:20]
51 try:
52 self._materialize(rule, by_table, table_name)
53 except Exception as exc:
54 self.skipped.append({'rule_id': rule['rule_id'], 'reason': 'context_query_error',
55 'error_type': type(exc).__name__})
56 continue
57 self.tables.append((table_name, rule))
58 self.stats['rules_materialized'] = len(self.tables)
59 if task:
60 task.close()

62 def _materialize(self, rule, by_table, table_name):
63 source, target = rule['source'], rule['target']
64 scope = rule.get('scope_bindings', {})
65 if len(set(scope.values())) != len(scope):
66 raise ValueError('Scope bindings collide')
67 query, params = association_match_sql(self.data, rule,
68 selector=rule.get('selector'), scope_bindings={v: k for k, v in scope.items()},
69 transform=rule.get('transform', {}).get('operator', 'identity'))
70 branches = []
71 for direction, local, remote in [('forward', source, target), ('reverse', target, source)]:
72 positions = sorted(by_table.get(local['table'], []))
73 if not positions:
74 continue
75 seed_col, neighbor_col = ('source_row_number', 'target_row_number') if direction == 'forward' else (
76 'target_row_number', 'source_row_number')
77 marks = ','.join('?' for _ in positions)
78 branches.append(f"SELECT '{direction}' AS direction, {seed_col} AS seed_row, "
79 f"{neighbor_col} AS neighbor_row FROM matched WHERE {seed_col} IN ({marks})")
80 params.extend(positions)
81 self.data.db.execute(f"""CREATE OR REPLACE TEMP TABLE {qi(table_name)} AS
82 WITH matched AS MATERIALIZED ({query}), adjacent AS ({' UNION ALL '.join(branches)}),
83 ranked AS (SELECT *, count(*) OVER (PARTITION BY direction, seed_row) AS neighbor_count,
84 row_number() OVER (PARTITION BY direction, seed_row ORDER BY neighbor_row) AS rank
85 FROM adjacent)
86 SELECT * FROM ranked WHERE rank <= ?""", [*params, self.limit])
87 self.data.db.execute(f"CREATE INDEX {qi(table_name+'_lookup')} ON {qi(table_name)} (direction, seed_row)")

89 def _record(self, table, row_number, rule, direction):
90 info = self.data.tables[table]
91 roles = _field_roles(info)
92 role_by_column = {field: role for role, fields in roles.items() for field in fields}
93 source_side = direction == 'reverse'
94 endpoint = rule['source' if source_side else 'target']
95 keys = {endpoint['field']}
96 if source_side:
97 keys.update(rule.get('selector', {}))
98 keys.update(rule.get('scope_bindings', {}))
99 else:
100 keys.update(rule.get('scope_bindings', {}).values())
101 columns = list(dict.fromkeys([*info.get('pk', []), *role_by_column, *sorted(keys)]))
102 selected = ','.join(qi(column) for column in columns)
103 values = self.data.db.execute(f'SELECT __r2_row, {selected} FROM {qi(info["sql_name"])} '
104 'WHERE __r2_row=?', [row_number]).fetchone()
105 if values is None:
106 raise ValueError('Matched source row no longer exists')
107 row = dict(zip(['__r2_row', *columns], values))
108 fields, omitted = defaultdict(list), []
109 for column in columns:
110 if column not in role_by_column and column not in keys:
111 continue
112 value = row.get(column)
113 if value is None or not str(value).strip():
114 continue
115 raw = str(value)
116 if len(raw) > self.max_chars:
117 omitted.append({'column': column, 'reason': 'whole_field_exceeds_context_limit',
118 'characters': len(raw)})
119 continue
120 fields[role_by_column.get(column, 'reference')].append({
121 'column': column, 'value': raw, 'truncated': False,
122 'schema_evidence_id': f'schema:{table}:{column}'})
123 rid = self.data.record_id(table, row)
124 source_card = self.index.db.execute(
125 'SELECT card_id FROM card_sources WHERE record_id=? ORDER BY card_id LIMIT 1',
126 (rid,)).fetchone()
127 return {'card_id': source_card['card_id'] if source_card else 'context:' + digest([self.data.snapshot_id, rid])[:24],
128 'record_id': rid, 'table': table, 'row_number': row_number,
129 'kind': 'related_context', 'context_role': 'related_context',
130 'eligible_for_exact_alignment': False, 'fields': dict(fields),
131 'name': next((v['value'] for v in fields.get('name', [])), ''),
132 'scope': {v['column']: v['value'] for v in fields.get('scope', [])},
133 'unit': next((v['value'] for v in fields.get('unit', [])), ''),
134 'omitted_fields': omitted,
135 'connection': {'rule_id': rule['rule_id'], 'direction': direction,
136 'source': rule['source'], 'target': rule['target'],
137 'selector': rule.get('selector', {}), 'scope_bindings': rule.get('scope_bindings', {}),
138 'transform': rule.get('transform', {'operator': 'identity'}),
139 'semantic_relation': 'unresolved', 'identity_claim': False}}

141 def for_seed(self, seed):
142 leads, available = [], 0
143 for table_name, rule in self.tables:
144 for direction, endpoint, remote in [('forward', rule['source'], rule['target']),
145 ('reverse', rule['target'], rule['source'])]:
146 if endpoint['table'] != seed['table']:
147 continue
148 rows = self.data.db.execute(f'SELECT neighbor_row, neighbor_count FROM {qi(table_name)} '
149 'WHERE direction=? AND seed_row=? ORDER BY neighbor_row',
150 [direction, seed['row_number']]).fetchall()
151 if rows:
152 available += rows[0][1]
153 leads.extend((remote['table'], row, rule, direction) for row, _ in rows)
154 records, seen, read_errors = [], set(), []
155 for table, row, rule, direction in leads:
156 key = (table, row)
157 if key in seen or (table == seed['table'] and row == seed['row_number']):
158 continue
159 seen.add(key)
160 if len(records) >= self.limit:
161 continue
162 try:
163 records.append(self._record(table, row, rule, direction))
164 except Exception as exc:
165 read_errors.append({'table': table, 'row': row, 'error_type': type(exc).__name__})
166 omitted = max(0, available-len(records))
167 coverage = {'seed_record_id': seed['record_id'], 'neighbor_edges_available': available,
168 'records_expanded': len(records), 'neighbor_edges_not_expanded': omitted,
169 'fields_not_expanded': sum(len(record['omitted_fields']) for record in records),
170 'read_errors': read_errors, 'max_records': self.limit,
171 'partial': bool(omitted or read_errors or any(record['omitted_fields'] for record in records))}
172 self.seed_coverage.append(coverage)
173 return records, coverage

175 def coverage(self):
176 return {**self.stats, 'skipped_rules': self.skipped, 'seeds': self.seed_coverage,
177 'records_expanded': sum(item['records_expanded'] for item in self.seed_coverage),
178 'neighbor_edges_not_expanded': sum(item['neighbor_edges_not_expanded'] for item in self.seed_coverage),
179 'partial': bool(self.skipped or self.stats['rules_not_materialized_due_to_limit']
180 or any(item['partial'] for item in self.seed_coverage)),
181 'identity_claim': False, 'scan_strategy': 'once_per_checked_rule_for_current_seed_window'}

184 for name, _ in self.tables:
