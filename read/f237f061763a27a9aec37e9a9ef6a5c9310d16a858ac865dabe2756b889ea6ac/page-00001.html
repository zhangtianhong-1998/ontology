147@pytest.mark.parametrize("checkpoint_hash, message", [
148 (None, "lacks implementation_code_hash"),
149 ("", "lacks implementation_code_hash"),

152def test_implementation_rejection_precedes_any_restoration(checkpoint_case, checkpoint_hash, message):
153 from copy import deepcopy
154 data, profile, plan, path = checkpoint_case
155 save_state(path, data, profile, {"plan": plan},
156 implementation_code_hash=IMPLEMENTATION_HASH)
157 current_contract = state_contract(data, profile, implementation_code_hash=IMPLEMENTATION_HASH)
158 saved = json.loads(path.read_text())
159 assert saved["contract"] == current_contract
160 if checkpoint_hash is None:
161 del saved["contract"]["implementation_code_hash"]
162 else:
163 saved["contract"]["implementation_code_hash"] = checkpoint_hash
164 # Every other contract field remains identical. None of these contents may
165 # be restored merely because prompts/profile/snapshot still match.
166 table = sorted(data.tables)[0]
167 saved["evidence"] = {"old-acceptance": {"raw_fragment": "stale evidence"}}
168 saved["column_roles"] = {table: [{"column": "old", "role": "name"}]}
169 saved["column_role_report"] = {"source_snapshot": data.snapshot_id, "coverage": {"partial": False}}
170 saved["plan"] = {"invalid_plan": "must never reach parsing"}
171 path.write_text(json.dumps(saved))
172 data.column_role_report = {"source_snapshot": data.snapshot_id, "coverage": {"partial": True}}
173 before = deepcopy((data.evidence, data.tables, data.column_role_report))
174 checkpoint_before = path.read_bytes()
175 with pytest.raises(ValueError, match=message):
176 restore_state(path, data, profile, implementation_code_hash=IMPLEMENTATION_HASH)
177 assert (data.evidence, data.tables, data.column_role_report) == before
178 assert path.read_bytes() == checkpoint_before

220def test_pipeline_passes_its_manifest_code_hash_to_every_checkpoint_call():
221 import ast
222 import inspect
223 from ontology_r2 import pipeline
224 tree = ast.parse(inspect.getsource(pipeline.build))
225 calls = [node for node in ast.walk(tree) if isinstance(node, ast.Call)
226 and isinstance(node.func, ast.Name) and node.func.id in {"restore_state", "save_state"}]
227 names = [node.func.id for node in calls]
228 assert names.count("restore_state") == 1
229 # Additional iterative stages may checkpoint too; every save must use
230 # the same implementation hash, rather than fixing the number of stages.
231 assert names.count("save_state") >= 2
232 for node in calls:
233 argument = next(item.value for item in node.keywords if item.arg == "implementation_code_hash")
234 assert isinstance(argument, ast.Name) and argument.id == "code_hash"
235 # The same value is already recorded in the run manifest; no parallel
236 # implementation-fingerprint algorithm is introduced by this patch.
237 manifest = next(node.value for node in ast.walk(tree) if isinstance(node, ast.Assign)
238 and any(isinstance(target, ast.Name) and target.id == "manifest" for target in node.targets))
239 code_hash = next(value for key, value in zip(manifest.keys, manifest.values)
240 if isinstance(key, ast.Constant) and key.value == "implementation_code_hash")
241 assert isinstance(code_hash, ast.Name) and code_hash.id == "code_hash"
