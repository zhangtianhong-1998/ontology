78 def source(self, table, row_number, expected):
79 schema_path = self.root / "schema/tables" / (table.split(".")[-1] + ".yaml")
80 result = {"table": table, "row_number": row_number, "csv_line": row_number + 1,
81 "expected_values": expected, "applicable": schema_path.exists()}
82 if not result["applicable"]:
83 return result
84 metadata = yaml.load(schema_path.read_text(), Loader=yaml.CSafeLoader)
85 constraint_path = self.root / "schema/constraints" / schema_path.name
86 result["snapshot_schema_verified"] = all(
87 self.manifest.get("input_files", {}).get(str(p.relative_to(self.root))) == sha256(p)
88 for p in (schema_path, constraint_path))
89 path = self.root / "data" / (metadata["table_name"] + ".csv")
90 scoped = self.root / "data" / metadata["schema"] / (metadata["table_name"] + ".csv")
91 if scoped.exists():
92 path = scoped
93 result["file"] = str(path)
94 rel = str(path.relative_to(self.root))
95 file_hash = sha256(path)
96 expected_hash = self.manifest.get("input_files", {}).get(rel)
97 result["snapshot_file_verified"] = bool(expected_hash and file_hash == expected_hash)
98 row = None
99 with path.open(newline="", encoding="utf-8-sig") as stream:
100 reader = csv.DictReader(stream)
101 for position, item in enumerate(reader, 1):
102 if position == row_number:
103 row = item
104 result["csv_line"] = reader.line_num
105 break
106 result["sample_values_match"] = row is not None and all(row.get(k) == v for k, v in expected.items())
107 if row is None:
108 return result
109 constraints = yaml.load(constraint_path.read_text(), Loader=yaml.CSafeLoader)
110 pk = []
111 for entry in constraints.get("constraints", []):
112 match = re.search(r"PRIMARY KEY\s*\(([^)]+)\)", entry.get("definition", ""), re.I)
113 if match:
114 pk = [field.strip().strip('"') for field in match[1].split(",")]
115 key = {k: row[k] for k in pk} if pk and all(row.get(k) not in (None, "") for k in pk) else {
116 "file": file_hash, "row": row_number}
117 result["record_id"] = table + ":" + digest([key, row_number, self.snapshot])[:24]
118 result["values"] = {k: row.get(k) for k in expected}
119 self.sources[result["record_id"]] = result
120 return result

143 def accepted(self, source):

146 Concept.source_refs and evidence_ids include related/dependency context.
