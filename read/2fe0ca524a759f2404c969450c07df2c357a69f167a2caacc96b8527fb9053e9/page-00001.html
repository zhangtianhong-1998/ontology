28def test_exact_compilation_retains_unit_case_and_rejects_mixed_units():
29 records = [power_record("r1", "MW"), power_record("r2", "mW")]
30 data = SimpleNamespace(snapshot_id="s", tables={}, evidence={})
31 concepts = [compile_concept(data, PROFILE, {"records": [record]}, power_decision([record]), {})[0]
32 for record in records]
33 assert {item["unit"] for item in concepts} == {"MW", "mW"}
34 assert len({item["ontology_type_id"] for item in concepts}) == 2
35 assert _unit_values(records[0]).isdisjoint(_unit_values(records[1]))
36 with pytest.raises(ValueError, match="Conflicting units"):
37 compile_concept(data, PROFILE, {"records": records}, power_decision(records), {})
