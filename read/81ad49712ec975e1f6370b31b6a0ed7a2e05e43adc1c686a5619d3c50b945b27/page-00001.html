14def test_static_rule_ids_stay_as_bindings_while_rule_contents_and_parameters_survive(tmp_path):
15 root = tmp_path / "input"
16 columns = {"business_rule_id": "合成业务字段", "rule_name": "规则名称", "definition": "规则定义",
17 "name_id": "", "alias_key": "", "expr_code": "", "rule_type": "",
18 "rule_json": "", "field_rule": "允许取值或过滤规则 JSON", "formula_column": "计算公式"}
19 base = {"rule_name": "库存阈值", "definition": "低于阈值提示库存风险", "rule_type": "THRESHOLD",
20 "rule_json": '{"min":10}', "field_rule": '["A"]', "formula_column": "收入"}
21 rows = [{**base, "business_rule_id": str(i), "name_id": "N" + str(i),
22 "alias_key": "A" + str(i), "expr_code": "E" + str(i)} for i in range(1, 6)]
23 rows[2]["rule_json"] = '{"min":20}'
24 rows[3]["field_rule"] = '["B"]'
25 rows[4]["rule_type"] = "RANGE"
26 _table(root, "definition", columns, rows, pk="business_rule_id")
27 work = tmp_path / "work"
28 work.mkdir()
29 data = Dataset(root, work)
30 try:
31 table = data.tables["fruit.definition"]
32 roles = _field_roles(table)
33 assert roles["name"] == ["rule_name"]
34 assert not roles.get("alias")
35 assert set(roles["formula"]) == {"rule_json", "field_rule", "formula_column"}
36 assert roles["scope"] == ["rule_type"]
37 # Keys remain excluded, but heuristic rule/formula contents still need
38 # bounded review. An unresolved review must not discard these fields.
39 review_columns = {"rule_json", "field_rule", "formula_column"}
40 llm = FakeLLM({"unresolved_columns": sorted(review_columns)})
41 limits = {"enabled": True, "max_tables": 1, "max_columns_per_table": 3,
42 "max_sample_rows": 3}
43 with patch.object(data, "db", wraps=data.db) as db:
44 report = asyncio.run(infer_column_role_candidates(data, llm, limits))
45 # Role review reads only the fixed source rows, without a new
46 # full scan or one query per column.
47 assert db.execute.call_count == 1
48 sql, positions = db.execute.call_args.args
49 assert "WHERE __r2_row IN (?, ?, ?)" in sql
50 assert positions == [1, 3, 5]
51 assert len(llm.packets) == report["coverage"]["model_calls_attempted"] == 1
52 packet = llm.packets[0]
53 assert {column["column"] for column in packet["columns"]} == review_columns
54 assert set(packet["role_review_columns"]) == review_columns
55 assert {row["row_number"] for row in packet["sample_rows"]} == {1, 3, 5}
56 assert all(set(row["values"]) == review_columns for row in packet["sample_rows"])
57 assert report["candidates"] == []
58 assert report["coverage"]["partial"]
59 assert _field_roles(table) == roles
60 assert len(report["tables"][0]["role_conflicts"]) == 5
61 built = build_semantic_cards(data, tmp_path / "cards.sqlite")
62 assert built["coverage"]["definition_patterns_indexed"] == 4
63 index = SemanticCardIndex(built["index_path"])
64 try:
65 cards = {card["row_number"]: card for card in index.all_cards(10)["cards"]}
66 assert len(cards) == 5
67 assert cards[1]["pattern_id"] == cards[2]["pattern_id"]
68 assert cards[1]["card_id"] != cards[2]["card_id"]
69 for card in cards.values():
70 assert {entry["column"] for entry in card["fields"]["reference"]} == {
71 "business_rule_id", "name_id", "alias_key", "expr_code"}
72 assert {entry["column"] for entry in card["fields"]["formula"]} == {
73 "rule_json", "field_rule", "formula_column"}
74 assert card["scope"]["rule_type"] in {"THRESHOLD", "RANGE"}
75 pairs = _semantic_columns(data, card)
76 assert ("formula", "business_rule_id") not in pairs
77 assert ("formula", "rule_type") not in pairs
78 assert ("scope", "rule_type") in pairs
79 assert all(item.get("schema_evidence_id") for item in card["role_conflicts"])
80 finally:
81 index.close()
82 finally:
