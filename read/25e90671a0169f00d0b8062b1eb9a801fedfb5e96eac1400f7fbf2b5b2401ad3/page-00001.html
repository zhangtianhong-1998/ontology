17_TIME = re.compile(r"(?:^|_)(?:year|quarter|month|period|date|time|week)(?:_|$)|年|季|月|期间|账期|会计期", re.I)
18_DIMENSION = re.compile(
19 r"(?:^|_)(?:dim|dimension|region|area|country|city|product|"
20 r"category|channel|customer|market|org)(?:_|$)|维度|地区|区域|国家|城市|品类|渠道|客户|市场|组织", re.I)
21_COORDINATE_KEY = re.compile(r"(?:^|_)(?:code|key)$|(?:编码|代码)$", re.I)
22_VALUE = re.compile(
23 r"(?:^|_)(?:amount|revenue|profit|cost|quantity|qty|price|volume|"
24 r"count|rate|ratio|total|sales|value)(?:_|$)|金额|收入|利润|成本|"
25 r"产量|销量|均价|数量|比率|占比|总额|度量值|指标值", re.I)
26_CONTROL = frozenset(("rule", "config", "anchor", "log", "reference", "ref",
27 "param", "input", "output", "tag", "connection", "member",

37def _numeric_value(column, profile):
38 sql_type = str(column.get("data_type") or "").casefold()
39 declared_numeric = bool(re.match(r"^(?:smallint|integer|bigint|int\d*|decimal|numeric|real|double|float)", sql_type))
40 usable = profile.get("usable_count") or 0
41 numeric = profile.get("numeric_shape_count") or 0
42 return declared_numeric or (usable >= 2 and numeric / usable >= 0.9)

45def _temporal_value(column, profile):
46 """Check observed value shape before treating a comment as a time column."""
47 sql_type = str(column.get("data_type") or "").casefold()
48 if "timestamp" in sql_type or sql_type.startswith(("date", "datetime")):
49 return True
50 samples = [str(value).strip() for value in profile.get("distinct_sample", ())
51 if value is not None and str(value).strip()]
52 return bool(samples) and sum(bool(_TEMPORAL_VALUE.fullmatch(value)) for value in samples) / len(samples) >= 0.8

55def classify_row_purpose(table, *, accepted_dimension_fields=()):
56 """Describe a *table's likely row purpose*, leaving mixed cases unresolved.
57
58 Strong fact evidence requires a numeric business value plus both a business
59 time and a dimension coordinate. A definition/formula in the same table
60 makes that inference ambiguous, so the table is not silently excluded.
61 """
62 roles = _field_roles(table)
63 accepted_dimension_fields = set(accepted_dimension_fields)
64 columns = {item["column_name"]: item for item in table["columns"]}
65 profiles = {item["column"]: item for item in table.get("profiles", ())}
66 safe = {item["column"] for item in classify_columns(table)
67 if item["role"] not in ("sensitive", "empty", "audit_time", "audit_metadata")}
68 pk = set(table.get("pk") or ())
69 names = []
70 times = []
71 dimensions = []
72 values = []
73 inferred = {}
74 for item in table.get("inferred_semantic_roles", ()):
75 if item.get("status") == "source_verified_role_candidate":
76 inferred.setdefault(item["column"], set()).add(item["role"])
77 identity_fields = set(roles.get("identity", ()))
78 for name in table["column_names"]:
79 if name not in safe or (name in pk and name not in identity_fields):
80 continue
81 if (name not in accepted_dimension_fields
82 and any(name in roles.get(role, ()) for role in CONTEXT_ROLES - {"identity"})):
83 continue
84 column = columns[name]
85 comment = str(column.get("column_comment") or "")
86 proposed = inferred.get(name, set())
87 # Match each source independently: appending an empty comment changes
88 # the boundary of plain names such as "value" and "region".
89 numeric_business_value = bool(name not in identity_fields and
90 (_VALUE.search(name) or _VALUE.search(comment)
91 or "numeric_business_value" in proposed) and
92 _numeric_value(column, profiles.get(name, {})))
93 temporal_shape = _temporal_value(column, profiles.get(name, {}))
94 # A value column may be described as "按月统计的销售金额". The comment
95 # specifies its grain, not the column's own values. Prefer observed
96 # temporal shape over lexical time words in a comment.
97 time_candidate = bool(_TIME.search(name) or (_TIME.search(comment) and temporal_shape)
98 or "business_time" in proposed)
99 if time_candidate and (temporal_shape or not numeric_business_value):
100 times.append(name)
101 # A non-PK business code is a grain candidate, regardless of the
102 # domain noun in its name. It is not an accepted Dimension or a join.
103 if (_DIMENSION.search(name) or _DIMENSION.search(comment)
104 or _COORDINATE_KEY.search(name) or _COORDINATE_KEY.search(comment)
105 or name in accepted_dimension_fields or name in identity_fields
106 or "dimension_coordinate" in proposed):
107 dimensions.append(name)
108 if numeric_business_value and name not in times:
109 values.append(name)
110 if name in roles.get("name", ()):
111 names.append(name)
112 tokens = set(table["table_name"].casefold().split("_"))
113 has_definition_text = bool(roles.get("description") or roles.get("formula"))
114 fact_shape = bool(times and dimensions and values)
115 model_fact_shape = all(any(role in rs for rs in inferred.values()) for role in
116 ("business_time", "dimension_coordinate", "numeric_business_value"))
117 if fact_shape and model_fact_shape:
118 purpose = "business_fact"
119 reason = "observed_shapes_and_source_checked_field_role_candidates"
120 elif tokens & _CONTROL and fact_shape:
121 purpose = "unresolved"
122 reason = "control_table_name_and_business_fact_evidence_coexist"
123 elif tokens & _CONTROL:
124 purpose = "configuration_data"
125 reason = "control_or_reference_table_name"
126 elif fact_shape and has_definition_text:
127 purpose = "unresolved"
128 reason = "business_fact_and_definition_evidence_coexist"
129 elif fact_shape:
130 purpose = "business_fact"
131 reason = "numeric_business_value_with_time_and_dimension"
132 elif names and (has_definition_text or tokens & _DEFINITION):
133 purpose = "definition_data"
134 reason = "named_definition_or_formula"
135 else:
136 purpose = "unresolved"
137 reason = "insufficient_or_mixed_row_purpose_evidence"
138 return {"purpose": purpose, "reason": reason,
139 "evidence_columns": {"name": names, "definition": list(roles.get("description", ())),
140 "formula": list(roles.get("formula", ())),
141 "business_time": times, "dimension_coordinate": dimensions,
142 "identity_coordinate": sorted(identity_fields & set(dimensions)),
143 "numeric_business_value": values},
144 "scope": "imported_csv_snapshot_only",
145 "classification_granularity": "table_level_candidate_from_column_evidence",
146 "authority": "candidate_not_business_fact_proof",
147 "accepted_dimension_fields_used": sorted(accepted_dimension_fields & set(dimensions))}

150def _candidate_groups(table, purpose, *, max_columns):
151 """Select a few meaningful sets; do not enumerate the power set."""
152 roles = _field_roles(table)
153 classified = classify_columns(table)
154 safe = {item["column"] for item in classified if item["role"] not in
155 ("sensitive", "empty", "audit_time", "audit_metadata")}
156 valid = lambda fields: tuple(dict.fromkeys(field for field in fields if field in safe))[:max_columns]
157 groups = []
158 def add(label, fields):
159 fields = valid(fields)
160 if fields and fields not in [old[1] for old in groups]:
161 groups.append((label, fields))
162 add("declared_primary_key", table.get("pk") or ())
163 codes = [item["column"] for item in classified
164 if item["join_eligible"] and item["column"] not in table.get("pk", ())
165 and item["column"] in safe]
166 for field in codes[:2]:
167 add("reference_key", [field])
168 names = roles.get("name", ())
169 scopes = roles.get("scope", ())
170 if names:
171 add("name", names[:1])
172 if scopes:
173 add("name_and_scope", [names[0], *scopes[:2]])
174 evidence = purpose["evidence_columns"]
175 if purpose["purpose"] == "business_fact":
176 coordinates = [*codes[:1], *evidence["dimension_coordinate"][:2],
177 *evidence["business_time"][:1]]
178 add("observation_coordinates", coordinates)
179 if evidence["numeric_business_value"]:
180 add("coordinates_and_value", [*coordinates[:max_columns - 1],
181 evidence["numeric_business_value"][0]])
182 return groups

185def profile_joint_distinct(data, table_name, purpose=None, *, max_groups=6, max_columns=4):
186 """Exact distinct counts for a bounded list of single and joint field sets.
187
188 Rows with a blank in any member are excluded from that candidate's count.
189 A 100% distinct ratio is only an observed uniqueness statement within the
190 imported snapshot, not a stable key assertion about future data.
191 """
192 if any(type(value) is not int or value < 1 for value in (max_groups, max_columns)):
193 raise ValueError("Joint-distinct limits must be positive integers")
194 table = data.tables[table_name]
195 purpose = purpose or classify_row_purpose(table)
196 choices = _candidate_groups(table, purpose, max_columns=max_columns)
197 selected = choices[:max_groups]
198 if not selected:
199 return {"table": table_name, "scan_scope": "full_input", "input_rows": table["rows"],
200 "candidate_sets": [], "candidate_sets_omitted_limit": 0,
201 "method": "bounded_exact_joint_distinct_no_cartesian_enumeration"}
202 expressions = []
203 for _, fields in selected:
204 usable = " AND ".join(f"{qi(field)} IS NOT NULL AND length(trim({qi(field)})) > 0"
205 for field in fields)
206 tuple_sql = qi(fields[0]) if len(fields) == 1 else "(" + ", ".join(map(qi, fields)) + ")"
207 expressions.extend([f"count(*) FILTER (WHERE {usable})",
208 f"count(DISTINCT {tuple_sql}) FILTER (WHERE {usable})"])
209 stats = data.db.execute(
210 f"SELECT {', '.join(expressions)} FROM {qi(table['sql_name'])}").fetchone()
211 candidate_sets = []
212 for position, (label, fields) in enumerate(selected):
213 complete, distinct = stats[2 * position:2 * position + 2]
214 candidate_sets.append({"label": label, "columns": list(fields),
215 "complete_rows": complete, "distinct_tuples": distinct,
216 "duplicate_rows": complete - distinct,
217 "blank_or_null_rows": table["rows"] - complete,
218 "unique_within_complete_input_rows": complete > 0 and complete == distinct,
219 "exact": True})
220 return {"table": table_name, "scan_scope": "full_input", "input_rows": table["rows"],
221 "candidate_sets": candidate_sets,
222 "candidate_sets_omitted_limit": max(0, len(choices) - len(selected)),
223 "method": "bounded_exact_joint_distinct_no_cartesian_enumeration",
224 "semantic_status": "candidate_grain_only_not_business_key_or_relation"}
