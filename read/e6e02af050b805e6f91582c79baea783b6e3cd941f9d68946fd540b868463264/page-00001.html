97def test_components_require_explicit_same_role_slot_connections():
98 data, bundle = setup()
99 proposed = decision()
100 proposed.slots[0].target_component = None
101 with pytest.raises(ValueError, match="Orphan projected components: operating_object") as error:
102 compile_projection(data, PROFILE, BuildPlan(), bundle, proposed)
103 assert "same-role slot.target_component" in str(error.value)
104 # An explicit reference to the wrong semantic role does not remove the gap.
105 proposed.slots[1].target_component = "operating_object"
106 with pytest.raises(ValueError, match="different semantic role"):
107 compile_projection(data, PROFILE, BuildPlan(), bundle, proposed)
108 proposed.slots[1].target_component = None
109 proposed.slots[0].target_component = "operating_object"
110 _, template, _ = compile_projection(data, PROFILE, BuildPlan(), bundle, proposed)
111 assert all(slot["target_type_id"] for slot in template["slots"] if slot["target_component"])
112 # Removing an ungrounded component is also valid; an untyped slot remains pending.
113 proposed.slots[0].target_component = None
114 proposed.components = [component for component in proposed.components if component.name != "operating_object"]
115 _, template, _ = compile_projection(data, PROFILE, BuildPlan(), bundle, proposed)
116 assert template["slots"][0]["target_type_id"] is None

183def test_online_reuse_avoids_second_model_request_and_keeps_separate_instances():
184 data, bundle = setup()
185 later = {"bundle_id": "b2", "task_kind": "concept_induction", "records": [record("row3", "芒果", "非洲")],
186 "exact_alignment_record_ids": ["row3"]}
187 class LLM:
188 def __init__(self):
189 self.calls = []
190
191 async def ask(self, task, payload, schema):
192 self.calls.append(task)
193 if task == "concept_bundle":
194 return ConceptBundleDecision(status="proposed", action="project_template", projection=decision())
195 assert task == "group_review"
196 return BundleReview(accepted=True)

199 output = asyncio.run(construct_from_bundles(data, PROFILE, BuildPlan(), [bundle, later], llm,
200 max_bundles=2, enable_template_projection=True))

249def test_type_identity_is_independent_of_matcher_column_names_and_checks_reuse_formula():
250 data, bundle = setup()
251 plan, first, _ = compile_projection(data, PROFILE, BuildPlan(), bundle, decision())
252 renamed = deepcopy(bundle)
253 proposed = decision().model_dump()
254 # Different physical name column, same definition template and quantity.
255 for record_ in renamed["records"]:
256 for entry in record_["fields"]["name"]:
257 if entry["column"] == "x1":
258 entry["column"] = "display_text"
259 proposed["field_templates"][0]["column"] = "display_text"
260 _, second, _ = compile_projection(data, PROFILE, plan, renamed,
261 TemplateProjectionDecision.model_validate(proposed))
262 assert first["object_type_id"] == second["object_type_id"]
263 assert first["template_id"] != second["template_id"]
264 proposed["existing_type_id"] = first["object_type_id"]
265 for rec in renamed["records"]:
266 rec["fields"]["formula"][0]["value"] = "failed / inspected"
267 rec["record_id"] = rec["record_id"].replace("1", "3").replace("2", "4")
268 def rename_citations(value):
269 if isinstance(value, dict):
270 for key, item in value.items():
271 if key == "record_id":
272 value[key] = item.replace("1", "3").replace("2", "4")
273 else:
274 rename_citations(item)
275 elif isinstance(value, list):
276 for item in value:
277 rename_citations(item)
278 rename_citations(proposed)

280 with pytest.raises(ValueError, match="formula"):
281 compile_projection(data, PROFILE, plan, renamed, TemplateProjectionDecision.model_validate(proposed))
