60def test_exact_evidence_signature_preserves_definition_and_scope():
61 base = {"kind": "definition", "root_hint": "Metric", "fields": {
62 "name": [{"column": "metric_name", "value": "利润"}],
63 "description": [{"column": "definition", "value": "收入减成本"}],
64 "scope": [{"column": "region", "value": "华东"}],
65 }}
66 changed = {**base, "fields": {**base["fields"],
67 "scope": [{"column": "region", "value": "华南"}]}}
68 assert _exact_evidence_signature(base) != _exact_evidence_signature(changed)
69 assert _exact_evidence_signature(base) == _exact_evidence_signature(base)
70 assert _exact_evidence_signature({**base, "table": "fruit.a"}) != _exact_evidence_signature(
71 {**base, "table": "fruit.b"})
72 truncated_a = {**base, "card_id": "card:a", "fields": {
73 **base["fields"], "description": [{"column": "definition", "value": "same prefix",
74 "truncated": True}]}}
75 truncated_b = {**truncated_a, "card_id": "card:b"}
76 assert _exact_evidence_signature(truncated_a) != _exact_evidence_signature(truncated_b)
