169 data = _dataset(tmp_path)

173 with sqlite3.connect(path) as db:

175 if legacy_cache:
176 db.execute("DROP INDEX card_sources_record")
177 card_ids = [row[0] for row in db.execute("SELECT card_id FROM cards ORDER BY card_id")]
178 record_id, row_number = db.execute("SELECT record_id,row_number FROM card_sources LIMIT 1").fetchone()
179 # A record may have several card bindings. Retain the former
180 # card-first covering index's first result after adding the index.
181 db.executemany("INSERT OR IGNORE INTO card_sources VALUES (?,?,?)",
182 [(card_id, record_id, row_number) for card_id in reversed(card_ids)])

186 tables = ("cards", "aliases", "card_sources", "card_fts", "metadata")
187 before = {table: db.execute(f"SELECT * FROM {table} ORDER BY rowid").fetchall()
188 for table in tables}
189 sources = {p: p.read_bytes() for p in (tmp_path / "input").rglob("*") if p.is_file()}
190 index = SemanticCardIndex(path)
191 try:
192 query = "SELECT card_id FROM card_sources WHERE record_id=? ORDER BY card_id LIMIT 1"
193 assert index.db.execute(query, (record_id,)).fetchone()[0] == expected == min(card_ids)
194 plan = " ".join(row[3] for row in index.db.execute("EXPLAIN QUERY PLAN " + query, (record_id,)))

196 assert "SCAN" not in plan and "TEMP B-TREE" not in plan
197 assert {table: [tuple(row) for row in index.db.execute(f"SELECT * FROM {table} ORDER BY rowid")]
198 for table in tables} == before
