103def _bounded_decision(packet):
104 decision = _concept_decision([packet['bundle']['records'][0]['record_id']]).model_dump()
105 # The reviewer must see this complete decision, even though its source
106 # bundle already fits in the proposal request.
107 decision['reason'] = '核验说明' * 500

111def _bounded_packets_and_limit(llm):
112 bundles = _batch_packets(3)
113 for bundle in bundles:
114 bundle['evidence_padding'] = '原始证据' * 1000
115 data = SimpleNamespace(snapshot_id='snap', evidence={})
116 # Budget the same complete payload sent in production, including names and
117 # role evidence. A duplicate fixture would drift when the contract grows.
118 packets = [bundle_request_payload(data, PROFILE, BuildPlan(), bundle) for bundle in bundles]
119 reviews = [{**packet, 'candidate': _bounded_decision(packet)} for packet in packets]
120 proposal_bytes = llm.request_bytes('concept_batch', {'packets': packets}, ConceptBatchDecision)
121 review_pair_bytes = llm.request_bytes('group_review_batch', {'packets': reviews[:2]}, BundleBatchReview)
122 limit = max(proposal_bytes, review_pair_bytes) + 1200
123 assert llm.request_bytes('group_review_batch', {'packets': reviews}, BundleBatchReview) > limit
124 # This is the actual regression boundary: the old 65% payload heuristic
125 # would unnecessarily break the valid three-proposal request apart.
126 assert len(json.dumps({'packets': packets}, ensure_ascii=False).encode()) > int(limit * .65)
127 return bundles, limit

130def _bounded_answer(calls):
131 def answer(body, number):
132 text = body['messages'][-1]['content']
133 if isinstance(text, list):
134 text = ''.join(item.get('text', '') for item in text)
135 task, raw = text.split('\n', 1)
136 payload = json.loads(raw)
137 calls.append((task, payload))
138 if task == 'concept_batch':
139 return 'submit_result', {'decisions': [
140 {'bundle_id': packet['bundle']['bundle_id'], 'decision': _bounded_decision(packet)}
141 for packet in reversed(payload['packets'])]}
142 if task == 'group_review_batch':
143 return 'submit_result', {'reviews': [
144 {'bundle_id': packet['bundle']['bundle_id'], 'review': {'accepted': True}}
145 for packet in reversed(payload['packets'])]}
146 assert task == 'group_review'
147 return 'submit_result', {'accepted': True}
148 return answer
