10class SmallDataset:
11 def __init__(self, tables):
12 self.db = duckdb.connect(":memory:")
13 self.snapshot_id = "synthetic-alias-snapshot"
14 self.tables = {}
15 for index, (name, (columns, rows)) in enumerate(tables.items()):
16 sql_name = f"src_{index}"
17 fields = ", ".join(f"{qi(column)} VARCHAR" for column in columns)
18 self.db.execute(f"CREATE TABLE {qi(sql_name)} (__r2_row BIGINT, {fields})")
19 self.db.executemany(
20 f"INSERT INTO {qi(sql_name)} VALUES ({','.join('?' for _ in range(len(columns) + 1))})",
21 [(number, *row) for number, row in enumerate(rows, 1)],
22 )
23 self.tables[name] = {"sql_name": sql_name, "column_names": columns,
24 "rows": len(rows)}
25
26 def close(self):
27 self.db.close()

34def test_standard_name_matches_alias_list_and_preserves_original_values():
35 data = SmallDataset({
36 "demo.metric": (["standard_name"], [("阿里云收入",), ("商业市场收入",)]),
37 "demo.dictionary": (["metric_aliases"], [("阿里云收入；阿里云营收",)]),
38 })
39 try:
40 result = _pair(data, ("demo.metric", "standard_name"),
41 ("demo.dictionary", "metric_aliases"))
42 candidate = result["candidates"][0]
43 assert candidate["orientation"] == "undetermined"
44 assert candidate["decision"] == {"status": "proposed", "semantic_relation": "unresolved"}
45 assert "alias_item_equal" in candidate["comparison_modes"]
46 example = candidate["examples"][0]
47 assert example["comparison_rule"] == "alias_item_equal"
48 assert {example["source"]["raw_value"], example["target"]["raw_value"]} == {
49 "阿里云收入", "阿里云收入；阿里云营收",
50 }
51 assert {example["source"]["rule"], example["target"]["rule"]} == {
52 "raw_equal", "alias_list_item_nfkc_whitespace_casefold",
53 }
54 metric_side = ("source" if candidate["source"]["table"] == "demo.metric"
55 else "target")
56 assert candidate["checks"][f"{metric_side}_raw_values_matched_in_sample"] == 1
57 assert candidate["checks"][f"{metric_side}_distinct_raw_values_sampled"] == 2
58 assert candidate["checks"][f"{metric_side}_sample_coverage"] == 0.5
59 assert any(c["raw_value"] == "商业市场收入" for c in candidate["counterexamples"])
60 assert any(c["matched_text"] == "阿里云营收" for c in candidate["counterexamples"])
61 finally:
62 data.close()
