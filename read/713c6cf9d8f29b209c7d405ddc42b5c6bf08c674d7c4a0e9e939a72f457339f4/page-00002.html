843class _BatchConceptLLM:
844 def __init__(self, *, corrupt_first=False, fail_batch_review=False):
845 self.calls = []
846 self.corrupt_first = corrupt_first
847 self.fail_batch_review = fail_batch_review
848 self.config = {"max_input_bytes": 100000}
849
850 async def ask(self, task, payload, schema):
851 self.calls.append(task)
852 if task == "concept_batch":
853 decisions = []
854 for i, packet in enumerate(payload["packets"]):
855 record_id = packet["bundle"]["records"][0]["record_id"]
856 if self.corrupt_first and i == 0:
857 record_id = payload["packets"][1]["bundle"]["records"][0]["record_id"]
858 decisions.append({"bundle_id": packet["bundle"]["bundle_id"],
859 "decision": _concept_decision([record_id]).model_dump()})
860 return schema.model_validate({"decisions": decisions})
861 if task == "group_review_batch":
862 if self.fail_batch_review:
863 from ontology_r2.llm import BudgetExceeded
864 raise BudgetExceeded("budget")
865 return schema.model_validate({"reviews": [
866 {"bundle_id": packet["bundle"]["bundle_id"], "review": {"accepted": True}}
867 for packet in payload["packets"]]})
868 if task == "concept_bundle":
869 return _concept_decision([payload["bundle"]["records"][0]["record_id"]])
870 if task == "group_review":
871 return BundleReview(accepted=True)
872 raise AssertionError(task)
