18def _attribute_rank(column):
19 name = column.casefold()
20 if any(part in name for part in ("name", "title", "description", "definition", "comment", "meaning", "formula", "label", "名称", "说明", "定义", "口径")):
21 return 0
22 if name in ("id", "code", "key", "sn", "number", "uuid") or name.endswith(("_id", "_code", "_no", "_number")):
23 return 1
24 if any(part in name for part in ("time", "date", "created", "updated", "modified", "时间", "日期")):
25 return 2
26 return 3

69def _instance_preview(db, ontology, max_nodes):
70 """Sample each semantic type, so a large record table cannot hide other types."""
71 type_ids = [item["id"] for item in ontology.get("object_types", [])
72 if item.get("category") == "business_type"]
73 counts = [{"type_id": type_id, "instance_kind": kind, "count": count}
74 for type_id, kind, count in db.execute(
75 "SELECT coalesce(json_extract(body,'$.ontology_type_id'),json_extract(body,'$.type')), "
76 "coalesce(json_extract(body,'$.instance_kind'),'legacy'), count(*) "
77 "FROM items WHERE kind='objects' GROUP BY 1,2")]
78 if not type_ids:
79 return [], [], counts
80 marks = ",".join("?" for _ in type_ids)
81 rows = db.execute(
82 "SELECT body FROM (SELECT body, row_number() OVER (PARTITION BY "
83 "coalesce(json_extract(body,'$.ontology_type_id'),json_extract(body,'$.type')), "
84 "coalesce(json_extract(body,'$.instance_kind'),'legacy') ORDER BY id) AS rank "
85 "FROM items WHERE kind='objects' AND "
86 "coalesce(json_extract(body,'$.ontology_type_id'),json_extract(body,'$.type')) "
87 f"IN ({marks})) WHERE rank<=3 ORDER BY rank LIMIT ?", type_ids + [max_nodes])
88 nodes = {item["id"]: item for item in (json.loads(row[0]) for row in rows)}
89 selected_ids, links = list(nodes), {}
90 for start in range(0, len(selected_ids), SQLITE_PARAMETERS_PER_QUERY):
91 batch = selected_ids[start:start + SQLITE_PARAMETERS_PER_QUERY]
92 placeholders = ",".join("?" for _ in batch)
93 for (raw,) in db.execute(
94 "SELECT body FROM items WHERE kind='assertions' "
95 "AND json_extract(body,'$.object') IS NOT NULL AND ("
96 f"json_extract(body,'$.subject') IN ({placeholders}) OR "
97 f"json_extract(body,'$.object') IN ({placeholders})) ORDER BY id LIMIT ?",
98 batch + batch + [max_nodes * 2]):
99 edge = json.loads(raw)
100 if len(links) >= max_nodes * 2:
101 break
102 needed = {edge["subject"], edge["object"]} - nodes.keys()
103 if len(nodes) + len(needed) > max_nodes:
104 continue
105 for key in needed:
106 row = db.execute("SELECT body FROM items WHERE kind='objects' AND id=?", (key,)).fetchone()
107 if row:
108 nodes[key] = json.loads(row[0])
109 if edge["subject"] in nodes and edge["object"] in nodes:
110 links[edge["id"]] = edge
111 return list(nodes.values()), list(links.values()), counts

172def _ontology_overview(ontology):
173 """Keep only accepted ontology types and witnessed type-level relations on the canvas."""
174 roots = ontology.get("object_roots", [])
175 derived = [item for item in ontology.get("object_types", [])
176 if item.get("category") == "business_type"]
177 nodes = roots + derived
178 visible = {item["id"] for item in nodes}
179 inheritance = [
180 {"source": item["id"], "target": item["parent"]}
181 for item in derived if item.get("parent") in visible
182 ]
183 relations = []
184 omitted = []
185 for relation in ontology.get("relation_types", []):
186 # A multi-valued domain/range is a set of constraints, not evidence for
187 # every pair in their Cartesian product. Record-level endpoints also
188 # must not appear as business ontology links.
189 if relation.get("category") != "business_relation_type":
190 omitted.append(relation["id"])
191 continue
192 domain, range_ = relation.get("domain") or [], relation.get("range") or []
193 if len(domain) == len(range_) == 1 and domain[0] in visible and range_[0] in visible:
194 relations.append({"id": relation["id"], "source": domain[0],
195 "target": range_[0], "label": relation.get("label") or relation["id"],
196 "parent": relation.get("parent")})
197 else:
198 omitted.append(relation["id"])
199 return {"nodes": nodes, "inheritance": inheritance, "relations": relations,
200 "omitted_relation_type_ids": omitted}

386def _sequence_preview(path, limit, *, key=None, byte_limit=8 * 1024 * 1024):
387 """Read only a bounded YAML prefix; never load a million-binding document."""
388 summary = {"file": path.name, "shown": 0, "truncated": False, "status": "missing"}
389 if not path.exists():
390 return [], summary
391 with path.open("rb") as stream:
392 raw = stream.read(byte_limit + 1)
393 summary.update(status="available", file_bytes=path.stat().st_size,
394 read_bytes=min(len(raw), byte_limit), truncated=len(raw) > byte_limit)
395 text = raw[:byte_limit].decode("utf-8", errors="ignore")
396 loader = UniqueLoader(text)
397 items = []
398
399 def skip_node():
400 event = loader.get_event()
401 depth = int(isinstance(event, (yaml.SequenceStartEvent, yaml.MappingStartEvent)))
402 while depth:
403 event = loader.get_event()
404 if isinstance(event, (yaml.SequenceStartEvent, yaml.MappingStartEvent)):
405 depth += 1
406 elif isinstance(event, (yaml.SequenceEndEvent, yaml.MappingEndEvent)):
407 depth -= 1
408
409 try:
410 loader.get_event() # stream
411 loader.get_event() # document
412 if key is not None:
413 if not isinstance(loader.get_event(), yaml.MappingStartEvent):
414 raise ValueError("Expected a mapping document")
415 while not loader.check_event(yaml.MappingEndEvent):
416 name = loader.construct_object(loader.compose_node(None, None), deep=True)
417 if name == key:
418 break
419 skip_node()
420 else:
421 return [], {**summary, "status": "missing_key"}
422 if not isinstance(loader.get_event(), yaml.SequenceStartEvent):
423 raise ValueError("Expected a sequence")
424 while len(items) < limit and not loader.check_event(yaml.SequenceEndEvent):
425 node = loader.compose_node(None, None)
426 # YAML may implicitly close an incomplete block mapping at EOF.
427 # Discard that tail instead of presenting it as a complete binding.
428 if len(raw) > byte_limit and node.end_mark.index >= len(text.rstrip()):
429 break
430 item = loader.construct_object(node, deep=True)
431 if not isinstance(item, dict):
432 raise ValueError("Expected mapping records")
433 items.append(item)
434 summary["truncated"] |= not loader.check_event(yaml.SequenceEndEvent)
435 except (ValueError, yaml.YAMLError) as exc:
436 # A cut YAML tail is expected. Preserve complete records already parsed.
437 if not summary["truncated"]:
438 summary.update(status="invalid", error=str(exc)[:300])
439 finally:
440 loader.dispose()
441 summary["shown"] = len(items)
442 return items, summary
