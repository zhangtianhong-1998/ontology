1"""Conservative source-subject checks; retrieval topics are not quantities.
2
3This guard can reject a service-purpose quote, or identify a separately quoted
4quantity definition. It never certifies the identity of an entire source row.
5Unknown wording stays unknown and still needs the caller's semantic checks.
6"""

14_SERVICE_NAME = re.compile(
15 r"(?:接口|服务|卡片|图表|仪表盘|展示)(?:名称|标题)|"
16 r"\b(?:api|interface|service|endpoint|card|dashboard|chart|display)[_\s]+(?:name|title|label)\b", re.I)
17_SERVICE_DESCRIPTION = re.compile(
18 r"(?:接口|服务|卡片|展示)(?:说明|描述|用途|功能)|"
19 r"\b(?:api|interface|service|endpoint|card|dashboard)[_\s]+(?:description|purpose|function)\b", re.I)
20_PURPOSE = re.compile(
21 r"^(?:(?:此|该|本)?(?:接口|服务|卡片|图表|页面|API)[，,:：\s]*)?"
22 r"(?:(?:提供|用于|用来|支持)[，,:：\s]*)?(?:查询|检索|展示|显示|返回|获取|调用|渲染)|"
23 r"^(?:(?:this\s+)?(?:api|interface|service|endpoint|card|dashboard|chart)\s+)?"
24 r"(?:(?:is\s+)?(?:used\s+to|designed\s+to)\s+)?"
25 r"(?:queries|query|retrieves|retrieve|fetches|fetch|displays|display|shows|show|"
26 r"returns|return|provides\s+access\s+to)\b|"
27 r"^(?:an?\s+)?(?:api|endpoint|dashboard|chart)\s+(?:for|to)\b", re.I)
28_URL = re.compile(r"^(?:https?|grpc|wss?)://[^\s]+$", re.I)
29_PATH = re.compile(r"^/[A-Za-z0-9_{}:./-]+$")
30_ADDRESS_DECLARATION = re.compile(r"接口地址|服务地址|端点地址|\b(?:api[_\s]*url|endpoint|request[_\s]*path)\b", re.I)
31_AUTH_DECLARATION = re.compile(r"(?:^|[_\s])(?:auth|authentication)(?:[_\s]|$)|认证|鉴权", re.I)
32_HTTP_METHODS = frozenset(("GET", "POST", "PUT", "PATCH", "DELETE", "HEAD", "OPTIONS"))
33_PROTOCOLS = frozenset(("HTTP", "HTTPS", "HTTP/1.1", "HTTP/2", "GRPC", "REST", "WEBSOCKET"))

36def _independent_definition(value, label):
37 """Recognize a literal named definition, without requiring arithmetic."""
38 if not label.strip():
39 return False
40 # A definition must start with its own quantity name. A topic appearing
41 # inside a service-purpose sentence is insufficient.
42 return bool(re.match(
43 r"^\s*[‘“\"']?" + re.escape(label.strip()) + r"[’”\"']?\s*"
44 r"(?:(?:的)?(?:定义为|是指|指的是|表示|等于|为|是|指)|[:：=]|"
45 r"\b(?:is(?:\s+defined\s+as)?|means|refers\s+to|represents|equals|counts|measures)\b)"
46 r"\s*\S.{3,}$", value, re.I))

49def quantity_subject_assessment(data, record, label, classification_quote):
50 """Assess a quoted Metric/Measure definition using one privacy-safe row.
51
52 ``blocks_exact_record_identity`` also blocks a mixed service record whose
53 independent quantity field would require a separate projection contract.
54 No table-name/root-hint classifier or model is invoked here.
55 """
56 table = getattr(data, "tables", {}).get(record.get("table"), {})
57 excluded = set(table.get("semantic_excluded_columns") or ())
58 columns = {c["column_name"]: c for c in table.get("columns", ())
59 if c["column_name"] not in excluded and not is_sensitive_column(c)
60 and not _AUTH_DECLARATION.search(c["column_name"] + " " + str(c.get("column_comment") or ""))}
61 entries = []
62 for role, values in record.get("fields", {}).items():
63 for item in values:
64 if item.get("column") in columns and item.get("value") is not None:
65 entries.append({**item, "role": role, "value": str(item["value"]).strip()})
66 quote = str(classification_quote or "").strip()
67 selected = [e for e in entries if e["role"] in ("description", "formula")
68 and not e.get("truncated") and quote and e["value"] == quote]
69 result = {"status": "unknown", "blocks_quantity_definition": False,
70 "blocks_exact_record_identity": False, "requires_definition_projection": False,
71 "record_identity_supported": False, "reasons": [], "evidence": [],
72 "selected_definition_columns": [e["column"] for e in selected]}
73 if not selected:
74 result["reasons"].append("complete_source_definition_quote_not_available")
75 # The classification quote may belong to a different exact record.
76 # Still inspect this record's own subject; another record cannot lend
77 # its quantity identity to a service row with the same label.
78
79 def declaration(column):
80 meta = columns[column]
81 return column.replace("_", " ") + " " + str(meta.get("column_comment") or "")
82
83 def evidence(kind, column, *, value=None, shape=None):
84 item = {"kind": kind, "table": record.get("table"), "record_id": record.get("record_id"),
85 "row_number": record.get("row_number"), "column": column,
86 "source_declaration": declaration(column)}
87 if value is not None:
88 item["quote"] = value
89 if shape is not None:
90 # URL credentials/query strings and unrelated payloads are never
91 # added to semantic context merely to establish service shape.
92 item["value_shape"] = shape
93 result["evidence"].append(item)
94
95 selected_service_name = False
96 for entry in entries:
97 if (entry["role"] in ("name", "alias") and not entry.get("truncated")
98 and entry["value"] == str(label).strip()
99 and _SERVICE_NAME.search(declaration(entry["column"]))):
100 selected_service_name = True
101 evidence("service_name_declaration", entry["column"], value=entry["value"])
102
103 # Do not load excluded/authentication columns even transiently. A single
104 # exact source row supplies shape clues omitted from the semantic card.
105 raw = {entry["column"]: entry["value"] for entry in entries}
106 row_number = record.get("row_number")
107 if (hasattr(data, "db") and table.get("sql_name") and type(row_number) is int
108 and row_number > 0 and columns):
109 names = list(columns)
110 row = data.db.execute(
111 f"SELECT {', '.join(qi(name) for name in names)} FROM {qi(table['sql_name'])} "
112 "WHERE __r2_row=? LIMIT 1", [row_number]).fetchone()
113 if row is not None:
114 raw.update({name: str(value).strip() for name, value in zip(names, row) if value is not None})
115
116 shapes = set()
117 for column, value in raw.items():
118 shape = ("service_address" if _URL.fullmatch(value) or
119 (_PATH.fullmatch(value) and _ADDRESS_DECLARATION.search(declaration(column))) else
120 "http_method" if value.upper() in _HTTP_METHODS else
121 "service_protocol" if value.upper() in _PROTOCOLS else None)
122 if shape:
123 shapes.add(shape)
124 evidence("service_structure", column, shape=shape)
125 structural_service = "service_address" in shapes and bool(shapes & {"http_method", "service_protocol"})
126 service_descriptions = []
127 description_columns = {e["column"] for e in entries
128 if e["role"] == "description" and not e.get("truncated")}
129 for column, value in raw.items():
130 if (_PURPOSE.search(value) and not _independent_definition(value, str(label))
131 and (column in description_columns or structural_service
132 or _SERVICE_DESCRIPTION.search(declaration(column)))):
133 service_descriptions.append({"column": column, "value": value})
134 evidence("operation_purpose_text", column, value=value)
135 selected_service_declaration = any(_SERVICE_DESCRIPTION.search(declaration(e["column"])) for e in selected)
136 independent = bool(selected) and _independent_definition(quote, str(label))
137 service_subject = selected_service_name or bool(service_descriptions) or (
138 structural_service and selected_service_declaration)
139
140 if independent:
141 result["status"] = "independent_quantity_definition"
142 result["reasons"].append("named_independent_definition_field")
143 for entry in selected:
144 evidence("quantity_definition_candidate", entry["column"], value=entry["value"])
145 if service_subject:
146 result.update(requires_definition_projection=True, blocks_exact_record_identity=True)
147 result["reasons"].append("mixed_record_requires_definition_projection")
148 elif service_subject:
149 result.update(status="service_subject", blocks_quantity_definition=True,
150 blocks_exact_record_identity=True)
151 result["reasons"].append("service_or_display_topic_is_not_quantity_definition")
152 elif selected:
153 result["reasons"].append("quantity_subject_not_established_by_this_guard")
154 return result
