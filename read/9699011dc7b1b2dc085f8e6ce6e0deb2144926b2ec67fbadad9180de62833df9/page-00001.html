25def _reference_sql(column_sql, source_path):
26 """Read an object-key JSON path; never evaluate supplied SQL or array indices.
27
28 Match the existing RelationPlan.source_path execution contract. Invalid
29 JSON, missing keys, containers and booleans are absent references, not text
30 values produced by serializing the entire document.
31 """
32 if not source_path:
33 return column_sql
34 if (not isinstance(source_path, list) or len(source_path) > 16
35 or any(not isinstance(key, str) or not key for key in source_path)):
36 raise ValueError("source_path must contain 1..16 nonempty object keys")
37 path = "$" + "".join("." + json.dumps(key, ensure_ascii=False) for key in source_path)
38 literal = "'" + path.replace("'", "''") + "'"
39 document = f"CASE WHEN json_valid({column_sql}) THEN {column_sql} ELSE '{{}}' END"
40 return (f"CASE WHEN json_type({document}, {literal}) = 'VARCHAR' "
41 f"THEN json_extract_string({document}, {literal}) END")

44def json_scalar_paths(value, *, max_depth=16, omissions=None):
45 """Yield complete nonempty string leaves from JSON objects, without flattening lists."""
46 try:
47 parsed = json.loads(value) if isinstance(value, str) else value
48 except (ValueError, TypeError):
49 return
50 def visit(node, path):
51 if len(path) > max_depth:
52 if omissions is not None:
53 omissions["paths_beyond_depth_limit"] = omissions.get("paths_beyond_depth_limit", 0) + 1
54 return
55 if isinstance(node, dict):
56 for key, item in sorted(node.items()):
57 if isinstance(key, str) and key:
58 yield from visit(item, [*path, key])
59 elif path and isinstance(node, str) and node.strip():
60 yield path, node
61 if isinstance(parsed, dict):
62 yield from visit(parsed, [])

65def discover_json_reference_candidates(data, *, source_tables=None, target_fields=None,
66 max_rows_per_table=128, max_paths=64,
67 max_target_fields=128, max_candidates=128,
68 max_document_chars=65536):
69 """Recall bounded JSON leaf→column links. Equality alone has no semantic status.
70
71 Samples discover paths only. validate_candidate/association_match_sql then
72 execute each path against the complete snapshot, including ambiguous keys.
73 Names need not match: observed exact scalar values perform the recall.
74 """
75 limits = (max_rows_per_table, max_paths, max_target_fields, max_candidates, max_document_chars)
76 if any(type(value) is not int or value <= 0 for value in limits):
77 raise ValueError("JSON reference discovery limits must be positive integers")
78 visible = {(table, item["column"]) for table, info in data.tables.items()
79 for item in classify_columns(info)
80 if item["role"] not in {"sensitive", "empty", "audit_time", "audit_metadata"}
81 and item["column"] not in info.get("semantic_excluded_columns", ())}
82 sources = sorted(set(source_tables) if source_tables is not None else data.tables)
83 if any(table not in data.tables for table in sources):
84 raise ValueError("Unknown JSON reference source table")
85 targets = sorted(set(map(tuple, target_fields)) if target_fields is not None else visible)
86 if any(field not in visible for field in targets):
87 raise ValueError("Unknown or invisible JSON reference target field")
88 omitted_targets = max(0, len(targets) - max_target_fields)
89 targets = targets[:max_target_fields]
90 paths, rows_unexamined, oversized, path_omissions = {}, 0, 0, {}
91 for table in sources:
92 columns = sorted(column for owner, column in visible if owner == table)
93 if not columns:
94 continue
95 info = data.tables[table]
96 total = data.db.execute(f"SELECT count(*) FROM {qi(info['sql_name'])}").fetchone()[0]
97 rows_unexamined += max(0, total - max_rows_per_table)
98 rows = data.db.execute(f"SELECT __r2_row, {', '.join(map(qi, columns))} "
99 f"FROM {qi(info['sql_name'])} ORDER BY __r2_row LIMIT ?",
100 [max_rows_per_table]).fetchall()
101 for values in rows:
102 for column, raw in zip(columns, values[1:]):
103 if not isinstance(raw, str) or not raw.lstrip().startswith("{"):
104 continue
105 if len(raw) > max_document_chars:
106 oversized += 1
107 continue
108 for path, value in json_scalar_paths(raw, omissions=path_omissions):
109 key = table, column, tuple(path)
110 entry = paths.setdefault(key, {"values": set(), "examples": []})
111 entry["values"].add(value)
112 if len(entry["examples"]) < 3:
113 entry["examples"].append({"row_number": values[0], "raw_document": raw,
114 "source_path": path, "value": value})
115 selected_paths = sorted(paths.items())[:max_paths]
116 candidates = []
117 for (table, column, path), entry in selected_paths:
118 values = sorted(entry["values"])
119 for target_table, target_column in targets:
120 if target_table == table:
121 continue
122 info = data.tables[target_table]
123 placeholders = ",".join("?" for _ in values)
124 shared = data.db.execute(f"SELECT DISTINCT {qi(target_column)} "
125 f"FROM {qi(info['sql_name'])} WHERE {qi(target_column)} IN ({placeholders})",
126 values).fetchall()
127 if not shared:
128 continue
129 source = {"table": table, "field": column, "source_path": list(path)}
130 target = {"table": target_table, "field": target_column}
131 candidates.append({"candidate_id": "json_reference:" + digest([data.snapshot_id, source, target])[:24],
132 "source": source, "target": target, "retrieval_channels": ["json_scalar_value_overlap"],
133 "numeric_overlap_only": all(_numeric_literal(row[0]) for row in shared),
134 "shared_sample_value_count": len(shared), "source_declared_pk": False,
135 "target_declared_pk": target_column in info.get("pk", []),
136 "examples": entry["examples"], "semantic_relation": "unresolved"})
137 omitted = max(0, len(candidates) - max_candidates)
138 return {"candidates": candidates[:max_candidates], "coverage": {
139 "paths_discovered": len(paths), "paths_examined": len(selected_paths),
140 "paths_not_examined": max(0, len(paths) - max_paths),
141 "source_rows_not_examined": rows_unexamined, "oversized_documents": oversized,
142 "paths_beyond_depth_limit": path_omissions.get("paths_beyond_depth_limit", 0),
143 "target_fields_not_examined": omitted_targets, "candidates_not_selected": omitted,
144 "candidates_emitted": min(len(candidates), max_candidates), "llm_calls": 0,
145 "scope": "sampled_json_paths_with_full_target_value_lookup",
146 "partial": bool(rows_unexamined or oversized or omitted_targets or omitted or path_omissions or len(paths) > max_paths)}}

176def _metadata_pair(source, target, data):
177 source_table, source_col = source
178 target_table, target_col = target
179 if source == target:
180 return False
181 source_definition = next((c.get("column_comment") or ""
182 for c in data.tables[source_table].get("columns", [])
183 if c.get("column_name") == source_col), "").casefold()
184 target_label = (data.tables[target_table].get("table_name", target_table.rsplit(".", 1)[-1])
185 + "." + target_col).casefold()
186 if target_label in source_definition or (target_table + "." + target_col).casefold() in source_definition:
187 return True
188 a, b = _stem(source_col), _stem(target_col)
189 if source_col.casefold() == target_col.casefold() and a:
190 return True
191 target_name = _table_stem(data.tables[target_table])
192 if _tokens(target_col) in (("id",), ("code",), ("key",), ("uuid",)):
193 return bool(target_name and a and (a == target_name or a == target_name[:-1]))
194 return bool(a and b and a == b and len(a) > 0)

211def _value_index_eligibility(data, field, max_length):
212 """Use observed shape to exclude clear payloads, not unfamiliar column names.
213
214 Missing/incomplete profiles stay eligible: absence of profile evidence must
215 not silently turn an opaque business key into an excluded field. The
216 expensive DISTINCT index is still bounded by the caller's field budget.
217 """
218 table, column = field
219 info = data.tables[table]
220 role = next((item for item in classify_columns(info)
221 if item["column"] == column), {})
222 tokens = set(_tokens(column))
223 named_key = bool(role.get("join_eligible") or
224 tokens & {"id", "uuid", "guid", "key", "code", "ref"})
225 profile = next((item for item in info.get("profiles", ())
226 if item.get("column") == column), None)
227 if not profile or profile.get("scan_scope") != "full_input":
228 return True, "profile_unavailable_or_incomplete"
229 usable = profile.get("usable_count")
230 if not isinstance(usable, int) or usable <= 0:
231 return True, "value_shape_unknown"
232 shortest = profile.get("min_chars_usable")
233 if isinstance(shortest, int) and shortest > max_length:
234 return False, "all_values_exceed_index_length"
235 if not named_key:
236 longest = profile.get("max_chars_usable")
237 short = profile.get("short_text_count")
238 if (isinstance(longest, int) and longest > max_length
239 and isinstance(short, int) and short / usable < 0.05):
240 return False, "predominantly_long_text"
241 # A high-cardinality integer may be an undocumented business code.
242 # Keep it in bounded value recall and mark numeric coincidence as risk.
243 return True, "short_or_unknown_value_shape"

246def _compatible_value_pair(data, source, target, *, nonnumeric_shared=False):
247 """Value equality is worth recalling only for plausible field roles."""
248 if source == target:
249 return False
250 if _metadata_pair(source, target, data):
251 return True
252 source_tokens, target_tokens = set(_tokens(source[1])), set(_tokens(target[1]))
253 source_key = bool(source_tokens & {"id", "key", "code", "ref", "field"})
254 target_key = bool(target_tokens & {"id", "key", "code", "ref", "field"})
255 if source_key and target_key:
256 return True
257 if source_tokens & {"name", "alias", "synonym"} and target_tokens & {"name", "alias", "synonym"}:
258 return True
259 # Distinct, nonnumeric raw values can recall differently named fields.
260 # This is only a technical lead; validate_candidate checks full rows and
261 # leaves its semantic_relation unresolved.
262 return source[0] != target[0]

265def _full_value_pairs(data, fields, max_length, max_fanout, *, on_step=None,
266 on_note=None):
267 """Build an on-disk DISTINCT dictionary; never load all values into Python.
268
269 DuckDB groups every eligible value from the imported snapshot. The only
270 loss here is explicitly reported: blank/oversized values and values shared
271 by too many fields. Matching source rows are checked later by
272 ``validate_candidate``; the dictionary is solely a recall structure.
273 """
274 index_name = qi("__r2_discovery_values")
275 data.db.execute(f"DROP TABLE IF EXISTS {index_name}")
276 data.db.execute(f"CREATE TEMP TABLE {index_name} (field_no INTEGER, value VARCHAR)")
277 oversized = {}
278 try:
279 for field_no, (table, column) in enumerate(fields):
280 if on_note is not None:
281 on_note(f"完整 distinct {table}.{column} 查询中")
282 sql_table, sql_column = _checked_field(data, table, column)
283 value_sql = (f"{sql_column} IS NOT NULL AND {sql_column} <> '' "
284 f"AND trim({sql_column}) <> ''")
285 oversized[f"{table}.{column}"] = data.db.execute(
286 f"SELECT count(*) FROM (SELECT DISTINCT {sql_column} "
287 f"FROM {sql_table} WHERE {value_sql} AND length({sql_column}) > ?)",
288 [max_length]).fetchone()[0]
289 data.db.execute(
290 f"INSERT INTO {index_name} SELECT ?, {sql_column} "
291 f"FROM {sql_table} WHERE {value_sql} AND length({sql_column}) <= ? "
292 f"GROUP BY {sql_column}", [field_no, max_length])
293 if on_step is not None:
294 on_step(f"完整 distinct {table}.{column}")
295 common = data.db.execute(
296 f"SELECT count(*) FROM (SELECT value FROM {index_name} "
297 f"GROUP BY value HAVING count(*) > ?)", [max_fanout]).fetchone()[0]
298 matches = data.db.execute(f"""
299 WITH usable AS (
300 SELECT value FROM {index_name} GROUP BY value
301 HAVING count(*) BETWEEN 2 AND ?
302 )
303 SELECT a.field_no, b.field_no, count(*) AS common_values,
304 bool_and(regexp_full_match(a.value, '[+-]?[0-9]+(\\.[0-9]+)?')) AS numeric_only
305 FROM usable v
306 JOIN {index_name} a ON a.value = v.value
307 JOIN {index_name} b ON b.value = v.value AND a.field_no < b.field_no
308 GROUP BY a.field_no, b.field_no
309 """, [max_fanout]).fetchall()
310 pairs = {}
311 for left, right, count, numeric in matches:
312 a, b = fields[left], fields[right]
313 for source, target in ((a, b), (b, a)):
314 if _compatible_value_pair(data, source, target,
315 nonnumeric_shared=not bool(numeric)):
316 pairs[(source, target)] = (count, bool(numeric))
317 return pairs, common, oversized
318 finally:
319 data.db.execute(f"DROP TABLE IF EXISTS {index_name}")

322def _conditioned_pairs(data, fields, max_values):
323 """Recall type-discriminated references as *candidates*, not foreign keys.
324
325 ``business_type/business_id`` and ``source_type/source_field`` are generic
326 patterns. Only discriminator literals observed in this snapshot are used;
327 a matching table/column token is a recall hint, never a semantic decision.
328 """
329 available = set(fields)
330 proposals = []
331 truncated = []
332 for table, type_field in fields:
333 if not type_field.endswith("_type"):
334 continue
335 prefix = type_field[:-5]
336 reference = next((name for name in (prefix + "_id", prefix + "_field")
337 if (table, name) in available), None)
338 if reference is None:
339 continue
340 source_table, source_type = _checked_field(data, table, type_field)
341 raw_values = data.db.execute(
342 f"SELECT DISTINCT {source_type} FROM {source_table} "
343 f"WHERE {source_type} IS NOT NULL AND trim({source_type}) <> '' "
344 f"ORDER BY {source_type} LIMIT ?", [max_values + 1]).fetchall()
345 if len(raw_values) > max_values:
346 truncated.append(f"{table}.{type_field}")
347 for (literal,) in raw_values[:max_values]:
348 cue = re.match(r"[a-zA-Z]+", literal)
349 if not cue:
350 continue
351 cue = cue.group().casefold()
352 for target_table, info in sorted(data.tables.items()):
353 target_tokens = _tokens(info.get("table_name", target_table.rsplit(".", 1)[-1]))
354 if target_table == table or cue not in target_tokens:
355 continue
356 columns = set(info["column_names"])
357 if reference.endswith("_id"):
358 targets = info.get("pk") or []
359 else:
360 targets = [name for name in info["column_names"]
361 if name == cue + "_name" or
362 set(_tokens(name)) & {"code", "key"}]
363 for target_field in targets:
364 if (target_table, target_field) not in available:
365 continue
366 proposals.append({
367 "source": (table, reference),
368 "target": (target_table, target_field),
369 "suggested_selector": {type_field: literal},
370 # Scope is learned from joint values below, never from
371 # a fixed table/column vocabulary such as dim/member.
372 "suggested_scope_bindings": {},
373 "target_role_priority": (0 if target_tokens[-1] == cue else
374 1 if target_tokens[-1] in
375 {"def", "definition", "detail"} else 2),
376 })
377 return proposals, truncated

380def _fair_source_order(items, source_of, quality):
381 """Interleave source tables and fields before applying a global budget."""
382 tables = defaultdict(lambda: defaultdict(list))
383 for item in items:
384 table, field = source_of(item)
385 tables[table][field].append(item)
386 for fields in tables.values():
387 for values in fields.values():
388 values.sort(key=quality, reverse=True)
389 uses = defaultdict(int)
390 ordered = []
391 while tables:
392 for table in sorted(list(tables)):
393 fields = tables[table]
394 field = min(fields, key=lambda f: (uses[(table, f)], quality(fields[f][-1]), f))
395 ordered.append(fields[field].pop())
396 uses[(table, field)] += 1
397 if not fields[field]:
398 del fields[field]
399 if not fields:
400 del tables[table]
401 return ordered

404def _conditioned_order(items, shared):
405 """Cover source, discriminator branch and target table before variants.
406
407 Existing value evidence orders families within each branch; it does not
408 validate the branch or erase the original selector, scope or risk flags.
409 """
410 grouped = defaultdict(lambda: defaultdict(lambda: defaultdict(list)))
411 def quality(item):
412 count = shared.get((item['source'], item['target']), (0, False))[0]
413 evidence = item.get('condition_evidence', {})
414 rows, matches = evidence.get('branch_rows', 0), evidence.get('branch_matches', 0)
415 return (-evidence.get('scoped_unique_gain', 0),
416 not bool(count or matches), -(matches / rows if rows else 0),
417 -count, item['target_role_priority'], item['target'])
418 for item in items:
419 branch = tuple(sorted(item['suggested_selector'].items()))
420 grouped[item['source']][branch][item['target'][0]].append(item)
421 local_rank = {}
422 for branches in grouped.values():
423 branch_queues = {}
424 for branch, families in branches.items():
425 for values in families.values():
426 values.sort(key=quality)
427 family_order = sorted(families, key=lambda key: (quality(families[key][0]), key))
428 queue = []
429 # A second field of one target table cannot consume another table's turn.
430 for offset in range(max(map(len, families.values()), default=0)):
431 queue.extend(families[key][offset] for key in family_order
432 if offset < len(families[key]))
433 branch_queues[branch] = queue
434 rank = 0
435 for offset in range(max(map(len, branch_queues.values()), default=0)):
436 for branch in sorted(branch_queues):
437 if offset < len(branch_queues[branch]):
438 local_rank[id(branch_queues[branch][offset])] = rank
439 rank += 1
440 return _fair_source_order(items, lambda item: item['source'],
441 lambda item: (local_rank[id(item)],))
