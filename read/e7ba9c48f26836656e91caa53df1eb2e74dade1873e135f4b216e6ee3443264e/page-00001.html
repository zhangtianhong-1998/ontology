63def test_40_fields_use_two_aggregate_queries_and_bounded_samples(tmp_path):
64 db = duckdb.connect()
65 columns = [f"c{i}" for i in range(40)]
66 db.execute("CREATE TABLE src AS SELECT row_number() OVER () AS __r2_row, * FROM (SELECT " + ", ".join(f"'v{i}' AS {name}" for i, name in enumerate(columns)) + ")")
67 counted = _CountingDb(db)
68 data = SimpleNamespace(root=tmp_path, db=counted, tables={"demo.wide": {
69 "sql_name": "src", "column_names": columns, "rows": 1,
70 "csv_path": str(tmp_path / "wide.csv"), "csv_hash": "synthetic",
71 }})
72 profiles = profile_fields(data, {"sql_columns_per_group": 64, "input_scope": "sample"})["demo.wide"]
73 assert len(profiles) == 40
74 assert len([q for q in counted.queries if "count(*)" in q]) == 2
75 assert len([q for q in counted.queries if "ORDER BY __r2_row LIMIT 2048" in q]) == 2
76 assert all(p["input_scope"] == "sample" and p["sample_exhaustive_in_input"] for p in profiles)
77 with pytest.raises(ValueError, match="positive"):
