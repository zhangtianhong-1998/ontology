1"""Replay definition membership against its complete source template and row.
2
3Shared semantics establish a type membership, never equality of records or
4inheritance of their reference relations. Callers must resolve edges separately.
5"""
6from .storage import digest, qi

9def replay_definition_membership(data, member, template, row, business_type, cache):
10 """Return proven member columns; reject stale or incomplete saved contracts."""
11 if (not template or member.get("snapshot_id") != data.snapshot_id
12 or member.get("status") != "definition_template_match"
13 or template.get("type_id") != business_type.id
14 or member.get("type_id", member.get("object_type_id")) != business_type.id
15 or member.get("table") != template.get("table")
16 or member.get("template_id") != template.get("id")
17 or member.get("entity_identity_claim") is not False
18 or member.get("relationship_inheritance") is not False):
19 raise ValueError("Definition membership contract is missing, stale or claims record identity")
20 table_name = template["table"]
21 table = data.tables.get(table_name)
22 fields = template.get("semantic_fields") or []
23 if (table is None or not fields or any(
24 field.get("column") not in table["column_names"] or not field.get("role")
25 for field in fields)):
26 raise ValueError("Definition template semantic columns are missing")
27 semantics = [[field["role"], field["column"], field.get("value")] for field in fields]
28 signature = digest(semantics)
29 verification = member.get("verification") or {}
30 columns = [field["column"] for field in fields]
31 if (template.get("semantic_fingerprint") != signature
32 or template["id"] != "definition_template:" + digest([
33 data.snapshot_id, table_name, business_type.id, semantics])[:24]
34 or verification.get("method") != "complete_original_source_field_equality"
35 or verification.get("semantic_fingerprint") != signature
36 or verification.get("columns") != columns):
37 raise ValueError("Definition membership complete semantic fingerprint differs")
38 if (row.get("__r2_row") != member.get("row_number")
39 or data.record_id(table_name, row) != member.get("record_id")
40 or [[field["role"], field["column"], row.get(field["column"])] for field in fields] != semantics):
41 raise ValueError("Definition member source identity or complete semantics changed")
42 cache_key = (template["id"], signature)
43 if cache_key not in cache:
44 number = template.get("representative_row_number")
45 if type(number) is not int or number < 1:
46 raise ValueError("Definition template representative row is missing")
47 cursor = data.db.execute(f"SELECT * FROM {qi(table['sql_name'])} WHERE __r2_row=?", [number])
48 values = cursor.fetchone()
49 representative = dict(zip([part[0] for part in cursor.description], values)) if values else {}
50 record_id = template.get("representative_record_id")
51 if (not representative or data.record_id(table_name, representative) != record_id
52 or [[field["role"], field["column"], representative.get(field["column"])]
53 for field in fields] != semantics):
54 raise ValueError("Definition template representative changed")
55 # The accepted type must still cite this representative's own complete
56 # definition, not merely an arbitrary row with equal strings.
57 supported = False
58 for prop in business_type.source_properties:
59 if prop.role not in {"description", "formula"} or prop.source_table != table_name:
60 continue
61 for evidence_id in prop.evidence_ids:
62 evidence = data.evidence.get(evidence_id) or {}
63 ref = evidence.get("source_ref") or {}
64 if (evidence_id in template.get("evidence_ids", [])
65 and evidence.get("origin") == "observed_record"
66 and not evidence.get("raw_fragment_truncated")
67 and ref.get("snapshot_id") == data.snapshot_id
68 and ref.get("record_id") == record_id
69 and ref.get("table") == table_name
70 and ref.get("column") == prop.source_column
71 and representative.get(prop.source_column) == evidence.get("raw_fragment")):
72 supported = True
73 if not supported:
74 raise ValueError("Definition template has no accepted representative source proof")
75 cache[cache_key] = True
76 return list(dict.fromkeys(columns))
