6def source_record_type(table_name, table):
7 comment = (table.get("table_comment") or "").strip()
8 # A table comment can identify a definition table. Mentioning a metric or
9 # dimension as a reference is not enough to classify its rows as that type.
10 described_root = None
11 for phrase, root in (("指标定义", "Metric"), ("指标业务定义", "Metric"),
12 ("度量定义", "Measure"), ("维度定义", "Dimension"),
13 ("术语定义", "Term")):
14 if phrase in comment:
15 described_root = root
16 break
17 type_id = "source_record_type:" + table_name
18 definition = "源表记录结构" + ("：" + comment if comment else "：" + table_name)
19 return DerivedType(
20 id=type_id, parent="GeneralObject", label=comment or table_name,
21 definition=definition, evidence_ids=["schema:" + table_name],
22 category="source_record_type",
23 ), {"basis": "explicit_table_comment" if described_root else "source_table_only",
24 "described_business_root_hint": described_root,
25 "business_concept_inferred": False}
