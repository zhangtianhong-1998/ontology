35def test_only_observed_tuples_are_emitted_with_conflict_and_lineage(tmp_path):
36 data = _dataset(tmp_path)
37 try:
38 output = build_fact_observation_candidates(data)
39 reports = {item["table"]: item for item in output["tables"]}
40 fact = reports["fruit.fruit_profit_fact"]
41 assert fact["row_purpose"] == "business_fact"
42 assert fact["scan_scope"] == "full_input_for_selected_value_columns"
43 assert fact["emitted_candidates"] == 3
44 assert fact["omitted_observed_tuples"] == 0
45 assert fact["value_fields"][0]["complete_rows"] == 4
46 assert fact["value_fields"][0]["observed_tuples"] == 3
47 assert all(item["candidate_status"] == "candidate_only"
48 and item["business_type_binding"] == "unresolved"
49 for item in fact["candidates"])
50 same = next(item for item in fact["candidates"]
51 if item["observed_value"] == "100")
52 assert same["source_row_count"] == 2
53 assert same["duplicate_source_rows"] == 1
54 assert (same["source_row_min"], same["source_row_max"]) == (1, 2)
55 assert same["source_row_range_is_exact_membership"] is False
56 assert same["distinct_values_at_selected_coordinates"] == 2
57 assert "multiple_values_at_selected_coordinates" in same["ambiguity"]
58 # APPLE/EAST and BANANA/SOUTH occur; the unobserved cross-products do not.
59 assert not any(item["coordinate_values"].get("fruit_code") == "APPLE"
60 and item["coordinate_values"].get("region_code") == "SOUTH"
61 for item in fact["candidates"])
62 assert reports["fruit.fruit_metric_definition"]["row_purpose"] == "definition_data"
63 assert reports["fruit.fruit_metric_definition"]["candidates"] == []
64 assert reports["fruit.opaque_table"]["row_purpose"] == "unresolved"
65 assert reports["fruit.opaque_table"]["reason"] == "mixed_or_insufficient_row_purpose_evidence"
66 assert output["coverage"]["unresolved_tables"] == 1
67 finally:
68 data.close()

133def test_accepted_dimension_link_ranks_candidate_before_column_order(tmp_path):
134 root = tmp_path / "input"
135 _table(root, "many_dim_fact", {
136 "fact_id": "记录 ID", "fruit_code": "水果编码", "region_code": "地区编码",
137 "channel_code": "渠道编码", "opaque_slot": "", "period": "会计期",
138 "profit_amount": "销售利润金额",
139 }, [{"fact_id": "1", "fruit_code": "APPLE", "region_code": "EAST",
140 "channel_code": "ONLINE", "opaque_slot": "D1",
141 "period": "2025Q1", "profit_amount": "100"},
142 {"fact_id": "2", "fruit_code": "BANANA", "region_code": "SOUTH",
143 "channel_code": "OFFLINE", "opaque_slot": "D2",
144 "period": "2025Q2", "profit_amount": "120"}],
145 pk="fact_id")
146 _table(root, "dim_definition", {"dim_code": "维度编码"},
147 [{"dim_code": name} for name in ("EAST", "SOUTH", "ONLINE", "OFFLINE", "D1", "D2")],
148 pk="dim_code")
149 work = tmp_path / "work"
150 work.mkdir()
151 data = Dataset(root, work)
152 try:
153 good = {"source_table": "fruit.many_dim_fact", "source_column": "region_code",
154 "target_table": "fruit.dim_definition", "target_column": "dim_code",
155 "target_root_type": "Dimension", "status": "accepted",
156 "witness_snapshot_id": data.snapshot_id, "full_input_verified": True,
157 "evidence_ids": ["record:checked"]}
158 bad = {**good, "source_column": "channel_code", "witness_snapshot_id": "stale"}
159 technical = {
160 "source": {"table": "fruit.many_dim_fact", "field": "channel_code"},
161 "target": {"table": "fruit.dim_definition", "field": "dim_code"},
162 "status": "checked_technical", "snapshot_id": data.snapshot_id,
163 "verification": {"scan_scope": "full_input", "checks": {
164 "eligible_references": 2, "unique_matches": 2, "missing_scope": 0}},
165 }
166 opaque_technical = {**technical, "source": {
167 "table": "fruit.many_dim_fact", "field": "opaque_slot"}}
168 output = build_fact_observation_candidates(
169 data, max_dimension_columns=2, accepted_dimension_links=[good, bad],
170 verified_technical_links=[technical, opaque_technical])
171 fact = next(item for item in output["tables"]
172 if item["table"] == "fruit.many_dim_fact")
173 assert fact["selected_columns"]["dimensions"][0] == "region_code"
174 assert fact["selected_columns"]["accepted_dimension_fields_selected"] == ["region_code"]
175 assert fact["selected_columns"]["technical_link_fields_selected"] == ["channel_code"]
176 assert "opaque_slot" in fact["verified_technical_link_fields"]
177 assert "opaque_slot" not in fact["selected_columns"]["dimensions"]
178 assert fact["selected_columns"]["omitted_dimensions"]
179 assert fact["coordinate_grain_status"] == "unresolved_omitted_coordinate_columns"
180 assert "coordinate_columns_omitted_by_limit" in fact["candidates"][0]["ambiguity"]
181 assert output["coverage"]["accepted_dimension_links_ignored"] == 1
182 assert output["coverage"]["verified_technical_links_ignored"] == 0
183 assert output["coverage"]["partial"] is True
184 finally:
185 data.close()
