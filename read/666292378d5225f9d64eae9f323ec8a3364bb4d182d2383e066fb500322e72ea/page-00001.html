17class SmallDataset:
18 def __init__(self, tables):
19 self.db = duckdb.connect(":memory:")
20 self.snapshot_id = "synthetic-snapshot"
21 self.tables = {}
22 for index, (name, spec) in enumerate(tables.items()):
23 columns, rows, pk = spec
24 sql_name = f"src_{index}"
25 fields = ", ".join(f"{qi(col)} VARCHAR" for col in columns)
26 self.db.execute(f"CREATE TABLE {qi(sql_name)} (__r2_row BIGINT, {fields})")
27 placeholders = ",".join("?" for _ in range(len(columns) + 1))
28 self.db.executemany(f"INSERT INTO {qi(sql_name)} VALUES ({placeholders})",
29 [(i, *row) for i, row in enumerate(rows, 1)])
30 self.tables[name] = {
31 "schema": name.split(".", 1)[0],
32 "table_name": name.split(".", 1)[1],
33 "sql_name": sql_name, "column_names": columns,
34 "pk": pk, "foreign_keys": [],
35 }
36
37 def close(self):
38 self.db.close()

41def find_candidate(result, source_table, source_field, target_table, target_field):
42 return next(c for c in result["candidates"]
43 if c["source"] == {"table": source_table, "field": source_field}
44 and c["target"] == {"table": target_table, "field": target_field})
