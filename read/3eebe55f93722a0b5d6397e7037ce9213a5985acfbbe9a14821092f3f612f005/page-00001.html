22def validate_bundle_options(options):
23 """Validate one shared budget contract before import or model work."""
24 if type(options.get("template_projection_enabled", False)) is not bool:
25 raise ValueError("template_projection_enabled must be a boolean")
26 defaults = {"max_concept_bundles": 12, "max_relation_bundles": 12,
27 "max_seeds_per_table": 8, "lexical_top_k": 8,
28 "max_candidates_per_bundle": 3, "max_bundle_bytes": 16000,
29 "max_vector_cards": 20000, "vector_top_k": 8,
30 "max_joint_pairs_per_rule": 2, "max_pattern_seed_pool": 1000,
31 "max_joined_context_records": 3, "max_joined_context_rules": 64,
32 "max_joined_context_value_chars": 2048}
33 limits = {key: options.get(key, value) for key, value in defaults.items()}
34 # Risk exploration is bounded against packet capacity, not the potentially
35 # much larger number of technically checked selector variants.
36 limits["max_exploration_rules"] = options.get(
37 "max_exploration_rules", (limits["max_relation_bundles"] + 6) // 7
38 if type(limits["max_relation_bundles"]) is int else 0)
39 if any(type(value) is not int or value < 0 for value in limits.values()):
40 raise ValueError("Instance bundle limits must be nonnegative integers")
41 if not 0 <= limits["max_joint_pairs_per_rule"] <= 2:
42 raise ValueError("max_joint_pairs_per_rule must be 0..2")
43 if limits["max_bundle_bytes"] < 1000 or limits["max_candidates_per_bundle"] < 1:
44 raise ValueError("Bundle bytes and candidate count are too small")
45 if not 1 <= limits["max_pattern_seed_pool"] <= 10000:
46 raise ValueError("max_pattern_seed_pool must be 1..10000")
47 if limits["max_concept_bundles"] > 10000:
48 raise ValueError("max_concept_bundles must be 0..10000")
49 seed_window_index = options.get("seed_window_index", 0)
50 if type(seed_window_index) is not int or not 0 <= seed_window_index <= 1000000:
51 raise ValueError("seed_window_index must be an integer in 0..1000000")
52 return limits

62def _exact_evidence_signature(card):
63 """Canonical *literal* evidence identity, never fuzzy concept identity.
64
65 The field names and complete values are retained. Thus equal labels with
66 different definitions, units, scopes or reference columns are not skipped.
67 This intentionally under-deduplicates across heterogeneous schemas.
68 """
69 fields = card.get("fields", {})
70 if any(entry.get("truncated") for entries in fields.values() for entry in entries):
71 # A preview can hide a distinguishing suffix. Never coalesce two
72 # truncated cards by their visible prefix alone.
73 return "incomplete_evidence:" + str(card.get("card_id") or card.get("record_id")
74 or digest(card))
75 canonical = [(role, sorted((entry["column"], _norm(str(entry["value"])),
76 bool(entry.get("truncated")))
77 for entry in entries))
78 for role, entries in sorted(fields.items())]
79 return "evidence:" + digest([card.get("table"), card.get("kind"),
80 card.get("root_hint"), canonical])[:32]

84 from .semantic_cards import _structured_semantic_candidate

118def _vector_pool(index, limits, embedding, enabled):
119 report = {"enabled": bool(enabled), "status": "disabled", "cards_total": 0,
120 "cards_encoded": 0, "patterns_total": 0,
121 "source_cards_total": 0, "model_sha256": None}
122 if not enabled:
123 return None, report
124 if embedding is None:
125 report["status"] = "model_unavailable"
126 return None, report
127 total = index.db.execute("SELECT count(DISTINCT pattern_id) FROM cards WHERE kind='definition'").fetchone()[0]
128 source_cards = index.db.execute("SELECT count(*) FROM cards WHERE kind='definition'").fetchone()[0]
129 report.update(cards_total=total, patterns_total=total, source_cards_total=source_cards)
130 report["model_sha256"] = embedding.model_sha256
131 if total > limits["max_vector_cards"]:
132 report["status"] = "over_pattern_cap"
133 return None, report
134 cards = []
135 window_index = 0
136 while len(cards) < total:
137 window = index.pattern_window(1000, window_index=window_index)
138 cards.extend(window["patterns"])
139 window_index += 1
140 if not window["patterns"]:
141 raise ValueError("Pattern window ended before its declared total")
142 if not cards:
143 report["status"] = "no_definition_cards"
144 return None, report
145 vectors = embedding.documents([_card_text(card) for card in cards])
146 queries = ({card["card_id"]: vector for card, vector in zip(
147 cards, embedding.queries([_card_text(card) for card in cards]))}
148 if hasattr(embedding, "queries") else {})
149 report.update(status="complete", cards_encoded=len(cards))
150 return (cards, vectors, queries), report

153def _vector_hits(seed, pool, embedding, limit):
154 if pool is None:
155 return []
156 cards, vectors, queries = pool
157 query = queries.get(seed["card_id"])
158 if query is None:
159 query = embedding.query(_card_text(seed))
160 # Filter after an enlarged local pool; exact ties keep indexed card order.
161 ranked = top_cosine(vectors, query, min(len(cards), max(limit * 8, 32)))
162 result = []
163 for position, score in ranked:
164 card = cards[position]
165 if not _compatible(seed, card):
166 continue
167 result.append((card, score))
168 if len(result) >= limit:
169 break
170 return result

173def _select_candidates(seed, lexical, vectors, limit):
174 fused = {}
175 for rank, card in enumerate(lexical, 1):
176 if not _compatible(seed, card):
177 continue
178 item = fused.setdefault(card["card_id"], {"card": card, "channels": [],
179 "ranks": {}, "rrf": 0.0})
180 item["channels"].extend(card["retrieval_channels"])
181 item["ranks"]["lexical"] = rank
182 item["rrf"] += 1 / (60 + rank)
183 for rank, (card, similarity) in enumerate(vectors, 1):
184 item = fused.setdefault(card["card_id"], {"card": card, "channels": [],
185 "ranks": {}, "rrf": 0.0})
186 item["channels"].append("vector")
187 item["ranks"]["vector"] = rank
188 item["cosine"] = similarity
189 item["rrf"] += 1 / (60 + rank)
190 return sorted(fused.values(), key=lambda item: (-item["rrf"], item["card"]["card_id"]))[:limit]

193def _select_seeds(pool, limit, *, population_by_root=None, population_by_table=None):
194 """One floor per available root/table, then sqrt-population fair scheduling.
195
196 Population comes from the *complete indexed definition corpus* when called
197 by the pipeline. A truncated retrieval pool only bounds which cards are
198 available; it is never used as the population estimate in that path.
199 """
200 if limit <= 0 or not pool:
201 return []
202 order = ("Metric", "Measure", "Dimension", "Term", "GeneralObject")
203 root_rank = {root: position for position, root in enumerate(order)}
204 buckets = defaultdict(list)
205 for card in pool:
206 buckets[(card.get("root_hint") or "GeneralObject", card["table"])].append(card)
207 for cards in buckets.values():
208 def quality(card):
209 fields = card.get("fields", {})
210 return (2 * bool(fields.get("description")) + 2 * bool(fields.get("formula"))
211 + bool(fields.get("unit")) + bool(fields.get("alias")))
212 cards.sort(key=lambda card: (-quality(card), card["card_id"]))
213 cards.reverse() # pop() retrieves the best card without shifting lists.
214 fallback_roots = defaultdict(int)
215 fallback_tables = defaultdict(int)
216 for (root, table), cards in buckets.items():
217 fallback_roots[root] += len(cards)
218 fallback_tables[table] += len(cards)
219 roots = sorted({root for root, _ in buckets}, key=lambda root: (root_rank.get(root, len(order)), root))
220 full_roots = population_by_root or fallback_roots
221 full_tables = population_by_table or fallback_tables
222 selected, root_uses, table_uses = [], defaultdict(int), defaultdict(int)
223
224 def choose(bucket):
225 root, table = bucket
226 selected.append(buckets[bucket].pop())
227 root_uses[root] += 1
228 table_uses[table] += 1
229
230 def table_weight(bucket):
231 return sqrt(max(1, full_tables.get(bucket[1], fallback_tables[bucket[1]])))
232
233 def best_table(root, *, unrepresented=False):
234 choices = [key for key, cards in buckets.items()
235 if key[0] == root and cards and (not unrepresented or table_uses[key[1]] == 0)]
236 if not choices:
237 return None
238 return min(choices, key=lambda key: (
239 (table_uses[key[1]] + 1) / table_weight(key), key[1]))
240
241 # A floor is useful even for a small Dimension/Term corpus. When the
242 # budget permits, follow it with one seed from every source table.
243 for root in roots:
244 if len(selected) >= limit:
245 break
246 table = best_table(root)
247 if table:
248 choose(table)
249 for root, table in sorted(buckets, key=lambda key: (
250 root_rank.get(key[0], len(order)), key[0], key[1])):
251 if len(selected) >= limit:
252 break
253 if buckets[(root, table)] and table_uses[table] == 0:
254 choose((root, table))
255
256 while len(selected) < limit:
257 available = [root for root in roots if best_table(root) is not None]
258 if not available:
259 break
260 root = min(available, key=lambda name: (
261 (root_uses[name] + 1) / sqrt(max(1, full_roots.get(name, fallback_roots[name]))),
262 root_rank.get(name, len(order)), name))
263 choose(best_table(root))
264 return selected
