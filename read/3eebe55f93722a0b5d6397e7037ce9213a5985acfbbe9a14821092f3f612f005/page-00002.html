492def _family_ordered_rules(rules, limit):
493 """Cover field families across source/target tables before their variants.
494
495 Numeric overlap remains risky evidence, but is not excluded from the first
496 family pass. Ordering grants a review opportunity, never business validity.
497 """
498 if limit <= 0:
499 return []
500 families = defaultdict(list)
501 for rule in rules:
502 if rule["status"] in ("checked_technical", "observed_subset"):
503 families[_rule_family(rule)].append(rule)
504 def variant_order(rule):
505 return (-_reference_priority(rule),
506 json.dumps(rule.get("selector") or {}, sort_keys=True, ensure_ascii=False),
507 rule["rule_id"])
508 for variants in families.values():
509 variants.sort(key=variant_order)
510 # Within each source, visit each target table before repeating a target.
511 table_pairs = defaultdict(lambda: defaultdict(list))
512 for family, variants in families.items():
513 table_pairs[family[0]][family[1]].append(family)
514 for targets in table_pairs.values():
515 for target, keys in targets.items():
516 targets[target] = deque(sorted(keys, key=lambda key: (variant_order(families[key][0]), key)))
517 target_order = {
518 source: deque(sorted(targets, key=lambda target:
519 (variant_order(families[targets[target][0]][0]), target)))
520 for source, targets in table_pairs.items()}
521 sources = sorted(table_pairs, key=lambda source:
522 (variant_order(families[table_pairs[source][target_order[source][0]][0]][0]), source))
523 family_order = []
524 while sources:
525 remaining_sources = []
526 for source in sources:
527 target = target_order[source].popleft()
528 family_order.append(table_pairs[source][target].popleft())
529 if table_pairs[source][target]:
530 target_order[source].append(target)
531 if target_order[source]:
532 remaining_sources.append(source)
533 sources = remaining_sources
534 # A heavily enumerated selector must not take a second slot while another
535 # field family still has no representative. No candidate is discarded.
536 active = deque(deque(families[key]) for key in family_order)
537 chosen = []
538 while active and len(chosen) < limit:
539 variants = active.popleft()
540 chosen.append(variants.popleft())
541 if variants:
542 active.append(variants)
543 return chosen

546def _rule_quality_tier(rule, data=None):
547 """Schema hints affect scheduling only; unknown numeric joins stay explorable."""
548 tables = getattr(data, "tables", {})
549 source, target = rule.get("source", {}), rule.get("target", {})
550 source_info = tables.get(source.get("table"), {})
551 field = source.get("field", "")
552 column = next((item for item in source_info.get("columns", ())
553 if item["column_name"] == field), {})
554 comment = str(column.get("column_comment") or "")
555 ordinal = bool(re.search(
556 r"排序|序号|\b(?:display|sort)[ _]+(?:order|index)\b|"
557 r"\b(?:row|sequence)[ _]+(?:number|index|no)\b", field + " " + comment, re.I))
558 # A generic "reference ID" comment does not identify *this* target. In
559 # dense numeric domains that hint otherwise promotes every coincident ID
560 # join to the high-priority queue, bypassing the bounded risk allowance.
561 target_table = target.get("table", "")
562 target_field = target.get("field", "")
563 qualified_targets = [f"{name}.{target_field}" for name in
564 (target_table, target_table.rsplit(".", 1)[-1])
565 if name and target_field]
566 reference = ("declared_fk" in rule.get("retrieval_channels", ()) or any(
567 re.search(r"(?<![A-Za-z0-9_.])" + re.escape(name) + r"(?![A-Za-z0-9_.])", comment, re.I)
568 for name in qualified_targets))
569 source_pk = field in source_info.get("pk", ()) or bool(rule.get("source_declared_pk"))
570 numeric = bool(rule.get("numeric_overlap_only"))
571 if ordinal or (numeric and source_pk and not reference):
572 return "risk_exploration"
573 checks = (rule.get("verification") or {}).get("checks") or {}
574 eligible, outside = (checks.get("eligible_references", 0),
575 checks.get("outside_eligible_references", 0))
576 # Contrast is measured on complete checked rows, not on column spelling,
577 # a guessed discriminator label, or the mere existence of a selector.
578 conditional = bool(rule.get("selector") and eligible and outside
579 and checks.get("selector_unknown", 0) == 0
580 and checks.get("unique_matches", 0) / eligible
581 > checks.get("outside_matched_references", 0) / outside)
582 if (checks.get("unique_matches", 0) > 0
583 and (reference or (not source_pk and conditional))):
584 return "structural_reference"
585 return "risk_exploration" if numeric else "value_match"

588def _round_robin_rules(rules, limit, *, data=None, max_exploration_rules=None):
589 """Weighted review opportunities: structural/value/risk = 4/2/1.
590
591 Each tier covers its distinct join families before repeating variants.
592 A bounded risk queue prevents surplus budget being filled by ID collisions.
593 """
594 tiers = defaultdict(list)
595 for rule in rules:
596 if rule["status"] in ("checked_technical", "observed_subset"):
597 tiers[_rule_quality_tier(rule, data)].append(rule)
598 queues = {tier: deque(_family_ordered_rules(population, len(population)))
599 for tier, population in tiers.items()}
600 if max_exploration_rules is not None and "risk_exploration" in queues:
601 queues["risk_exploration"] = deque(list(queues["risk_exploration"])[:max_exploration_rules])
602 cycle = ("structural_reference", "value_match", "risk_exploration",
603 "structural_reference", "value_match", "structural_reference", "structural_reference")
604 chosen = []
605 while any(queues.values()) and len(chosen) < limit:
606 for tier in cycle:
607 if queues.get(tier):
608 chosen.append(queues[tier].popleft())
609 if len(chosen) >= limit:
610 break
611 return chosen

614def _reference_priority(rule):
615 """Prioritize observed unique witnesses without inferring field semantics."""
616 verification = rule.get("verification") or {}
617 checks = verification.get("checks") or {}
618 eligible = checks.get("eligible_references", 0)
619 unique = checks.get("unique_matches", 0)
620 score = 4 * min(1, unique / eligible) if eligible > 0 else 0
621 score += verification.get("scan_scope") == "full_input"
622 score += (checks.get("target_rows_with_complete_key", 0) > 0
623 and checks.get("target_max_multiplicity") == 1)
624 # Prefer an actually exercised conditional branch, not an untested label.
625 score += bool(rule.get("selector") and eligible > 0
626 and checks.get("selector_true", 0) > 0
627 and checks.get("selector_false", 0) > 0
628 and checks.get("selector_unknown", 0) == 0)
629 score -= 0.25 * bool(rule.get("numeric_overlap_only"))
630 return score
