1"""Run one local experiment in a terminal; write its exit state, never poll it.
2
3Credentials are read in memory. Environment paths and model names may be logged;
4secret values and provider exception bodies are deliberately not printed.
5"""
6from __future__ import annotations
7
8import argparse
9import asyncio
10import datetime as dt
11import hashlib
12import json
13import os
14import signal
15from pathlib import Path
16import subprocess
17import sys
18import time
19import traceback
20
21from dotenv import dotenv_values
22from ontology_r2.pipeline import build, load_config
23
24
25def main():
26 parser = argparse.ArgumentParser(description=__doc__)
27 parser.add_argument('--config', type=Path, required=True)
28 parser.add_argument('--output', type=Path, required=True)
29 parser.add_argument('--env-file', type=Path)
30 parser.add_argument('--expect-model', default='deepseek-v4-flash')
31 args = parser.parse_args()
32 if args.output.exists():
33 parser.error('Output exists; choose a new directory')
34 config = load_config(args.config)
35 environment = args.env_file or Path(config.get('env_file', '.env'))
36 values = dotenv_values(environment) if environment.is_file() else {}
37 if config.get('llm', {}).get('mode') != 'mock':
38 for suffix in ('MODEL', 'BASE_URL', 'API_KEY'):
39 key = 'ONTOLOGY_LLM_' + suffix
40 value = values.get(key) or values.get('LLM_' + suffix) or os.environ.get(key)
41 if not value:
42 parser.error(f'Missing {key}; configure the local environment file')
43 os.environ[key] = value
44 if os.environ['ONTOLOGY_LLM_MODEL'] != args.expect_model:
45 parser.error('Configured model does not match --expect-model')
46 root = Path(__file__).resolve().parents[2]
47 def git(*command):
48 return subprocess.check_output(['git', *command], cwd=root, text=True).strip()
49 snapshot = {'commit': git('rev-parse', 'HEAD'), 'branch': git('branch', '--show-current'),
50 'working_tree_clean': not bool(git('status', '--porcelain')),
51 'config': str(args.config.resolve()),
52 'config_sha256': hashlib.sha256(args.config.read_bytes()).hexdigest(),
53 'started_at': dt.datetime.now(dt.timezone.utc).isoformat(),
54 'pid': os.getpid(), 'model': os.environ.get('ONTOLOGY_LLM_MODEL', 'mock')}
55 print(json.dumps({'experiment_started': snapshot, 'output': str(args.output.resolve())}, ensure_ascii=False), flush=True)
56 started = time.monotonic()
57 # Some model SDKs swallow coroutine cancellation. Termination must not
58 # silently continue to the next paid request; keep the last checkpoint.
59 def stop(signum, _frame):
60 if args.output.exists():
61 (args.output / 'experiment_snapshot.json').write_text(json.dumps(snapshot, ensure_ascii=False, indent=2))
62 (args.output / 'process_exit.json').write_text(json.dumps({
63 'status': 'interrupted', 'exit_code': 128 + signum,
64 'signal': signum, 'elapsed_seconds': round(time.monotonic() - started, 3),
65 'finished_at': dt.datetime.now(dt.timezone.utc).isoformat()}, indent=2))
66 print(f'Experiment stopped by signal {signum}; last checkpoint retained.', flush=True)
67 os._exit(128 + signum)
68 signal.signal(signal.SIGINT, stop)
69 signal.signal(signal.SIGTERM, stop)
70 exit_code = 1
71 result = {}
72 try:
73 result = asyncio.run(build(config, args.output))
74 exit_code = 1 if result.get('status') == 'failed' else 0
75 print(json.dumps({key: result.get(key) for key in ('status', 'input_tables', 'input_records', 'elapsed_seconds', 'partial_reasons', 'llm')}, ensure_ascii=False), flush=True)
76 except KeyboardInterrupt:
77 result = {'status': 'interrupted'}
78 exit_code = 130
79 except Exception as exc:
80 result = {'status': 'failed', 'error_type': type(exc).__name__,
81 'frames': [{'file': frame.filename, 'line': frame.lineno, 'function': frame.name}
82 for frame in traceback.extract_tb(exc.__traceback__)]}
83 print(json.dumps(result), flush=True)
84 finally:
85 if args.output.exists():
86 (args.output / 'experiment_snapshot.json').write_text(json.dumps(snapshot, ensure_ascii=False, indent=2))
87 (args.output / 'process_exit.json').write_text(json.dumps({
88 'status': result.get('status'), 'exit_code': exit_code,
89 'elapsed_seconds': round(time.monotonic() - started, 3),
90 'finished_at': dt.datetime.now(dt.timezone.utc).isoformat()}, indent=2))
91 print(f'Experiment exited with code {exit_code}.', flush=True)
92 if (args.output / 'viewer.html').exists():
93 print(f'Viewer: {args.output / "viewer.html"}', flush=True)
94 return exit_code

98 sys.exit(main())
