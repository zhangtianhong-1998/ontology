10def _case(formula="利润 = 收入 - 成本", *, duplicate=False, scope=None):
11 evidence, types, concepts, alignments = {}, [], [], []
12 definitions = [("profit", "利润", formula), ("revenue", "收入", None), ("cost", "成本", None)]
13 if duplicate:
14 definitions.append(("gross-revenue", "收入", None))
15 for key, name, expression in definitions:
16 record_id = "record:" + key
17 props = []
18 for role, value in (("name", name), ("formula", expression)):
19 if value is None:
20 continue
21 evidence_id = key + ":" + role
22 evidence[evidence_id] = {"origin": "observed_record", "raw_fragment": value,
23 "raw_fragment_truncated": False, "source_ref": {
24 "snapshot_id": "snap", "record_id": record_id, "table": "definitions", "column": role}}
25 props.append({"role": role, "source_table": "definitions", "source_column": role,
26 "evidence_ids": [evidence_id]})
27 types.append(DerivedType(id=key, parent="Metric" if key == "profit" else "Measure",
28 label=name, definition=name, evidence_ids=[key + ":name"], category="business_type",
29 applicability_scope=(scope or {}) if key == "revenue" else {}, source_properties=props))
30 concepts.append({"id": "concept:" + key, "ontology_type_id": key})
31 alignments.append({"mapping_kind": "exact", "source_record_id": record_id, "concept_id": "concept:" + key})
32 return SimpleNamespace(snapshot_id="snap", evidence=evidence, tables={}), BuildPlan(object_types=types), {
33 "concepts": concepts, "record_alignments": alignments}
