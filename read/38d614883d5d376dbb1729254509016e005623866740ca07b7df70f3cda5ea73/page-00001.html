1"""Locate reviewed source identities without rescanning a table for every plan.
2
3Locators are hints: actual rows must reproduce the snapshot-local record ID.
4Only requested IDs are cached; incomplete evidence falls back once per table.
5"""
6from collections import Counter, defaultdict
7
8
9class WitnessLocator:
10 def __init__(self, data, plans):
11 self.data = data
12 self.wanted = defaultdict(set)
13 self.evidence_ids = defaultdict(set)
14 self.locations = {}
15 self.stats = {"evidence_rows_fetched": 0, "evidence_records_verified": 0,
16 "fallback_tables": 0, "fallback_rows_scanned": 0,
17 "witness_rows_fetched": 0, "witness_rows_delivered": 0,
18 "locator_issues": {}}
19 self.issues = Counter()
20 for plan in plans:
21 if plan.witnessed_pairs and plan.witness_snapshot_id != data.snapshot_id:
22 raise ValueError("Relation witness belongs to another input snapshot")
23 self.wanted[plan.source_table].update(pair.source_record_id for pair in plan.witnessed_pairs)
24 self.evidence_ids[plan.source_table].update(plan.evidence_ids)
25
26 def _issue(self, reason):
27 self.issues[reason] += 1
28 self.stats["locator_issues"] = dict(sorted(self.issues.items()))
29
30 def _fetched(self, phase, count):
31 # A batch may be fetched before the executor stops at its budget. Keep
32 # database return counts separate from records delivered or examined.
33 self.stats[phase + "_rows_fetched"] += count
34
35 def resolve(self, table):
36 if table in self.locations:
37 return self.locations[table]
38 wanted = self.wanted[table]
39 candidates = defaultdict(set)
40 evidence = getattr(self.data, "evidence", {})
41 for evidence_id in sorted(self.evidence_ids[table]):
42 ref = evidence.get(evidence_id, {}).get("source_ref", {})
43 record_id = ref.get("record_id")
44 if record_id not in wanted:
45 continue
46 if ref.get("table") != table or ref.get("snapshot_id") != self.data.snapshot_id:
47 self._issue("evidence_table_or_snapshot_mismatch")
48 continue
49 number = ref.get("row")
50 if type(number) is not int or number < 1:
51 self._issue("evidence_row_missing_or_invalid")
52 continue
53 if number > self.data.tables[table]["rows"]:
54 self._issue("evidence_row_out_of_range")
55 continue
56 candidates[number].add(record_id)
57 found = {}
58 visited = set()
59 for row in self.data.rows_at(
60 table, candidates, on_fetch=lambda count: self._fetched("evidence", count)):
61 number = row["__r2_row"]
62 visited.add(number)
63 actual = self.data.record_id(table, row)
64 if actual in candidates[number]:
65 found[actual] = number
66 self.issues["evidence_record_id_mismatch"] += len(candidates[number] - {actual})
67 if set(candidates) - visited:
68 self._issue("evidence_row_not_found")
69 self.stats["evidence_records_verified"] += len(found)
70 self.stats["locator_issues"] = dict(sorted((key, count) for key, count in self.issues.items() if count))
71 missing = wanted - set(found)
72 if missing:
73 # One pass serves all plans for this table, including IDs absent from
74 # the snapshot. Cache absence too; later plans must not scan again.
75 self.stats["fallback_tables"] += 1
76 for row in self.data.rows(table):
77 self.stats["fallback_rows_scanned"] += 1
78 actual = self.data.record_id(table, row)
79 if actual in missing:
80 found[actual] = row["__r2_row"]
81 self.locations[table] = found
82 return found
83
84 def rows(self, table, wanted, coverage):
85 """Preserve original source order and pre-budget outside-witness counts."""
86 locations = self.resolve(table)
87 numbers = {locations[record_id]: record_id for record_id in wanted if record_id in locations}
88 yielded = 0
89 for row in self.data.rows_at(
90 table, numbers, on_fetch=lambda count: self._fetched("witness", count)):
91 number = row["__r2_row"]
92 # Recheck at consumption, even though evidence resolution checked it.
93 if self.data.record_id(table, row) != numbers[number]:
94 raise ValueError("Witness locator no longer matches the source snapshot")
95 # Dataset import guarantees contiguous 1-based row numbers. A budget
96 # break sees exactly the nonwitness prefix the old iterator scanned.
97 coverage["outside_witness"] = number - 1 - yielded
98 self.stats["witness_rows_delivered"] += 1
99 yield row
100 yielded += 1
101 if yielded != len(numbers):
102 raise ValueError("Witness locator row disappeared from the source snapshot")
103 coverage["outside_witness"] = self.data.tables[table]["rows"] - yielded
