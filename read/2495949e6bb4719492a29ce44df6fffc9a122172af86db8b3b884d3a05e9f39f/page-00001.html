1"""Publish uniquely bound calculation contracts as typed ontology dependencies."""
2from .calculation_contracts import calculation_relation_errors, enrich_calculation_contracts
3from .models import DerivedType
4from .relation_contract import canonical_relation_id
5from .validation import validate_plan

7
8def compile_calculation_stage(data, profile, plan, groups, index=None):
9 result = enrich_calculation_contracts(data, plan, groups, index)
10 candidate = plan.model_copy(deep=True)
11 # Recompute only this stage's published edges. New accepted definitions can
12 # make an old operand ambiguous; an earlier proof must not survive that change.
13 previous_calculations = {item.id: item for item in candidate.relation_types
14 if item.endpoint_basis == 'calculation_binding'
15 and item.evidence_scope == 'complete_calculation_definition'
16 and item.predicate_name == 'calculation_dependency'}
17 existing = {item.id: item for item in candidate.relation_types
18 if item.id not in previous_calculations}
19 published_ids = set()
20 proof_updates = []
21 supported_roles = {'minuend', 'subtrahend', 'numerator', 'denominator', 'addend', 'factor'}
22 for dependency in result['dependencies']:
23 role = dependency['operand_role']
24 params = {'operand_role': role} if role in supported_roles else {}
25 relation_id = canonical_relation_id(
26 'depends_on', dependency['source_type_id'], dependency['target_type_id'],
27 predicate_name='calculation_dependency', semantic_parameters=params)
28 dependency['relation_type_id'] = relation_id
29 proposed = DerivedType(
30 id=relation_id, label='calculation_dependency', parent='depends_on',
31 predicate_name='calculation_dependency', semantic_parameters=params,
32 category='business_relation_type',
33 definition='The source calculation uses the target definition as an operand; the original expression preserves its role.',
34 domain=[dependency['source_type_id']], range=[dependency['target_type_id']],
35 endpoint_basis='calculation_binding', evidence_scope='complete_calculation_definition',
36 evidence_ids=dependency['evidence_ids'])
37 if relation_id in existing:
38 old = existing[relation_id]
39 if (old.parent != proposed.parent or old.predicate_name != proposed.predicate_name
40 or old.semantic_parameters != proposed.semantic_parameters
41 or old.domain != proposed.domain or old.range != proposed.range):
42 raise ValueError('Existing calculation relation has a conflicting signature')
43 if relation_id in published_ids:
44 proposed.evidence_ids = sorted(set(old.evidence_ids + proposed.evidence_ids))
45 if old.endpoint_basis != proposed.endpoint_basis or old.evidence_scope != proposed.evidence_scope:
46 proof_updates.append({'relation_type_id': relation_id,
47 'previous_endpoint_basis': old.endpoint_basis, 'previous_evidence_scope': old.evidence_scope,
48 'previous_evidence_ids': old.evidence_ids,
49 'new_endpoint_basis': proposed.endpoint_basis, 'new_evidence_scope': proposed.evidence_scope})
50 existing[relation_id] = proposed
51 else:
52 existing[relation_id] = proposed
53 published_ids.add(relation_id)
54 retracted = []
55 current_complete = {item['source_type_id'] for item in result['calculations']
56 if item['status'] == 'accepted'}
57 for relation_id in sorted(previous_calculations.keys() - published_ids):
58 old = previous_calculations[relation_id]
59 retracted.append({'relation_type_id': relation_id,
60 'previous_endpoint_basis': old.endpoint_basis, 'previous_evidence_scope': old.evidence_scope,
61 'previous_evidence_ids': old.evidence_ids,
62 'reasons': calculation_relation_errors(data, old, candidate.object_types)
63 or ['not_reproduced_by_current_definition_bindings']})
64 if not old.domain or old.domain[0] not in current_complete:
65 result['coverage']['partial'] = True
66 result['coverage']['retracted_calculation_relations'] = len(retracted)
67 candidate.relation_types = list(existing.values())
68 errors = validate_plan(candidate, data, profile)
69 if errors:
70 raise ValueError('Invalid calculation contracts: ' + '; '.join(errors))
71 return {**result, 'plan': candidate, 'proof_updates': proof_updates,
72 'retracted_relations': retracted}
