16def _candidate(column, value, role="formula"):
17 return {"column": column, "role": role, "status": "source_verified_role_candidate",
18 "semantic_status": "unjudged", "schema_evidence_id": f"schema:sample:{column}",
19 "observations": [{"column": column, "row_number": 1,

23@pytest.mark.parametrize("name,comment,value,expected", [

30def test_declared_formula_fragment_retains_original_source_evidence(name, comment, value, expected):
31 table = {"columns": [{"column_name": name, "column_comment": comment}]}
32 candidate = _candidate(name, value)
33 before = deepcopy(candidate)
34 fragment = classify_formula_fragment(table, candidate)
35 assert candidate == before
36 assert fragment["role"] == fragment["proposed_role"] == "formula"
37 assert fragment["effective_role"] == expected
38 assert fragment["formula_status"] == "fragment"
39 assert fragment["observations"] == before["observations"]
40 assert fragment["schema_evidence_id"] == before["schema_evidence_id"]
