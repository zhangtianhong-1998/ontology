109def scope_mock_plan(response, unit):
110 """Project the explicitly hand-authored whole-fixture plan into a unit."""
111 result = deepcopy(response)
112 result["tables"] = [t for t in result.get("tables", []) if t["table"] == unit]
113 result["relations"] = [r for r in result.get("relations", []) if r["source_table"] == unit]
114 needed = {t["object_type"] for t in result["tables"]} | {r["predicate"] for r in result["relations"]}
115 all_types = result.get("object_types", []) + result.get("relation_types", [])
116 while True:
117 expanded = needed | {t["parent"] for t in all_types if t["id"] in needed}
118 if expanded == needed:
119 break
120 needed = expanded
121 for key in ("object_types", "relation_types"):
122 result[key] = [t for t in result.get(key, []) if t["id"] in needed]
123 result["knowledge_questions"] = []
124 return result

249 def reserve_calls(self, count, *, stage):
250 """Keep part of the shared budget for later stages, including retries.
251
252 Cached answers remain usable because they do not pass through admission.
253 This is a per-run ceiling, not extra calls or a promise to spend the tail.
254 """
255 total = self.config.get("max_calls", 100)
256 validate_call_reservation(count, total)

259 "shared_call_ceiling": ceiling, "calls_before": self.calls,

262 self.call_reservations.append(entry)
263 try:
264 yield
265 finally:
266 entry["calls_after"] = self.calls
267 self._call_ceilings.pop()

269 def admit(self, request):
270 raw = json.dumps(request, ensure_ascii=False, default=str).encode()
271 if len(raw) > self.config.get("max_input_bytes", 100000):
272 raise InputBudgetExceeded(len(raw), self.config.get("max_input_bytes", 100000))
273 reserved = len(raw) + self.config.get("max_output_tokens", 4096)
274 if self.calls >= self.config.get("max_calls", 100) or self.reserved_tokens + reserved > self.config.get("max_reserved_tokens", 2000000):
275 raise BudgetExceeded("LLM call/token reservation budget exhausted")
276 for ceiling in self._call_ceilings:
277 if self.calls >= ceiling["shared_call_ceiling"]:
278 ceiling["blocked_admissions"] += 1
279 raise BudgetExceeded(
280 f"Stage {ceiling['stage']} reached its shared call ceiling; "
281 f"{ceiling['reserved_calls']} calls are reserved for later stages")
282 self.calls += 1
283 self.reserved_tokens += reserved
