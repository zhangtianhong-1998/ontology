1-- Synthetic design probes. No user data, LLM call or semantic acceptance.
2CREATE TEMP TABLE unrelated_left AS SELECT i::VARCHAR AS code FROM range(1, 4) t(i);
3CREATE TEMP TABLE unrelated_right AS SELECT i::VARCHAR AS code FROM range(1, 4) t(i);
4
5CREATE TEMP TABLE partial_source AS
6SELECT i AS row_id,
7 CASE WHEN i <= 20 THEN 'linked' ELSE 'local' END AS kind,
8 'K' || i::VARCHAR AS ref
9FROM range(1, 1001) t(i);
10CREATE TEMP TABLE partial_target AS
11SELECT 'K' || i::VARCHAR AS code FROM range(1, 21) t(i);
12
13CREATE TEMP TABLE scoped_source(ns VARCHAR, ref VARCHAR);
14INSERT INTO scoped_source VALUES ('north', 'A'), (NULL, 'A');
15CREATE TEMP TABLE scoped_target(ns VARCHAR, code VARCHAR);
16INSERT INTO scoped_target VALUES ('north', 'A'), ('south', 'A');
17
18CREATE TEMP TABLE duplicate_target(code VARCHAR);
19INSERT INTO duplicate_target VALUES ('A'), ('A');
20
21CREATE TEMP TABLE skewed_source AS
22SELECT CASE WHEN i < 1000 THEN 'A' ELSE 'Z' END AS ref FROM range(1, 1001) t(i);
23CREATE TEMP TABLE skewed_target(code VARCHAR);
24INSERT INTO skewed_target VALUES ('A');
25
26CREATE TEMP TABLE raw_codes(code VARCHAR);
27INSERT INTO raw_codes VALUES ('001'), ('1');
28CREATE TEMP TABLE tuple_keys(a VARCHAR, b VARCHAR);
29INSERT INTO tuple_keys VALUES ('a|b', 'c'), ('a', 'b|c');
30
31CREATE TEMP TABLE nested_source AS
32SELECT i AS row_id,
33 CASE WHEN i <= 980 THEN '{"note":"independent"}'
34 ELSE '{"binding":{"code":"K' || (i - 980)::VARCHAR || '"}}' END AS payload
35FROM range(1, 1001) t(i);
36
37CREATE TEMP TABLE design_measurements AS
38SELECT 'unrelated_numeric_overlap' AS name,
39 (SELECT count(*)::DOUBLE FROM unrelated_left l SEMI JOIN unrelated_right r USING(code)) / 3 AS value
40UNION ALL SELECT 'whole_column_containment',
41 (SELECT count(*)::DOUBLE FROM partial_source s SEMI JOIN partial_target t ON s.ref=t.code) / 1000
42UNION ALL SELECT 'conditional_containment',
43 (SELECT count(*)::DOUBLE FROM partial_source s SEMI JOIN partial_target t ON s.ref=t.code WHERE s.kind='linked') /
44 (SELECT count(*) FROM partial_source WHERE kind='linked')
45UNION ALL SELECT 'unscoped_target_matches',
46 (SELECT count(*) FROM scoped_source s JOIN scoped_target t ON s.ref=t.code WHERE s.ns='north')
47UNION ALL SELECT 'scoped_target_matches',
48 (SELECT count(*) FROM scoped_source s JOIN scoped_target t ON s.ref=t.code AND s.ns=t.ns WHERE s.ns='north')
49UNION ALL SELECT 'unknown_scope_records',
50 (SELECT count(*) FROM scoped_source WHERE ns IS NULL)
51UNION ALL SELECT 'duplicate_target_multiplicity',
52 (SELECT count(*) FROM duplicate_target WHERE code='A')
53UNION ALL SELECT 'skewed_row_hit_ratio',
54 (SELECT count(*)::DOUBLE FROM skewed_source s SEMI JOIN skewed_target t ON s.ref=t.code) / 1000
55UNION ALL SELECT 'skewed_distinct_containment',
56 (SELECT count(DISTINCT ref)::DOUBLE FROM skewed_source s SEMI JOIN skewed_target t ON s.ref=t.code) /
57 (SELECT count(DISTINCT ref) FROM skewed_source)
58UNION ALL SELECT 'raw_code_matches', (SELECT count(*) FROM raw_codes WHERE code='001')
59UNION ALL SELECT 'numeric_normalized_matches', (SELECT count(*) FROM raw_codes WHERE try_cast(code AS BIGINT)=1)
60UNION ALL SELECT 'unsafe_concatenated_keys', (SELECT count(DISTINCT a || '|' || b) FROM tuple_keys)
61UNION ALL SELECT 'safe_tuple_keys', (SELECT count(DISTINCT (a,b)) FROM tuple_keys)
62UNION ALL SELECT 'json_path_present',
63 (SELECT count(*) FROM nested_source WHERE json_extract_string(payload, '$.binding.code') IS NOT NULL)
64UNION ALL SELECT 'json_path_matched',
65 (SELECT count(*) FROM nested_source s SEMI JOIN partial_target t ON json_extract_string(s.payload, '$.binding.code')=t.code)
66UNION ALL SELECT 'empty_denominator_ratio', 0.0 / NULLIF(0,0);
67
68SELECT name, value FROM design_measurements ORDER BY name;
