15@contextmanager
16def _case(values, *, roles=None, comments=None, excluded=(), table='arbitrary.changed_source'):
17 comments = comments or {}
18 roles = roles or {'name': ['n'], 'description': ['d']}
19 db = duckdb.connect(':memory:')
20 db.execute('CREATE TABLE src (__r2_row BIGINT, ' + ', '.join(qi(c) + ' VARCHAR' for c in values) + ')')
21 db.execute('INSERT INTO src VALUES (' + ','.join('?' for _ in range(len(values) + 1)) + ')',
22 [1, *values.values()])
23
24 class Recorder:
25 queries = []
26
27 def execute(self, sql, params):
28 self.queries.append(sql)
29 return db.execute(sql, params)
30
31 recorder = Recorder()
32 data = SimpleNamespace(db=recorder, tables={table: {
33 'name': table, 'sql_name': 'src', 'semantic_excluded_columns': list(excluded),
34 'columns': [{'column_name': c, 'column_comment': comments.get(c, '')} for c in values]}})
35 record = {'table': table, 'record_id': 'source:1', 'row_number': 1,
36 'fields': {role: [{'column': c, 'value': values[c], 'truncated': False} for c in cols]
37 for role, cols in roles.items()}}
38 try:
39 yield data, record, recorder
40 finally:
41 db.close()
