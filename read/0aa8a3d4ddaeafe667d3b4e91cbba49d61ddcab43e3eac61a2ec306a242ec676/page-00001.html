37def test_native_identity_empty_filter_equals_python_strip_and_preserves_raw_values():
38 whitespace = discovery._PYTHON_STRIP_CHARACTERS
39 # Fail on a future Python Unicode update instead of silently changing identity.
40 assert set(whitespace) == {chr(i) for i in range(0x110000) if chr(i).isspace()}
41 values = [None, "", whitespace, "1", "01", "１", "A", "a", "\u200b", "\ufeff"]
42 for char in whitespace:
43 values.extend([char, char * 2, char + "A" + char])
44 db = duckdb.connect(":memory:")
45 try:
46 actual = db.execute("SELECT ref FROM unnest(?) t(ref) WHERE trim(ref, ?) <> ''",
47 [values, whitespace]).fetchall()
48 expected = [(key,) for value in values for key in discovery._match_keys(value, "identity")]
49 assert actual == expected
50 finally:
51 db.close()

54@pytest.mark.parametrize("conditional", [False, True])
55def test_native_identity_matches_legacy_links_statistics_and_ambiguity(monkeypatch, conditional):
56 values = [None, "", " ", "\t", "\n", "\u3000", "\x1c", " \t\u00a0",
57 " A ", "A", "a", "1", "01", "１", "\u200b", "duplicate", "missing"]
58 source = [(value, "linked", "north") for value in values]
59 source += [("A", "outside", "north"), ("A", None, "north"),
60 ("A", "linked", "south"), ("A", "linked", None)]
61 target = [(value, "north") for value in values if value != "missing"]
62 target += [("duplicate", "north"), ("A", "south")]
63 options = {"selector": {"kind": "linked"} if conditional else {},
64 "scope_bindings": {"scope": "scope"} if conditional else {}, "transform": "identity"}
65 with rows(source, target) as data:
66 sql, params = discovery.association_match_sql(data, candidate(), **options)
67 assert "__r2_match_keys" not in sql
68 assert data.db.execute("SELECT count(*) FROM duckdb_functions() "
69 "WHERE function_name='__r2_match_keys'").fetchone()[0] == 0
70 native_links = sorted(data.db.execute(sql, params).fetchall())
71 native_check = discovery._validate_transformed_candidate(
72 data, candidate(), sample_limit=100, **options)
73 assert native_check["checks"]["ambiguous_matches"] > 0
74 assert native_check["checks"]["missing_in_input"] > 0
75
76 original = discovery._matching_ctes
77 data.db.create_function("__r2_match_keys", discovery._match_keys,
78 ["VARCHAR", "VARCHAR"], "VARCHAR[]")
79
80 def legacy(*args, **kwargs):
81 query, parameters, group = original(*args, **kwargs)
82 # Restore only the former token CTEs; all conditions, multiplicity
83 # checks and link assembly remain identical to the native path.
84 query, replacements = re.subn(
85 r"source_tokens AS \(.*?\), target_tokens AS \(.*?\), target_keys AS",
86 "source_tokens AS (SELECT *, unnest(__r2_match_keys(ref, 'identity')) AS key "
87 "FROM source_values), target_tokens AS (SELECT *, "
88 "unnest(__r2_match_keys(ref, 'identity')) AS key FROM target_values), target_keys AS",
89 query, flags=re.S)
90 assert replacements == 1
91 return query, parameters, group
92
93 monkeypatch.setattr(discovery, "_matching_ctes", legacy)
94 legacy_sql, legacy_params = discovery.association_match_sql(data, candidate(), **options)
95 assert sorted(data.db.execute(legacy_sql, legacy_params).fetchall()) == native_links
96 assert discovery._validate_transformed_candidate(
97 data, candidate(), sample_limit=100, **options) == native_check
