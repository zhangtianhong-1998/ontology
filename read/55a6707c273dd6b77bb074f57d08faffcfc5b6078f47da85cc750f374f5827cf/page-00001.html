133class Rows:
134 def __init__(self, tables, snapshot="snapshot-A"):
135 self.db = duckdb.connect(":memory:")
136 self.snapshot_id = snapshot
137 self.tables = {}
138 for index, (name, (columns, values)) in enumerate(tables.items()):
139 sql_name = f"src_{index}"
140 fields = ", ".join(f"{qi(column)} VARCHAR" for column in columns)
141 self.db.execute(f"CREATE TABLE {qi(sql_name)} (__r2_row BIGINT, {fields})")
142 self.db.executemany(
143 f"INSERT INTO {qi(sql_name)} VALUES ({','.join('?' for _ in range(len(columns) + 1))})",
144 [(i, *row) for i, row in enumerate(values, 1)],
145 )
146 self.tables[name] = {
147 "sql_name": sql_name,
148 "column_names": columns,
149 "columns": [{"column_name": c, "column_comment": ""} for c in columns],
150 }
151
152 def close(self):
153 self.db.close()
