40def parse_calculation(expression):
41 """Parse a bounded arithmetic DSL and retain symbol positions and roles."""
42 if not isinstance(expression, str) or not expression.strip():
43 return {"formula_type": "empty", "status": "unresolved", "reason": "empty_formula"}
44 if len(expression) > 32768:
45 return {"formula_type": "unparsed", "status": "unresolved", "reason": "formula_size_limit"}
46 normalized = unicodedata.normalize("NFKC", expression).strip()
47 assignment = None
48 if "=" in normalized:
49 if normalized.count("=") != 1:
50 return {"formula_type": "unsupported", "status": "unresolved", "reason": "unsupported_comparison"}
51 assignment, normalized = (part.strip() for part in normalized.split("=", 1))
52 normalized = re.sub(r"(?<![\w.])(\d+(?:\.\d+)?)\s*%", r"(\1 / 100)", normalized)
53 try:
54 tree = ast.parse(normalized, mode="eval")
55 except (SyntaxError, ValueError, RecursionError):
56 return {"formula_type": "natural_language", "status": "unresolved", "reason": "not_parseable_arithmetic"}
57 if sum(1 for _ in ast.walk(tree)) > 1024:
58 return {"formula_type": "unsupported", "status": "unresolved", "reason": "expression_node_limit"}
59 symbols = []
60
61 def encode(node, path="root", role="operand"):
62 position = {"line": node.lineno, "start_utf8_byte": node.col_offset,
63 "end_line": node.end_lineno, "end_utf8_byte": node.end_col_offset}
64 if isinstance(node, ast.Name):
65 occurrence = {"symbol": node.id, "path": path, "operand_role": role,
66 "position_in_normalized_expression": position}
67 symbols.append(occurrence)
68 return {"kind": "symbol", **occurrence}
69 if (isinstance(node, ast.Constant) and type(node.value) in (int, float)
70 and (isinstance(node.value, int) or math.isfinite(node.value))):
71 return {"kind": "constant", "value": node.value}
72 if isinstance(node, ast.BinOp) and type(node.op) in _OPERATORS:
73 roles = _ROLES.get(type(node.op), ("operand", "operand"))
74 return {"kind": "operation", "operator": _OPERATORS[type(node.op)], "arguments": [
75 encode(node.left, path + ".0", roles[0]), encode(node.right, path + ".1", roles[1])]}
76 if isinstance(node, ast.UnaryOp) and isinstance(node.op, (ast.UAdd, ast.USub)):
77 return {"kind": "operation", "operator": "negate" if isinstance(node.op, ast.USub) else "positive",
78 "arguments": [encode(node.operand, path + ".0", role)]}
79 if (isinstance(node, ast.Call) and isinstance(node.func, ast.Name)
80 and node.func.id.casefold() in _FUNCTIONS and not node.keywords and node.args):
81 return {"kind": "function", "operator": node.func.id.casefold(),
82 "arguments": [encode(arg, path + f".{i}", "argument") for i, arg in enumerate(node.args)]}
83 raise ValueError("unsupported_formula_syntax_or_function")
84
85 try:
86 encoded = encode(tree.body)
87 except (ValueError, RecursionError) as exc:
88 return {"formula_type": "unsupported", "status": "unresolved", "reason": str(exc)}
89 return {"formula_type": "symbol_reference" if isinstance(tree.body, ast.Name) else "mathematical",
90 "status": "parsed", "normalized_expression": normalized, "assignment": assignment,
91 "expression_signature": ast.dump(tree, include_attributes=False),
92 "expression_tree": encoded, "symbols": symbols}

95def _root(item, types):
96 parent, seen = item.parent, set()
97 while parent in types and parent not in seen:

103def _internal_parameters(data, target):
104 """Source-proved operand algorithms/units are local, not caller constraints."""
105 if data is None:
106 return set()
107 from .column_role_inference import classify_formula_fragment
108 internal = set()
109 records = {data.evidence.get(eid, {}).get("source_ref", {}).get("record_id")
110 for prop in target.source_properties for eid in prop.evidence_ids}
111 for prop in target.source_properties:
112 for _, proof in _evidence(data, prop, records):
113 raw = proof["raw_fragment"]
114 for key, value in target.definition_parameters.items():
115 if value != raw:
116 continue
117 if prop.role in ("formula", "unit") and key in (prop.role, prop.source_column):
118 internal.add(key)
119 elif prop.role == "scope" and key == prop.source_column:
120 fragment = classify_formula_fragment(data.tables.get(prop.source_table, {}), {
121 "column": prop.source_column, "role": "formula",
122 "status": "source_verified_role_candidate", "observations": [{"value": raw}]})
123 if fragment and fragment["effective_role"] == "calculation_operator":
124 internal.add(key)
125 return internal

128def _target_parameters_supported(source, target, data=None):
129 """A referenced definition may require no parameter absent from its caller.
130
131 Exact agreement is required: a missing period or month/year mismatch is
132 not evidence for an implicit aggregation or grain conversion.
133 """
134 internal = _internal_parameters(data, target)
135 return all(source.definition_parameters.get(key) == value
136 for key, value in target.definition_parameters.items() if key not in internal)

177def _definition_code_names(data, index, record_id, *, include_self_keys=True):
178 """Self keys and explicitly declared symbol aliases can name operands."""
179 if index is None:
180 return []
181 rows = index.db.execute(
182 "SELECT c.table_name,c.fields_json,s.row_number FROM cards c JOIN card_sources s USING(card_id) "
183 "WHERE s.record_id=? AND c.kind='definition'", (record_id,)).fetchall()
184 result = []
185 for row in rows:
186 table_name = row["table_name"]
187 table = data.tables.get(table_name, {})
188 keys = _declared_single_keys(table) if include_self_keys else set()
189 aliases = _declared_symbol_fields(table)
190 entries = [entry for values in json.loads(row["fields_json"]).values() for entry in values]
191 if aliases:
192 fields = sorted(aliases)
193 original = data.db.execute(
194 f"SELECT {', '.join(qi(field) for field in fields)} FROM {qi(table['sql_name'])} WHERE __r2_row=?",
195 [row["row_number"]]).fetchone()
196 if original:
197 entries.extend({"column": field, "value": value, "truncated": False}
198 for field, value in zip(fields, original) if value not in (None, ""))
199 for entry in entries:
200 if entry.get("column") not in keys | aliases or entry.get("truncated"):
201 continue
202 if entry["column"] not in aliases:
203 matches = data.lookup(table_name, ((entry["column"], str(entry["value"])),))
204 if len(matches) != 1 or data.record_id(table_name, matches[0]) != record_id:
205 continue
206 evidence_id = "record:" + digest([data.snapshot_id, record_id, entry["column"]])[:24]
207 source_ref = {"record_id": record_id, "table": table_name, "column": entry["column"],
208 "snapshot_id": data.snapshot_id, "row": row["row_number"]}
209 data.evidence[evidence_id] = {"id": evidence_id, "origin": "observed_record",
210 "raw_fragment": str(entry["value"]), "raw_fragment_truncated": False,
211 "source_ref": source_ref}
212 result.append((str(entry["value"]), {"evidence_id": evidence_id,
213 "source_ref": source_ref, "basis": "declared_formula_symbol_alias"
214 if entry["column"] in aliases else "declared_unique_definition_key"}))
215 return result

218def _formula_binding_tasks(calculations, index=None):
219 """Schedule only missing symbols; retrieval results never assert a binding."""
220 tasks = {}
221 for calculation in calculations:
222 if calculation.get("formula_type") != "mathematical":
223 continue
224 for binding in calculation.get("bindings", []):
225 if binding.get("status") == "bound":
226 continue
227 key = (calculation["source_type_id"], binding["symbol"],
228 digest(calculation.get("applicability_scope", {})),
229 digest(calculation.get("definition_parameters", {})))
230 if key in tasks:
231 tasks[key]["occurrences"].append(binding["path"])
232 continue
233 task = {"id": "formula_binding_task:" + digest(key)[:24],
234 "task": "resolve_formula_operand_definition", "status": "unresolved",
235 "source_type_id": calculation["source_type_id"], "symbol": binding["symbol"],
236 "occurrences": [binding["path"]], "reason": binding.get("reason"),
237 "candidate_type_ids": binding.get("candidate_type_ids", []),
238 "scope": calculation.get("applicability_scope", {}),
239 "definition_parameters": calculation.get("definition_parameters", {}),
240 "evidence_ids": calculation["evidence_ids"], "definition_candidates": [],
241 "proof_required": "complete symbol alias or parameter mapping, compatible unit and scope",
242 "automatic_relation_created": False}
243 if index is not None and hasattr(index, "search") and len(binding["symbol"]) <= 512:
244 hits = index.search(binding["symbol"], limit=8, kind="definition")
245 task["definition_candidates"] = [
246 {key: hit.get(key) for key in
247 ("card_id", "record_id", "table", "name", "root_hint", "retrieval_channels")}
248 for hit in hits]
249 tasks[key] = task
250 return list(tasks.values())

253def enrich_calculation_contracts(data, plan, group_result, index=None):
254 """Return calculation trees and dependency bindings without changing the plan."""
255 if (index is not None and index.coverage.get("snapshot_id") != data.snapshot_id):
256 raise ValueError("Calculation definition index snapshot differs from current input")
257 plan = BuildPlan.model_validate(plan)
258 types = {item.id: item for item in plan.object_types}
259 selected = {key: item for key, item in types.items()
260 if item.category == "business_type" and _root(item, types) in ("Metric", "Measure")}
261 concepts = {item["id"]: item for item in group_result.get("concepts", [])}
262 records_by_type = defaultdict(set)
263 for alignment in group_result.get("record_alignments", []):
264 concept = concepts.get(alignment.get("concept_id"), {})
265 type_id = concept.get("ontology_type_id")
266 if alignment.get("mapping_kind") == "exact" and type_id in selected:
267 records_by_type[type_id].add(alignment["source_record_id"])
268 # A verified projection owns field-level definition evidence without
269 # claiming that its entire physical source row is the projected concept.
270 for type_id, item in selected.items():
271 if item.derivation_kind == "template_projection":
272 for prop in item.source_properties:
273 for evidence_id in prop.evidence_ids:
274 proof = data.evidence.get(evidence_id, {})
275 ref = proof.get("source_ref", {})
276 if (proof.get("origin") == "observed_record"
277 and not proof.get("raw_fragment_truncated")
278 and ref.get("snapshot_id") == data.snapshot_id
279 and ref.get("table") == prop.source_table
280 and ref.get("column") == prop.source_column and ref.get("record_id")):
281 records_by_type[type_id].add(ref["record_id"])
282 registry = defaultdict(lambda: defaultdict(list))
283 formulas = defaultdict(list)
284 variable_fields = defaultdict(set)
285 for projection in group_result.get("template_projections", []):
286 for field in projection.get("field_templates", []):
287 if "{" in field.get("template", ""):
288 variable_fields[projection.get("object_type_id")].add(field["column"])
289 for type_id, item in selected.items():
290 records = records_by_type[type_id]
291 for prop in item.source_properties:
292 for evidence_id, evidence in _evidence(data, prop, records):
293 raw = evidence["raw_fragment"]
294 if prop.role in ("name", "alias"):
295 if prop.source_column in variable_fields[type_id]:
296 continue
297 if item.derivation_kind == "template_projection" and prop.role == "name" and _norm(raw) != _norm(item.label):
298 continue
299 for name in (_aliases(raw) if prop.role == "alias" else [raw]):
300 registry[_norm(name)][type_id].append({"evidence_id": evidence_id,
301 "role": prop.role, "source_ref": evidence["source_ref"]})
302 elif prop.role == "formula":
303 formulas[type_id].append((raw, evidence_id, evidence["source_ref"]))
304 for record_id in records:
305 for name, source in _definition_code_names(
306 data, index, record_id, include_self_keys=item.derivation_kind != "template_projection"):
307 registry[_norm(name)][type_id].append(source)
308 calculations, dependencies = [], []
309 for type_id in sorted(formulas):
310 item = selected[type_id]
311 unique = {}
312 for raw, evidence_id, source_ref in formulas[type_id]:
313 unique.setdefault(raw, {"evidence_ids": [], "source_refs": []})
314 unique[raw]["evidence_ids"].append(evidence_id)
315 unique[raw]["source_refs"].append(source_ref)
316 normalized_variants = {parse_calculation(raw).get("expression_signature", raw) for raw in unique}
317 for raw, evidence in unique.items():
318 parsed = parse_calculation(raw)
319 calculation = {"id": "calculation:" + digest([type_id, raw])[:24],
320 "source_type_id": type_id, "source_label": item.label,
321 "raw_formula": raw, **evidence, **parsed,
322 "applicability_scope": item.applicability_scope,
323 "definition_parameters": item.definition_parameters,
324 "bindings": [], "identity_scope": "definition"}
325 if parsed["status"] == "parsed":
326 for occurrence in parsed["symbols"]:
327 candidates = registry.get(_norm(occurrence["symbol"]), {})
328 compatible = {candidate: proof for candidate, proof in candidates.items()
329 if candidate != type_id and _target_scope_supported(data, item, selected[candidate])
330 and _target_parameters_supported(item, selected[candidate], data)}
331 binding = {**occurrence, "candidate_type_ids": sorted(candidates),
332 "compatible_type_ids": sorted(compatible), "status": "unresolved"}
333 if len(compatible) == 1:
334 target_id = next(iter(compatible))
335 binding.update(status="bound", target_type_id=target_id,
336 binding_evidence=compatible[target_id],
337 target_internal_parameters=sorted(_internal_parameters(data, selected[target_id])))
338 else:
339 binding["reason"] = ("ambiguous_definition" if len(compatible) > 1 else
340 "self_reference" if type_id in candidates else
341 "definition_parameters_not_proven" if any(
342 not _target_parameters_supported(item, selected[candidate], data)
343 for candidate in candidates) else
344 "scope_not_proven" if candidates else "symbol_has_no_accepted_definition")
345 calculation["bindings"].append(binding)
346 if len(normalized_variants) > 1:
347 calculation.update(status="unresolved", reason="conflicting_source_formula_variants")
348 elif all(binding["status"] == "bound" for binding in calculation["bindings"]):
349 calculation["status"] = "accepted"
350 else:
351 calculation.update(status="unresolved", reason="unresolved_formula_operands")
352 if parsed["formula_type"] == "symbol_reference" and not registry.get(_norm(raw)):
353 calculation["formula_type"] = "natural_language"
354 calculations.append(calculation)
355 # Dependency cycles require a domain rule (for example a time offset). The
356 # arithmetic DSL does not yet express one, so do not accept circular bindings.
357 graph = defaultdict(set)
358 for calc in calculations:
359 if calc["status"] == "accepted":
360 graph[calc["source_type_id"]].update(binding["target_type_id"] for binding in calc["bindings"])
361 def cyclic(start):
362 stack, visited = list(graph[start]), set()
363 while stack:
364 current = stack.pop()
365 if current == start:
366 return True
367 if current not in visited:
368 visited.add(current)
369 stack.extend(graph[current])
370 return False
371 cycle_types = {node for node in list(graph) if cyclic(node)}
372 for calc in calculations:
373 if calc["source_type_id"] in cycle_types and calc["status"] == "accepted":
374 calc.update(status="unresolved", reason="cyclic_calculation_dependency")
375 if calc["status"] != "accepted":
376 continue
377 for binding in calc["bindings"]:
378 # Local operator and caller-scope evidence participates in the
379 # compatibility check and must travel with the published edge.
380 context_proofs = [eid for owner in (selected[calc["source_type_id"]], selected[binding["target_type_id"]])
381 for prop in owner.source_properties if prop.role in ("scope", "unit")
382 for eid in prop.evidence_ids if eid in data.evidence]
383 dependencies.append({"id": "calculation_dependency:" + digest([
384 calc["id"], binding["path"], binding["target_type_id"]])[:24],
385 "calculation_id": calc["id"], "source_type_id": calc["source_type_id"],
386 "target_type_id": binding["target_type_id"], "parent_relation": "depends_on",
387 "predicate_name": "calculation_dependency", "operand_role": binding["operand_role"],
388 "expression_path": binding["path"], "symbol": binding["symbol"],
389 "evidence_ids": sorted(set(calc["evidence_ids"] + context_proofs + [p["evidence_id"]
390 for p in binding["binding_evidence"] if "evidence_id" in p])),
391 "target_internal_parameters": binding.get("target_internal_parameters", []),
392 "binding_evidence": binding["binding_evidence"], "status": "accepted"})
393 statuses = Counter(item["status"] for item in calculations)
394 binding_tasks = _formula_binding_tasks(calculations, index)
395 return {"calculations": calculations, "dependencies": dependencies,
396 "binding_tasks": binding_tasks,
397 "coverage": {"accepted_quantitative_types": len(selected), "types_with_formula": len(formulas),
398 "types_without_formula": len(selected) - len(formulas), "calculations": len(calculations),
399 "accepted": statuses["accepted"], "unresolved": statuses["unresolved"],
400 "dependencies": len(dependencies), "partial": bool(statuses["unresolved"]),
401 "targeted_binding_tasks": len(binding_tasks),
402 "input_scope": "accepted_definition_evidence_and_definition_index",
403 "business_observation_rows_read": 0, "llm_calls": 0,
404 "numeric_evaluation": "not_performed"}}
