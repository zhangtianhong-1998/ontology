79def classify_formula_fragment(table, candidate):
80 """Keep source-backed calculation fragments distinct from full formulas.
81
82 A legacy formula proposal is reclassified only when both the declaration
83 and observed values support a fragment. Identifier-shaped values alone do
84 not disprove a direct mapping in an explicitly declared formula column.
85 This does not resolve operand targets or construct an expression.
86 """
87 profile_candidate = candidate.get("status") == "profile_supported_role_candidate"
88 if candidate.get("status") != "source_verified_role_candidate" and not profile_candidate:
89 return None
90 proposed = candidate.get("role")
91 if proposed not in {"formula", *_FRAGMENT_ROLES}:
92 return None
93 column = next((item for item in table.get("columns", ())
94 if item.get("column_name") == candidate.get("column")), None)
95 observations = candidate.get("observations") or []
96 if column is None or not observations:
97 return None
98 values = [str(item.get("value", "")).strip() for item in observations]
99 if any(not value for value in values):
100 return None
101 effective = proposed
102 reason = "source_checked_fragment_role_candidate"
103 validation_scope = "source_checked_observations"
104 # An operator proposal must also survive all already available source
105 # samples. A model citing only SUM cannot hide SUM(amount) in the same field.
106 profile = next((item for item in table.get("profiles", ())
107 if item.get("column") == candidate["column"]), {})
108 values.extend(str(value).strip() for value in profile.get("distinct_sample", ())
109 if value is not None and str(value).strip())
110 if proposed in _FRAGMENT_ROLES and not all(
111 formula_fragment_value(proposed, value) for value in values):
112 return None
113 if proposed == "formula":
114 name = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", column["column_name"]).casefold()
115 comment = str(column.get("column_comment") or "")
116 if (_EXPLICIT_FORMULA_NAME.search(name)
117 or name in {"calculation", "caliber"}
118 or _EXPLICIT_FORMULA_COMMENT.search(comment)):
119 return None
120 # Use every available profile sample as a contradiction check, not
121 # just the few values cited by the model. This remains sample-scoped.
122 validation_scope = "source_checked_observations_and_available_profile_samples"
123 if ((_OPERATOR_NAME.search(name) or _OPERATOR_COMMENT.search(comment))
124 and all(value.casefold() in _OPERATOR_VALUES for value in values)):
125 effective = "calculation_operator"
126 reason = "declared_operator_field_with_observed_operator_values"
127 elif ((_OPERAND_NAME.search(name) or _OPERAND_COMMENT.search(comment))
128 and all(_IDENTIFIER_VALUE.fullmatch(value) for value in values)):
129 effective = "operand_reference"
130 reason = "declared_operand_field_with_observed_identifier_values"
131 else:
132 return None
133 if profile_candidate:
134 validation_scope = "source_declaration_and_available_profile_samples"
135 return {**candidate, "column": candidate["column"], "proposed_role": proposed,
136 "effective_role": effective, "formula_status": "fragment", "reason": reason,
137 "validation_scope": validation_scope,
138 "expression_constructed": False, "operand_target_verified": False}

141def formula_fragment_value(role, value):
142 """Check each indexed row too; profile samples are not full-column proof."""
143 text = str(value or "").strip()
144 return bool(text) and (text.casefold() in _OPERATOR_VALUES if role == "calculation_operator"
145 else bool(_IDENTIFIER_VALUE.fullmatch(text)))

148def formula_fragment_context(table):
149 """Keep declared operators out of formulas without new scans/model calls."""
150 fragments = {}
151 profiles = {item["column"]: item for item in table.get("profiles", ())}
152 safe = {item["column"] for item in classify_columns(table)
153 if item["include_in_semantic_prompt"]}
154 table_name = table.get("name") or ".".join(filter(None, (
155 table.get("schema"), table.get("table_name"))))
156 for column in table.get("columns", ()):
157 name = column["column_name"]
158 profile = profiles.get(name, {})
159 samples = profile.get("distinct_sample") or []
160 if name not in safe or profile.get("scan_scope") != "full_input" or not samples:
161 continue
162 # Profile values are genuine source observations, but do not have
163 # record identities. Preserve that distinction instead of inventing rows.
164 candidate = {"column": name, "role": "formula",
165 "status": "profile_supported_role_candidate", "semantic_status": "unjudged",
166 "schema_evidence_id": f"schema:{table_name}:{name}",
167 "evidence_kind": "source_declaration_and_profile_values",
168 "profile_field_id": profile.get("field_id"),
169 "profile_sample_scope": profile.get("sample_scope"),
170 "sample_exhaustive_in_input": profile.get("sample_exhaustive_in_input", False),
171 "observations": [{"column": name, "value": str(value),
172 "source": "profile_distinct_sample"}
173 for value in samples if value is not None and str(value).strip()]}
174 fragment = classify_formula_fragment(table, candidate)
175 if fragment is not None and fragment["effective_role"] == "calculation_operator":
176 fragments[name] = fragment
177 for candidate in table.get("inferred_semantic_roles", ()):
178 fragment = classify_formula_fragment(table, candidate)
179 if fragment is not None:
180 fragments[fragment["column"]] = fragment
181 return fragments

191def _safe_columns(table):
192 """Return only source columns eligible for sampled semantic inspection."""
193 blocked = {"sensitive", "empty", "audit_time", "audit_metadata",
194 "technical_identifier"}
195 profiles = {item["column"]: item for item in table.get("profiles", ())}
196 roles = _field_roles(table)
197 settled_fragments = formula_fragment_context(table)
198 checked_roles = {item["column"] for item in table.get("inferred_semantic_roles", ())
199 if item.get("status") == "source_verified_role_candidate"}
200 # A metadata cue saying "calculation" is not a verified formula role.
201 # Reuse the existing bounded table packet to review these columns first.
202 review_formulas = set(roles.get("formula", ())) - set(settled_fragments) - checked_roles
203 already_named = {name for fields in roles.values() for name in fields} - review_formulas
204 already_named.update(settled_fragments)
205 # Strong identifier/content conflicts are already resolved to bindings;
206 # do not spend another model call trying to rename a known source key.
207 already_named.update(item["column"] for item in field_role_conflicts(table)
208 if item["binding_only"])
209 selected = []
210 for item in classify_columns(table):
211 name = item["column"]
212 profile = profiles.get(name, {})
213 usable = profile.get("usable_count")
214 if (item["role"] in blocked or name in already_named
215 or type(usable) is not int or usable <= 0):
216 continue
217 selected.append(name)
218 return sorted(selected, key=lambda name: name not in review_formulas)

221def _sample(data, table_name, columns, *, max_rows, max_value_chars):
222 """Read a fixed spread of actual row positions, never a value product."""
223 table = data.tables[table_name]
224 total = table["rows"]
225 count = min(total, max_rows)
226 positions = sorted({1 + ((total - 1) * index) // max(1, count - 1)
227 for index in range(count)})
228 if not positions:
229 return [], {}, {"over_length": 0, "sensitive_shape": 0}
230 placeholders = ", ".join("?" for _ in positions)
231 select = ", ".join(qi(name) for name in columns)
232 sql = (f"SELECT __r2_row, {select} FROM {qi(table['sql_name'])} "
233 f"WHERE __r2_row IN ({placeholders}) ORDER BY __r2_row")
234 rows = data.db.execute(sql, positions).fetchall()
235 packet_rows, observed = [], {}
236 omissions = {"over_length": 0, "sensitive_shape": 0}
237 for values in rows:
238 row_number = values[0]
239 fragments = {}
240 for column, raw in zip(columns, values[1:]):
241 if raw is None or not str(raw).strip():
242 continue
243 text = str(raw)
244 if len(text) > max_value_chars:
245 omissions["over_length"] += 1
246 continue
247 if _SENSITIVE_VALUE.search(text):
248 omissions["sensitive_shape"] += 1
249 continue
250 fragments[column] = text
251 observed[(column, row_number)] = text
252 if fragments:
253 packet_rows.append({"row_number": row_number, "values": fragments})
254 return packet_rows, observed, omissions

257def _validate_response(data, table_name, response, eligible, observed):
258 accepted, rejected = [], []
259 seen = set()
260 for proposal in response.proposals:
261 column, role = proposal.column, proposal.role
262 reason = None
263 if column not in eligible:
264 reason = "column_not_eligible_or_not_in_source"
265 elif (column, role) in seen or any(item["column"] == column for item in accepted):
266 reason = "duplicate_or_conflicting_role"
267 elif not proposal.observations:
268 reason = "no_source_observation"
269 elif any(observed.get((column, item.row_number)) != item.value
270 for item in proposal.observations):
271 reason = "sample_value_or_row_not_in_source"
272 if reason:
273 rejected.append({"column": column, "role": role, "reason": reason})
274 continue
275 seen.add((column, role))
276 evidence = []
277 for item in proposal.observations[:2]:
278 info = data.tables[table_name]
279 cursor = data.db.execute(
280 f"SELECT * FROM {qi(info['sql_name'])} WHERE __r2_row = ?",
281 [item.row_number])
282 names = [part[0] for part in cursor.description]
283 values = cursor.fetchone()
284 row = dict(zip(names, values)) if values is not None else {}
285 if row.get(column) != item.value:
286 reason = "sample_changed_before_validation"
287 break
288 evidence.append({"row_number": item.row_number,
289 "record_id": data.record_id(table_name, row),
290 "value": item.value,
291 "column": column})
292 if reason:
293 rejected.append({"column": column, "role": role, "reason": reason})
294 continue
295 candidate = {"column": column, "role": role,
296 "status": "source_verified_role_candidate",
297 "semantic_status": "unjudged",
298 "schema_evidence_id": f"schema:{table_name}:{column}",
299 "evidence_kind": "source_values_supported_role_candidate",
300 "observations": evidence}
301 resolution = resolve_context_role(data.tables[table_name], candidate)
302 if resolution:
303 candidate.update(role=resolution["effective_role"], proposed_role=role,
304 effective_role=resolution["effective_role"],
305 role_resolution=resolution["reason"], role_evidence=resolution)
306 fragment = classify_formula_fragment(data.tables[table_name], candidate)
307 if role in _FRAGMENT_ROLES and fragment is None:
308 rejected.append({"column": column, "role": role,
309 "reason": "fragment_role_conflicts_with_observed_values"})
310 continue
311 accepted.append(fragment or candidate)
312 return accepted, rejected
