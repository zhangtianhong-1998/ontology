19def _fixture(*, descriptions=None, formulas=None, units=("元", "元"),
20 scopes=None, roots=("Metric", "Metric"), truncated=False):
21 descriptions = descriptions or (
22 "阿里云业务销售所得营业收入", "商业市场业务销售所得营业收入")
23 scopes = scopes or ({"business": "阿里云"}, {"business": "商业市场"})
24 evidence = {}
25 children = []
26 for i, (name, description) in enumerate(zip(("阿里云", "商业市场"), descriptions)):
27 props = []
28 sources = [("description", description)]
29 if formulas is not None and formulas[i] is not None:
30 sources.append(("formula", formulas[i]))
31 for role, fragment in sources:
32 evidence_id = f"record:e{i}-{role}"
33 evidence[evidence_id] = {
34 "id": evidence_id, "origin": "observed_record",
35 "raw_fragment": fragment,
36 "raw_fragment_truncated": truncated if role == "description" else False,
37 "source_ref": {"table": "fruit.measure_def", "column": role,
38 "record_id": f"r{i}", "snapshot_id": "snap"},
39 }
40 props.append(SourceProperty(role=role, source_table="fruit.measure_def",
41 source_column=role, evidence_ids=[evidence_id]))
42 children.append(DerivedType(
43 id=f"type:{name}", parent=roots[i], label=f"{name}收入",
44 definition=description, category="business_type",
45 unit=units[i], applicability_scope=scopes[i],
46 derivation_kind="exact_definition", evidence_scope="definition_record",
47 evidence_ids=[p.evidence_ids[0] for p in props],
48 source_properties=props, source_concept_ids=[f"concept:{i}"],
49 ))
50 return SimpleNamespace(snapshot_id="snap", evidence=evidence, tables={}), BuildPlan(object_types=children)

69def test_generalization_reparents_two_accepted_types_and_keeps_sources():
70 data, core = _fixture(descriptions=(
71 "阿里云水果销售所得营业收入", "商业市场水果销售所得营业收入"))
72 for name, child in zip(("阿里云", "商业市场"), core.object_types):
73 child.label = f"{name}水果收入"
74 decision = _decision()
75 decision.update(label="水果收入", definition="水果销售所得营业收入",
76 business_object_quote="水果")
77 for child, support in zip(core.object_types, decision["children"]):
78 support.update(shared_quote="水果销售所得营业收入",
79 specialization_quote=child.definition)
80 candidates = propose_generalization_candidates(core, data, 3)
81 assert len(candidates) == 1 and candidates[0]["status"] == "candidate_only"
82 compiled, parent = compile_generalization(data, PROFILE, core, decision)
83 assert parent.label == "水果收入" and parent.parent == "Metric"
84 assert parent.derivation_kind == "shared_supertype"
85 assert parent.induced_from_type_ids == ["type:商业市场", "type:阿里云"]
86 assert parent.evidence_scope == "multiple_definition_records"
87 assert len(parent.evidence_ids) == 2
88 assert {item.parent for item in compiled.object_types if item.id != parent.id} == {parent.id}
89 assert {item.parent for item in core.object_types} == {"Metric"}
90 assert compiled.relation_types == core.relation_types
91 repeated, same_parent = compile_generalization(data, PROFILE, compiled, decision)
92 assert same_parent.id == parent.id and len(repeated.object_types) == 3

143@pytest.mark.parametrize("change, error", [
144 ("unit", "units conflict"),
145 ("root", "mismatched root"),
146 ("formula", "formulas are incompatible"),
147 ("parent_qualifier", "Parent contains a child specialization"),
148 ("scope", "common applicability scope"),
149 ("truncated", "complete definition fragment"),
150 ("same_child", "two distinct child types"),
151 ("short_anchor", "too short"),
152])
153def test_rejects_unsupported_generalization(change, error):
154 kwargs = {}
155 if change == "unit":
156 kwargs["units"] = ("元", "吨")
157 elif change == "root":
158 kwargs["roots"] = ("Metric", "Measure")
159 elif change == "formula":
160 kwargs["formulas"] = ("收入=销量+均价", "收入=销量-均价")
161 elif change == "truncated":
162 kwargs["truncated"] = True
163 data, core = _fixture(**kwargs)
164 decision = _decision()
165 if change == "parent_qualifier":
166 decision["definition"] = "阿里云业务销售所得营业收入"
167 elif change == "scope":
168 decision["parent_scope"] = {"business": "阿里云"}
169 elif change == "same_child":
170 decision["children"][1]["type_id"] = decision["children"][0]["type_id"]
171 elif change == "short_anchor":
172 decision["shared_anchor"] = "收"
173 with pytest.raises(ValueError, match=error):
174 compile_generalization(data, PROFILE, core, decision)

250def test_shared_parameters_are_inherited_hashed_and_rechecked_on_replay():
251 data, core, decision = _parameterized_case()
252 candidate = propose_generalization_candidates(core, data, 1)[0]
253 compiled, parent = compile_generalization(data, PROFILE, core, decision)
254 assert candidate["definition_parameters"] == parent.definition_parameters == {
255 "period_scope": "公历年", "budget": "1"}
256 assert all(item.definition_parameters == parent.definition_parameters for item in compiled.object_types)
257 repeated, same = compile_generalization(data, PROFILE, BuildPlan.model_validate(compiled.model_dump()), decision)
258 assert same.id == parent.id and len(repeated.object_types) == 3
259 altered = compiled.model_copy(deep=True)
260 next(item for item in altered.object_types if item.id == parent.id).definition_parameters = {}
261 with pytest.raises(ValueError, match="parent ID conflicts"):
262 compile_generalization(data, PROFILE, altered, decision)
263 for item in core.object_types:
264 item.definition_parameters = {"period_scope": "公历月", "budget": "1"}
265 _, monthly = compile_generalization(data, PROFILE, core, decision)
266 assert monthly.id != parent.id
