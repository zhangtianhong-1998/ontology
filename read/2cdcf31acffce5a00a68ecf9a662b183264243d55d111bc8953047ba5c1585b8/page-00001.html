71 def advance(self, n: int = 1, detail: str | None = None) -> None:
72 if self.closed:
73 raise RuntimeError("progress task is closed")
74 if isinstance(n, bool) or not isinstance(n, int) or n < 0:
75 raise ValueError("progress advance must be a non-negative integer")

110 def _emit(self, status: str, *, now: float | None = None) -> None:
111 now = _monotonic() if now is None else now
112 elapsed = max(0.0, now - self.started_at)
113 rate = self.count / elapsed if elapsed > 0 else 0.0
114 count = f"{self.count:,}" if self.total is None else f"{self.count:,}/{self.total:,}"
