71def _explicit_direction(quote, first_anchors, second_anchors, relation, direction,
72 predicate_name=None):
73 """Accept a complete positive sentence; unmatched modifiers stay unresolved.
74
75 Substring endpoint ordering loses negation and conditions. Until condition
76 compilation exists, accepting only a full supported clause is deliberate.
77 """
78 compact = lambda value: re.sub(r"\s+", "", _norm(value)).rstrip("。.!！")
79 phrase = compact(quote)
80 name = canonical_relation_label(relation, predicate_name)
81 from .relation_contract import _DERIVED_CUES
82 cues = _DERIVED_CUES[name] if name != relation else _CUES[relation]
83 pairs = ((first_anchors, second_anchors) if direction == "source_to_target"
84 else (second_anchors, first_anchors))
85 allowed = {compact(first) + compact(cue) + compact(second)
86 for first in pairs[0] for second in pairs[1] for cue in cues
87 if first and second and len(compact(first)) >= 2 and len(compact(second)) >= 2}
88 return phrase in allowed

91def _read_witness(data, candidate):
92 table_name = candidate.get("configuration_table")
93 table = data.tables.get(table_name)
94 row_number = candidate.get("configuration_witness_row_number")
95 if table is None or type(row_number) is not int or row_number < 1:
96 raise ValueError("Configuration witness table or row is invalid")
97 cursor = data.db.execute(
98 f"SELECT * FROM {qi(table['sql_name'])} WHERE __r2_row=?", [row_number])
99 names = [item[0] for item in cursor.description]
100 raw = cursor.fetchone()
101 if raw is None:
102 raise ValueError("Configuration witness row is absent")
103 row = dict(zip(names, raw))
104 if data.record_id(table_name, row) != candidate.get("configuration_record_id"):
105 raise ValueError("Configuration witness record identity differs from the snapshot")
106 for side in ("source", "target"):
107 column = candidate.get(side + "_code_column")
108 if column not in table["column_names"] or row[column] != candidate.get(side + "_code"):
109 raise ValueError("Configuration witness code differs from its candidate")
110 _check_column(data, table_name, column)
111 return row

114def _endpoint(data, plan, candidate, side, aligned):
115 stated = candidate.get(side + "_definition") or {}
116 table_name, code_column = stated.get("table"), stated.get("code_column")
117 code = candidate.get(side + "_code")
118 if (table_name not in data.tables or code_column not in data.tables[table_name]["column_names"]
119 or code in (None, "")):
120 raise ValueError("Definition endpoint lacks a valid code binding")
121 _check_column(data, table_name, code_column)
122 rows = data.lookup(table_name, ((code_column, code), *sorted(stated.get("lookup_scope", {}).items())))
123 if len(rows) != 1:
124 raise ValueError("Definition code is missing or ambiguous in the full snapshot")
125 record_id = data.record_id(table_name, rows[0])
126 matched = aligned.get(record_id)
127 type_id = candidate.get(side + "_type_id")
128 if (record_id != stated.get("record_id") or matched is None
129 or matched["type_id"] != type_id
130 or matched.get("snapshot_id") not in (None, data.snapshot_id)
131 or matched["concept_id"] != stated.get("concept_id")):
132 raise ValueError("Definition row lacks the cited current accepted business type binding")
133 business_type = next((item for item in plan.object_types
134 if item.id == type_id and item.category == "business_type"), None)
135 if business_type is None:
136 raise ValueError("Configuration endpoint is not an accepted business type")
137 # The accepted mapping owns this row; the type owns the admissible
138 # definition columns. Read those cells together, even when the type was
139 # originally induced from a different template witness.
140 proofs = set(matched.get("evidence_ids", []))
141 for prop in business_type.source_properties:
142 if prop.role not in ("description", "formula") or prop.source_table != table_name:
143 continue
144 _check_column(data, table_name, prop.source_column)
145 value = rows[0].get(prop.source_column)
146 if not isinstance(value, str) or not value.strip() or len(value) > 1024:
147 continue
148 eid = "record:" + digest([data.snapshot_id, record_id, prop.source_column])[:24]
149 existing = data.evidence.get(eid)
150 if existing and (existing.get("raw_fragment") != value or existing.get("raw_fragment_truncated")
151 or existing.get("source_ref", {}).get("snapshot_id") != data.snapshot_id):
152 raise ValueError("Current endpoint definition evidence differs from its complete snapshot record")
153 data.evidence[eid] = {"id": eid, "origin": "observed_record", "raw_fragment": value,
154 "raw_fragment_truncated": False, "source_ref": {"snapshot_id": data.snapshot_id,
155 "table": table_name, "column": prop.source_column, "record_id": record_id,
156 "row": rows[0]["__r2_row"]}}
157 proofs.add(eid)
158 matched = {**matched, "evidence_ids": sorted(proofs)}
159 return business_type, matched, record_id
