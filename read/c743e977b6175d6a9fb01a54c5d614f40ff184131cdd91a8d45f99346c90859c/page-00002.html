415def _balanced_configuration_candidates(candidates, limit):
416 """Visit each spec and semantic pattern before spending on another witness.
417
418 Scheduling never accepts the other rows in a pattern. Their exact endpoint
419 pairs and complete text remain separate candidates and in coverage.
420 """
421 specs = defaultdict(lambda: defaultdict(deque))
422 seen, seen_pairs, duplicates = set(), set(), 0
423 # A separately discovered path may use a different candidate ID for the
424 # same witness. Retain its replayable proof before deduplicating routes.
425 for item in sorted(candidates, key=lambda value: not bool(value.get("auxiliary_path_proof"))):
426 if item["id"] in seen:
427 duplicates += 1
428 continue
429 seen.add(item["id"])
430 pair = tuple(item.get(side + "_definition", {}).get("record_id") for side in ("source", "target"))
431 witness = item.get("configuration_record_id")
432 if witness and all(pair):
433 # Alternative checked join paths to the same current endpoints
434 # are proof alternatives, not fresh semantic decisions.
435 key = (witness, pair, item.get("source_type_id"), item.get("target_type_id"),
436 item.get("relation_text_column"), item.get("predicate_literal"),
437 item.get("direction_status"), item.get("predicate_name"),
438 digest(item.get("semantic_parameters", {})))
439 if key in seen_pairs:
440 duplicates += 1
441 continue
442 seen_pairs.add(key)
443 literal = str(item.get("predicate_literal") or "")
444 # Code substitutions affect scheduling only; original text is retained
445 # verbatim for every decision and replay check.
446 for side in ("source", "target"):
447 code = item.get(side + "_code")
448 if isinstance(code, str) and len(code) >= 2:
449 literal = literal.replace(code, "{" + side + "_code}")
450 pattern = digest([item.get("source_type_id"), item.get("target_type_id"), literal,
451 item.get("selector"), item.get("source_scope_bindings"),
452 item.get("target_scope_bindings")])
453 specs[item.get("spec_id", "")][pattern].append(item)
454 queues = {}
455 for spec_id, patterns in specs.items():
456 queue = deque()
457 while any(patterns.values()):
458 for pattern in sorted(patterns):
459 if patterns[pattern]:
460 queue.append(patterns[pattern].popleft())
461 queues[spec_id] = queue
462 selected = []
463 while len(selected) < limit and any(queues.values()):
464 for key in sorted(queues):
465 if queues[key] and len(selected) < limit:
466 selected.append(queues[key].popleft())
467 return selected, duplicates

470def _explicit_clause_candidate(candidate, plan):
471 """Route endpoint-led statements to their own predicate/qualifier checks.
472
473 This is routing, not acceptance. Negation and conditions still pass intact
474 to the explicit compiler, which can reject them; a surrounding config
475 description beginning with "this mapping" is handled as a purpose instead.
476 """
477 literal = _norm(candidate.get("predicate_literal"))
478 types = {item.id: item for item in plan.object_types}
479 anchors = []
480 for side in ("source", "target"):
481 item = types.get(candidate.get(side + "_type_id"))
482 anchors.append([_norm(value) for value in (candidate.get(side + "_code"), item.label if item else None)
483 if value and len(_norm(value)) >= 2])
484 return any(literal.startswith(first) and second in literal[len(first):]
485 and any(_norm(cue) in literal[len(first):] for cues in _CUES.values() for cue in cues)
486 for left, right in (anchors, anchors[::-1]) for first in left for second in right)

626def _definition_previews(data, business_type, record_id, binding_evidence=()):
627 """Read the current mapped record, not just the template's seed record.
628
629 Binding evidence can supply a different record's complete value only for
630 a definition/formula column declared by that accepted type. This does not
631 borrow text from another endpoint, snapshot or unrelated field.
632 """
633 previews = []
634 for prop in business_type.source_properties:
635 if prop.role not in ("description", "formula"):
636 continue
637 for evidence_id in sorted(set(prop.evidence_ids) | set(binding_evidence)):
638 evidence = data.evidence.get(evidence_id) or {}
639 ref = evidence.get("source_ref", {})
640 if (evidence.get("origin") == "observed_record"
641 and not evidence.get("raw_fragment_truncated")
642 and ref.get("snapshot_id") == data.snapshot_id
643 and ref.get("record_id") == record_id
644 and ref.get("table") == prop.source_table
645 and ref.get("column") == prop.source_column):
646 value = evidence.get("raw_fragment")
647 if isinstance(value, str) and value and len(value) <= 1024:
648 previews.append({"role": prop.role, "quote": value,
649 "evidence_id": evidence_id})
650 return previews
