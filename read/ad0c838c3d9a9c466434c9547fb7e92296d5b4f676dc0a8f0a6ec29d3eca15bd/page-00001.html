31def make_data():
32 data = Dataset.__new__(Dataset)
33 data.db = duckdb.connect()
34 data.db.execute('SET threads=1')
35 data.indexes, data.evidence, data.tables = set(), {}, {}
36 data.snapshot_id = 'fixed-test-snapshot'

38 ('s1','a','CN','yes'), ('s2','a','CN','yes'),
39 ('s3','missing','CN','yes'), ('s4','b','CN','yes'),
40 ('s5','a','CN','no'), ('s6','a','','yes'),

47 sql = table + '_data'

49 data.db.execute(f'CREATE TABLE {sql} (__r2_row BIGINT, id VARCHAR, code VARCHAR, scope VARCHAR, kind VARCHAR)')
50 data.db.executemany(f'INSERT INTO {sql} VALUES (?,?,?,?,?)', [(i, *r) for i,r in enumerate(rows,1)])

56def make_plan(data, source_positions=(2,9), count=1, evidence=True):
57 src, tgt = list(data.rows('source')), list(data.rows('target'))
58 evidence_ids = []
59 if evidence:
60 for row in src:
61 rid = data.record_id('source',row)
62 evidence_id = 'e:'+rid
63 data.evidence[evidence_id] = {'source_ref': {'table':'source', 'record_id':rid,
64 'row': row['__r2_row'], 'snapshot_id':data.snapshot_id}}
65 evidence_ids.append(evidence_id)
66 relations = [RelationPlan(id=f'p{i}', source_table='source', target_table='target',
67 source_column='code', target_column='code', mode='identifier', predicate='points_to',
68 selector=Condition(op='eq',field='kind',value='yes'), scope_bindings={'scope':'scope'},
69 evidence_scope='sample_semantic_with_full_technical_check', witness_snapshot_id=data.snapshot_id,
70 witnessed_pairs=[WitnessedPair(source_record_id=data.record_id('source',src[n-1]),
71 target_record_id=data.record_id('target',tgt[0])) for n in source_positions],
72 evidence_ids=evidence_ids) for i in range(count)]
73 return BuildPlan(tables=[TablePlan(table=t, object_type='source_record_type:'+t,
74 label_column='id', evidence_ids=[]) for t in ('source','target')], relations=relations)

77def execute(data, plan, extractor=Extractor, cap=1000):
78 sink = Sink.__new__(Sink)
79 sink.db = sqlite3.connect(':memory:')
80 sink.db.execute('CREATE TABLE items (kind TEXT, id TEXT, body TEXT, PRIMARY KEY(kind,id))')
81 try:
82 stats = asyncio.run(extractor(data,sink,plan,NoLLM(),{'max_relation_records':cap}).execute())
83 items = [(kind,identity,json.loads(body)) for kind,identity,body in
84 sink.db.execute('SELECT kind,id,body FROM items ORDER BY kind,id')]
85 return stats, items
86 finally:
87 sink.db.close()
