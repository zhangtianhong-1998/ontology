37def test_configuration_edges_execute_only_the_witness_paths(tmp_path):
38 data, plan, concepts, alignments, candidate, rules = _configuration_case(tmp_path, "度量提供指标的量定义")
39 try:
40 accepted = asyncio.run(adjudicate_configuration_relations(data, PROFILE, plan, [candidate],

43 record_alignments=alignments, ontology_bindings=accepted["bindings"])

67 assert any(item["reason"] == "mapping_has_no_source_evidence" for item in replay["pending"])

72def _template_case(tmp_path):
73 root = tmp_path / "input"
74 _table(root, "result_definition", {"id": "ID", "name": "名称", "definition": "定义",
75 "formula": "完整公式", "measure_ref": "量名称引用"}, [
76 {"id": str(i), "name": region + "利润", "definition": "利润由收入减成本，地区为" + region,
77 "formula": "收入 - 成本", "measure_ref": "收入"} for i, region in ((1, "东"), (2, "西"))])
78 _table(root, "amount_definition", {"id": "ID", "name": "名称", "definition": "定义"}, [
79 {"id": "11", "name": "收入", "definition": "收入的通用量定义"},
80 {"id": "12", "name": "成本", "definition": "成本的通用量定义"}])
81 work = tmp_path / "work"
82 work.mkdir()
83 data = Dataset(root, work)
84 types, concepts, alignments, records = [], [], [], []
85 for table in data.tables:
86 for row in data.rows(table):
87 rid = data.record_id(table, row)
88 key = "profit" if row.get("formula") else row["name"]
89 fields, props = {}, []
90 for role, column in (("name", "name"), ("description", "definition"),
91 ("formula", "formula"), ("reference", "measure_ref")):
92 if not row.get(column):
93 continue
94 eid = "record:" + digest([data.snapshot_id, rid, column])[:24]
95 data.evidence[eid] = {"id": eid, "origin": "observed_record", "raw_fragment": row[column],
96 "source_ref": {"table": table, "row": row["__r2_row"], "record_id": rid,
97 "column": column, "snapshot_id": data.snapshot_id}}
98 fields.setdefault(role, []).append({"column": column, "value": row[column]})
99 if role != "reference":
100 props.append({"role": role, "source_table": table, "source_column": column, "evidence_ids": [eid]})
101 if key not in {item.id for item in types}:
102 types.append(DerivedType(id=key, parent="Metric" if key == "profit" else "Measure",
103 label="利润" if key == "profit" else key, definition="可复用定义", category="business_type",
104 derivation_kind="template_projection" if key == "profit" else None,
105 source_properties=props, evidence_ids=[eid for prop in props for eid in prop["evidence_ids"]]))
106 if key == "profit":
107 records.append({"table": table, "record_id": rid, "row_number": row["__r2_row"], "fields": fields})
108 else:
109 concepts.append({"id": key, "ontology_type_id": key, "ontology_level": "type"})
110 alignments.append({"id": "alignment:" + key, "source_record_id": rid,
111 "concept_id": key, "mapping_kind": "exact", "evidence_ids": types[-1].evidence_ids})
112 slot_evidence = "record:" + digest([data.snapshot_id, records[0]["record_id"], "measure_ref"])[:24]
113 template = {"template_id": "projection:p", "contract_hash": "p", "object_type_id": "profit",
114 "source_table": records[0]["table"], "snapshot_id": data.snapshot_id, "status": "accepted", "root_type": "Metric",
115 "semantic_columns": ["name", "definition", "formula"], "invariants": {"formula": "收入 - 成本"},
116 "slots": [{"name": "region", "role": "dimension", "evidence_ids": [slot_evidence]},
117 {"name": "amount", "role": "measure", "target_type_id": "收入", "fixed_value": "收入", "evidence_ids": [slot_evidence]}],
118 "field_templates": [{"column": "name", "template": "{region}利润"},
119 {"column": "definition", "template": "利润由收入减成本，地区为{region}"},
120 {"column": "formula", "template": "收入 - 成本"}, {"column": "measure_ref", "template": "{amount}"}]}
121 bound = [bind_projection(data, template, record) for record in records]
122 semantic = compile_template_relations(data, PROFILE, BuildPlan(object_types=types), [template], bound)
123 calculation = compile_calculation_stage(data, PROFILE, semantic["plan"], {
124 "concepts": concepts, "record_alignments": alignments, "template_projections": [template]})
125 return data, calculation, {"template_projections": [template], "template_bindings": bound,
126 "ontology_bindings": semantic["bindings"], "calculation_contracts": calculation,
127 "concepts": concepts, "record_alignments": alignments}

130def test_template_slots_and_formula_instances_keep_actual_coordinates(tmp_path):
131 data, calculation, inputs = _template_case(tmp_path)
132 try:
133 result = materialize_ontology_instances(data, calculation["plan"], **inputs)
134 assert result["coverage"]["definition_record_instances"] == 4
135 assert result["coverage"]["slot_occurrences"] == 2
136 assert len(result["assertions"]) == 6 # 2 source slots + 2 exact formula operands per source.
137 primary = [node for node in result["objects"] if node["type"] == "profit"]
138 assert {node["slot_values"]["region"] for node in primary} == {"东", "西"}
139 assert len({node["id"] for node in result["objects"]}) == 6
140 assert {edge["subject"] for edge in result["assertions"]} == {node["id"] for node in primary}
141 assert all(edge["object"] in {node["id"] for node in result["objects"]} for edge in result["assertions"])
142 assert {item["reason"] for item in result["pending"]} == {"template_slot_has_no_accepted_relation"}
143 assert result["coverage"]["llm_calls"] == result["coverage"]["cartesian_products_created"] == 0
144 # Changing a saved coordinate without changing the source must fail replay.
145 forged = deepcopy(inputs)
146 forged["template_bindings"][0]["slot_values"]["region"] = "北"
147 rejected = materialize_ontology_instances(data, calculation["plan"], **forged)
148 assert rejected["coverage"]["definition_record_instances"] == 3
149 assert any(item["reason"] == "template_replay_differs_from_saved_binding" for item in rejected["pending"])
150 limited = materialize_ontology_instances(data, calculation["plan"], max_assertions=0, **inputs)
151 assert limited["assertions"] == []
152 finally:
153 data.close()

156def test_future_variable_measure_value_cannot_inherit_initial_target(tmp_path):
157 data, calculation, inputs = _template_case(tmp_path)
158 try:
159 template = inputs["template_projections"][0]
160 template["template_id"], template["contract_hash"] = "projection:variable", "variable"
161 template["slots"][1].pop("fixed_value")
162 target = next(item for item in calculation["plan"].object_types if item.id == "收入")
163 template["slots"][1]["target_value_contract"] = _source_target_names(data, target)
164 for binding in inputs["template_bindings"]:
165 binding["template_id"] = template["template_id"]
166 # The initial reviewed witness binds only the source value 收入.
167 accepted = compile_template_relations(data, PROFILE, calculation["plan"], [template],
168 inputs["template_bindings"][:1])
169 inputs["ontology_bindings"] = accepted["bindings"]
170 table = data.tables[template["source_table"]]
171 data.db.execute(f"UPDATE {qi(table['sql_name'])} SET measure_ref=? WHERE __r2_row=2", ["成本"])
172 inputs["template_bindings"][1]["slot_values"]["amount"] = "成本"
173 result = materialize_ontology_instances(data, accepted["plan"], **inputs)
174 # A different quantity invalidates the complete template mapping before
175 # it can materialize a definition instance or inherit its formula edges.
176 assert result["coverage"]["definition_record_instances"] == 3
177 assert result["coverage"]["slot_occurrences"] == 1
178 assert len(result["assertions"]) == 3
179 assert any(item["reason"] == "template_replay_differs_from_saved_binding" for item in result["pending"])
180 # A fresh compile also cannot publish a universal target over both values.
181 retried = compile_template_relations(data, PROFILE, calculation["plan"], [template],
182 inputs["template_bindings"])
183 assert retried["bindings"] == []
184 assert any("Varying measure" in item["reason"] for item in retried["pending"])
185 finally:
186 data.close()
