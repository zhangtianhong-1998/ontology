90def generate(output):
91 """Create one new dataset with deterministic records and no declared FK."""
92 output.mkdir(parents=True, exist_ok=False)
93 for table, comment, key, columns, rows in TABLES:
94 base = {"schema": SCHEMA, "table_name": table}
95 metadata = {
96 "tables": {**base, "table_comment": comment, "relkind": "r", "estimated_rows": len(rows),
97 "columns": [{"column_name": name, "ordinal_position": i + 1,
98 "data_type": "numeric" if name.endswith("_amount") else "text",
99 "is_not_null": True, "default_value": None, "column_comment": text}
100 for i, (name, text) in enumerate(columns.items())]},
101 "constraints": {**base, "constraints": [{"constraint_name": table + "_pk", "constraint_type": "p",
102 "constraint_type_name": "PRIMARY KEY", "definition": f"PRIMARY KEY ({key})"}]},
103 "foreign_keys": {**base, "foreign_keys": []},
104 }
105 for kind, document in metadata.items():
106 path = output / "schema" / kind / (table + ".yaml")
107 path.parent.mkdir(parents=True, exist_ok=True)
108 path.write_text(yaml.safe_dump(document, allow_unicode=True, sort_keys=False), encoding="utf-8")
109 path = output / "data" / (table + ".csv")
110 path.parent.mkdir(exist_ok=True)
111 with path.open("w", newline="", encoding="utf-8") as stream:
112 writer = csv.DictWriter(stream, fieldnames=columns)
113 writer.writeheader()
114 writer.writerows(rows)
115 return {"tables": len(TABLES), "records": sum(len(t[-1]) for t in TABLES), "output": str(output.resolve())}
