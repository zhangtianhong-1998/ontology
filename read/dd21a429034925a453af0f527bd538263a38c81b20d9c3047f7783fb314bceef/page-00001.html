1"""Conservative superclass induction from accepted, source-grounded types.
2
3This is intentionally narrower than semantic similarity. A pair becomes a
4subclass assertion only when its complete definition fragments differ by an
5explicitly cited specialization term. Other pairs remain retrieval candidates.
6"""

23ROOTS = frozenset(("GeneralObject", "Measure", "Metric", "Dimension", "Term"))
24_HAN = re.compile(r"[\u3400-\u9fff]+")
25_WORD = re.compile(r"[a-z0-9]{4,}")
26_PUNCT = re.compile(r"[\s\W_]+", re.UNICODE)

49def _norm(value):
50 return unicodedata.normalize("NFKC", str(value or "")).casefold().strip()
51
52
53def _plain(value):
54 return _PUNCT.sub("", _norm(value))

57def _template(value):
58 # Keep arithmetic, comparison and tax/ratio punctuation: a purported
59 # shared definition must not turn x+y and x-y into the same template.
60 return re.sub(r"\s+", "", _norm(value))

63def _semantic_fragments(data, item):
64 """Yield complete, same-snapshot description/formula evidence only."""
65 for prop in item.source_properties:
66 if prop.role not in ("description", "formula"):
67 continue
68 for evidence_id in prop.evidence_ids:
69 evidence = data.evidence.get(evidence_id) or {}
70 source = evidence.get("source_ref") or {}
71 if (evidence.get("origin") != "observed_record"
72 or evidence.get("raw_fragment_truncated")
73 or source.get("table") != prop.source_table
74 or source.get("column") != prop.source_column
75 or source.get("snapshot_id") != data.snapshot_id):
76 continue
77 fragment = evidence.get("raw_fragment")
78 if isinstance(fragment, str) and fragment.strip():
79 yield evidence_id, prop.role, fragment, source.get("record_id")

82def _terms(value):
83 terms = set()
84 for run in _HAN.findall(_norm(value)):
85 for size in (2, 3):
86 terms.update(run[i:i + size] for i in range(len(run) - size + 1))
87 terms.update(_WORD.findall(_norm(value)))
88 return terms

91def generalization_eligible_types(plan, data):
92 """One eligibility contract shared by retrieval and its coverage report."""
93 return [item for item in plan.object_types
94 if item.category == "business_type" and item.parent in ROOTS
95 and item.parent != "Measure"
96 and (item.derivation_kind in (None, "exact_definition")
97 or (item.derivation_kind == "template_projection"
98 and item.evidence_scope == "projected_definition_template"))
99 and any(_semantic_fragments(data, item))]

102def propose_generalization_candidates(plan: BuildPlan, data, max_pairs: int):
103 """Recall bounded same-root pairs; this never asserts a superclass.
104
105 Only accepted business types with complete definition/formula evidence
106 participate. Common n-grams are retrieval signals, not identity evidence.
107 """
108 if type(max_pairs) is not int or max_pairs < 0:
109 raise ValueError("max_pairs must be a nonnegative integer")
110 if not max_pairs:
111 return []
112 eligible = generalization_eligible_types(plan, data)
113 by_group = defaultdict(list)
114 for item in eligible:
115 fragments = list(_semantic_fragments(data, item))
116 if fragments:
117 by_group[(item.parent, unit_identity(item.unit), tuple(sorted(item.definition_parameters.items())),
118 tuple(sorted(item.identity_qualifiers.items())),
119 tuple(sorted(item.classification_attributes.items())))].append((item, fragments))
120 ranked = []
121 for (root, unit, parameters, qualifiers, classifications), entries in by_group.items():
122 inverted = defaultdict(list)
123 for index, (_, fragments) in enumerate(entries):
124 for term in _terms(" ".join(fragment for _, _, fragment, _ in fragments)):
125 inverted[term].append(index)
126 scores = defaultdict(int)
127 pair_budget = max(64, max_pairs * 64)
128 # Rare terms rank first, but even a lone two-Han-character term such as
129 # 收入 may recall a pair. Per-posting fanout and the global budget avoid
130 # building a quadratic number of pairs when that word is widespread.
131 for _, indexes in sorted(inverted.items(), key=lambda item: (len(item[1]), item[0])):
132 if len(scores) >= pair_budget:
133 break
134 for position, left in enumerate(indexes):
135 if len(scores) >= pair_budget:
136 break
137 for right in indexes[position + 1:position + 9]:
138 if len(scores) >= pair_budget:
139 break
140 scores[(left, right)] += 1
141 for (left, right), score in scores.items():
142 a, b = entries[left][0], entries[right][0]
143 ids = sorted((a.id, b.id))
144 identity = [*ids, dict(parameters)] if parameters else ids
145 ranked.append({"child_type_ids": ids, "root_type": root,
146 "unit": unit or None, "lexical_overlap": score,
147 "definition_parameters": dict(parameters),
148 "identity_qualifiers": dict(qualifiers),
149 "classification_attributes": dict(classifications),
150 "status": "candidate_only",
151 "candidate_id": "generalization:" + digest(identity)[:24]})
152 return sorted(ranked, key=lambda x: (-x["lexical_overlap"], x["child_type_ids"]))[:max_pairs]

155def _supporting_fragment(data, child, support):
156 """Both quotes must occur in one complete definition record fragment."""
157 for evidence_id, role, fragment, record_id in _semantic_fragments(data, child):
158 if (support.shared_quote in fragment
159 and support.specialization_quote in fragment
160 and support.specialization_term in support.specialization_quote):
161 return evidence_id, role, fragment, record_id
162 raise ValueError("Shared and specialization quotes lack one complete definition fragment")

165def _formula_templates(data, child, term):
166 # Operators and parentheses are meaningful here: x+y != x-y.
167 formulas = {re.sub(r"\s+", "", _norm(fragment).replace(_norm(term), ""))
168 for _, role, fragment, _ in _semantic_fragments(data, child)
169 if role == "formula"}
170 return formulas

173def compile_generalization(data, profile, core: BuildPlan, decision):
174 """Atomically add one superclass and reparent its cited children.
175
176 Returns ``(candidate_plan, parent_type)`` or ``None`` for a nonproposal.
177 Passing checks records source support, not independently verified truth.
178 """
179 decision = GeneralizationDecision.model_validate(decision)
180 if decision.status != "proposed":
181 return None
182 if (not decision.label.strip() or not decision.definition.strip()
183 or not decision.shared_anchor.strip() or decision.root_type not in ROOTS):
184 raise ValueError("Generalization lacks label, definition, shared anchor or root")
185 if len(decision.children) < 2 or len({x.type_id for x in decision.children}) != len(decision.children):
186 raise ValueError("Generalization needs two distinct child types")
187 anchor = _plain(decision.shared_anchor)
188 if len(anchor) < 2 or (len(anchor) < 4 and not _HAN.fullmatch(anchor)):
189 raise ValueError("Shared anchor is too short to ground generalization")
190 # Two-character anchors are allowed, but the parent still needs a real
191 # definition. The bare label 收入, even quoted twice, proves no superclass.
192 if len(_plain(decision.definition)) < max(8, len(anchor) + 4):
193 raise ValueError("Shared parent definition lacks semantic detail")
194 existing = {item.id: item for item in core.object_types}
195 children = []
196 for support in decision.children:
197 child = existing.get(support.type_id)
198 if child is None or child.category != "business_type":
199 raise ValueError("Generalization child is not an accepted business type")
200 children.append(child)
201 # A generic reusable quantity might be the basis of several Metrics, but
202 # that is a cross-root semantic relation, not an automatic subclass edge.
203 # Measure-to-Measure specialization needs a separate scope contract.
204 if decision.root_type == "Measure":
205 raise ValueError("Measure generalization requires a reusable-quantity scope contract")
206 parameters = dict(children[0].definition_parameters)
207 qualifiers = dict(children[0].identity_qualifiers)
208 classifications = dict(children[0].classification_attributes)
209 if any(item.classification_attributes != classifications for item in children[1:]):
210 raise ValueError("Child classification attributes differ; classification-dropping generalization is unsupported")
211 if any(item.identity_qualifiers != qualifiers for item in children[1:]):
212 raise ValueError("Child identity qualifiers differ; cross-namespace generalization is unsupported")
213 if any(item.definition_parameters != parameters for item in children[1:]):
214 raise ValueError("Child definition parameters differ; parameter-dropping generalization is unsupported")
215 units = {unit_identity(item.unit) for item in children}
216 if len(units) != 1 or unit_identity(decision.unit) not in units:
217 raise ValueError("Child units conflict with the proposed parent unit")
218 if decision.root_type == "Metric" and not next(iter(units)):
219 raise ValueError("Quantitative generalization requires known compatible units")
220 common_scope = {key: value for key, value in children[0].applicability_scope.items()
221 if all(item.applicability_scope.get(key) == value for item in children[1:])}
222 if decision.parent_scope != common_scope:
223 raise ValueError("Parent scope must equal the common applicability scope")
224 scope_keys = {frozenset(item.applicability_scope) for item in children}
225 if len(scope_keys) != 1:
226 raise ValueError("Child applicability scope keys differ")
227
228 label = " ".join(_norm(decision.label).split())
229 definition = " ".join(_norm(decision.definition).split())
230 identity = ["shared_supertype", decision.root_type, label,
231 definition, common_scope, unit_identity(decision.unit)]
232 # Empty parameters retain old IDs, allowing pre-parameter plans to replay.
233 if parameters:
234 identity.append(parameters)
235 if qualifiers:
236 identity.append({"identity_qualifiers": qualifiers})
237 if classifications:
238 identity.append({"classification_attributes": classifications})
239 parent_id = "type:" + digest(identity)[:24]
240 old_parent = existing.get(parent_id)
241 if old_parent and (old_parent.category != "business_type"
242 or old_parent.derivation_kind != "shared_supertype"
243 or old_parent.parent != decision.root_type
244 or old_parent.definition != decision.definition.strip()
245 or old_parent.definition_parameters != parameters
246 or old_parent.classification_attributes != classifications):
247 raise ValueError("Generalized parent ID conflicts with an existing type")
248
249 evidence_ids, templates, terms, record_ids = set(), [], set(), set()
250 for child, support in zip(children, decision.children):
251 if child.parent not in (decision.root_type, parent_id if old_parent else decision.root_type):
252 # The one-parent model cannot silently add a second business parent.
253 raise ValueError("Child already has a derived parent or mismatched root")
254 if not support.shared_quote.strip() or not support.specialization_quote.strip():
255 raise ValueError("Generalization lacks child source quotes")
256 term = support.specialization_term.strip()
257 if not _plain(term) or _plain(term) in terms:
258 raise ValueError("Specialization terms must be distinct")
259 terms.add(_plain(term))
260 if (_plain(term) in _plain(decision.label)
261 or _plain(term) in _plain(decision.definition)):
262 raise ValueError("Parent contains a child specialization")
263 if (_plain(term) not in _plain(child.label or "")
264 and not any(_plain(term) in _plain(value)
265 for value in child.applicability_scope.values())):
266 raise ValueError("Specialization term is not a child label or scope qualifier")
267 if _plain(decision.label) not in _plain(_norm(child.label or "").replace(_norm(term), "")):
268 raise ValueError("Parent label is absent from a specialized child label")
269 for key, value in child.applicability_scope.items():
270 if key not in common_scope and _plain(term) not in _plain(value):
271 raise ValueError("A varying scope value lacks the cited specialization")
272 evidence_id, _, fragment, record_id = _supporting_fragment(data, child, support)
273 if not record_id or record_id in record_ids:
274 raise ValueError("Generalization needs distinct source definition records")
275 record_ids.add(record_id)
276 if _plain(decision.shared_anchor) not in _plain(support.shared_quote):
277 raise ValueError("Shared anchor is absent from a child semantic quote")
278 template = _template(_norm(fragment).replace(_norm(term), ""))
279 if not template or _plain(decision.shared_anchor) not in template:
280 raise ValueError("Child fragment loses shared meaning after specialization removal")
281 templates.append(template)
282 evidence_ids.add(evidence_id)
283 if len(set(templates)) != 1 or _template(decision.definition) != templates[0]:
284 raise ValueError("Child definition templates or proposed parent definition conflict")
285 formula_sets = [_formula_templates(data, child, support.specialization_term.strip())
286 for child, support in zip(children, decision.children)]
287 if any(len(formulas) > 1 for formulas in formula_sets) or any(
288 formulas != formula_sets[0] for formulas in formula_sets[1:]):
289 raise ValueError("Child calculation formulas are incompatible")
290 evidence_ids.update(evidence_id for child in children
291 for evidence_id, role, _, _ in _semantic_fragments(data, child)
292 if role == "formula")
293 # Parameter witnesses can live outside description/formula columns; retain
294 # the accepted children's supporting evidence when inheriting their settings.
295 if parameters:
296 evidence_ids.update(evidence_id for child in children for evidence_id in child.evidence_ids)
297
298 if decision.root_type == "Metric":
299 anchor = _plain(decision.business_object_quote)
300 if (len(anchor) < 2 or anchor == _plain(decision.label)
301 or anchor not in _plain(decision.label)
302 or anchor not in _plain(decision.definition)
303 or not all(any(anchor in _plain(fragment)
304 for _, role, fragment, _ in _semantic_fragments(data, child)
305 if role == "description")
306 for child in children)):
307 raise ValueError("Metric parent requires a shared source-quoted business object")
308
309 child_ids = {item.id for item in children}
310 if old_parent:
311 if (set(old_parent.induced_from_type_ids) == child_ids
312 and all(item.parent == parent_id for item in children)
313 and set(old_parent.evidence_ids) == evidence_ids):
314 return core.model_copy(deep=True), old_parent
315 raise ValueError("Existing generalized parent has different children or evidence")
316 parent = DerivedType(
317 id=parent_id, parent=decision.root_type, label=decision.label.strip(),
318 definition=decision.definition.strip(), category="business_type",
319 unit=decision.unit, derivation_kind="shared_supertype",
320 induced_from_type_ids=sorted(item.id for item in children),
321 applicability_scope=common_scope,
322 definition_parameters=parameters,
323 identity_qualifiers=qualifiers,
324 classification_attributes=classifications,
325 evidence_scope="multiple_definition_records",
326 evidence_ids=sorted(evidence_ids),
327 source_concept_ids=sorted({concept_id for item in children
328 for concept_id in item.source_concept_ids}),
329 )
330 candidate = core.model_copy(deep=True)
331 candidate.object_types.append(parent)
332 candidate.object_types = [item.model_copy(update={"parent": parent_id})
333 if item.id in child_ids else item
334 for item in candidate.object_types]
335 errors = validate_plan(candidate, data, profile)
336 if errors:
337 raise ValueError("Generalized plan invalid: " + "; ".join(errors))
338 return candidate, parent
