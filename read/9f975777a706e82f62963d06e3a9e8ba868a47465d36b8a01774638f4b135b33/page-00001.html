15@lru_cache(maxsize=4096)
16def formula_symbols(expression):
17 tree = ast.parse(expression, mode="eval")
18 allowed = (ast.Expression, ast.BinOp, ast.UnaryOp, ast.Add, ast.Sub, ast.Mult, ast.Div, ast.USub, ast.UAdd, ast.Name, ast.Load, ast.Constant, ast.Call)
19 functions = {"sum", "avg", "min", "max", "abs"}
20 symbols = []
21 for node in ast.walk(tree):
22 if not isinstance(node, allowed):
23 raise ValueError("unsupported_formula_syntax")
24 if isinstance(node, ast.Call):
25 if not isinstance(node.func, ast.Name) or node.func.id.casefold() not in functions or node.keywords:
26 raise ValueError("unsupported_formula_function")
27 if isinstance(node, ast.Constant) and (isinstance(node.value, bool) or not isinstance(node.value, (int, float))):
28 raise ValueError("unsupported_formula_literal")
29 function_nodes = {id(n.func) for n in ast.walk(tree) if isinstance(n, ast.Call)}
30 for node in ast.walk(tree):
31 if isinstance(node, ast.Name) and id(node) not in function_nodes:
32 symbols.append(node.id)
33 return tuple(dict.fromkeys(symbols))

44def condition_fields(condition):
45 if condition is None:

50def terms(value):
51 text = str(value).casefold()
52 words = re.findall(r"[a-z0-9_]+|[\u4e00-\u9fff]", text)
53 return sorted(set(words))
