91def _replace_first_csv_row(path, **changes):
92 with path.open(newline="", encoding="utf-8") as stream:
93 reader = csv.DictReader(stream)
94 fields, rows = reader.fieldnames, list(reader)
95 rows[0].update(changes)
96 with path.open("w", newline="", encoding="utf-8") as stream:
97 writer = csv.DictWriter(stream, fieldnames=fields)
98 writer.writeheader()
99 writer.writerows(rows)
