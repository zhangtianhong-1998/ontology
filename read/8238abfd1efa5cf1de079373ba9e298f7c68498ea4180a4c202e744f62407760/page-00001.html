94def attach_concept_evidence(data, result):
95 """Register sampled definition fragments so any derived type can cite a row."""
96 for candidate in result["candidates"]:
97 for record in candidate["records"]:
98 for entries in record["fields"].values():
99 for entry in entries:
100 evidence_id = "record:" + digest([record["record_id"], entry["column"]])[:24]
101 entry["record_evidence_id"] = evidence_id
102 data.evidence[evidence_id] = {
103 "id": evidence_id, "origin": "observed_record",
104 "raw_fragment": entry["value"],
105 "raw_fragment_truncated": entry["truncated"],
106 "source_ref": {"table": record["table"], "record_id": record["record_id"],
107 "row": record["row_number"], "column": entry["column"],
108 "snapshot_id": data.snapshot_id},
109 }

112def add_fact_value_attributes(ontology, field_bindings):
113 """Add only value properties whose source field has an accepted type binding."""
114 attributes = {}
115 for binding in field_bindings:
116 if (binding.get("status") != "accepted_field_binding"
117 or binding.get("instances_created", 0) < 1):
118 continue
119 table, column, type_id = (binding["table"], binding["value_column"],
120 binding["type_id"])
121 attribute_id = "fact_value_property:" + digest([type_id, table, column])[:24]
122 attributes[(type_id, table, column)] = attribute_id
123 ontology["attributes"].append({
124 "id": attribute_id, "label": column, "relation_type": "has",
125 "domain": [type_id], "literal_type": "decimal",
126 "storage_type": "string", "source_table": table,
127 "source_column": column,
128 "evidence_scope": "observed_fact_value_in_source_snapshot",
129 "mapping_status": "observed_fact_value",
130 "evidence_ids": [f"schema:{table}:{column}",
131 binding["definition_evidence_id"]],
132 })
133 return attributes

158 canonical_map = result["canonical_map"]
159 groups = {}
160 for type_id, canonical_id in canonical_map.items():
161 groups.setdefault(canonical_id, set()).add(type_id)
162 for item in ontology.get("object_types", []):

164 continue

166 item["canonical_type_id"] = canonical_id
167 item["equivalent_type_ids"] = sorted(groups[canonical_id] - {item["id"]})
168 ontology["type_equivalences"] = [
169 assertion for assertion in result["equivalence_assertions"]
170 if canonical_map.get(assertion["source_type_id"])
171 == canonical_map.get(assertion["target_type_id"])
172 == assertion["canonical_type_id"]
