27def projection_match_report(template, record):
28 """Describe checked fields without promoting a partial match to identity."""
29 if record.get("table") != template.get("source_table"):
30 return None
31 values, roles = _fields(record)
32 semantic = {key for key in values if roles[key] & _SEMANTIC
33 and key not in template.get("instance_columns", {})}
34 matched, missing = set(), {}
35 for column in set(template["semantic_columns"]) ^ semantic:
36 missing[column] = "semantic_field_set_changed"
37 for column, expected in template["invariants"].items():
38 if values.get(column) == expected:
39 matched.add(column)
40 else:
41 missing[column] = "invariant_changed"
42 verified_invariants = sorted(matched)
43 slots = {item["name"]: item for item in template["slots"]}
44 captures = {name: slot["fixed_value"] for name, slot in slots.items() if "fixed_value" in slot}
45 pending = list(template["field_templates"])
46 while pending:
47 progressed = False
48 for field in pending[:]:
49 column = field["column"]
50 try:
51 found = (_match_text(field["template"], values[column], slots, values, captures)
52 if column in values else None)
53 except ValueError as error:
54 if str(error) == "Adjacent unbound slots are ambiguous":
55 continue
56 found = None
57 pending.remove(field)
58 progressed = True
59 if found is None:
60 missing[column] = "field_contract_mismatch"
61 else:
62 matched.add(column)
63 captures = found
64 if not progressed:
65 for field in pending:
66 missing[field["column"]] = "unbound_slot_ambiguity"
67 break
68 try:
69 complete = match_projection(template, record)
70 except ValueError:
71 complete = None
72 class_quote = (template.get("class_definition") or {}).get("quote") or template.get("definition", "")
73 class_supported = bool(class_quote and any(class_quote in values[column]
74 for column in values if "description" in roles[column]))
75 if template.get("class_definition_evidence"):
76 class_supported = all(values.get(item["column"]) == item["quote"]
77 for item in template["class_definition_evidence"])
78 for name in _slot_target_mismatches(template["slots"], captures):
79 missing["slot:" + name] = "captured_value_not_supported_by_target_name_or_alias"
80 protected = {column for column in values if roles[column] & {"formula", "unit", "identity"}}
81 unresolved_slots = [{"name": slot["name"], "role": slot["role"], "label": slot["label"],
82 "observed_value": captures.get(slot["name"])}
83 for slot in template["slots"]
84 if slot["role"] in SEMANTIC_SLOTS and not slot.get("target_type_id")]
85 return {"template_id": template["template_id"], "object_type_id": template["object_type_id"],
86 "status": "complete_match" if complete else "partial_match",
87 "identity_claim": "complete_contract_only" if complete else "none",
88 "source_record_id": record["record_id"], "matched_columns": sorted(matched - set(missing)),
89 "unresolved_fields": [{"column": key, "reason": value} for key, value in sorted(missing.items())],
90 "verified_invariant_columns": verified_invariants,
91 "protected_fields_changed": sorted(column for column in protected
92 if column in template["invariants"]
93 and values.get(column) != template["invariants"][column]),
94 "class_quote_supported": class_supported, "unresolved_slots": unresolved_slots,
95 "slot_values": complete["slot_values"] if complete else captures}

99 """Rank all accepted contracts by source checks, not insertion recency."""
100 record = representative_record(bundle)
101 if record is None or limit <= 0:
102 return []

113 # Keep partial evidence visible but do not call shared units or words

116 "template_id", "object_type_id", "source_table", "root_type", "label", "definition",

218def _class_declaration_candidates(record):
219 """Recall whole, explicitly declarative clauses; never invent a class name.
220
221 These source fragments are scheduling candidates only. The source fields
222 remain complete in the packet and the normal compiler/reviewer decides
223 whether a quoted clause really declares the proposed business class.
224 """
225 result = []
226 for entry in record.get("fields", {}).get("description", []):
227 if entry.get("truncated"):
228 continue
229 raw = str(entry["value"])
230 for clause in re.split(r"[。！？；;\n]|(?<!\d)\.(?:\s+|$)|[!?](?:\s+|$)", raw):
231 quote = clause.strip()
232 chinese = re.fullmatch(r"([^，,]{1,80}?)(?:是一(?:类|种)|是用于|是用来).+", quote)
233 english = re.fullmatch(r"(.{1,80}?)\s+is\s+(?:a|an)\s+(?:class|type|kind|category)\s+of\s+.+",
234 quote, flags=re.I)
235 matched = chinese or english
236 if matched and not re.search(r"(?:不|并非|不是|非)$", matched[1]):
237 result.append({"column": entry["column"], "quote": quote,
238 "label_fragment": matched[1].strip(), "status": "source_clause_candidate"})
239 return result
