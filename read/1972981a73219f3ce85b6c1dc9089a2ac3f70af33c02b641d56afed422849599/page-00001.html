14def run(dataset, output, *, max_cards=100000, max_concept_bundles=80,
15 max_pattern_seed_pool=1000, seed_window_index=0):
16 output = Path(output)
17 output.mkdir(parents=True, exist_ok=False)
18 (output / "work").mkdir()
19 start = time.monotonic()
20 data = Dataset(Path(dataset), output / "work", "2GB")
21 try:
22 facts = build_fact_observation_candidates(data)
23 cards = build_semantic_cards(
24 data, output / "work" / "semantic_cards.sqlite", max_cards=max_cards)
25 index = SemanticCardIndex(cards["index_path"])
26 try:
27 packets = build_instance_bundles(
28 data, index, {"rules": []},
29 {"max_concept_bundles": max_concept_bundles,
30 "max_relation_bundles": 0,
31 "max_pattern_seed_pool": max_pattern_seed_pool,
32 "seed_window_index": seed_window_index})
33 finally:
34 index.close()
35 card_coverage, bundle_coverage = cards["coverage"], packets["coverage"]
36 summary = {
37 "input_scope": "imported_csv_snapshot; synthetic status is caller-supplied",
38 "tables": len(data.tables),
39 "rows": sum(table["rows"] for table in data.tables.values()),
40 "definition_cards": bundle_coverage["definition_cards_indexed"],
41 "definition_patterns": bundle_coverage["definition_patterns_indexed"],
42 "concept_bundles": bundle_coverage["bundles_by_task"]["concept_induction"],
43 "seed_window_index": seed_window_index,
44 "seed_roots": bundle_coverage["seed_roots"],
45 "definition_cards_not_seeded": bundle_coverage["definition_cards_not_seeded"],
46 "definition_patterns_unprocessed": bundle_coverage["definition_patterns_unprocessed"],
47 "business_fact_tables": facts["coverage"]["business_fact_tables"],
48 "business_fact_candidates": facts["coverage"]["emitted_candidates"],
49 "business_fact_instances_created": card_coverage["business_fact_instances_created"],
50 "llm_calls": 0, "elapsed_seconds": round(time.monotonic() - start, 2),
51 "semantic_quality": "unjudged; no ontology induction or relation judgment",
52 }
53 write_yaml(output / "semantic_card_coverage.yaml", card_coverage)
54 write_yaml(output / "instance_bundle_coverage.yaml", bundle_coverage)
55 write_yaml(output / "fact_observation_coverage.yaml", facts["coverage"])
56 write_yaml(output / "fact_observation_candidates.yaml", facts)
57 write_yaml(output / "summary.yaml", summary)
58 return summary
59 finally:
60 data.close()
