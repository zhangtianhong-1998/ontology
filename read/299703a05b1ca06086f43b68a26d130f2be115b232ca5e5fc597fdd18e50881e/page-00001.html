17ROLE_CUES = {
18 "alias": (r"(?:^|_)(?:alias|aka|synonym)(?:_|$)", r"别名|同义名|曾用名"),
19 "name": (r"(?:^|_)(?:name|title|label)(?:_|$)", r"名称|名字|标题"),
20 "description": (r"(?:^|_)(?:definition|description|desc|meaning|comment|explanation)(?:_|$)", r"定义|说明|描述|含义|解释"),
21 "formula": (r"(?:^|_)(?:formula|calculation|expression|expr|caliber|rule)(?:_|$)", r"公式|计算|口径"),
22 "unit": (r"(?:^|_)(?:unit|uom)(?:_|$)", r"单位|量纲"),
23 "scope": (r"(?:^|_)(?:scope|domain|region|area|range|applicable|period)(?:_|$)", r"范围|适用|业务域|地区|区域|周期"),
24}
25_IDENTIFIER_SUFFIX = re.compile(r"(?:^|_)(?:id|code|key|uuid|guid|no|number)$", re.I)
26_IDENTIFIER_DECLARATION = re.compile(
27 r"(?:编号|编码|标识符|引用键|主键|外键)(?:[，,;；。.\s]|$)|"
28 r"\b(?:identifier|primary key|foreign key|reference key)\b", re.I)
29_RULE_PARAMETER = re.compile(
30 r"(?:^|_)(?:rule|formula|expression|calculation)_(?:type|kind|mode|operator)$", re.I)
31_REFERENCE_SELECTOR_DECLARATION = re.compile(
32 r"(?:引用|关联|目标)(?:类型|种类|类别)|"

36def _raw_field_roles(table):
37 """Collect metadata and source-checked candidates before conflict resolution."""
38 profiles = {item["column"]: item for item in table.get("profiles", [])}
39 excluded = set(table.get("semantic_excluded_columns") or ())
40 selected = defaultdict(list)
41 for column in table["columns"]:
42 name = column["column_name"]
43 if name in excluded or is_sensitive_column(column):
44 continue
45 profile = profiles.get(name, {})
46 if profile.get("scan_scope") == "full_input" and profile.get("usable_count") == 0:
47 continue
48 comment = str(column.get("column_comment") or "")
49 # An explicit column name is stronger than a comment mentioning several roles.
50 role = next((key for key, (pattern, _) in ROLE_CUES.items()
51 if re.search(pattern, name.casefold())), None)
52 if role is None and not re.search(r"(?:^|_)(?:id|code|key|uuid|time|date|no|number)(?:_|$)", name.casefold()):
53 role = next((key for key, (_, pattern) in ROLE_CUES.items()
54 if re.search(pattern, comment)), None)
55 if role:
56 selected[role].append(name)
57 context_role = declared_context_role(column)
58 if context_role:
59 # A governance/source field may carry a readable name; retain that
60 # source text, but never route its context candidate into scope.
61 if name in selected.get("scope", ()):
62 selected["scope"].remove(name)
63 selected[context_role].append(name)
64 # Competing source-backed role candidates coexist for recall. Neither a
65 # metadata cue nor a model proposal establishes the final business role.
66 for item in table.get("inferred_semantic_roles", ()):
67 name, role = item.get("column"), item.get("role")
68 if (item.get("status") != "source_verified_role_candidate"
69 or role not in {*ROLE_CUES, *CONTEXT_ROLES, "reference", "unknown"} or name not in profiles
70 or name in excluded or name in selected.get(role, ())):
71 continue
72 column = next((c for c in table["columns"] if c["column_name"] == name), None)
73 if (column is None or is_sensitive_column(column)
74 or profiles[name].get("scan_scope") != "full_input"
75 or profiles[name].get("usable_count", 0) == 0):
76 continue
77 resolution = resolve_context_role(table, item)
78 effective_role = resolution["effective_role"] if resolution else role
79 if effective_role in CONTEXT_ROLES and name in selected.get("scope", ()):
80 selected["scope"].remove(name)
81 if name not in selected[effective_role]:
82 selected[effective_role].append(name)
83 return dict(selected)

86def field_role_conflicts(table, roles=None):
87 """Resolve strong declaration conflicts without discarding source fields.
88
89 A key's semantic-looking prefix does not make its value a name or formula.
90 Only identifier declarations/suffixes trigger this boundary; a primary key
91 alone and a numeric-looking value do not. Rule contents remain candidates.
92 """
93 roles = _raw_field_roles(table) if roles is None else roles
94 columns = {item["column_name"]: item for item in table["columns"]}
95 table_name = table.get("name") or ".".join(filter(None, (
96 table.get("schema"), table.get("table_name"))))
97 conflicts = []
98 for candidate in table.get("inferred_semantic_roles", ()):
99 if candidate.get("status") != "source_verified_role_candidate":
100 continue
101 resolution = resolve_context_role(table, candidate)
102 if resolution and resolution["effective_role"] != resolution["proposed_role"]:
103 conflicts.append(resolution)
104 for role in ("name", "alias", "formula", "description"):
105 for field in roles.get(role, ()):
106 column = columns[field]
107 normalized = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", field).casefold()
108 identifier_name = bool(_IDENTIFIER_SUFFIX.search(normalized))
109 identifier_declared = bool(_IDENTIFIER_DECLARATION.search(str(column.get("column_comment") or "")))
110 reference_selector = bool(_REFERENCE_SELECTOR_DECLARATION.search(
111 str(column.get("column_comment") or "")))
112 base = {"column": field, "proposed_role": role,
113 "schema_evidence_id": f"schema:{table_name}:{field}",
114 "resolution_origin": "source_declaration_conflict",
115 "semantic_identity_claim": False}
116 if identifier_name or identifier_declared or reference_selector:
117 conflicts.append({**base, "effective_role": "reference", "binding_only": True,
118 "reason": "identifier_declaration_does_not_support_name_alias_or_formula",
119 "identifier_name": identifier_name, "identifier_declared": identifier_declared,
120 "reference_selector_declared": reference_selector,
121 "declared_primary_key": field in table.get("pk", ())})
122 elif role == "formula" and _RULE_PARAMETER.search(normalized):
123 conflicts.append({**base, "effective_role": "scope", "binding_only": False,
124 "formula_status": "fragment", "parameter_role": "rule_parameter",
125 "reason": "rule_type_or_mode_is_a_parameter_not_complete_rule_content"})
126 return conflicts

129def _field_roles(table):
130 """Retain all supported roles, resolving strong key/content conflicts first.
131
132 No per-role slots discard business fields. Resolved bindings remain in
133 card references; rule parameters remain in scope and pattern identity.
134 """
135 selected = _raw_field_roles(table)
136 for conflict in field_role_conflicts(table, selected):
137 field, role = conflict["column"], conflict["proposed_role"]
138 if field in selected.get(role, ()):
139 selected[role].remove(field)
140 if conflict["effective_role"] == "scope" and field not in selected.setdefault("scope", []):
141 selected["scope"].append(field)
142 return {role: fields for role, fields in selected.items() if fields}

145def _name_cues(value):
146 normalized = unicodedata.normalize("NFKC", value).casefold().strip()
147 pieces = re.findall(r"[a-z0-9]+|[\u4e00-\u9fff]+", normalized)
148 cues = set()
149 for piece in pieces:
150 if re.fullmatch(r"[\u4e00-\u9fff]+", piece):
151 cues.update("gram:" + piece[i:i + 2] for i in range(len(piece) - 1))
152 elif len(piece) >= 3:
153 cues.add("term:" + piece)
154 if normalized:
155 cues.add("exact:" + normalized)
156 return cues

159def _record_card(data, table_name, row, roles, max_field_chars):
160 values = {}
161 for role, columns in roles.items():
162 entries = []
163 for column in columns:
164 raw = row.get(column)
165 if raw is None or not str(raw).strip():
166 continue
167 value = str(raw)
168 entries.append({"column": column, "value": value[:max_field_chars],
169 "truncated": len(value) > max_field_chars,
170 "schema_evidence_id": f"schema:{table_name}:{column}"})
171 if entries:
172 values[role] = entries
173 if not values.get("name"):
174 return None
175 return {"table": table_name, "record_id": data.record_id(table_name, row),
176 "row_number": row["__r2_row"], "fields": values}

179def _sample_definition_rows(data, table_name, roles, limit, max_field_chars):
180 table = data.tables[table_name]
181 columns = list(dict.fromkeys([field for values in roles.values() for field in values] + table["pk"]))
182 names = [f"nullif(trim({qi(field)}), '')" for field in roles["name"]]
183 name_key = "coalesce(" + ", ".join([*names, "''"]) + ")"
184 sql = (f"SELECT __r2_row, {', '.join(map(qi, columns))} FROM {qi(table['sql_name'])} "
185 f"WHERE {name_key} <> '' ORDER BY md5({name_key}), __r2_row LIMIT ?")
186 rows = data.db.execute(sql, [limit + 1]).fetchall()
187 cards = []
188 for values in rows[:limit]:
189 row = dict(zip(["__r2_row", *columns], values))
190 card = _record_card(data, table_name, row, roles, max_field_chars)
191 if card is not None:
192 cards.append(card)
193 return cards, len(rows) > limit

201def recall_concept_candidates(data, *, max_definition_tables=16,
202 max_records_per_table=64, max_total_records=1024,
203 max_name_cue_fanout=32, max_pair_comparisons=10000,
204 max_candidates=200, max_field_chars=256):
205 """Recall bounded record pairs from names/aliases; return evidence and coverage.
206
207 SQL scans the imported CSV snapshot and returns at most the configured row
208 budget. The result is only a list of records worth semantic comparison.
209 """
210 limits = (max_definition_tables, max_records_per_table, max_total_records,
211 max_name_cue_fanout, max_pair_comparisons, max_candidates,
212 max_field_chars)
213 if any(type(value) is not int or value <= 0 for value in limits):
214 raise ValueError("Concept candidate limits must be positive integers")
215
216 choices = []
217 skipped = []
218 for table_name, table in sorted(data.tables.items()):
219 roles = _field_roles(table)
220 if not roles.get("name"):
221 skipped.append({"table": table_name, "reason": "no_usable_name_field"})
222 continue
223 choices.append((table_name, roles))
224 choices.sort(key=lambda pair: (-sum(bool(pair[1].get(role)) for role in
225 ("description", "formula", "unit", "scope")), pair[0]))
226 skipped.extend({"table": name, "reason": "table_budget"}
227 for name, _ in choices[max_definition_tables:])
228 cards = []
229 coverage = []
230 for table_name, roles in choices[:max_definition_tables]:
231 remaining = max_total_records - len(cards)
232 if remaining <= 0:
233 skipped.append({"table": table_name, "reason": "record_budget"})
234 continue
235 limit = min(max_records_per_table, remaining)
236 sampled, truncated = _sample_definition_rows(data, table_name, roles, limit, max_field_chars)
237 cards.extend(sampled)
238 coverage.append({"table": table_name, "input_rows": data.tables[table_name]["rows"],
239 "selected_records": len(sampled), "sample_truncated": truncated,
240 "selected_columns": roles})
241
242 index = defaultdict(set)
243 for position, card in enumerate(cards):
244 names = [entry["value"] for role in ("name", "alias")
245 for entry in card["fields"].get(role, [])]
246 for name in names:
247 for cue in _name_cues(name):
248 index[cue].add(position)
249 pairs = defaultdict(set)
250 skipped_fanout = 0
251 comparisons = 0
252 comparison_budget_exhausted = False
253 for cue, holders in sorted(index.items(), key=lambda item: (len(item[1]), item[0])):
254 if len(holders) > max_name_cue_fanout:
255 skipped_fanout += 1
256 continue
257 for left, right in combinations(sorted(holders), 2):
258 if comparisons >= max_pair_comparisons:
259 comparison_budget_exhausted = True
260 break
261 comparisons += 1
262 pairs[(left, right)].add(cue)
263 if comparison_budget_exhausted:
264 break
265
266 ranked = sorted(pairs, key=lambda pair: (
267 not any(cue.startswith("exact:") for cue in pairs[pair]),
268 -len(pairs[pair]), cards[pair[0]]["record_id"], cards[pair[1]]["record_id"]))
269 candidates = []
270 for left, right in ranked[:max_candidates]:
271 a, b = cards[left], cards[right]
272 units_a, units_b = _unit_values(a), _unit_values(b)
273 cues = pairs[(left, right)]
274 candidates.append({
275 "candidate_id": digest([data.snapshot_id, a["record_id"], b["record_id"]])[:24],
276 "source_snapshot": data.snapshot_id,
277 "records": [a, b],
278 "shared_name_cues": sorted(cues),
279 "retrieval_channels": (["normalized_exact_name"] if any(c.startswith("exact:") for c in cues) else [])
280 + (["name_or_alias_fragment"] if any(not c.startswith("exact:") for c in cues) else []),
281 "unit_observation": "different" if units_a and units_b and units_a.isdisjoint(units_b)
282 else "same" if units_a and units_b else "unknown",
283 "decision": {"status": "proposed", "same_concept": "unresolved"},
284 })
285 return {"candidates": candidates, "coverage": {
286 "input_scope": "imported_csv_snapshot_only", "definition_tables_considered": len(choices),
287 "definition_tables_sampled": len(coverage), "tables": coverage,
288 "tables_skipped": skipped, "definition_records_sampled": len(cards),
289 "name_cues_skipped_high_fanout": skipped_fanout,
290 "pair_comparisons": comparisons,
291 "pair_comparison_budget_exhausted": comparison_budget_exhausted,
292 "candidate_pairs_found": len(ranked), "candidates_returned": len(candidates),
293 "candidate_pairs_not_returned": max(0, len(ranked) - len(candidates)),
294 "partial": (bool(skipped) or any(t["sample_truncated"] for t in coverage)
295 or skipped_fanout > 0 or comparison_budget_exhausted
296 or len(ranked) > len(candidates)),
297 "semantic_status": "unjudged; name overlap and units do not prove identity",
298 }}
