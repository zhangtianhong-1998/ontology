79def _checked_links(data, table_name, association_context):
80 """Return current checked links, preserving their conditions and direction."""
81 graph = getattr(association_context, "graph", association_context) or {}
82 if not isinstance(graph, dict):
83 return [], set()
84 owners = {edge["target"]: edge["source"] for edge in graph.get("edges", [])
85 if edge.get("type") == "table_has_column"}
86 neighbors = set()
87 links = []
88 for edge in graph.get("edges", []):
89 if (edge.get("type") != "technical_link" or edge.get("status") != "checked_technical"
90 or edge.get("snapshot_id") != data.snapshot_id):
91 continue
92 a, b = owners.get(edge.get("source")), owners.get(edge.get("target"))
93 if table_name in (a, b):
94 neighbors.update(name for name in (a, b) if name and name != table_name)
95 # These are recall leads, not type identity evidence. Keep their
96 # risks and full verification in both the prompt and reusable
97 # contract fingerprint; a changed risk must invalidate reuse.
98 links.append({key: deepcopy(edge[key]) for key in (
99 "source", "target", "status",
100 "selector", "scope_bindings", "transform", "semantic_relation",
101 "numeric_overlap_only", "risk_flags", "verification", "meaning",
102 "lineage_inferred",
103 ) if key in edge})
104 return sorted(links, key=digest), neighbors

123def _packet(data, report, field, core, association_context, max_samples):
124 table = data.tables[report["table"]]
125 columns = {item["column_name"]: item for item in table["columns"]}
126 name, coordinates = field["column"], list(report.get("coordinate_columns") or [])
127 safe = {item["column"] for item in classify_columns(table)
128 if item["role"] not in ("sensitive", "empty", "audit_time", "audit_metadata")}
129 if name not in safe or not set(coordinates) <= safe:
130 raise ValueError("Fact field or coordinate is excluded from semantic inspection")
131 declaration = _declaration(data, table, columns[name])
132 # DISTINCT is applied to complete observed rows, preserving co-occurrence.
133 selected = list(dict.fromkeys([*coordinates, name]))
134 fields = ", ".join(qi(value) for value in selected)
135 rows = data.db.execute(f"SELECT {fields}, min(__r2_row) FROM {qi(table['sql_name'])} "
136 f"GROUP BY {fields} ORDER BY min(__r2_row) LIMIT ?", [max_samples + 1]).fetchall()
137 samples = [{"row_number": row[-1], "values": dict(zip(selected, row[:-1]))}
138 for row in rows[:max_samples]]
139 units = [value for column in columns if column in safe
140 if (value := field_unit_evidence(data, table, column)) is not None]
141 links, neighbors = _checked_links(data, table["name"], association_context)
142 context = _definition_context(data, core, neighbors)
143 roles = [{"column": item["column"], "role": item["role"], "status": item["status"],
144 "semantic_status": item.get("semantic_status", "unjudged")}
145 for item in table.get("inferred_semantic_roles", [])
146 if item.get("status") == "source_verified_role_candidate" and item.get("column") in safe]
147 packet = {"table": table["name"], "table_comment": table.get("table_comment") or "",
148 "value_column": name, "source_declaration": declaration,
149 "coordinate_columns": coordinates,
150 "coordinate_declarations": [columns[value] for value in coordinates],
151 "source_checked_role_candidates": roles, "sample_rows": samples,
152 "sample_scope": "bounded_actual_distinct_rows_only",
153 "more_sample_rows_available": len(rows) > max_samples,
154 "unit_columns": units, "checked_association_conditions": links,
155 "declared_unit_quote_contract": {
156 "recognized_units": sorted(_unit_from_comment(declaration["value"])),
157 "required_unit_quote_if_declared": declaration["value"],
158 "rule": "When unit comes from this field declaration, unit_quote MUST equal the entire required_unit_quote_if_declared, not just a unit word or substring."
159 },
160 "associated_definitions": context,
161 "contract": "Declare only a source-named business quantity. Samples do not name a Metric. "
162 "Missing formulas and units stay unknown; scope keys must map to actual coordinate columns. "
163 "Do not turn row-varying tenant values into types. identity_qualifiers require a literal "
164 "key=value (or key: value) clause in the complete field declaration, an explicit "
165 "scope_bindings coordinate, and the same nonblank value across the full input. "
166 "Preserve every explicit identity clause; omitting one or returning an empty object "
167 "cannot remove the source restriction. Unsupported identity clauses remain unresolved."}
168 # Current record evidence IDs are excluded from the reusable contract, but
169 # the complete semantic statements and unit completeness still participate.
170 context_signature = [{key: value for key, value in item.items() if key != "source_definitions"} |
171 {"source_definitions": [(value["role"], value["value"]) for value in item["source_definitions"]]}
172 for item in context]
173 unit_signature = [{key: value for key, value in item.items() if key != "evidence_id"} for item in units]
174 packet["contract_fingerprint"] = digest([field_contract_fingerprint(table, name, coordinates),
175 context_signature, unit_signature, links])
176 return packet

179def _declared_identity_qualifiers(table, raw):
180 """Read explicit identity clauses independently of the proposed decision.
181
182 Only recognized identity keys or source-checked identity columns qualify.
183 A business clause such as limit=100 is not automatically type identity.
184 Unsupported or conflicting identity clauses stay unresolved.
185 """
186 identity_fields = {item["column_name"] for item in table["columns"]
187 if declared_context_role(item) == "identity"}
188 identity_fields.update(item["column"] for item in table.get("inferred_semantic_roles", [])
189 if item.get("status") == "source_verified_role_candidate"
190 and item.get("role") == "identity")
191 names = {_norm(name) for name in identity_fields}
192 names.update(_norm(re.sub(r"_(?:id|code|name|number)$", "", name, flags=re.I))
193 for name in identity_fields)
194 clauses = re.finditer(r"(?<!\w)(?P<key>[^\W\d]\w*)\s*(?P<operator>[:：=]|[<>!]=?)"
195 r"\s*(?P<value>[^;；,，\n)）]*)", raw)
196 declared = {}
197 for clause in clauses:
198 key, value = clause["key"], clause["value"].strip()
199 identity_key = (_norm(key) in names or declared_context_role({"column_name": key,
200 "column_comment": key}) == "identity" or key in {"租户", "命名空间", "定义版本", "定义修订"})
201 if not identity_key:
202 continue
203 if (clause["operator"] not in {"=", ":", "："}
204 or not re.fullmatch(r"[^\s=<>!:：()（）]+", value)
205 or key in declared and declared[key] != value):
206 raise ValueError("Identity qualifier declaration is unsupported or conflicting")
207 declared[key] = value
208 return declared

211def _checked_identity_qualifiers(data, table, raw, decision, coordinates):
212 """Qualify a source-field type only by declared, constant source identity."""
213 evidence_ids = []
214 qualifiers = dict(sorted(decision.identity_qualifiers.items()))
215 declared = _declared_identity_qualifiers(table, raw)
216 if qualifiers != declared:
217 raise ValueError("Identity qualifiers must preserve every explicit source identity declaration")
218 for key, value in qualifiers.items():
219 column = decision.scope_bindings.get(key)
220 clause = (re.escape(key) + r"\s*[=:：]\s*" + re.escape(value)
221 + r"(?=$|[\s,，。;；)）])")
222 if (not key.strip() or not value.strip() or not re.search(clause, raw)
223 or column not in coordinates or column not in table["column_names"]):
224 raise ValueError("Identity qualifier requires a literal field declaration and an explicit coordinate binding")
225 # Inspect complete input, never infer constancy from packet samples.
226 values = data.db.execute(f"SELECT DISTINCT {qi(column)} FROM {qi(table['sql_name'])} LIMIT 2").fetchall()
227 if values != [(value,)]:
228 raise ValueError("Identity qualifier coordinate is not one declared constant across the full input")
229 evidence_id = "field_identity:" + digest([data.snapshot_id, table["name"], column, key, value])[:24]
230 data.evidence[evidence_id] = {"id": evidence_id, "origin": "observed_identity_column",
231 "raw_fragment": value, "raw_fragment_truncated": False,
232 "source_ref": {"table": table["name"], "column": column, "snapshot_id": data.snapshot_id},
233 "verification": {"scan_scope": "full_input", "distinct_nonblank_values": 1,
234 "identity_key": key, "source_field_declaration": raw}}
235 evidence_ids.append(evidence_id)
236 return qualifiers, evidence_ids

239def compile_fact_schema(data, core, report, packet, decision):
240 """Compile a source-schema Metric and an auditable field binding template."""
241 decision = FactSchemaDecision.model_validate(decision)
242 if decision.status != "proposed":
243 return None
244 declaration = packet["source_declaration"]
245 name = packet["value_column"]
246 table = data.tables[packet["table"]]
247 if (decision.identity_evidence_id != declaration["evidence_id"]
248 or decision.identity_quote != declaration["value"]):
249 raise ValueError("Metric identity requires the full original field declaration")
250 object_, quantity, label = (decision.business_object_quote.strip(), decision.quantity_quote.strip(), decision.label.strip())
251 raw = declaration["value"]
252 if (not label or any(len(_surface(value)) < 2 or value not in raw
253 or _surface(value) in _GENERIC for value in (object_, quantity))
254 or _surface(object_) == _surface(quantity)
255 or object_ in quantity or quantity in object_):
256 raise ValueError("Metric identity requires distinct source-declared business object and quantity")
257 if (_surface(label) not in _surface(raw) or _surface(object_) not in _surface(label)
258 or _surface(quantity) not in _surface(label)):
259 raise ValueError("Metric label must retain both literal source identity anchors")
260 coordinates = packet["coordinate_columns"]
261 if (not set(decision.scope_bindings.values()) <= set(coordinates)
262 or len(set(decision.scope_bindings.values())) != len(decision.scope_bindings)
263 or any(not key.strip() for key in decision.scope_bindings)):
264 raise ValueError("Scope binding must explicitly name unique observed coordinate columns")
265 qualifiers, identity_evidence_ids = _checked_identity_qualifiers(data, table, raw, decision, coordinates)
266 observed_units = _unit_from_comment(raw)
267 unit_evidence_ids = []
268 if decision.unit_column:
269 observed = next((item for item in packet["unit_columns"] if item["column"] == decision.unit_column), None)
270 if (observed is None or observed["status"] != "verified_single_unit"
271 or observed["value"] != decision.unit_quote or _unit_key(observed["value"]) != _unit_key(decision.unit)):
272 raise ValueError("Unit must quote a complete current-snapshot unit column")
273 unit_evidence_ids.append(observed["evidence_id"])
274 observed_units.add(_unit_key(observed["value"]))
275 elif decision.unit and (decision.unit_quote != raw or observed_units != {_unit_key(decision.unit)}):
276 raise ValueError("Unit must appear explicitly in the complete field declaration")
277 if observed_units != ({_unit_key(decision.unit)} if decision.unit else set()):
278 raise ValueError("Unit is missing or conflicts with source declarations")
279 unit = decision.unit.strip() if decision.unit else None
280 # The text is deterministic: a model cannot smuggle in an unquoted formula.
281 definition = f"{raw}；来源事实字段 {table['name']}.{name} 的观测量，计算口径未提供。"
282 identity_signature = [table["name"], name, raw, label, object_, quantity, unit]
283 if qualifiers:
284 identity_signature.append(qualifiers)
285 type_id = "type:fact_field:" + digest(identity_signature)[:24]
286 evidence_ids = [declaration["evidence_id"], *unit_evidence_ids, *identity_evidence_ids]
287 item = DerivedType(id=type_id, parent="Metric", label=label, definition=definition,
288 category="business_type", evidence_scope="source_schema", unit=unit,
289 identity_qualifiers=qualifiers,
290 evidence_ids=evidence_ids,
291 semantic_parameters={"business_object_quote": object_, "quantity_quote": quantity,
292 "calculation_status": "unknown", "identity_basis": "source_field_declaration"},
293 source_properties=[{"role": "name", "source_table": table["name"],
294 "source_column": name, "evidence_ids": [declaration["evidence_id"]]}])
295 binding = FactFieldBindingDecision(status="bind", type_id=item.id, source_column_quote=raw,
296 type_definition_quote=definition,
297 source_definition_evidence_id=declaration["evidence_id"], type_source_quote=raw,
298 scope_bindings=decision.scope_bindings, unit_column=decision.unit_column)
299 template = {"id": "field_binding_template:" + digest([table["name"], name, packet["contract_fingerprint"], qualifiers])[:24],
300 "status": "accepted_field_template", "table": table["name"], "value_column": name,
301 "source_snapshot_id": data.snapshot_id, "type_id": item.id,
302 "type_signature": digest(item.model_dump()),
303 "schema_fingerprint": field_contract_fingerprint(table, name, coordinates),
304 "contract_fingerprint": packet["contract_fingerprint"],
305 "scope_bindings": decision.scope_bindings, "unit_status": "source_verified" if unit else "unknown",
306 "identity_qualifiers": qualifiers, "identity_evidence_ids": identity_evidence_ids,
307 "calculation_status": "unknown", "evidence_ids": evidence_ids,
308 "decision": decision.model_dump(), "binding_decision": binding.model_dump()}
309 return item, template

323def _fact_schema_packets(packet, llm, max_packet_bytes):
324 """Partition complete context units; every page retains identity/unit rules.
325
326 All pages must agree before the original complete contract can compile.
327 A single oversized declaration, row or link is left unresolved, never cut.
328 """
329 input_limit = getattr(llm, "config", {}).get("max_input_bytes", 100000)
330
331 def sizes(value):
332 payload_bytes = len(json.dumps(value, ensure_ascii=False).encode())
333 request_bytes = (llm.request_bytes("fact_schema_induction", value, FactSchemaDecision)
334 if hasattr(llm, "request_bytes") else payload_bytes)
335 return payload_bytes, request_bytes
336
337 def fits(value):
338 payload_bytes, request_bytes = sizes(value)
339 return payload_bytes <= max_packet_bytes and request_bytes <= input_limit
340
341 def oversized(value):
342 payload_bytes, request_bytes = sizes(value)
343 error = (InputBudgetExceeded(payload_bytes, max_packet_bytes, budget_name="max_packet_bytes")
344 if payload_bytes > max_packet_bytes else InputBudgetExceeded(request_bytes, input_limit))
345 error.packet_bytes = payload_bytes
346 error.structured_request_bytes = request_bytes
347 return error
348
349 if fits(packet):
350 return [packet]
351 keys = ("checked_association_conditions", "associated_definitions", "sample_rows")
352 base = {key: deepcopy(value) for key, value in packet.items() if key not in keys}
353 units, counts = [], {}
354 for key in keys:
355 # Identical JSON evidence is losslessly interned; distinct conditions,
356 # directions, source records and complete verification stay separate.
357 unique = {digest(value): value for value in packet.get(key, [])}
358 counts[key] = {"original": len(packet.get(key, [])), "distinct": len(unique)}
359 units.extend((key, value) for value in unique.values())
360 maximum_pages = max(1, len(units))
361
362 def page(values, number, total=maximum_pages):
363 return {**base, **{key: [deepcopy(value) for kind, value in values if kind == key]
364 for key in keys},
365 "context_partition": {"page_index": number, "page_count": total,
366 "complete_units": counts,
367 "contract": "This is one complete-unit context page for the same field declaration. "
368 "No page alone accepts a type: every page must propose the same identity, "
369 "scope and unit before compilation. Do not invent or truncate evidence."}}
370
371 grouped, current = [], []
372 for unit in units:
373 if fits(page([*current, unit], maximum_pages)):
374 current.append(unit)
375 continue
376 if current:
377 grouped.append(current)
378 current = []
379 if not fits(page([unit], maximum_pages)):
380 raise oversized(page([unit], maximum_pages))
381 current.append(unit)
382 if current or not grouped:
383 grouped.append(current)
384 pages = [page(values, number, len(grouped)) for number, values in enumerate(grouped, 1)]
385 if not all(fits(value) for value in pages):
386 raise oversized(next(value for value in pages if not fits(value)))
387 return pages
