20def build_projection_bindings(data, index, templates, *, max_records=100000, max_gap_examples=32):
21 """Apply accepted projection contracts to observed rows, outside LLM budgets.
22
23 The index narrows source tables only. Complete original values are fetched
24 in fixed batches, including independent record identifiers. No Cartesian
25 product, name-based identity merge or business-relation inheritance occurs.
26 """
27 if type(max_records) is not int or max_records < 0:
28 raise ValueError("Projection binding limit must be nonnegative")
29 if type(max_gap_examples) is not int or max_gap_examples < 0:
30 raise ValueError("Projection gap example limit must be nonnegative")
31 if index.coverage.get("snapshot_id") != data.snapshot_id:
32 raise ValueError("Projection binding snapshot differs from its source index")
33 by_table = defaultdict(list)
34 for template in templates:
35 if template.get("snapshot_id") != data.snapshot_id or template.get("status") != "accepted":
36 raise ValueError("Projection contract is not accepted for this snapshot")
37 by_table[template["source_table"]].append(template)
38 bindings, counts, unexamined = [], Counter(), []
39 gap_counts, gap_bundles = Counter(), {}
40 total = 0
41 for table, table_templates in sorted(by_table.items()):
42 table_count = index.db.execute(
43 "SELECT count(*) FROM card_sources s JOIN cards c ON c.card_id=s.card_id "
44 "WHERE c.table_name=? AND c.kind='definition'", (table,)).fetchone()[0]
45 total += table_count
46 report = index.coverage.get("by_table", {}).get(table, {})
47 if report.get("unknown_columns_not_examined"):
48 unexamined.append({"table": table, "records": table_count,
49 "reason": "unexamined_semantic_columns"})
50 continue
51 cursor = index.db.execute(
52 "SELECT s.card_id, s.record_id, s.row_number FROM card_sources s "
53 "JOIN cards c ON c.card_id=s.card_id WHERE c.table_name=? "
54 "AND c.kind='definition' ORDER BY s.row_number, s.record_id", (table,))
55 while counts["records_scanned"] < max_records:
56 batch = cursor.fetchmany(min(256, max_records - counts["records_scanned"]))
57 if not batch:
58 break
59 cards = {item["card_id"]: index.get(item["card_id"]) for item in batch}
60 columns = sorted({entry["column"] for card in cards.values()
61 for entries in card["fields"].values() for entry in entries})
62 rows = _read_rows(data, table, [item["row_number"] for item in batch], columns)
63 for source in batch:
64 counts["records_scanned"] += 1
65 row = rows.get(source["row_number"])
66 if row is None or data.record_id(table, row) != source["record_id"]:
67 counts["identity_changed"] += 1
68 continue
69 card = deepcopy(cards[source["card_id"]])
70 card.update(record_id=source["record_id"], row_number=source["row_number"])
71 for role, entries in list(card["fields"].items()):
72 card["fields"][role] = [{**entry, "value": str(row[entry["column"]]), "truncated": False}
73 for entry in entries if row.get(entry["column"]) is not None]
74 try:
75 binding = reuse_projection(data, table_templates, {"records": [card]})
76 except ValueError:
77 counts["unverifiable_source"] += 1
78 continue
79 if binding is None:
80 counts["uncovered_or_ambiguous"] += 1
81 if max_gap_examples:
82 reports = template_reuse_context(table_templates, {"records": [card]}, limit=1)
83 report = reports[0] if reports else {}
84 gap_key = digest([table, report.get("template_id"), report.get("unresolved_fields"),
85 report.get("protected_fields_changed"),
86 report.get("conflicting_accepted_endpoints")])[:24]
87 gap_counts[gap_key] += 1
88 if gap_key not in gap_bundles and len(gap_bundles) < max_gap_examples:
89 gap_bundles[gap_key] = {
90 "bundle_id": "projection_gap:" + digest([data.snapshot_id, gap_key])[:24],
91 "task_kind": "concept_induction", "records": [card],
92 "exact_alignment_record_ids": [card["record_id"]],
93 "projection_gap": {"gap_id": gap_key, "source_match_report": report,
94 "grouping_is_not_identity": True}}
95 else:
96 bindings.append(binding)
97 omitted = total - counts["records_scanned"]
98 for gap_key, bundle in gap_bundles.items():
99 bundle["projection_gap"]["records_with_same_gap_shape"] = gap_counts[gap_key]
100 return {"bindings": bindings, "repair_bundles": list(gap_bundles.values()), "coverage": {
101 "scope": "indexed_definition_rows_of_template_source_tables", "llm_calls": 0,
102 "eligible_source_records": total, "records_scanned": counts["records_scanned"],
103 "bindings_emitted": len(bindings), "records_not_scanned": omitted,
104 "uncovered_or_ambiguous": counts["uncovered_or_ambiguous"],
105 "identity_changed": counts["identity_changed"],
106 "unverifiable_source": counts["unverifiable_source"], "unexamined_tables": unexamined,
107 "max_records": max_records, "cartesian_products_created": 0,
108 "uncovered_gap_groups": len(gap_counts), "gap_examples_retained": len(gap_bundles),
109 "max_gap_examples": max_gap_examples,
110 "partial": bool(omitted or counts["uncovered_or_ambiguous"]
111 or counts["identity_changed"] or counts["unverifiable_source"])}}

114def _read_rows(data, table_name, row_numbers, columns):
115 """Fetch one fixed-size batch using original row positions and parameters."""
116 info = data.tables[table_name]
117 fields = list(dict.fromkeys([*info.get("pk", []), *columns]))
118 if not set(fields) <= set(info["column_names"]):
119 raise ValueError("Template refers to an unknown source column")
120 placeholders = ",".join("?" for _ in row_numbers)
121 selected = ", " + ", ".join(qi(field) for field in fields) if fields else ""
122 cursor = data.db.execute(f"SELECT __r2_row{selected} FROM {qi(info['sql_name'])} "
123 f"WHERE __r2_row IN ({placeholders})", row_numbers)
124 names = [item[0] for item in cursor.description]
125 return {row[0]: dict(zip(names, row)) for row in cursor.fetchall()}

128def _semantic_columns(data, card):
129 """Respect compiled card roles and retain known empty semantic fields."""
130 # Start from the card compiler's effective fields. Supplementing known
131 # empty fields must never restore a metadata/model role rejected by that
132 # compiler as an identifier binding or a rule parameter.
133 pairs = {(role, entry["column"]) for role, entries in card["fields"].items()
134 if role in _SEMANTIC_ROLES for entry in entries}
135 roles = _field_roles(data.tables[card["table"]])
136 pairs.update((role, field) for role, fields in roles.items() for field in fields
137 if role in _SEMANTIC_ROLES)
138 conflicts = {(item.get("proposed_role"), item.get("column"))
139 for item in card.get("role_conflicts", [])
140 if item.get("binding_only") and item.get("effective_role") == "reference"
141 and item.get("proposed_role") in ("name", "alias", "formula", "description", "identity", "scope")}
142 conflicts.update(("formula", item.get("column")) for item in card.get("role_conflicts", [])
143 if item.get("proposed_role") == "formula" and item.get("effective_role") == "scope"
144 and item.get("formula_status") == "fragment")
145 # Older cards named only alias exclusions. Do not infer a name conflict
146 # from a column-wide list; a valid name or scope on that column survives.
147 if not card.get("role_conflicts"):
148 conflicts.update(("alias", field) for field in card.get(
149 "binding_columns_excluded_from_semantic_pattern", []))
150 # Only the card's checked fragment contract can demote a formula role.
151 # An ordinary scope annotation alone never removes a genuine formula.
152 conflicts.update(("formula", item["column"]) for item in card.get("calculation_fragments", [])
153 if item.get("column") and item.get("formula_status") == "fragment"
154 and item.get("effective_role") in ("calculation_operator", "operand_reference"))
155 pairs.difference_update(conflicts)
156 return sorted(pairs)

163def _template(data, index, concept, alignment):
164 card_id = alignment.get("source_card_id")
165 source = index.db.execute("SELECT card_id, row_number FROM card_sources WHERE record_id=?",
166 (alignment["source_record_id"],)).fetchone()
167 if source is None or (card_id and source["card_id"] != card_id):
168 raise ValueError("accepted_source_not_in_current_index")
169 card = index.get(source["card_id"])
170 if card["kind"] != "definition" or card["snapshot_id"] != data.snapshot_id:
171 raise ValueError("source_is_not_current_definition_card")
172 if any(entry.get("truncated") for role, entries in card["fields"].items()
173 if role in _SEMANTIC_ROLES for entry in entries):
174 raise ValueError("semantic_preview_truncated")
175 if index.coverage.get("by_table", {}).get(card["table"], {}).get("unknown_columns_not_examined"):
176 raise ValueError("semantic_columns_unexamined")
177 pairs = sorted(set(_semantic_columns(data, card)) | {
178 ("unknown", field) for field in index.coverage.get("by_table", {}).get(card["table"], {}).get("unknown_columns_considered", [])})
179 # A representative can have an empty reference that later members fill.
180 # Use the complete table-level selection, not just its nonempty previews.
181 references = sorted(set(index.coverage.get("by_table", {}).get(card["table"], {}).get(
182 "reference_columns_considered", [])) | {
183 entry["column"] for entry in card["fields"].get("reference", [])})
184 if index.coverage.get("by_table", {}).get(card["table"], {}).get("reference_columns_not_examined"):
185 raise ValueError("reference_columns_unexamined")
186 rows = _read_rows(data, card["table"], [source["row_number"]],
187 [field for _, field in pairs] + references)
188 row = rows.get(source["row_number"])
189 if row is None or data.record_id(card["table"], row) != alignment["source_record_id"]:
190 raise ValueError("accepted_source_identity_changed")
191 for role, entries in card["fields"].items():
192 if role in _SEMANTIC_ROLES:
193 if any(row.get(entry["column"]) != entry["value"] for entry in entries):
194 raise ValueError("accepted_source_text_changed")
195 if not any(role in {"description", "formula"} and str(row.get(field) or "").strip()
196 for role, field in pairs):
197 raise ValueError("name_only_template_cannot_propagate")
198 evidence_ids = sorted(set(concept.get("evidence_ids", [])) | set(alignment.get("evidence_ids", [])))
199 if not evidence_ids or any(eid not in data.evidence for eid in evidence_ids):
200 raise ValueError("accepted_template_evidence_missing")
201 semantics = _raw_semantics(row, pairs)
202 template_id = "definition_template:" + digest([
203 data.snapshot_id, card["table"], concept["ontology_type_id"], semantics])[:24]
204 return {"id": template_id, "type_id": concept["ontology_type_id"],
205 "concept_id": concept["id"], "concept_reference_role": "accepted_type_template",
206 "table": card["table"], "pattern_id": card["pattern_id"],
207 "representative_record_id": alignment["source_record_id"],
208 "representative_card_id": card["card_id"], "representative_row_number": source["row_number"],
209 "semantic_fields": [{"role": role, "column": field, "value": value}
210 for role, field, value in semantics],
211 "semantic_fingerprint": digest(semantics),
212 "binding_columns_excluded_from_semantic_pattern": list(card.get(
213 "binding_columns_excluded_from_semantic_pattern", [])),
214 "role_conflicts": list(card.get("role_conflicts", [])),
215 "calculation_fragments": list(card.get("calculation_fragments", [])),
216 "reference_values": {field: row.get(field) for field in references},
217 "scope": card["scope"], "unit": card["unit"],
218 "evidence_ids": evidence_ids,
219 "identity_claim": "none; template reuse does not merge source entities"}

222def build_definition_memberships(data, index, group_result, *, max_records=100000):
223 """Compile accepted templates and inspect bounded source records with zero LLM calls.
224
225 Input ``index`` is an open SemanticCardIndex. The return keeps independent
226 source record identities and pending reference variants. ``max_records``
227 bounds inspected records, including rejected records; all remainder counts
228 refer to the complete indexed definition population, not the whole database.
229 """
230 if type(max_records) is not int or max_records < 0:
231 raise ValueError("definition_memberships.max_records must be nonnegative")
232 if (group_result.get("snapshot_id") != data.snapshot_id
233 or index.coverage.get("snapshot_id") != data.snapshot_id):
234 raise ValueError("Definition membership snapshots differ")
235 concepts = {item["id"]: item for item in group_result.get("concepts", [])
236 if item.get("ontology_level") == "type" and item.get("ontology_type_id")
237 and item.get("decision") == "accepted_by_automatic_checks"}
238 exact = defaultdict(set)
239 for alignment in group_result.get("record_alignments", []):
240 if alignment.get("mapping_kind") == "exact":
241 exact[alignment["source_record_id"]].add(alignment["concept_id"])
242 templates, errors = {}, []
243 for alignment in group_result.get("record_alignments", []):
244 concept = concepts.get(alignment.get("concept_id"))
245 if not concept or alignment.get("mapping_kind") != "exact":
246 continue
247 if exact[alignment["source_record_id"]] != {concept["id"]}:
248 errors.append({"record_id": alignment["source_record_id"], "reason": "conflicting_exact_alignment"})
249 continue
250 try:
251 template = _template(data, index, concept, alignment)
252 except ValueError as exc:
253 errors.append({"record_id": alignment["source_record_id"], "reason": str(exc)})
254 continue
255 key = (template["pattern_id"], template["semantic_fingerprint"], template["type_id"])
256 templates.setdefault(key, template)
257 by_pattern = defaultdict(list)
258 for template in templates.values():
259 by_pattern[template["pattern_id"]].append(template)
260 total = index.db.execute("SELECT count(*) FROM card_sources s JOIN cards c ON c.card_id=s.card_id "
261 "WHERE c.kind='definition'").fetchone()[0]
262 eligible = 0
263 for pattern in by_pattern:
264 eligible += index.db.execute("SELECT count(*) FROM card_sources s JOIN cards c ON c.card_id=s.card_id "
265 "WHERE c.pattern_id=? AND c.kind='definition'", (pattern,)).fetchone()[0]
266 memberships, pending, rejections = [], [], []
267 counts = Counter()
268 for pattern, pattern_templates in sorted(by_pattern.items()):
269 if counts["records_scanned"] >= max_records:
270 break
271 table = pattern_templates[0]["table"]
272 pairs = sorted({(entry["role"], entry["column"]) for template in pattern_templates
273 for entry in template["semantic_fields"]})
274 card_cache = {}
275 cursor = index.db.execute("SELECT s.card_id, s.record_id, s.row_number FROM card_sources s "
276 "JOIN cards c ON c.card_id=s.card_id WHERE c.pattern_id=? "
277 "AND c.kind='definition' ORDER BY s.row_number, s.record_id", (pattern,))
278 while counts["records_scanned"] < max_records:
279 batch = cursor.fetchmany(min(256, max_records-counts["records_scanned"]))
280 if not batch:
281 break
282 for source in batch:
283 if source["card_id"] not in card_cache:
284 card_cache[source["card_id"]] = index.get(source["card_id"])
285 reference_columns = sorted({column for template in pattern_templates
286 for column in template["reference_values"]} | {
287 entry["column"] for card in card_cache.values()
288 for entry in card["fields"].get("reference", [])})
289 rows = _read_rows(data, table, [source["row_number"] for source in batch],
290 [field for _, field in pairs] + reference_columns)
291 for source in batch:
292 counts["records_scanned"] += 1
293 row = rows.get(source["row_number"])
294 reason = None
295 card = card_cache[source["card_id"]]
296 if row is None or data.record_id(table, row) != source["record_id"]:
297 reason = "source_record_identity_changed"
298 matching = []
299 else:
300 signature = digest(_raw_semantics(row, pairs))
301 matching = [template for template in pattern_templates
302 if template["table"] == card["table"]
303 and template["semantic_fingerprint"] == signature
304 and template["scope"] == card["scope"]
305 and template["unit"] == card["unit"]]
306 if not matching:
307 reason = "full_semantic_values_differ"
308 accepted_concepts = exact.get(source["record_id"], set())
309 if matching and accepted_concepts:
310 matching = [template for template in matching if accepted_concepts == {template["concept_id"]}]
311 if not matching:
312 reason = "conflicting_exact_alignment"
313 if len({template["type_id"] for template in matching}) > 1:
314 reason = "conflicting_type_templates"
315 if reason:
316 counts[reason] += 1
317 rejections.append({"record_id": source["record_id"], "row_number": source["row_number"],
318 "card_id": source["card_id"], "reason": reason})
319 continue
320 template = min(matching, key=lambda item: item["id"])
321 refs = {column: row.get(column) for column in reference_columns}
322 # Preserve the complete variant. It is deliberately not an
323 # inferred relation, exact alignment, or same-as assertion.
324 variant = refs != {column: template["reference_values"].get(column) for column in reference_columns}
325 member = {"id": "definition_membership:" + digest([
326 data.snapshot_id, source["record_id"], template["type_id"]])[:24],
327 "kind": "definition_type_membership", "type_id": template["type_id"],
328 "concept_id": template["concept_id"], "concept_reference_role": "accepted_type_template",
329 "template_id": template["id"], "record_id": source["record_id"],
330 "table": table, "row_number": source["row_number"], "card_id": source["card_id"],
331 "pattern_id": pattern, "snapshot_id": data.snapshot_id,
332 "reference_values": refs, "reference_variants_pending": variant,
333 "mapping_kind": "shares_definition_type_template", "entity_identity_claim": False,
334 "status": "definition_template_match",
335 "verification_scope": "complete_source_semantics_only; reference_meaning_requires_separate_check",
336 "relationship_inheritance": False, "evidence_ids": template["evidence_ids"],
337 "verification": {"method": "complete_original_source_field_equality",
338 "semantic_fingerprint": template["semantic_fingerprint"],
339 "columns": [field for _, field in pairs]}}
340 memberships.append(member)
341 if variant:
342 pending.append({"record_id": source["record_id"], "table": table,
343 "row_number": source["row_number"], "card_id": source["card_id"],
344 "template_id": template["id"], "reference_values": refs,
345 "status": "unresolved_reference_variant",
346 "reason": "shared_definition_type_does_not_establish_reference_relations"})
347 missing = total-eligible
348 unscanned = eligible-counts["records_scanned"]
349 index_partial = bool(index.coverage.get("partial") or index.coverage.get("rows_omitted_cap", 0))
350 inspection_complete = not (missing or unscanned or rejections or errors or index_partial)
351 return {"snapshot_id": data.snapshot_id, "templates": list(templates.values()),
352 "memberships": memberships, "reference_variants_pending": pending,
353 "rejections": rejections, "template_errors": errors,
354 "coverage": {"scope": "indexed_definition_source_records_only", "llm_calls": 0,
355 "indexed_definition_records": total, "records_with_accepted_template": eligible,
356 "records_without_accepted_template": missing, "max_records": max_records,
357 "records_scanned": counts["records_scanned"], "records_not_scanned_due_to_limit": unscanned,
358 "memberships_emitted": len(memberships), "templates_compiled": len(templates),
359 "reference_variants_pending": len(pending), "rejected_records": len(rejections),
360 "rejection_reasons": {key: value for key, value in counts.items() if key != "records_scanned"},
361 "type_membership_complete": inspection_complete,
362 "reference_inspection_complete": inspection_complete,
363 "reference_linkage_complete": inspection_complete and not pending,
364 "reference_linkage_scope": "complete indexed definition inspection with no pending reference variants; does not certify business relations",
365 "source_index_partial": index_partial,
366 "index_rows_omitted_cap": index.coverage.get("rows_omitted_cap", 0),
367 "partial": not inspection_complete or bool(pending)}}
