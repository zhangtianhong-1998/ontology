96def test_configuration_projection_reads_current_witness_definition_not_seed(tmp_path):
97 data, plan, concepts, alignments, spec = _input(tmp_path)
98 try:
99 memberships = []
100 for index, alignment in enumerate(alignments):
101 item = plan.object_types[index]
102 current_eid = item.source_properties[0].evidence_ids[0]
103 old_eid = "other-seed:" + str(index)
104 data.evidence[old_eid] = deepcopy(data.evidence[current_eid])
105 data.evidence[old_eid]["source_ref"]["record_id"] = "different-seed-record"
106 item.source_properties[0].evidence_ids = [old_eid]
107 memberships.append({"id": "template:" + str(index), "object_type_id": item.id,
108 "record_id": alignment["source_record_id"], "mapping_kind": "template_instance",
109 "status": "accepted", "snapshot_id": data.snapshot_id, "evidence_ids": [current_eid]})
110 candidate = next(item for item in discover_configuration_relations(data, plan, concepts, [], [spec],
111 memberships=memberships)["candidates"] if item["status"] == "endpoint_verified_candidate")
112 # Complete row values may exist in the snapshot before any caller has
113 # registered their evidence IDs. Endpoint binding must read them here.
114 for membership in memberships:
115 data.evidence.pop(membership["evidence_ids"][0], None)
116 compiled, assertion = compile_configuration_relation(data, PROFILE, plan, candidate, _decision(),
117 concepts, [], memberships=memberships)
118 assert assertion["subject"] == candidate["source_definition"]["record_id"]
119 assert assertion["endpoint_scope"] == "typed_record_witness"
120 assert assertion["endpoint_mapping_kinds"] == {"source": "template_instance", "target": "template_instance"}
121 assert compiled.relation_types[0].evidence_scope == "one_configuration_witness_with_type_bindings"
122 assert validate_plan(compiled, data, PROFILE) == []
123 data.evidence[memberships[0]["evidence_ids"][0]]["source_ref"]["snapshot_id"] = "stale"
124 with pytest.raises(ValueError, match="complete snapshot record"):
125 compile_configuration_relation(data, PROFILE, plan, candidate, _decision(), concepts, [], memberships=memberships)
126 finally:
127 data.close()

205def test_configuration_assertion_materializes_only_its_replayed_record_pair(tmp_path):
206 data, plan, concepts, alignments, spec = _input(tmp_path)
207 try:
208 candidate = next(item for item in discover_configuration_relations(data, plan, concepts, alignments, [spec])[
209 "candidates"] if item["status"] == "endpoint_verified_candidate")
210 compiled, assertion = compile_configuration_relation(data, PROFILE, plan, candidate, _decision(), concepts, alignments)
211
212 def materialize(saved_assertion, candidates):
213 return materialize_ontology_instances(data, compiled, concepts=concepts,
214 record_alignments=alignments, record_relations=[saved_assertion],
215 configuration_candidates=candidates)
216
217 # Explicit configuration assertions have no physical RelationPlan;
218 # their source candidate and complete decision authorize this path.
219 assert not assertion.get("source_relation_plan_id")
220 accepted = materialize(assertion, [candidate])
221 assert accepted["pending"] == []
222 assert len(accepted["objects"]) == 2
223 assert len(accepted["assertions"]) == 1
224 edge = accepted["assertions"][0]
225 nodes = {item["id"]: item for item in accepted["objects"]}
226 assert nodes[edge["subject"]]["source_ref"]["record_id"] == assertion["source_record_pair"]["source"]
227 assert nodes[edge["object"]]["source_ref"]["record_id"] == assertion["source_record_pair"]["target"]
228 assert edge["configuration_candidate_id"] == candidate["id"]
229 assert edge["instance_scope"] == "definition_graph_not_observed_business_values"
230 assert accepted["coverage"]["cartesian_products_created"] == 0
231 assert accepted["coverage"]["llm_calls"] == 0
232
233 tampered = deepcopy(assertion)
234 tampered["configuration_decision"]["direction"] = "target_to_source"
235 for saved, candidates, reason in (
236 (assertion, [], "source candidate is absent"),
237 (tampered, [candidate], "predicate and direction"),
238 ):
239 rejected = materialize(saved, candidates)
240 assert len(rejected["objects"]) == 2
241 assert rejected["assertions"] == []
242 assert len(rejected["pending"]) == 1
243 assert rejected["pending"][0]["reason"] == "record_relation_replay_failed"
244 assert reason in rejected["pending"][0]["detail"]
245 finally:
246 data.close()
