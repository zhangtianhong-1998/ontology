1"""Revalidate recorded role responses and count patterns without a model client.
2
3This is a regression/cost preflight, not a fresh model evaluation. The source
4snapshot must match the recorded requests; missing responses fail explicitly.
5"""
6
7import argparse
8import asyncio
9import json
10import time
11from pathlib import Path
12
13from ontology_r2.column_role_inference import infer_column_role_candidates
14from ontology_r2.pipeline import load_config
15from ontology_r2.progress import ProgressReporter
16from ontology_r2.semantic_cards import SemanticCardIndex, build_semantic_cards
17from ontology_r2.storage import Dataset, digest, write_yaml
18
19
20class RecordedRoles:
21 """Replay source-bound responses; intentionally has no provider or network."""
22
23 def __init__(self, trace_path):
24 self.responses = {}
25 self.calls = 0
26 pending = None
27 for line in Path(trace_path).open(encoding="utf-8"):
28 event = json.loads(line)
29 if event.get("task") != "column_role_inference":
30 continue
31 if event.get("stage") == "llm_request":
32 pending = event["input"]
33 elif event.get("stage") == "llm_response" and pending:
34 self.responses[(pending["source_snapshot"], pending["table"])] = event["result"]
35 pending = None
36
37 async def ask(self, task, packet, schema):
38 if task != "column_role_inference":
39 raise AssertionError("This preflight only replays column roles")
40 self.calls += 1
41 return schema.model_validate(self.responses[(packet["source_snapshot"], packet["table"])])
42
43
44def main():
45 parser = argparse.ArgumentParser(description=__doc__)
46 parser.add_argument("--config", required=True)
47 parser.add_argument("--trace", required=True)
48 parser.add_argument("--output", required=True)
49 args = parser.parse_args()
50 config = load_config(args.config)
51 output = Path(args.output).resolve()
52 output.mkdir(parents=True, exist_ok=False)
53 (output / "work").mkdir()
54 started = time.monotonic()
55 progress = ProgressReporter(config.get("progress"))
56 replay = RecordedRoles(args.trace)
57 data = Dataset(config["dataset"], output / "work", config.get("memory_limit", "1GB"),
58 config.get("profiling"), progress=progress, privacy_config=config.get("privacy"))
59 try:
60 missing = sorted(table for table in data.tables
61 if (data.snapshot_id, table) not in replay.responses)
62 if missing:
63 raise ValueError("Missing recorded responses for the exact snapshot: " + ", ".join(missing))
64 roles = asyncio.run(infer_column_role_candidates(data, replay, config.get("column_role_inference")))
65 if any(table.get("error_type") for table in roles["tables"]):
66 raise ValueError("Role replay failed; inspect responses against the new contract")
67 data.column_role_report = roles
68 write_yaml(output / "column_role_candidates.yaml", roles)
69 options = config.get("instance_bundles", {})
70 cards = build_semantic_cards(
71 data, output / "work" / "semantic_cards.sqlite",
72 max_cards=options.get("max_cards", 200000),
73 max_field_chars=options.get("max_field_chars", 512),
74 max_unknown_fields_per_table=options.get("max_unknown_fields_per_table", 4),
75 progress=progress)
76 write_yaml(output / "semantic_card_coverage.yaml", cards["coverage"])
77 index = SemanticCardIndex(output / "work" / "semantic_cards.sqlite")
78 try:
79 counts = [dict(zip(("table", "kind", "cards", "patterns"), row)) for row in index.db.execute(
80 "SELECT table_name, kind, count(*), count(DISTINCT pattern_id) "
81 "FROM cards GROUP BY table_name, kind ORDER BY table_name, kind")]
82 finally:
83 index.close()
84 patterns = sum(item["patterns"] for item in counts if item["kind"] == "definition")
85 limit = min(options.get("max_concept_bundles", 0), options.get("max_pattern_seed_pool", 0))
86 manifest = {
87 "status": "offline_role_replay_complete", "llm_calls": 0,
88 "recorded_responses_replayed": replay.calls, "source_snapshot": data.snapshot_id,
89 "input_tables": len(data.tables), "input_records": sum(t["rows"] for t in data.tables.values()),
90 "config_hash": digest(config), "definition_patterns": patterns,
91 "concept_window_limit": limit, "patterns_outside_first_window": max(0, patterns - limit),
92 "by_table_and_kind": counts, "elapsed_seconds": round(time.monotonic() - started, 3),
93 "limits": ["Recorded responses, not fresh model predictions",
94 "Pattern count is a scheduling count, not a predicted LLM call count",
95 "Relations, semantic acceptance and template reuse are not measured here"],
96 }
97 write_yaml(output / "manifest.yaml", manifest)
98 print(json.dumps({key: value for key, value in manifest.items()
99 if key != "by_table_and_kind"}, ensure_ascii=False, indent=2))
100 finally:
101 data.close()
102
103
104if __name__ == "__main__":
105 main()
