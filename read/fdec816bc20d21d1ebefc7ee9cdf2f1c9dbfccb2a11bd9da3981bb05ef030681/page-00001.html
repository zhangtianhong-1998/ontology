14def _compile(data, key, name, scope, roles, *, metric=False, formula=None, source_scope=True):
15 definition = name + ("是苹果经营业务的收入减去成本。" if metric else "是可用于不同经营对象的金额合计。")
16 fields = {
17 "name": [{"column": "name", "value": name}],
18 "description": [{"column": "definition", "value": definition}],
19 "unit": [{"column": "unit", "value": "元"}],
20 "scope": [{"column": column, "value": value} for column, value in scope.items()]
21 if source_scope else [],
22 }
23 if formula:
24 fields["formula"] = [{"column": "formula", "value": formula}]
25 record = {"record_id": key, "table": "definitions", "kind": "definition",
26 "row_number": 1, "scope": scope, "unit": "元", "fields": fields}
27 decision = ConceptBundleDecision(
28 status="proposed", label=name, definition=definition,
29 root_type="Metric" if metric else "Measure", ontology_level="type",
30 classification_basis="business_driven_metric" if metric else "reusable_measure",
31 classification_quote=definition, business_object_quote="苹果" if metric else "",
32 scope_roles=roles,
33 alignments=[RecordAlignmentDecision(record_id=key, mapping_kind="exact", quote=name)],
34 )
35 concept, alignments = compile_concept(data, PROFILE, {"records": [record]}, decision, {})
36 return concept, alignments, _compiled_object_type(concept)

39@pytest.mark.parametrize("value", ["不限", "不限制", "不限定经营对象、地区或年份", "无限制", "all", "UNRESTRICTED"])
40def test_complete_unrestricted_declaration_retains_evidence_without_equality_condition(value):
41 data = SimpleNamespace(snapshot_id="snap", evidence={}, tables={})
42 concept, _, item = _compile(data, "revenue", "收入", {"limit": value, "currency": "CNY"},
43 {"limit": "unrestricted", "currency": "applicability"})
44 assert item.applicability_scope == {"currency": "CNY"}
45 assert concept["scope"] == {"limit": value, "currency": "CNY"}
46 assert concept["scope_roles"]["limit"] == "unrestricted"
47 assert concept["unrestricted_scope"] == {"limit": value}
48 source = next(source for source in item.source_properties if source.source_column == "limit")
49 assert data.evidence[source.evidence_ids[0]]["raw_fragment"] == value

68def test_metric_calculation_binds_unrestricted_measures_without_erasing_real_scopes():
69 data = SimpleNamespace(snapshot_id="snap", evidence={}, tables={})
70 groups = {"concepts": [], "record_alignments": []}
71 types = []
72 for key, name in (("revenue", "收入"), ("cost", "成本")):
73 concept, alignments, item = _compile(data, key, name,
74 {"limit": "不限定经营对象、地区或年份"}, {"limit": "unrestricted"})
75 groups["concepts"].append(concept)
76 groups["record_alignments"].extend(alignments)
77 types.append(item)
78 concept, alignments, metric = _compile(data, "profit", "苹果经营利润",
79 {"business_object": "苹果", "region_scope": "全国", "period_scope": "公历年"},
80 {"business_object": "applicability", "region_scope": "applicability", "period_scope": "applicability"},
81 metric=True, formula="收入 - 成本")
82 groups["concepts"].append(concept)
83 groups["record_alignments"].extend(alignments)
84 plan = BuildPlan(object_types=[*types, metric])
85 result = enrich_calculation_contracts(data, plan, groups)
86 assert result["calculations"][0]["status"] == "accepted"
87 assert {edge["target_type_id"] for edge in result["dependencies"]} == {item.id for item in types}
88 assert metric.applicability_scope["region_scope"] == "全国"
89 assert metric.applicability_scope["period_scope"] == "公历年"
90 # A genuinely narrower operand still cannot become a global dependency.
91 types[0].applicability_scope = {"region_scope": "仅华东"}
92 rejected = enrich_calculation_contracts(data, plan, groups)
93 assert rejected["calculations"][0]["status"] == "unresolved"
94 assert any(item.get("reason") == "scope_not_proven" for item in rejected["calculations"][0]["bindings"])
