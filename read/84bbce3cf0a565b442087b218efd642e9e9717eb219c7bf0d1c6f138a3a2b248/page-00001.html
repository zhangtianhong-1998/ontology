13def _index(tmp_path, *, partial=False, rows=None):
14 path = tmp_path / "cards.sqlite"
15 with sqlite3.connect(path) as db:
16 db.executescript("""
17 CREATE TABLE cards(card_id TEXT PRIMARY KEY, table_name TEXT, kind TEXT, snapshot_id TEXT);
18 CREATE TABLE card_sources(card_id TEXT, record_id TEXT, row_number INTEGER);
19 CREATE TABLE metadata(key TEXT PRIMARY KEY, value TEXT);
20 CREATE INDEX by_record ON card_sources(record_id);
21 """)
22 rows = rows if rows is not None else [
23 ("api", 1), ("api", 2), ("api", 3),
24 ("measure", 1), ("measure", 2), ("measure", 3),
25 ("wide", 1), ("wide", 2), ("zero_binding", 1),
26 ]
27 for i, (table, row) in enumerate(rows):
28 db.execute("INSERT INTO cards VALUES (?,?,?,?)", (f"c{i}", table, "definition", "s"))
29 db.execute("INSERT INTO card_sources VALUES (?,?,?)", (f"c{i}", f"{table}:{row}", row))
30 # A second card for the same source row must not increase the denominator.
31 if rows:
32 table, row = rows[0]
33 db.execute("INSERT INTO cards VALUES ('duplicate',?,'definition','s')", (table,))
34 db.execute("INSERT INTO card_sources VALUES ('duplicate',?,?)", (f"{table}:{row}", row))
35 db.execute("INSERT INTO cards VALUES ('reference','reference','reference','s')")
36 db.execute("INSERT INTO card_sources VALUES ('reference','reference:1',1)")
37 db.execute("INSERT INTO metadata VALUES ('coverage',?)", (json.dumps({
38 "snapshot_id": "s", "partial": partial, "rows_omitted_cap": 7 if partial else 0,
39 "rows_without_card": 19,
40 }),))
41 return path

74def test_unique_source_union_includes_exact_only_api_and_zero_binding_template_tables(tmp_path):
75 path = _index(tmp_path)
76 before = hashlib.sha256(path.read_bytes()).hexdigest()
77 memberships = ({"record_id": f"measure:{i}", "table": "measure", "row_number": i,
78 "type_id": "type", "mapping_kind": "shares_definition_type_template",
79 "status": "definition_template_match", "reference_variants_pending": True}
80 for i in (1, 2))
81 result = summarize_definition_coverage(path, _group(), plan=PLAN,
82 definition_memberships=memberships,
83 template_bindings=iter([_binding("measure", 2), _binding("measure", 2),
84 _binding("wide", 1), _binding("measure", 3, status="rejected")]),
85 template_projections=[_projection(t) for t in ("measure", "wide", "zero_binding")])
86 total = result["totals"]
87 assert total["indexed_definition_rows"] == 9
88 assert total["exact_alignment_rows"] == total["membership_rows"] == total["template_binding_rows"] == 2
89 assert total["rows_with_type_binding"] == 4
90 assert total["rows_without_type_binding"] == 5
91 assert total["rows_with_multiple_binding_paths"] == 2
92 assert total["type_binding_fraction"] == 4 / 9
93 assert total["template_scope_rows"] == 6
94 assert total["template_binding_fraction_within_template_scope"] == 2 / 6
95 assert total["rows_outside_template_scope"] == 3
96 assert total["bound_rows_outside_template_scope"] == 1
97 assert result["tables_outside_template_scope"] == ["api"]
98 assert result["by_table"]["api"]["rows_with_type_binding"] == 1
99 assert result["by_table"]["zero_binding"]["in_template_scope"] is True
100 assert result["by_table"]["zero_binding"]["rows_without_type_binding"] == 1
101 assert result["artifact_counts"]["template_accepted"] == 3 # duplicate evidence, one source row
102 assert hashlib.sha256(path.read_bytes()).hexdigest() == before
