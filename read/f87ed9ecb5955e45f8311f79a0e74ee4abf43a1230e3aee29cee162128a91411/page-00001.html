1"""Conservative equivalence of accepted business types from full source definitions.
2
3Equal names are candidate retrieval only. The compiler rechecks both complete
4definition records and never edits the accepted ontology plan; callers may use
5the returned canonical map only for pairs whose entire same-name group passed.
6"""

21ROOTS = frozenset(("GeneralObject", "Measure", "Metric", "Dimension", "Term"))
22_SPACE = re.compile(r"\s+")
23_PROSE_PUNCT = re.compile(r"[\s，,。.;；:：、]+")

26class EquivalenceDecision(Strict):
27 status: Literal["proposed", "no_change", "unresolved"]
28 source_type_id: str = ""
29 target_type_id: str = ""
30 source_description_evidence_id: str = ""
31 target_description_evidence_id: str = ""
32 # Full, byte-for-byte source descriptions. Short substrings could hide a
33 # qualifier or negation in the omitted portion.
34 source_description_quote: str = ""
35 target_description_quote: str = ""
36 source_formula_evidence_id: str | None = None
37 target_formula_evidence_id: str | None = None
38 source_formula_quote: str | None = None
39 target_formula_quote: str | None = None
40 semantic_equivalence_explanation: str = ""
41 reason: str = ""

44def _norm(value):
45 return unicodedata.normalize("NFKC", str(value or "")).casefold().strip()
46

48def _description_key(value):
49 return _PROSE_PUNCT.sub("", _norm(value))

52def _formula_key(value):
53 expression = _SPACE.sub("", _norm(value))
54 # The written name left of an assignment is not a calculation operand.
55 assignment = re.fullmatch(r"[\w\u3400-\u9fff.]+=([^=].*)", expression)
56 return assignment.group(1) if assignment else expression

59def _root_of(item: DerivedType, by_id):
60 parent, seen = item.parent, {item.id}
61 while parent in by_id:
62 if parent in seen:
63 return None
64 seen.add(parent)
65 parent = by_id[parent].parent
66 return parent if parent in ROOTS else None

69def _fragments(data, item):
70 """Return all complete description/formula fragments or ``None``.
71
72 One invalid fragment makes the type ineligible; we cannot selectively hide
73 a second, contradictory source definition from the comparison packet.
74 """
75 result = {"description": [], "formula": []}
76 cited = set(item.evidence_ids)
77 for prop in item.source_properties:
78 if prop.role not in result:
79 continue
80 if not prop.evidence_ids:
81 return None
82 for evidence_id in prop.evidence_ids:
83 evidence = data.evidence.get(evidence_id) or {}
84 ref = evidence.get("source_ref") or {}
85 raw = evidence.get("raw_fragment")
86 if (evidence_id not in cited
87 or evidence.get("origin") != "observed_record"
88 or evidence.get("raw_fragment_truncated")
89 or not isinstance(raw, str) or not raw.strip()
90 or ref.get("table") != prop.source_table
91 or ref.get("column") != prop.source_column
92 or ref.get("snapshot_id") != data.snapshot_id
93 or not ref.get("record_id")):
94 return None
95 result[prop.role].append({
96 "evidence_id": evidence_id, "value": raw,
97 "record_id": ref["record_id"], "source_ref": ref,
98 })
99 if not result["description"]:
100 return None
101 for role in result:
102 result[role].sort(key=lambda entry: entry["evidence_id"])
103 return result

106def _eligible_groups(plan, data):
107 by_id = {item.id: item for item in plan.object_types}
108 groups = defaultdict(list)
109 for item in plan.object_types:
110 if (item.category != "business_type"
111 or (item.derivation_kind, item.evidence_scope) not in {
112 ("exact_definition", "definition_record"),
113 ("template_projection", "projected_definition_template")}
114 or not item.label or not item.definition):
115 continue
116 root = _root_of(item, by_id)
117 if root is None:
118 continue
119 unit = unit_identity(item.unit)
120 # Two unknown units do not establish a compatible quantitative unit.
121 if root == "Metric" and not unit:
122 continue
123 # A Measure may carry a unit and need not declare one of the small
124 # built-in operators. The complete definition is checked below;
125 # operator equality remains part of the candidate signature.
126 fragments = _fragments(data, item)
127 if fragments is None:
128 continue
129 scope = tuple(sorted((_norm(k), _norm(v))
130 for k, v in item.applicability_scope.items()))
131 # Parameters preserve literal source values. Normalizing punctuation,
132 # case or whitespace here could erase a meaningful calculation setting.
133 parameters = tuple(sorted(item.definition_parameters.items()))
134 key = (root, _norm(item.label), unit, scope, item.aggregation_operator, parameters,
135 tuple(sorted(item.identity_qualifiers.items())),
136 tuple(sorted(item.classification_attributes.items())))
137 groups[key].append((item, fragments))
138 for values in groups.values():
139 values.sort(key=lambda pair: pair[0].id)
140 return groups

143def _candidate(data, group_key, source, target):
144 left, right = source[0], target[0]
145 identity = [data.snapshot_id, left.id, right.id]
146 if left.definition_parameters:
147 identity.append(left.definition_parameters)
148 return {
149 "candidate_id": "equivalence:" + digest(identity)[:24],
150 "source_type_id": left.id, "target_type_id": right.id,
151 "root_type": group_key[0], "label": left.label,
152 "unit": left.unit, "aggregation_operator": left.aggregation_operator,
153 "applicability_scope": dict(group_key[3]),
154 "definition_parameters": dict(left.definition_parameters),
155 "classification_attributes": dict(left.classification_attributes),
156 "status": "candidate_only",
157 }

160def propose_equivalence_candidates(plan: BuildPlan, data, max_pairs: int):
161 """Bounded same-name recall; a candidate is never an identity assertion."""
162 if type(max_pairs) is not int or max_pairs < 0:
163 raise ValueError("max_pairs must be a nonnegative integer")
164 groups = _eligible_groups(plan, data)
165 result = []
166 for key, members in sorted(groups.items(), key=lambda entry: repr(entry[0])):
167 for source, target in combinations(members, 2):
168 if len(result) >= max_pairs:
169 return result
170 result.append(_candidate(data, key, source, target))
171 return result

174def _selected_fragment(fragments, role, evidence_id, quote):
175 if not evidence_id or not quote:
176 raise ValueError(f"Missing full {role} source quotation")
177 found = next((fragment for fragment in fragments[role]
178 if fragment["evidence_id"] == evidence_id), None)
179 if found is None or found["value"] != quote:
180 raise ValueError(f"{role} quotation must equal its complete source fragment")
181 return found

184def _single_meaning(fragments, role, key):
185 values = {key(fragment["value"]) for fragment in fragments[role]}
186 if len(values) > 1:
187 raise ValueError(f"A type has conflicting complete {role} sources")
188 return next(iter(values), None)

191def compile_equivalence(data, core: BuildPlan, candidate, decision):
192 """Validate one model decision and emit source-backed pair evidence.
193
194 No plan mutation occurs. The stage decides whether a whole same-name
195 group forms a complete equivalence class and may publish a canonical map.
196 """
197 decision = EquivalenceDecision.model_validate(decision)
198 if decision.status != "proposed":
199 return None
200 source_id, target_id = candidate.get("source_type_id"), candidate.get("target_type_id")
201 if (not source_id or not target_id or source_id == target_id
202 or decision.source_type_id != source_id
203 or decision.target_type_id != target_id):
204 raise ValueError("Decision does not name the exact candidate pair")
205 current_types = {item.id: item for item in core.object_types}
206 if (source_id in current_types and target_id in current_types
207 and current_types[source_id].classification_attributes != current_types[target_id].classification_attributes):
208 raise ValueError("Source classification attributes differ; exact equivalence is unsupported")
209 if (source_id in current_types and target_id in current_types
210 and current_types[source_id].identity_qualifiers != current_types[target_id].identity_qualifiers):
211 raise ValueError("Definition identity qualifiers differ; cross-namespace equivalence is unsupported")
212 if (source_id in current_types and target_id in current_types
213 and current_types[source_id].definition_parameters != current_types[target_id].definition_parameters):
214 raise ValueError("Definition parameters differ; exact equivalence is unsupported")
215 groups = _eligible_groups(core, data)
216 group = next((members for members in groups.values()
217 if {source_id, target_id} <= {item.id for item, _ in members}), None)
218 if group is None:
219 raise ValueError("Pair lacks same-root, same-name, same-unit, same-scope complete definitions")
220 by_id = {item.id: (item, fragments) for item, fragments in group}
221 source, source_fragments = by_id[source_id]
222 target, target_fragments = by_id[target_id]
223 group_key = next(key for key, members in groups.items()
224 if {source_id, target_id} <= {item.id for item, _ in members})
225 expected = _candidate(data, group_key, by_id[source_id], by_id[target_id])
226 if candidate.get("candidate_id") != expected["candidate_id"]:
227 raise ValueError("Candidate identity changed")
228 if candidate.get("definition_parameters", {}) != source.definition_parameters:
229 raise ValueError("Candidate definition parameters changed")
230 if candidate.get("classification_attributes", {}) != source.classification_attributes:
231 raise ValueError("Candidate classification attributes changed")
232 left = _selected_fragment(source_fragments, "description",
233 decision.source_description_evidence_id,
234 decision.source_description_quote)
235 right = _selected_fragment(target_fragments, "description",
236 decision.target_description_evidence_id,
237 decision.target_description_quote)
238 if len(decision.semantic_equivalence_explanation.strip()) < 6:
239 raise ValueError("Equivalence requires a substantive semantic explanation")
240 source_description = _single_meaning(source_fragments, "description", _description_key)
241 target_description = _single_meaning(target_fragments, "description", _description_key)
242 source_formula = _single_meaning(source_fragments, "formula", _formula_key)
243 target_formula = _single_meaning(target_fragments, "formula", _formula_key)
244 one_formula_missing = (source_formula is None) != (target_formula is None)
245 if one_formula_missing:
246 # A formula written in a *description* is still source text. Accept
247 # this narrow case only if the missing-side complete description is
248 # exactly the same assignment/expression after whitespace removal.
249 present_formula = source_formula or target_formula
250 missing_descriptions = (target_fragments["description"] if source_formula
251 else source_fragments["description"])
252 if (not present_formula or not all(
253 _formula_key(x["value"]) == present_formula
254 for x in missing_descriptions)):
255 raise ValueError("A formula exists on only one side without the same full expression in its definition")
256 if source_formula is not None:
257 _selected_fragment(source_fragments, "formula",
258 decision.source_formula_evidence_id,
259 decision.source_formula_quote)
260 elif decision.source_formula_evidence_id or decision.source_formula_quote:
261 raise ValueError("Formula quotation supplied without source formula")
262 if target_formula is not None:
263 _selected_fragment(target_fragments, "formula",
264 decision.target_formula_evidence_id,
265 decision.target_formula_quote)
266 elif decision.target_formula_evidence_id or decision.target_formula_quote:
267 raise ValueError("Formula quotation supplied without target formula")
268 if source_formula is not None and target_formula is not None and source_formula != target_formula:
269 raise ValueError("Complete calculation formulas differ")
270 if source_description != target_description:
271 # No fixed qualifier vocabulary can cover all scope distinctions. An
272 # identical formula with e.g. 直营 vs 加盟 is still a different type.
273 # Paraphrase alignment needs a separate, evidence-backed mechanism;
274 # this canonical map accepts only equal complete descriptions.
275 raise ValueError("Complete source descriptions differ")
276 source_records = {(x["source_ref"]["table"], x["record_id"])
277 for x in source_fragments["description"]}
278 target_records = {(x["source_ref"]["table"], x["record_id"])
279 for x in target_fragments["description"]}
280 if source_records & target_records:
281 raise ValueError("Two types cite the same definition record; classification conflict")
282 canonical = min(source_id, target_id)
283 identity = [data.snapshot_id, source_id, target_id]
284 if source.definition_parameters:
285 identity.append(source.definition_parameters)
286 return {
287 "id": "type_equivalence:" + digest(identity)[:24],
288 "source_type_id": source_id, "target_type_id": target_id,
289 "canonical_type_id": canonical,
290 "root_type": group_key[0], "label": source.label,
291 "unit": source.unit, "aggregation_operator": source.aggregation_operator,
292 "applicability_scope": source.applicability_scope,
293 "definition_parameters": dict(source.definition_parameters),
294 "classification_attributes": dict(source.classification_attributes),
295 "source_description_evidence_ids": [x["evidence_id"] for x in source_fragments["description"]],
296 "target_description_evidence_ids": [x["evidence_id"] for x in target_fragments["description"]],
297 "source_formula_evidence_ids": [x["evidence_id"] for x in source_fragments["formula"]],
298 "target_formula_evidence_ids": [x["evidence_id"] for x in target_fragments["formula"]],
299 "source_description_quote": left["value"],
300 "target_description_quote": right["value"],
301 "source_formula_quote": decision.source_formula_quote,
302 "target_formula_quote": decision.target_formula_quote,
303 "semantic_equivalence_explanation": decision.semantic_equivalence_explanation.strip(),
304 "evidence_ids": sorted({x["evidence_id"] for values in
305 (source_fragments, target_fragments)
306 for role in ("description", "formula") for x in values[role]}),
307 "identity_scope": "accepted_type_equivalence_in_source_snapshot",
308 }
