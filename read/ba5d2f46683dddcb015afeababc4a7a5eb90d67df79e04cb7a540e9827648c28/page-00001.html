8def repair_settings(options, total_calls):
9 options = options or {}
10 if not isinstance(options, dict):
11 raise ValueError('targeted_repair must be a mapping')
12 defaults = {'enabled': False, 'max_rounds': 2, 'max_tasks_per_round': 24,
13 'max_examples_per_task': 3, 'max_calls_per_round': min(100, total_calls),
14 'reserve_calls_for_followup': 0}
15 if set(options) - set(defaults):
16 raise ValueError('Unknown targeted_repair settings')
17 result = {**defaults, **options}
18 if type(result['enabled']) is not bool:
19 raise ValueError('targeted_repair.enabled must be boolean')
20 bounds = {'max_rounds': (1, 5), 'max_tasks_per_round': (1, 1000),
21 'max_examples_per_task': (1, 10), 'max_calls_per_round': (0, total_calls),
22 'reserve_calls_for_followup': (0, total_calls)}
23 for key, (lo, hi) in bounds.items():
24 if type(result[key]) is not int or not lo <= result[key] <= hi:
25 raise ValueError(f'targeted_repair.{key} must be an integer in {lo}..{hi}')
26 return result

66 'observed_value_tasks_not_selected': 0}

88 tasks['coverage']['groups_not_selected'] = tasks['coverage'].get('groups_not_selected', 0) + extra
89 tasks['coverage']['unchanged_tasks_deferred'] = deferred
90 tasks['coverage']['input_blocked_tasks_deferred'] = blocked_count
91 # Discovery is refreshed after every changed round. Count its current
92 # omissions, rather than retaining gaps a later round already repaired.
93 pending = {'repair_tasks_not_attempted': 0,
94 'groups_not_selected': tasks['coverage'].get('groups_not_selected', 0),
95 'observed_value_tasks_not_selected': tasks['coverage'].get(
96 'observed_value_tasks_not_selected', 0),
97 'unchanged_tasks_deferred': deferred,
98 'input_blocked_tasks_deferred': blocked_count}

102 report['observed_value_tasks'] = list(observed.values())
103 if index is not None and report['observed_value_tasks']:

124 data, profile, result['plan'], tasks['bundles'], llm, prior_result=result,
125 max_bundles=len(tasks['bundles']), review=bundle_options.get('review', True),
126 max_repairs_per_bundle=bundle_options.get('max_repairs_per_bundle', 0),

128 progress=progress, on_checkpoint=on_checkpoint)
129 attempted_ids = {step['bundle_id'] for step in result.get('steps', [])

131 attempted_fingerprints.update(repair_evidence_fingerprint(bundle)

154 'discovery_limit' if any(pending.values()) else
155 'no_semantic_change')

158 # New group checkpoints retain the original source inventory and may prove

162 result['partial'] = bool(result.get('partial') or result['bundles_skipped'] or any(pending.values()))
163 result.setdefault('coverage', {})['first_pass'] = first_coverage

169 'model_calls': llm.calls - start_calls,
170 'first_pass_unattempted': first_skipped,
171 'observed_value_tasks': len(report['observed_value_tasks']),
