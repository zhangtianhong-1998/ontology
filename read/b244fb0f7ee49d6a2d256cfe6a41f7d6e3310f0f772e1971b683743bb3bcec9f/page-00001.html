150def validate_semantics(root):
151 """Check relationships independently from the expected ontology output."""
152 root = Path(root)
153 def read(table):
154 with (root / "data" / f"{table}.csv").open(newline="", encoding="utf-8") as f:
155 yield from csv.DictReader(f)
156 measures = list(read("fruit_measure_def"))
157 symbols = {r["parameter_name"] for r in measures}
158 names = {r["measure_name"] for r in measures}
159 by_code = {r["measure_code"]: r for r in measures}
160 formulas = 0
161 for table, field in (("fruit_metric_detail", "calculation_formula"), ("fruit_metric_attr", "metric_formula"), ("fruit_measure_def", "source_field")):
162 for row in read(table):
163 expr = row[field]
164 if not expr:
165 continue
166 rhs = expr.split("=", 1)[-1].strip()
167 tree = ast.parse(rhs, mode="eval")
168 unknown = {n.id for n in ast.walk(tree) if isinstance(n, ast.Name)} - symbols
169 if unknown:
170 raise ValueError(f"Unbound formula symbols in {table}: {sorted(unknown)}")
171 formulas += 1
172 metrics = {r["metric_code"]: r for r in read("fruit_metric_detail")}
173 for row in read("fruit_metric_attr"):
174 source = metrics.get(row["metric_code"])
175 if source is None or source["calculation_formula"] != row["metric_formula"] or source["unit"] != row["unit"]:
176 raise ValueError("Metric attributes conflict with their definition")
177 if row["measure_code"] not in by_code:
178 raise ValueError("Metric attribute measure reference has no definition")
179 for row in read("fruit_dashboard_card"):
180 if row["metric_code"] not in metrics or row["measure_name"] not in names:
181 raise ValueError("Dashboard references have no definitions")
182 for row in measures:
183 if row["measure_name"] == "合格率" and (row["parameter_name"] != "qualified_rate" or "tested_quantity" not in row["source_field"]):
184 raise ValueError("Qualification rate has an inconsistent calculation")
185 report = {"semantic_fixture_version": 3, "formulas_with_resolved_symbols": formulas,
186 "distinct_quantity_symbols": len(symbols), "metric_definition_rows": len(metrics),
187 "checks": ["formula_symbols", "metric_attribute_formula_unit", "quantity_references", "qualification_formula"],
188 "boundary": "Input consistency only; not extraction accuracy or an ontology gold standard"}
189 print(json.dumps(report, ensure_ascii=False, indent=2), flush=True)
190 return report
