530def _order_alias_proposals(proposals, candidates, alias_leads):
531 """Share a fixed validation budget across source fields and target tables.
532
533 Both orientations have independent turns. Sampled value support only
534 orders leads; full-input uniqueness is still checked by the normal DSL.
535 """
536 leads = {lead['candidate_id']: lead for lead in alias_leads}
537 def object_payload(lead):
538 for example in lead.get('examples', []):
539 for side in ('source', 'target'):
540 value = example.get(side, {}).get('raw_value', '')
541 if not isinstance(value, str) or not value.lstrip().startswith('{'):
542 continue
543 try:
544 if isinstance(json.loads(value), dict):
545 return True
546 except (ValueError, TypeError):
547 pass
548 return False
549 payload_leads = {cid for cid, lead in leads.items() if object_payload(lead)}
550 def source(item):
551 value = candidates[item[0].candidate_id]['source']
552 return value['table'], value['field']
553 def quality(item):
554 candidate = candidates[item[0].candidate_id]
555 lead = leads[candidate['alias_recall_candidate_id']]
556 stats = lead.get('checks', {})
557 forward = candidate['source'] == lead['source']
558 target_side, source_side = ('target', 'source') if forward else ('source', 'target')
559 keys = stats.get('shared_normalized_keys_in_sample', 0)
560 matched = stats.get(target_side + '_raw_values_matched_in_sample', 0)
561 # Several target raw values per key is a collision hint, not proof of
562 # ambiguous target records. Repetition across source records is harmless.
563 # Splitting JSON documents at commas is weaker alias evidence than a
564 # plain lexical list. Keep these leads for later technical verification.
565 return (lead['candidate_id'] in payload_leads,
566 bool(stats.get('numeric_overlap_only')), matched > keys,
567 -keys, -stats.get(source_side + '_sample_coverage', 0),
568 candidate['target']['table'], candidate['target']['field'], item[0].candidate_id)
569 by_source = defaultdict(lambda: defaultdict(list))
570 for item in proposals:
571 by_source[source(item)][candidates[item[0].candidate_id]['target']['table']].append(item)
572 ranks = {}
573 for families in by_source.values():
574 for items in families.values():
575 items.sort(key=quality)
576 keys = sorted(families, key=lambda key: (quality(families[key][0]), key))
577 rank = 0
578 for offset in range(max(map(len, families.values()), default=0)):
579 for key in keys:
580 if offset < len(families[key]):
581 ranks[families[key][offset][0].candidate_id] = rank
582 rank += 1
583 return _fair_source_order(proposals, source,
584 lambda item: (ranks[item[0].candidate_id], quality(item)))

587def _alias_selection_coverage(proposals, candidates, rules):
588 def groups(values):
589 return ({v['source']['table'] for v in values},
590 {(v['source']['table'], v['source']['field']) for v in values},
591 {(v['source']['table'], v['source']['field'], v['target']['table']) for v in values})
592 full = groups([candidates[p.candidate_id] for p, _ in proposals])
593 # Explicit/Agent proposals may have verified the same canonical rule first.
594 # Origin is provenance, not identity; scope/selector/transform must all agree.
595 alias_rule_ids = {_rule_id(candidates[p.candidate_id], p) for p, _ in proposals}
596 verified = groups([r for r in rules if r['rule_id'] in alias_rule_ids
597 and r['verification']['scan_scope'] == 'full_input'
598 and r['verification'].get('error') is None
599 and isinstance(r['verification'].get('checks'), dict)
600 and 'eligible_references' in r['verification']['checks']])
601 return {'policy': 'source_table_field_target_table_direction_round_robin',
602 'sample_support_is_not_verification': True,
603 **{name: {'available': len(all_), 'verified': len(some),
604 'unverified': len(all_ - some)}
605 for name, all_, some in zip(('source_tables', 'source_fields',
606 'target_families'), full, verified)}}
