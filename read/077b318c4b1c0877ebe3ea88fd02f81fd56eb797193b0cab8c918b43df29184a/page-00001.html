13def direct_mapping(data):

17 record_types.append(record_type)

29def traversal(data):
30 parents = {name: {f["referenced_schema"] + "." + f["referenced_table"] for f in t["foreign_keys"]
31 if f["referenced_schema"] + "." + f["referenced_table"] in data.tables
32 and f["referenced_schema"] + "." + f["referenced_table"] != name}
33 for name, t in data.tables.items()}
34 pending, order = set(parents), []
35 while pending:
36 ready = sorted(n for n in pending if not parents[n].intersection(pending))
37 if not ready:
38 break
39 order.extend(ready)
40 pending.difference_update(ready)
41 return order + sorted(pending), sorted(pending)

51def _prompt_table(data, name, *, sample=False, focus_fields=(), compact=False):
52 table = data.tables[name]
53 roles = classify_columns(table)
54 semantic = {item["column"] for item in roles if item["include_in_semantic_prompt"]}
55 visible = semantic
56 if compact:
57 # Neighbor tables are context, not another unit to plan. Keep the
58 # checked join keys and a few fields that explain their meaning.
59 visible = (semantic & set(_example_columns(table, None, limit=6))) | set(focus_fields)
60 profiles = {item["column"]: item for item in table["profiles"]}
61 result = {
62 "name": name, "comment": table.get("table_comment"),
63 "columns": [column for column in table["columns"] if column["column_name"] in visible],
64 "deterministic_bindings": [item["deterministic_binding"] for item in roles
65 if item["role"] in ("audit_time", "audit_metadata", "technical_identifier")
66 and (not compact or item["column"] in visible)],
67 "empty_in_input": [item["column"] for item in roles if item["role"] == "empty"],
68 # Schema/profile evidence remains in the local artifacts. Repeating its
69 # per-column counters in every table packet adds no semantic context.
70 "column_roles": [{"column": item["column"], "role": item["role"]}
71 for item in roles if item["column"] in visible],
72 "constraints": table["constraints"], "foreign_keys": table["foreign_keys"],
73 "profiles": [{"column": column, "usable_count": profiles[column]["usable_count"],
74 "approx_distinct_usable": profiles[column]["approx_distinct_usable"]}
75 for column in table["column_names"] if column in visible],
76 }
77 if compact:
78 result["semantic_columns_not_expanded"] = len(semantic - visible)
79 if sample:
80 sample_columns = _example_columns(table, None, limit=12)
81 result["sampled_columns"] = sample_columns
82 result["sample_columns_not_shown"] = [column for column in table["column_names"]
83 if column in semantic and column not in sample_columns]
84 result["sample"] = [{
85 "record_id": data.record_id(name, row), "row_number": row["__r2_row"],
86 "values": {column: row[column][:160] if row[column] is not None else None
87 for column in sample_columns},
88 "truncated_columns": [column for column in sample_columns
89 if row[column] is not None and len(row[column]) > 160],
90 } for row in data.rows(name, limit=3)]
91 return result

94def _example_columns(table, key, limit=4):
95 roles = {item["column"]: item for item in classify_columns(table)}
96 ranked = []
97 for index, column in enumerate(table["columns"]):
98 name = column["column_name"]
99 if name == key or not roles[name]["include_in_semantic_prompt"]:
100 continue
101 description = name + " " + str(column.get("column_comment") or "")
102 priority = 0 if re.search(r"name|alias|definition|description|formula|meaning|scope|unit|名称|别名|定义|描述|公式|口径|单位", description, re.I) else 1
103 ranked.append((priority, index, name))
104 return [name for _, _, name in sorted(ranked)[:limit]]

107def unit_context(data, name, candidate_targets=(), candidate_fields=None):
108 targets = {f["referenced_schema"] + "." + f["referenced_table"] for f in data.tables[name]["foreign_keys"]}
109 candidate_targets = [target for target in dict.fromkeys(candidate_targets)
110 if target != name and target in data.tables]
111 relevant = {name} | targets.intersection(data.tables) | set(candidate_targets[:3]).intersection(data.tables)
112 catalog = []
113 for table_name, table in data.tables.items():
114 hints = [column for column, role in zip(table["columns"], classify_columns(table))

116 catalog.append({"name": table_name, "comment": table.get("table_comment"),
117 "column_hints": [{"name": col["column_name"], "comment": col.get("column_comment")}
118 for col in hints[:2]],
119 "other_semantic_column_count": max(0, len(hints) - 2)})
120 candidate_fields = candidate_fields or {}
121 return {"tables": [_prompt_table(data, table_name, sample=table_name == name,
122 compact=table_name != name,
123 focus_fields=candidate_fields.get(table_name, ()))
124 for table_name in sorted(relevant)],
125 "table_catalog": catalog,
126 "candidate_targets_not_expanded": candidate_targets[3:],
127 "evidence": [{"id": item["id"]} for item in data.evidence.values()
128 if item.get("origin") != "observed_record"
129 and item["source_ref"].get("table") in relevant]}

176def merge_delta(core, delta, unit, data, profile):
177 """No deletion, no mutation of core, and no overwrite of another unit."""
178 errors = []
179 if any(t.table != unit for t in delta.tables):
180 errors.append("delta cannot change another table")
181 if any(r.source_table != unit for r in delta.relations):
182 errors.append("delta relation must originate in current table")
183 candidate = core.model_copy(deep=True)
184 for key in ("object_types", "relation_types", "relations"):
185 existing = {t.id: t for t in getattr(candidate, key)}
186 for item in getattr(delta, key):
187 if item.id in existing and existing[item.id] != item:
188 errors.append(f"conflicting {key} id: {item.id}")
189 elif item.id not in existing:
190 getattr(candidate, key).append(item.model_copy(deep=True))
191 existing[item.id] = item
192 for update in delta.tables:
193 current = next((t for t in candidate.tables if t.table == update.table), None)
194 if current is None:
195 errors.append("unknown delta table: " + update.table)
196 continue
197 attrs = dict(current.attributes)
198 for key, value in update.attributes.items():
199 key = key if key.startswith(f"column:{unit}.") else f"attribute:{unit}:{key}"
200 if key in attrs and attrs[key] != value:
201 errors.append("cannot overwrite attribute mapping: " + key)
202 else:
203 attrs[key] = value
204 replacement = update.model_copy(update={"attributes": attrs})
205 candidate.tables[candidate.tables.index(current)] = replacement
206 candidate.knowledge_questions = []
207 errors += validate_plan(candidate, data, profile)
208 for t in candidate.tables:
209 if not set(data.tables[t.table]["column_names"]) <= set(t.attributes.values()):
210 errors.append("direct column mapping lost: " + t.table)
211 if errors:
212 raise ValueError("; ".join(errors))
213 return candidate
